# Numeric prefix sorts these to the top of the filter list.

cm_location_rgo_bonus = {
	scope = location
	tag = building
	range = {
		min = 0
		# 0.25 = max modifier stack: 0.1 base (in_game/common/auto_modifiers/country.txt:100)
		# + 0.05 parliament agenda + 2x0.025 advances + 0.05 local_quality_standards_modifier
		# (main_menu/common/static_modifiers/location.txt:3804).
		max = 0.25
		step = 0.01
		format = search_filter_cm_location_rgo_bonus_format
	}
	trigger = {
		cm_rgob_location_bonus >= scope:min_value
		OR = {
			scope:max_value >= 0.25
			cm_rgob_location_bonus <= scope:max_value
		}
	}
}
