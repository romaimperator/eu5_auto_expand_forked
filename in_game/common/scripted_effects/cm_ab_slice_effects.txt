# Auto-Build market slice: one cycle covers a rotating subset of the country's markets
# instead of all of them, so a very large nation's staging cost matches a medium one's.
# Markets are the slice unit because they partition the country's locations.
# Markets are stored as variable-map keys rather than list targets, the shape the food
# histograms already use.

# Market-locations admitted per cycle. A country at or under this size takes every market
# every cycle, which is what it did before the slice existed.
@cm_ab_slice_location_budget = 300

# Picks this cycle's markets. A cursor walks the capital's market, then each market holding a
# governor, then one market from the rotation, and repeats, so the priority markets come round
# again after every single other market rather than once per full rotation. Root is the country.
cm_ab_pick_slice = {
	clear_variable_map = cm_ab_slice_now
	# Temporary because cm_ab_slice_has_unseen saves the same name, and a temporary target
	# over a permanent one of the same name is a logged warning.
	save_temporary_scope_as = cm_ab_slice_country
	if = {
		limit = { NOT = { cm_ab_slice_has_unseen = yes } }
		clear_variable_map = cm_ab_slice_seen
	}
	cm_ab_slice_build_priority = yes
	set_local_variable = { name = cm_ab_slice_np value = 0 }
	every_in_list = {
		variable = cm_ab_slice_prio
		change_local_variable = { name = cm_ab_slice_np add = 1 }
	}
	if = {
		limit = { NOT = { has_variable = cm_ab_slice_cursor } }
		set_variable = { name = cm_ab_slice_cursor value = 0 }
	}
	# The rotation slot sits one past the priority markets, and losing a governor can leave
	# the cursor beyond the shortened sequence.
	if = {
		limit = { var:cm_ab_slice_cursor > local_var:cm_ab_slice_np }
		set_variable = { name = cm_ab_slice_cursor value = 0 }
	}
	set_local_variable = { name = cm_ab_slice_room value = @cm_ab_slice_location_budget }
	if = {
		limit = { var:cm_ab_slice_cursor < local_var:cm_ab_slice_np }
		cm_ab_slice_take_priority = yes
	}
	else = {
		cm_ab_slice_take_rotation = yes
	}
	change_variable = { name = cm_ab_slice_cursor add = 1 }
	cm_ab_slice_fill_budget = yes
}

# The capital's market followed by each distinct market holding a governor. A variable list
# holds each target once, so a governor sharing the capital's market collapses into it.
# Root is the country, expects scope:cm_ab_slice_country.
cm_ab_slice_build_priority = {
	clear_variable_list = cm_ab_slice_prio
	if = {
		limit = { exists = capital }
		capital.market = {
			save_scope_as = cm_ab_slice_mkt
			scope:cm_ab_slice_country = { add_to_variable_list = { name = cm_ab_slice_prio target = scope:cm_ab_slice_mkt } }
		}
	}
	every_owned_location = {
		limit = { cm_location_has_governor = yes }
		market = {
			save_scope_as = cm_ab_slice_mkt
			scope:cm_ab_slice_country = { add_to_variable_list = { name = cm_ab_slice_prio target = scope:cm_ab_slice_mkt } }
		}
	}
}

# Admits the priority market the cursor is on. The index test is an inner if, not the
# iterator's limit, because a limit does not see the increments its own body makes.
# Root is the country, expects scope:cm_ab_slice_country and the cm_ab_slice_* locals.
cm_ab_slice_take_priority = {
	set_local_variable = { name = cm_ab_slice_i value = 0 }
	every_in_list = {
		variable = cm_ab_slice_prio
		if = {
			limit = { local_var:cm_ab_slice_i = scope:cm_ab_slice_country.var:cm_ab_slice_cursor }
			cm_ab_slice_admit = yes
		}
		change_local_variable = { name = cm_ab_slice_i add = 1 }
	}
}

# Admits the next market the rotation still owes a visit.
# Root is the country, expects scope:cm_ab_slice_country and the cm_ab_slice_* locals.
cm_ab_slice_take_rotation = {
	set_local_variable = { name = cm_ab_slice_rot value = 0 }
	every_market_present_in_country = {
		if = {
			limit = {
				local_var:cm_ab_slice_rot = 0
				cm_ab_market_unseen_this_rotation = yes
			}
			set_local_variable = { name = cm_ab_slice_rot value = 1 }
			cm_ab_slice_admit = yes
		}
	}
}

# Spends whatever the cursor's market left of the location budget, which is what keeps a
# country small enough to fit inside it staging every market every cycle.
# Root is the country, expects scope:cm_ab_slice_country and the cm_ab_slice_* locals.
cm_ab_slice_fill_budget = {
	every_market_present_in_country = {
		if = {
			limit = {
				cm_ab_market_unseen_this_rotation = yes
				local_var:cm_ab_slice_room >= { value = market_location_count }
			}
			cm_ab_slice_admit = yes
		}
	}
}

# Root is the market, expects scope:cm_ab_slice_country and local_var:cm_ab_slice_room.
cm_ab_slice_admit = {
	save_scope_as = cm_ab_slice_mkt
	if = {
		limit = {
			scope:cm_ab_slice_country = {
				NOT = { is_key_in_variable_map = { name = cm_ab_slice_now target = scope:cm_ab_slice_mkt } }
			}
		}
		change_local_variable = { name = cm_ab_slice_room subtract = market_location_count }
		scope:cm_ab_slice_country = {
			add_to_variable_map = { name = cm_ab_slice_now key = scope:cm_ab_slice_mkt value = 1 }
			add_to_variable_map = { name = cm_ab_slice_seen key = scope:cm_ab_slice_mkt value = 1 }
			# This cycle re-scores the market, so what its last visit recorded goes now rather than
			# at the rotation boundary.
			cm_ab_ind_drop_market_memory = yes
		}
	}
}

# A new tag's markets are different, so the rotation restarts rather than carrying over.
# Root is the country.
cm_ab_clear_slice_state = {
	clear_variable_map = cm_ab_slice_now
	clear_variable_map = cm_ab_slice_seen
	clear_variable_list = cm_ab_slice_prio
	remove_variable = cm_ab_slice_cursor
	clear_variable_map = cm_ab_ind_gd_anchor
	clear_variable_map = cm_ab_ind_gd_loc
	clear_variable_map = cm_ab_ind_gd_bt
	clear_variable_map = cm_ab_ind_mkt_best
	clear_variable_map = cm_ab_ind_mkt_cost
	clear_variable_map = cm_ab_ind_mkt_bt
}
