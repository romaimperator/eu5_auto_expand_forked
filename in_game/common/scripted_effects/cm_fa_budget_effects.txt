# Full-automation budget engine: monthly per-category pools set from the treasury above
# the reserve, and the per-category dispatch that runs each drain and passes its unspent
# pool down the player's category order. A category buys its best pick whenever the treasury
# covers it, and what it spent past its own allotment becomes a debt it settles out of the
# allotments that follow. Pools and debts are country variables and persist in saves.

# Sets each enabled category's pool to its share of split = max(0, gold - reserve). A debt holds
# no gold, so it takes nothing off the top. Root is the country.
cm_fa_monthly_budget_refresh = {
	if = {
		# A reorganization holds the treasury until every governor its plan calls for stands, so no
		# category gets a pool this month. Urban Rights spends government power and Building Cleanup
		# spends nothing, so both keep running.
		limit = { exists = var:cm_ln_hold }
		cm_fa_zero_pools = yes
	}
	else = {
		set_local_variable = { name = cm_fa_split value = gold }
		change_local_variable = { name = cm_fa_split subtract = cm_fa_get_reserve }
		# Held allotments come off the top, so no other category is handed gold one of them is
		# saving for.
		change_local_variable = { name = cm_fa_split subtract = cm_fa_reserved_total }
		change_local_variable = { name = cm_fa_split max = 0 }
		cm_fa_set_category_pool = { cat = industry }
		cm_fa_set_category_pool = { cat = collapsar }
		cm_fa_set_category_pool = { cat = food }
		cm_fa_set_category_pool = { cat = utilities }
		cm_fa_set_category_pool = { cat = roads }
		cm_fa_set_category_pool = { cat = military }
		# Governors drains ahead of the shared categories.
		if = {
			limit = { exists = var:cm_fa_governors_enabled }
			set_variable = { name = cm_fa_pool_governors value = local_var:cm_fa_split }
		}
	}
}

# pool = pct * split, plus whatever the category held back last cycle.
# $cat$ = industry | collapsar | food | utilities | roads | military. Root is the country.
cm_fa_set_category_pool = {
	if = {
		limit = { exists = var:cm_fa_$cat$_enabled }
		set_local_variable = { name = cm_fa_inc value = local_var:cm_fa_split }
		change_local_variable = { name = cm_fa_inc multiply = var:cm_fa_$cat$_budget }
		change_local_variable = { name = cm_fa_inc multiply = 0.01 }
		if = {
			# The hold is handed back whole and re-made by the drain if the pick is still out of
			# reach, so it grows month on month until the pick is bought.
			limit = { exists = var:cm_fa_resv_$cat$ }
			change_local_variable = { name = cm_fa_inc add = var:cm_fa_resv_$cat$ }
			remove_variable = cm_fa_resv_$cat$
		}
		set_variable = { name = cm_fa_pool_$cat$ value = local_var:cm_fa_inc }
	}
}

# pool = min(pool, pct * the live treasury above the reserve). Called on budget-slider change.
# $cat$ = category id. Root is the country.
cm_fa_clamp_pool = {
	if = {
		limit = { has_variable = cm_fa_pool_$cat$ }
		set_local_variable = { name = cm_fa_cap value = gold }
		change_local_variable = { name = cm_fa_cap subtract = cm_fa_get_reserve }
		change_local_variable = { name = cm_fa_cap max = 0 }
		change_local_variable = { name = cm_fa_cap multiply = var:cm_fa_$cat$_budget }
		change_local_variable = { name = cm_fa_cap multiply = 0.01 }
		change_variable = { name = cm_fa_pool_$cat$ min = local_var:cm_fa_cap }
	}
}

# Root is the country.

	cm_fa_clear_debts = yes
	cm_fa_clear_reservations = yes
	cm_fa_clear_slice_state = yes
	cm_fa_rd_clear_plan = yes
}

# Root is the country.
cm_fa_zero_pools = {
	set_variable = { name = cm_fa_pool_industry value = 0 }
	set_variable = { name = cm_fa_pool_collapsar value = 0 }
	set_variable = { name = cm_fa_pool_food value = 0 }
	set_variable = { name = cm_fa_pool_utilities value = 0 }
	set_variable = { name = cm_fa_pool_roads value = 0 }
	set_variable = { name = cm_fa_pool_military value = 0 }
	set_variable = { name = cm_fa_pool_governors value = 0 }
}

# Writes off what every category owes. Root is the country.
cm_fa_clear_debts = {
	remove_variable = cm_fa_debt_industry
	remove_variable = cm_fa_debt_collapsar
	remove_variable = cm_fa_debt_food
	remove_variable = cm_fa_debt_utilities
	remove_variable = cm_fa_debt_roads
	remove_variable = cm_fa_debt_military
}

# Releases every held allotment back into the split. Root is the country.
cm_fa_clear_reservations = {
	remove_variable = cm_fa_resv_industry
	remove_variable = cm_fa_resv_collapsar
	remove_variable = cm_fa_resv_food
	remove_variable = cm_fa_resv_utilities
	remove_variable = cm_fa_resv_roads
	remove_variable = cm_fa_resv_military
}

	cm_fa_log_balance = { cat = food ACTION = cm_log_fa_bal_food }
	cm_fa_log_balance = { cat = collapsar ACTION = cm_log_fa_bal_collapsar }
	cm_fa_log_balance = { cat = roads ACTION = cm_log_fa_bal_roads }
	cm_fa_log_balance = { cat = industry ACTION = cm_log_fa_bal_industry }
	cm_fa_log_balance = { cat = utilities ACTION = cm_log_fa_bal_utilities }
	cm_fa_log_balance = { cat = military ACTION = cm_log_fa_bal_military }
	cm_dbg_log_decimal = { CAT = budget action = cm_log_fa_bal_total value = local_var:cm_fa_dbg_total }
}

# One category's balance, positive where it is holding an allotment and negative where it owes;
# a category can never be doing both, since owing is what stops it before it can hold.
# $cat$ = category id, $ACTION$ = its log key. Root is the country, accumulates the probe's
# local cm_fa_dbg_total.
cm_fa_log_balance = {
	set_local_variable = { name = cm_fa_dbg value = 0 }
	if = {
		limit = { exists = var:cm_fa_resv_$cat$ }
		set_local_variable = { name = cm_fa_dbg value = var:cm_fa_resv_$cat$ }
	}
	if = {
		limit = { exists = var:cm_fa_debt_$cat$ }
		change_local_variable = { name = cm_fa_dbg subtract = var:cm_fa_debt_$cat$ }
	}
	change_local_variable = { name = cm_fa_dbg_total add = local_var:cm_fa_dbg }
	cm_dbg_log_decimal = { CAT = budget action = $ACTION$ value = local_var:cm_fa_dbg }
}

# $v$ = variable name. Root is the new country, expects scope:old_country.
cm_fa_copy_budget_var = {
	if = {
		limit = { scope:old_country = { exists = var:$v$ } }
		set_variable = { name = $v$ value = scope:old_country.var:$v$ }
	}
}

# Approves one staged full-automation build at this location under its category's pool,
# then consumes its keys; a key missing from cm_fa_b_costs means the engine blocked the
# build at scoring and the item is skipped. Callers iterate their own category's entries.
# $cat$ = category code, $POOL$ = pool variable, $GOODS_GATE$ = the market
# construction-goods gate trigger, $SLOTS_GATE$ = the open-construction-slot gate trigger.
# Root is the location. Expects scope:cm_country, scope:cm_building_type.
cm_fa_try_approve_build = {
	if = {
		limit = {
			# A category that stopped on cost approves nothing else this cycle.
			# cm_fa_no_hold lifts both halves for the fort line.
			OR = {
				local_var:cm_fa_cost_blocked = 0
				local_var:cm_fa_no_hold = 1
			}
			trigger_if = {
				limit = {
					has_variable_map = cm_fa_b_cats
					is_key_in_variable_map = { name = cm_fa_b_cats target = scope:cm_building_type }
				}
				"variable_map(cm_fa_b_cats|scope:cm_building_type)" = $cat$
			}
			trigger_else = { always = no }
			has_variable_map = cm_fa_b_costs
			is_key_in_variable_map = { name = cm_fa_b_costs target = scope:cm_building_type }
		}
		save_temporary_scope_value_as = { name = cm_cost value = "variable_map(cm_fa_b_costs|scope:cm_building_type)" }
		market = { save_temporary_scope_as = cm_market }
		if = {
			# Approved by another feature this cycle: consume silently.
			limit = { NOT = { is_target_in_variable_list = { name = cm_q_approved_build_types target = scope:cm_building_type } } }
			if = {
				limit = {
					$SLOTS_GATE$ = yes
					$GOODS_GATE$ = yes
					cm_fa_can_afford = { POOL = $POOL$ }
				}
				cm_approve_gui_build = yes
				scope:cm_country = { change_variable = { name = $POOL$ subtract = scope:cm_cost } }
				if = {
					limit = { cm_fa_type_is_single_level = no }
					cm_fa_fill_build_slots = { POOL = $POOL$ }
				}
			}
			else_if = {
				# The two gates pass here, so the cost is the only thing that failed.
				limit = {
					$SLOTS_GATE$ = yes
					$GOODS_GATE$ = yes
					local_var:cm_fa_no_hold = 0
				}
				cm_fa_stop_on_cost = { COST = scope:cm_cost }
			}
		}
		remove_from_variable_map = { name = cm_fa_b_costs key = scope:cm_building_type }
		remove_from_variable_map = { name = cm_fa_b_cats key = scope:cm_building_type }
	}
}

# Full-automation entry to cm_q_fill_build_slots: the batch draws its category pool down in a
# scratch variable so one affordability trigger serves every category, then writes the balance
# back. $POOL$ = the category pool variable name. Root is the location.
# Expects scope:cm_country, scope:cm_building_type, scope:cm_cost.
cm_fa_fill_build_slots = {
	scope:cm_country = { set_variable = { name = cm_fa_copy_pool value = var:$POOL$ } }
	cm_q_fill_build_slots = { AFFORD = cm_fa_copy_can_afford CLAIM = cm_fa_claim_build_copy }
	scope:cm_country = {
		set_variable = { name = $POOL$ value = var:cm_fa_copy_pool }
		remove_variable = cm_fa_copy_pool
	}
}

# Per-level claim for a full-automation batch: the shared claim plus the pool draw.
# Root is the location. Expects scope:cm_country, scope:cm_cost.
cm_fa_claim_build_copy = {
	cm_q_claim_build_copy = yes
	scope:cm_country = { change_variable = { name = cm_fa_copy_pool subtract = scope:cm_cost } }
}

# Runs one category's drain under its pool, taking in the funds earlier categories left and
# handing on whatever it does not spend. Root is the country, expects scope:cm_fa_cat.
cm_fa_drain_category = {
	switch = {
		trigger = scope:cm_fa_cat
		flag:cm_fa_cat_industry = { cm_fa_drain_one = { CAT = industry PASS = cm_fa_pass_industry } }
		flag:cm_fa_cat_collapsar = { cm_fa_drain_one = { CAT = collapsar PASS = cm_fa_pass_collapsar } }
		flag:cm_fa_cat_food = { cm_fa_drain_one = { CAT = food PASS = cm_fa_pass_food } }
		flag:cm_fa_cat_utilities = { cm_fa_drain_one = { CAT = utilities PASS = cm_fa_pass_utilities } }
		flag:cm_fa_cat_roads = { cm_fa_drain_one = { CAT = roads PASS = cm_fa_pass_roads } }
		# Military takes the carry, and runs a ledger it can owe against, only once its manpower
		# chain has a tier it can build. Until then the category is the fort line alone, which
		# its own share covers.
		flag:cm_fa_cat_military = {
			if = {
				limit = { exists = var:cm_ln_bk_active }
				cm_fa_drain_one = { CAT = military PASS = cm_fa_pass_military }
			}
			else = {
				cm_fa_pass_military = yes
			}
		}
	}
}

# Settles what the category owes out of its own allotment, then spends what is left of it plus
# the carry, or stands the category down for the cycle while it still owes. Each category answers
# only for its own stop, so the signal is cleared per category rather than once for the loop.
# $CAT$ = category id, $PASS$ = its drain effect. Root is the country.
cm_fa_drain_one = {
	set_local_variable = { name = cm_fa_cost_blocked value = 0 }
	set_local_variable = { name = cm_fa_hold_cost value = 0 }
	if = {
		limit = { NOT = { has_variable = cm_fa_pool_$CAT$ } }
		set_variable = { name = cm_fa_pool_$CAT$ value = 0 }
	}
	set_local_variable = { name = cm_fa_paid value = 0 }
	if = {
		# Settled before the carry arrives, so only the category's own allotment pays it down.
		limit = { exists = var:cm_fa_debt_$CAT$ }
		set_local_variable = { name = cm_fa_paid value = var:cm_fa_pool_$CAT$ }
		change_local_variable = { name = cm_fa_paid min = var:cm_fa_debt_$CAT$ }
		change_variable = { name = cm_fa_debt_$CAT$ subtract = local_var:cm_fa_paid }
		change_variable = { name = cm_fa_pool_$CAT$ subtract = local_var:cm_fa_paid }
		if = {
			limit = { var:cm_fa_debt_$CAT$ <= 0 }
			remove_variable = cm_fa_debt_$CAT$
		}
	}
	if = {
		# Still owing, so it buys nothing and the carry passes it by.
		limit = { exists = var:cm_fa_debt_$CAT$ }
		change_variable = { name = cm_fa_carry add = local_var:cm_fa_paid }
	}
	else = {
		change_variable = { name = cm_fa_pool_$CAT$ add = var:cm_fa_carry }
		# The settled gold goes down the order, so it is handed on only after the pool has taken
		# the incoming carry.
		set_variable = { name = cm_fa_carry value = local_var:cm_fa_paid }
		set_local_variable = { name = cm_fa_ledger value = 1 }
		$PASS$ = yes
		set_local_variable = { name = cm_fa_ledger value = 0 }
		if = {
			# It bought past its allotment, and the overspend is what it now owes.
			limit = { var:cm_fa_pool_$CAT$ < 0 }
			set_variable = { name = cm_fa_debt_$CAT$ value = 0 }
			change_variable = { name = cm_fa_debt_$CAT$ subtract = var:cm_fa_pool_$CAT$ }
		}
		else_if = {
			# It stopped on a pick it could not reach, so it holds what that pick costs and hands
			# down only the rest.
			limit = {
				local_var:cm_fa_cost_blocked = 1
				local_var:cm_fa_hold_cost > 0
				var:cm_fa_pool_$CAT$ > 0
			}
			set_local_variable = { name = cm_fa_keep value = var:cm_fa_pool_$CAT$ }
			change_local_variable = { name = cm_fa_keep min = local_var:cm_fa_hold_cost }
			set_variable = { name = cm_fa_resv_$CAT$ value = local_var:cm_fa_keep }
			change_variable = { name = cm_fa_pool_$CAT$ subtract = local_var:cm_fa_keep }
			change_variable = { name = cm_fa_carry add = var:cm_fa_pool_$CAT$ }
		}
		else = {
			change_variable = { name = cm_fa_carry add = var:cm_fa_pool_$CAT$ }
		}
	}
	set_variable = { name = cm_fa_pool_$CAT$ value = 0 }
}

# Stands the category down on a pick it could not pay for, recording what that pick costs so
# cm_fa_drain_one holds an allotment sized to it. $COST$ = the pick's cost.
cm_fa_stop_on_cost = {
	set_local_variable = { name = cm_fa_cost_blocked value = 1 }
	set_local_variable = { name = cm_fa_hold_cost value = $COST$ }
}

# Runs one category's staging. Root is the country, expects scope:cm_fa_cat.
cm_fa_stage_category = {
	switch = {
		trigger = scope:cm_fa_cat
		flag:cm_fa_cat_industry = { cm_fa_run_industry = yes }
		flag:cm_fa_cat_collapsar = { cm_fa_run_collapsar = yes }
		flag:cm_fa_cat_food = { cm_fa_run_food = yes }
		flag:cm_fa_cat_utilities = { cm_fa_run_utilities = yes }
		flag:cm_fa_cat_urban_rights = { cm_fa_run_urban_rights = yes }
		flag:cm_fa_cat_governors = { cm_fa_run_governors = yes }
		flag:cm_fa_cat_military = { cm_ln_mil_run = yes }
	}
}
