# Full-automation budget engine: monthly per-category pools set from the treasury above
# the reserve, and the per-category dispatch that runs each drain and passes its unspent
# pool down the player's category order. Pools are country variables and persist in saves.

# Sets each enabled category's pool to its share of this month's available spending.
# available = max(0, gold - reserve)
# Root is the country.
cm_fa_monthly_budget_refresh = {
	set_local_variable = { name = cm_fa_avail value = gold }
	change_local_variable = { name = cm_fa_avail subtract = cm_fa_get_reserve }
	change_local_variable = { name = cm_fa_avail max = 0 }
	# A reorganization holds the treasury while it pays for a road out to a planned
	# governor site. Urban Rights spends government power and Building Cleanup spends
	# nothing, so both keep running.
	if = {
		limit = { exists = var:cm_ln_hold }
		set_local_variable = { name = cm_fa_avail value = 0 }
	}
	cm_fa_set_category_pool = { cat = industry }
	cm_fa_set_category_pool = { cat = collapsar }
	cm_fa_set_category_pool = { cat = food }
	cm_fa_set_category_pool = { cat = utilities }
	cm_fa_set_category_pool = { cat = roads }
	cm_fa_set_category_pool = { cat = forts }
	# Governors drains ahead of the shared categories.
	if = {
		limit = { exists = var:cm_fa_governors_enabled }
		set_variable = { name = cm_fa_pool_governors value = local_var:cm_fa_avail }
	}
}

# pool = pct * available. $cat$ = industry | collapsar | food | utilities | roads | forts.
# Root is the country.
cm_fa_set_category_pool = {
	if = {
		limit = { exists = var:cm_fa_$cat$_enabled }
		set_local_variable = { name = cm_fa_inc value = local_var:cm_fa_avail }
		change_local_variable = { name = cm_fa_inc multiply = var:cm_fa_$cat$_budget }
		change_local_variable = { name = cm_fa_inc multiply = 0.01 }
		set_variable = { name = cm_fa_pool_$cat$ value = local_var:cm_fa_inc }
	}
}

# pool = min(pool, pct * the live treasury above the reserve). Called on budget-slider
# change. $cat$ = category id. Root is the country.
cm_fa_clamp_pool = {
	if = {
		limit = { has_variable = cm_fa_pool_$cat$ }
		set_local_variable = { name = cm_fa_cap value = gold }
		change_local_variable = { name = cm_fa_cap subtract = cm_fa_get_reserve }
		change_local_variable = { name = cm_fa_cap max = 0 }
		change_local_variable = { name = cm_fa_cap multiply = var:cm_fa_$cat$_budget }
		change_local_variable = { name = cm_fa_cap multiply = 0.01 }
		change_variable = { name = cm_fa_pool_$cat$ min = local_var:cm_fa_cap }
	}
}

# Root is the country.
cm_fa_clear_budget_state = {
	set_variable = { name = cm_fa_pool_industry value = 0 }
	set_variable = { name = cm_fa_pool_collapsar value = 0 }
	set_variable = { name = cm_fa_pool_food value = 0 }
	set_variable = { name = cm_fa_pool_utilities value = 0 }
	set_variable = { name = cm_fa_pool_roads value = 0 }
	set_variable = { name = cm_fa_pool_forts value = 0 }
	set_variable = { name = cm_fa_pool_governors value = 0 }
	cm_fa_clear_slice_state = yes
	cm_fa_rd_clear_plan = yes
}

# Copies the budget state from scope:old_country on tag switch; pools are non-CMM state
# CMF does not carry. Root is the new country.
cm_fa_copy_budget_state = {
	cm_fa_copy_budget_var = { v = cm_fa_pool_industry }
	cm_fa_copy_budget_var = { v = cm_fa_pool_collapsar }
	cm_fa_copy_budget_var = { v = cm_fa_pool_food }
	cm_fa_copy_budget_var = { v = cm_fa_pool_utilities }
	cm_fa_copy_budget_var = { v = cm_fa_pool_roads }
	cm_fa_copy_budget_var = { v = cm_fa_pool_forts }
	cm_fa_copy_budget_var = { v = cm_fa_pool_governors }
	cm_fa_copy_budget_var = { v = cm_fa_rd_n }
}

# $v$ = variable name. Root is the new country, expects scope:old_country.
cm_fa_copy_budget_var = {
	if = {
		limit = { scope:old_country = { exists = var:$v$ } }
		set_variable = { name = $v$ value = scope:old_country.var:$v$ }
	}
}

# Approves one staged full-automation build at this location under its category's pool,
# then consumes its keys; a key missing from cm_fa_b_costs means the engine blocked the
# build at scoring and the item is skipped. Callers iterate their own category's entries.
# $cat$ = category code, $POOL$ = pool variable, $RESERVE$ = the category's gold floor,
# $GOODS_GATE$ = the market construction-goods gate trigger, $SLOTS_GATE$ = the
# open-construction-slot gate trigger.
# Root is the location. Expects scope:cm_country, scope:cm_building_type.
cm_fa_try_approve_build = {
	if = {
		limit = {
			trigger_if = {
				limit = {
					has_variable_map = cm_fa_b_cats
					is_key_in_variable_map = { name = cm_fa_b_cats target = scope:cm_building_type }
				}
				"variable_map(cm_fa_b_cats|scope:cm_building_type)" = $cat$
			}
			trigger_else = { always = no }
			has_variable_map = cm_fa_b_costs
			is_key_in_variable_map = { name = cm_fa_b_costs target = scope:cm_building_type }
		}
		save_temporary_scope_value_as = { name = cm_cost value = "variable_map(cm_fa_b_costs|scope:cm_building_type)" }
		market = { save_temporary_scope_as = cm_market }
		if = {
			# Approved by another feature this cycle: consume silently.
			limit = { NOT = { is_target_in_variable_list = { name = cm_q_approved_build_types target = scope:cm_building_type } } }
			if = {
				limit = {
					$SLOTS_GATE$ = yes
					$GOODS_GATE$ = yes
					cm_fa_can_afford = { POOL = $POOL$ RESERVE = $RESERVE$ }
				}
				cm_approve_gui_build = yes
				scope:cm_country = { change_variable = { name = $POOL$ subtract = scope:cm_cost } }
				if = {
					limit = { cm_fa_type_is_single_level = no }
					cm_fa_fill_build_slots = { POOL = $POOL$ RESERVE = $RESERVE$ }
				}
			}
		}
		remove_from_variable_map = { name = cm_fa_b_costs key = scope:cm_building_type }
		remove_from_variable_map = { name = cm_fa_b_cats key = scope:cm_building_type }
	}
}

# Full-automation entry to cm_q_fill_build_slots: the batch draws its category pool down in a
# scratch variable so one affordability trigger serves every category, then writes the balance
# back. $POOL$ = the category pool variable name, $RESERVE$ = its gold floor.
# Root is the location. Expects scope:cm_country, scope:cm_building_type, scope:cm_cost.
cm_fa_fill_build_slots = {
	scope:cm_country = {
		set_variable = { name = cm_fa_copy_pool value = var:$POOL$ }
		set_variable = { name = cm_fa_copy_reserve value = $RESERVE$ }
	}
	cm_q_fill_build_slots = { AFFORD = cm_fa_copy_can_afford CLAIM = cm_fa_claim_build_copy }
	scope:cm_country = {
		set_variable = { name = $POOL$ value = var:cm_fa_copy_pool }
		remove_variable = cm_fa_copy_pool
		remove_variable = cm_fa_copy_reserve
	}
}

# Per-level claim for a full-automation batch: the shared claim plus the pool draw.
# Root is the location. Expects scope:cm_country, scope:cm_cost.
cm_fa_claim_build_copy = {
	cm_q_claim_build_copy = yes
	scope:cm_country = { change_variable = { name = cm_fa_copy_pool subtract = scope:cm_cost } }
}

# Runs one category's drain under its pool, taking in the funds earlier categories left
# and handing on whatever it does not spend. Root is the country, expects scope:cm_fa_cat.
cm_fa_drain_category = {
	switch = {
		trigger = scope:cm_fa_cat
		flag:cm_fa_cat_industry = { cm_fa_drain_one = { CAT = industry PASS = cm_fa_pass_industry } }
		flag:cm_fa_cat_collapsar = { cm_fa_drain_one = { CAT = collapsar PASS = cm_fa_pass_collapsar } }
		flag:cm_fa_cat_food = { cm_fa_drain_one = { CAT = food PASS = cm_fa_pass_food } }
		flag:cm_fa_cat_utilities = { cm_fa_drain_one = { CAT = utilities PASS = cm_fa_pass_utilities } }
		flag:cm_fa_cat_roads = { cm_fa_drain_one = { CAT = roads PASS = cm_fa_pass_roads } }
		# Forts drains against its own pool without taking the carry, so its 5 percent share
		# stays 5 percent even though it runs last.
		flag:cm_fa_cat_forts = { cm_fa_pass_forts = yes }
	}
}

# $CAT$ = category id, $PASS$ = its drain effect. Root is the country.
cm_fa_drain_one = {
	if = {
		limit = { NOT = { has_variable = cm_fa_pool_$CAT$ } }
		set_variable = { name = cm_fa_pool_$CAT$ value = 0 }
	}
	change_variable = { name = cm_fa_pool_$CAT$ add = var:cm_fa_carry }
	set_variable = { name = cm_fa_carry value = 0 }
	$PASS$ = yes
	change_variable = { name = cm_fa_carry add = var:cm_fa_pool_$CAT$ }
	set_variable = { name = cm_fa_pool_$CAT$ value = 0 }
}

# Runs one category's staging. Root is the country, expects scope:cm_fa_cat.
cm_fa_stage_category = {
	switch = {
		trigger = scope:cm_fa_cat
		flag:cm_fa_cat_industry = { cm_fa_run_industry = yes }
		flag:cm_fa_cat_collapsar = { cm_fa_run_collapsar = yes }
		flag:cm_fa_cat_food = { cm_fa_run_food = yes }
		flag:cm_fa_cat_utilities = { cm_fa_run_utilities = yes }
		flag:cm_fa_cat_urban_rights = { cm_fa_run_urban_rights = yes }
		flag:cm_fa_cat_governors = { cm_fa_run_governors = yes }
		flag:cm_fa_cat_forts = { cm_ln_ft_run = yes }
	}
}
