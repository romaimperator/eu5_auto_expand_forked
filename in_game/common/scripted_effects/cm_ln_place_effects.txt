# Lucky Nations governor placement: three passes a month. Every planned site below city rank is
# urbanized in parallel, one road corridor is laid at a time, and any site with both behind it
# takes its governor. A site that needs an existing seat freed is served last, and that
# replacement is free, because the seat it replaces was a placement the country could have
# made correctly in the first place.

# Months to wait on a road order before re-issuing it.
@cm_ln_road_retry_months = 24
# Re-issues before the site is given up.
@cm_ln_road_max_tries = 5
# Months a site may be refused its rank upgrade before it is given up.
@cm_ln_rank_max_months = 60
# Placements a site may swallow without a governor appearing before it is given up.
@cm_ln_place_max_tries = 5

# Places at the best-ranked pending site that needs neither a road nor a rank, so a site still
# urbanizing does not hold up the ones behind it. $FREE$ = yes grants permission to give up a seat
# to make room. Root is the country, scope:cm_country set.
cm_ln_place_next = {
	ordered_in_list = {
		variable = cm_ln_pending
		order_by = var:cm_ln_plan_ord
		position = 0
		check_range_bounds = no
		limit = {
			cm_ln_target_needs_road = no
			cm_ln_target_needs_rank = no
		}
		save_scope_as = cm_ln_target
	}
	if = {
		limit = { exists = scope:cm_ln_target }
		cm_ln_resolve_target_type = yes
		cm_ln_do_placement = { FREE = $FREE$ }
	}
}

# The one corridor a month, laid to the best-ranked pending site that has no road to the capital.
# Root is the country, scope:cm_country set.
cm_ln_road_pass = {
	ordered_in_list = {
		variable = cm_ln_pending
		order_by = var:cm_ln_plan_ord
		position = 0
		check_range_bounds = no
		limit = { cm_ln_target_needs_road = yes }
		save_scope_as = cm_ln_road_target
	}
	if = {
		limit = { exists = scope:cm_ln_road_target }
		cm_ln_road_step = yes
	}
}

# Every pending site short of city rank, so the plan urbanizes in parallel with the one road.
# Root is the country, scope:cm_country set.
cm_ln_rank_pass = {
	every_in_list = {
		variable = cm_ln_pending
		save_scope_as = cm_ln_rank_target
		if = {
			limit = { cm_ln_target_needs_rank = yes }
			# The walk puts the location in root, and the step writes country variables.
			scope:cm_country = { cm_ln_rank_step = yes }
		}
	}
}

# Root is the country. Sets scope:cm_building_type to the Found building the site's rank calls for.
# Expects scope:cm_ln_rank_target.
cm_ln_resolve_rank_type = {
	if = {
		limit = { scope:cm_ln_rank_target = { location_rank = location_rank:rural_settlement } }
		building_type:cm_upgrade_rural_to_town_building = { save_scope_as = cm_building_type }
	}
	else = {
		building_type:cm_upgrade_town_to_city_building = { save_scope_as = cm_building_type }
	}
}

# The urbanize pass gates on population, the RGO ignore list, the food reservation and min_discount,
# so a planned site can sit at rank indefinitely; this path carries only the rank, already-built,
# already-under-construction, engine-placement and gold checks. Root is the country,
# scope:cm_country and scope:cm_ln_rank_target set.
cm_ln_rank_step = {
	cm_ln_resolve_rank_type = yes
	scope:cm_ln_rank_target = { save_temporary_scope_as = cm_location }
	if = {
		limit = {
			scope:cm_ln_rank_target = {
				NOT = { has_building = scope:cm_building_type }
				NOT = {
					any_buildings_in_location = {
						building_type = scope:cm_building_type
						building_levels_under_construction > 0
					}
				}
			}
		}
		if = {
			limit = { scope:cm_ln_rank_target = { cm_location_allowed_to_build = { target = scope:cm_building_type } } }
			if = {
				limit = { cm_check_country_gold_above = { threshold = cm_ln_rank_gold_threshold } }
				change_variable = { name = cm_ln_tick_spent add = scope:cm_ln_rank_target.cm_ln_build_cost_value }
				scope:cm_ln_rank_target = {
					construct_building = {
						building_type = scope:cm_building_type
						owner = scope:cm_country
						payer = scope:cm_country
					}
					remove_variable = cm_ln_rank_wait
				}
				cm_dbg_log = { cat = governors action = cm_ln_log_gov_rank_up }
			}
			else = {
				set_local_variable = { name = cm_ln_gold_wait value = 1 }
			}
		}
		else = {
			# The engine refuses the upgrade, which saving gold cannot fix, so the site counts months
			# toward the cap the prune drops it at, rather than sitting in the plan until the abort.
			scope:cm_ln_rank_target = {
				if = {
					limit = { NOT = { exists = var:cm_ln_rank_wait } }
					set_variable = { name = cm_ln_rank_wait value = 0 }
				}
				change_variable = { name = cm_ln_rank_wait add = 1 }
			}
		}
	}
}

# Also the single removal point for a site the engine keeps refusing its rank upgrade, since
# cm_ln_rank_pass cannot drop one mid-walk without mutating the list it is iterating.
# Root is the country, scope:cm_country set.
cm_ln_prune_pending = {
	clear_variable_list = cm_ln_pending_keep
	every_in_list = {
		variable = cm_ln_pending
		save_scope_as = cm_ln_p
		if = {
			limit = {
				scope:cm_ln_p = {
					owner ?= scope:cm_country
					cm_location_has_governor = no
					trigger_if = {
						limit = { exists = var:cm_ln_rank_wait }
						var:cm_ln_rank_wait < @cm_ln_rank_max_months
					}
				}
			}
			scope:cm_country = { add_to_variable_list = { name = cm_ln_pending_keep target = scope:cm_ln_p } }
		}
		else_if = {
			# Lost, or refused its rank upgrade for good. Either way the plan gives the site up so the
			# confirm step can send the search after a replacement for its slot.
			limit = { scope:cm_ln_p = { cm_location_has_governor = no } }
			scope:cm_country = {
				if = {
					limit = { scope:cm_ln_p = { owner ?= scope:cm_country } }
					cm_dbg_log = { cat = governors action = cm_ln_log_given_up_rank }
				}
				else = {
					cm_dbg_log = { cat = governors action = cm_ln_log_given_up_lost }
				}
				cm_ln_drop_plan_site = { SITE = scope:cm_ln_p }
			}
		}
	}
	clear_variable_list = cm_ln_pending
	every_in_list = {
		variable = cm_ln_pending_keep
		save_scope_as = cm_ln_p
		scope:cm_country = { add_to_variable_list = { name = cm_ln_pending target = scope:cm_ln_p } }
	}
	clear_variable_list = cm_ln_pending_keep
}

# Codes the condition the refused site fails, so a refusal in the log says which one instead of only
# that one happened. Mirrors the two vanilla governor allow and location_potential blocks:
# in_game/common/building_types/capital_buildings.txt local_governor :607 and :640,
# naval_governor :760 and :798. The slot cap is tested last of them all, so a verdict of 1 means it
# is the only condition left standing in the way, which is what cm_ln_do_placement gives up a seat
# for. Root is the country, scope:cm_country and scope:cm_ln_target set.
# Leaves local_var:cm_ln_fail.
cm_ln_place_fail_reason = {
	set_local_variable = { name = cm_ln_fail value = 6 }
	if = {
		# cm_ln_target_needs_rank keeps a site below city rank out of the placement, so reaching here
		# means the rank went away after the plan committed to it.
		limit = { scope:cm_ln_target = { cm_ln_target_needs_rank = yes } }
		set_local_variable = { name = cm_ln_fail value = 3 }
	}
	else_if = {
		limit = { scope:cm_ln_target = { has_variable = cm_ln_plan_naval } }
		if = {
			limit = { scope:cm_ln_target = { cm_ln_site_is_naval = no } }
			set_local_variable = { name = cm_ln_fail value = 5 }
		}
		else_if = {
			limit = { cm_fa_gov_has_naval_room = no }
			set_local_variable = { name = cm_ln_fail value = 1 }
		}
	}
	else_if = {
		limit = { scope:cm_ln_target = capital }
		set_local_variable = { name = cm_ln_fail value = 2 }
	}
	else_if = {
		limit = { scope:cm_ln_target = { NOT = { is_connected_to = owner.capital } } }
		set_local_variable = { name = cm_ln_fail value = 4 }
	}
	else_if = {
		limit = {
			cm_fa_gov_has_land_room = no
			scope:cm_ln_target = { NOT = { has_building = building_type:seljuk_mint } } # allow: :634
		}
		set_local_variable = { name = cm_ln_fail value = 1 }
	}
}

# Counts one placement at a planned site that produced no governor and routes the site: back onto
# cm_ln_pending for another go, or onto cm_ln_plan_drop once the tries run out, so the slot ends up
# somewhere the engine will actually take it. An order the engine refused and an order that left no
# building behind are the same kind of failure and share the count.
# Root is the country, scope:cm_country set. $SITE$ = the location.
cm_ln_place_retry_or_drop = {
	$SITE$ = {
		if = {
			limit = { NOT = { exists = var:cm_ln_place_tries } }
			set_variable = { name = cm_ln_place_tries value = 0 }
		}
		change_variable = { name = cm_ln_place_tries add = 1 }
	}
	if = {
		limit = { $SITE$ = { var:cm_ln_place_tries < @cm_ln_place_max_tries } }
		add_to_variable_list = { name = cm_ln_pending target = $SITE$ }
	}
	else = {
		cm_dbg_log = { cat = governors action = cm_ln_log_given_up_placements }
		add_to_variable_list = { name = cm_ln_plan_drop target = $SITE$ }
	}
}

# Applies the give-ups staged on cm_ln_plan_drop. Staged rather than immediate because the confirm
# step decides them while iterating cm_ln_plan, which cm_ln_drop_plan_site removes from.
# Root is the country, scope:cm_country set.
cm_ln_apply_plan_drops = {
	every_in_list = {
		variable = cm_ln_plan_drop
		save_scope_as = cm_ln_p
		scope:cm_country = {
			remove_list_variable = { name = cm_ln_pending target = scope:cm_ln_p }
			cm_ln_drop_plan_site = { SITE = scope:cm_ln_p }
		}
	}
	clear_variable_list = cm_ln_plan_drop
}

# Gives up on a planned site. The reject mark keeps this run's searches off it, so the replacement
# the confirm step sends them after is somewhere else. cm_ln_pending is the caller's to handle,
# because cm_ln_prune_pending rebuilds it rather than mutating it mid-walk.
# Root is the country, scope:cm_country set. $SITE$ = the location.
cm_ln_drop_plan_site = {
	remove_list_variable = { name = cm_ln_plan target = $SITE$ }
	remove_list_variable = { name = cm_ln_seats target = $SITE$ }
	$SITE$ = {
		remove_variable = cm_ln_plan_ord
		remove_variable = cm_ln_plan_naval
		remove_variable = cm_ln_rank_wait
		remove_variable = cm_ln_place_tries
		if = {
			limit = { owner ?= scope:cm_country }
			set_variable = { name = cm_ln_probe_reject value = 1 }
		}
	}
}

# Root is the country, expects scope:cm_ln_target. Sets scope:cm_building_type.
cm_ln_resolve_target_type = {
	if = {
		limit = { scope:cm_ln_target = { has_variable = cm_ln_plan_naval } }
		building_type:naval_governor = { save_scope_as = cm_building_type }
	}
	else = {
		building_type:local_governor = { save_scope_as = cm_building_type }
	}
}

# The kind the engine will take at the site as it stands now, rather than the kind the plan recorded.
# Root is the country, scope:cm_country set. $SITE$ = the location. Sets scope:cm_building_type.
cm_ln_resolve_placeable_type = {
	if = {
		limit = {
			cm_fa_gov_has_naval_room = yes
			$SITE$ = { cm_location_allowed_to_build = { target = building_type:naval_governor } }
		}
		building_type:naval_governor = { save_scope_as = cm_building_type }
	}
	else = {
		building_type:local_governor = { save_scope_as = cm_building_type }
	}
}

# One corridor at a time, so the whole held treasury stands behind it. The order's duration is
# not script-readable, so the retry is a timer rather than a check.
# Root is the country, scope:cm_country and scope:cm_ln_road_target set.
cm_ln_road_step = {
	if = {
		# The corridor in flight was ordered for another site, so it either landed or left the queue.
		# Its timer and its tries belong to that site and must not age out this one's order.
		limit = {
			exists = var:cm_ln_road_issued
			NOT = { var:cm_ln_road_site = scope:cm_ln_road_target }
		}
		cm_ln_clear_road_state = yes
	}
	if = {
		# road_building is the advance that unlocks gravel_road:
		# in_game/common/advances/3_road_unlocks.txt:1
		limit = { NOT = { has_advance = road_building } }
		remove_list_variable = { name = cm_ln_pending target = scope:cm_ln_road_target }
		cm_dbg_log = { cat = governors action = cm_ln_log_given_up_road }
		cm_ln_drop_plan_site = { SITE = scope:cm_ln_road_target }
		cm_ln_clear_road_state = yes
	}
	else_if = {
		limit = {
			exists = capital
			NOT = { exists = var:cm_ln_road_issued }
		}
		capital = { save_temporary_scope_as = cm_fa_rd_from }
		scope:cm_ln_road_target = { save_temporary_scope_as = cm_fa_rd_to }
		save_temporary_scope_value_as = { name = cm_fa_rd_tier value = cm_ln_road_tier_cost_value }
		if = {
			limit = { cm_check_country_gold_above = { threshold = cm_ln_road_gold_threshold } }
			construct_road = {
				road_type = road_type:gravel_road
				from = capital
				to = scope:cm_ln_road_target
			}
			change_variable = { name = cm_ln_tick_spent add = scope:cm_fa_rd_from.cm_fa_rd_cost_value }
			set_variable = { name = cm_ln_road_issued value = 1 }
			set_variable = { name = cm_ln_road_site value = scope:cm_ln_road_target }
			set_variable = { name = cm_ln_road_age value = 0 }
		}
		else = {
			set_local_variable = { name = cm_ln_gold_wait value = 1 }
		}
	}
	else_if = {
		limit = { exists = var:cm_ln_road_issued }
		change_variable = { name = cm_ln_road_age add = 1 }
		if = {
			limit = { var:cm_ln_road_age >= @cm_ln_road_retry_months }
			remove_variable = cm_ln_road_issued
			set_variable = { name = cm_ln_road_age value = 0 }
			change_variable = { name = cm_ln_road_tries add = 1 }
			if = {
				limit = { var:cm_ln_road_tries >= @cm_ln_road_max_tries }
				remove_list_variable = { name = cm_ln_pending target = scope:cm_ln_road_target }
				cm_dbg_log = { cat = governors action = cm_ln_log_given_up_road }
				cm_ln_drop_plan_site = { SITE = scope:cm_ln_road_target }
				cm_ln_clear_road_state = yes
			}
		}
	}
}

# Root is the country.
cm_ln_clear_road_state = {
	remove_variable = cm_ln_road_issued
	remove_variable = cm_ln_road_site
	set_variable = { name = cm_ln_road_age value = 0 }
	set_variable = { name = cm_ln_road_tries value = 0 }
}

# $FREE$ = yes grants permission to give up a seat when the engine refuses the site for want of one,
# which is what the relocation phase passes; it does not make the placement itself free. The paid
# seat count does that.
# Root is the country, scope:cm_country, scope:cm_ln_target and scope:cm_building_type set.
cm_ln_do_placement = {
	scope:cm_ln_target = { save_temporary_scope_as = cm_location }
	if = {
		limit = { scope:cm_ln_target = { cm_location_allowed_to_build = { target = scope:cm_building_type } } }
		if = {
			limit = { cm_ln_gov_seat_is_prepaid = yes }
			cm_ln_place_allowed_target = { PREPAID = yes }
		}
		else = {
			cm_ln_place_allowed_target = { PREPAID = no }
		}
	}
	else = {
		cm_ln_place_fail_reason = yes
		if = {
			# The governor cap lives inside vanilla's own allow block, so the engine keeps refusing
			# until the seat the plan does not want is gone: in_game/common/building_types/
			# capital_buildings.txt local_governor allow :622.
			limit = { local_var:cm_ln_fail = 1 }
			if = {
				# The placement phase spends no try on a cap it is not allowed to clear; the
				# relocation phase behind it is where the seat comes from.
				limit = { NOT = { always = $FREE$ } }
				set_variable = { name = cm_ln_phase value = 6 }
			}
			else = {
				cm_ln_free_a_seat = yes
				if = {
					limit = { local_var:cm_ln_freed = 1 }
					cm_dbg_log = { cat = governors action = cm_ln_log_gov_moved }
					# The seat given up was one of the paid ones, so this is prepaid by arithmetic rather
					# than by a read the destroy may not have reached yet. And it is the room the
					# placement was short of, which construct_building would not have read anyway.
					cm_ln_place_allowed_target = { PREPAID = yes }
				}
				else = {
					cm_ln_log_place_refused = yes
					cm_ln_place_retry_or_drop = { SITE = scope:cm_ln_target }
				}
			}
		}
		else = {
			cm_ln_log_place_refused = yes
			cm_ln_place_retry_or_drop = { SITE = scope:cm_ln_target }
		}
	}
}

# The placement itself, once the engine has agreed to the site: free where the country is holding
# fewer governors than it has bought, paid otherwise. Root is the country, scope:cm_country,
# scope:cm_ln_target and scope:cm_building_type set. $PREPAID$ = yes where the country is owed this
# governor.
cm_ln_place_allowed_target = {
	if = {
		limit = { always = $PREPAID$ }
		cm_ln_construct_governor_free = yes
		cm_dbg_log = { cat = governors action = cm_ln_log_gov_replaced }
		remove_list_variable = { name = cm_ln_pending target = scope:cm_ln_target }
		add_to_variable_list = { name = cm_ln_seats target = scope:cm_ln_target }
	}
	else_if = {
		limit = { cm_check_country_gold_above = { threshold = cm_ln_placement_gold_threshold } }
		scope:cm_ln_target = {
			construct_building = {
				building_type = scope:cm_building_type
				owner = scope:cm_country
				payer = scope:cm_country
			}
		}
		cm_dbg_log = { cat = governors action = cm_ln_log_gov_placed }
		remove_list_variable = { name = cm_ln_pending target = scope:cm_ln_target }
		add_to_variable_list = { name = cm_ln_seats target = scope:cm_ln_target }
	}
	else = {
		set_local_variable = { name = cm_ln_gold_wait value = 1 }
	}
}

# The count of governors this country has bought, which only ever rises: it takes in the ones it
# started with, the ones the plan paid for and any the vanilla AI put up itself, simply by never
# falling below how many are seated. Everything the placement chain needs follows from that one
# number, since a seated count below it is a governor already paid for and not standing, whatever
# took it - a capital move stranding a seat, vanilla's own remove_if, an order that came to nothing,
# or a seat the plan gave up on purpose. Root is the country.
cm_ln_sync_paid_seats = {
	if = {
		limit = { NOT = { exists = var:cm_ln_gov_paid_seats } }
		set_variable = { name = cm_ln_gov_paid_seats value = 0 }
	}
	change_variable = { name = cm_ln_gov_paid_seats max = cm_ln_gov_seated_value }
}

# Root is the country. Expects local_var:cm_ln_fail. One key per reason rather than a trailing code,
# because the log pane truncates the entry text before a value.
cm_ln_log_place_refused = {
	if = {
		limit = { local_var:cm_ln_fail = 1 }
		cm_dbg_log = { cat = governors action = cm_ln_log_refused_slot }
	}
	else_if = {
		limit = { local_var:cm_ln_fail = 2 }
		cm_dbg_log = { cat = governors action = cm_ln_log_refused_capital }
	}
	else_if = {
		limit = { local_var:cm_ln_fail = 3 }
		cm_dbg_log = { cat = governors action = cm_ln_log_refused_rank }
	}
	else_if = {
		limit = { local_var:cm_ln_fail = 4 }
		cm_dbg_log = { cat = governors action = cm_ln_log_refused_route }
	}
	else_if = {
		limit = { local_var:cm_ln_fail = 5 }
		cm_dbg_log = { cat = governors action = cm_ln_log_refused_naval }
	}
	else = {
		cm_dbg_log = { cat = governors action = cm_ln_log_refused_unknown }
	}
}

# Instant, so an AI that cancels its own construction cannot take this one away either.
# Root is the country, scope:cm_country, scope:cm_ln_target and scope:cm_building_type set.
cm_ln_construct_governor_free = {
	scope:cm_ln_target = {
		construct_building = {
			building_type = scope:cm_building_type
			cost_multiplier = 0
			cost_multiplier_reason = cm_ln_gov_paid_replacement
			instant = yes
			owner = scope:cm_country
			payer = scope:cm_country
		}
	}
}

# Gives up the one seat the site being placed takes over from, the seat having been a placement the
# country could have made correctly in the first place. Only a seat of
# the kind being placed frees the cap the placement is short of, since land and naval governors
# count against separate pools. Among those the plan's own governor scores rank them, kept by
# cm_ln_record_seat_scores because cm_ln_strip_finder_outputs removes cm_gf_score at classify, and
# the seat nearest the new site goes first where the run scored several the same, which the seats a
# plan site supersedes all are.
# Root is the country, scope:cm_country, scope:cm_ln_target and scope:cm_building_type set.
# Leaves local_var:cm_ln_freed.
cm_ln_free_a_seat = {
	set_local_variable = { name = cm_ln_freed value = 0 }
	# Under the -1000000 the rank bottoms out at, so the first seat the walk reaches always takes it
	# and the value doubles as the found-one test.
	set_local_variable = { name = cm_ln_best value = -2000000 }
	every_in_list = {
		variable = cm_ln_moves
		# cm_ln_moves is built at classify and a seat can lose its governor afterwards, so only one
		# still holding the kind being placed can make the room.
		limit = {
			owner ?= scope:cm_country
			has_building = scope:cm_building_type
		}
		save_temporary_scope_as = cm_ln_seat_c
		if = {
			limit = { cm_ln_seat_give_up_rank > local_var:cm_ln_best }
			set_local_variable = { name = cm_ln_best value = cm_ln_seat_give_up_rank }
			save_scope_as = cm_ln_seat
		}
	}
	if = {
		limit = { local_var:cm_ln_best > -2000000 }
		set_local_variable = { name = cm_ln_freed value = 1 }
		scope:cm_ln_seat = { destroy_building = "building(scope:cm_building_type|scope:cm_country)" }
		remove_list_variable = { name = cm_ln_moves target = scope:cm_ln_seat }
		remove_list_variable = { name = cm_ln_seats target = scope:cm_ln_seat }
	}
}

# Rebuilds the governors that have gone missing from the seats this country was given, at no cost
# because it already paid for each of them once. A seat is watched until it is given up on purpose,
# so one the engine refuses for years still comes back the month it stops refusing. Stands aside
# while a reorganization runs, which gives seats up on purpose and re-places them itself.
# Root is the country.
cm_ln_seat_restore_pass = {
	if = {
		limit = {
			has_variable_list = cm_ln_seats
			OR = {
				NOT = { exists = var:cm_ln_phase }
				var:cm_ln_phase = 0
			}
		}
		save_scope_as = cm_country
		every_in_list = {
			variable = cm_ln_seats
			save_scope_as = cm_ln_target
			# The walk puts the location in root, and the step reads the country's slot room.
			scope:cm_country = { cm_ln_seat_restore_step = yes }
		}
	}
}

# A site the country no longer owns fails the placement check, both governor types carrying
# owner = scope:actor in their allow block, so it waits here against a reconquest.
# Root is the country, scope:cm_country and scope:cm_ln_target set.
cm_ln_seat_restore_step = {
	if = {
		limit = {
			scope:cm_ln_target = {
				cm_location_has_governor = no
				cm_fa_gov_location_placeable = yes
			}
		}
		cm_ln_resolve_placeable_type = { SITE = scope:cm_ln_target }
		cm_ln_construct_governor_free = yes
		cm_dbg_log = { cat = governors action = cm_ln_log_seat_restored }
	}
}

# Root is the country.
cm_ln_clear_seats = {
	clear_variable_list = cm_ln_seats
}
