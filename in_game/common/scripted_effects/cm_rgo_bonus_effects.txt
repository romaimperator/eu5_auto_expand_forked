# Runtime R.G.O. Bonus coverage: the hidden window's rgob waves classify every building
# type's worth-using production methods into global pair and row maps plus the good and
# method index chains, then the classification completion branch arms a batched sweep that
# precomputes per-province coverage. The filters read only the province cm_rgob_cov map.

# Provinces swept per driver tick. Sized to fit inside one frame: raising it shortens the
# tail, but a batch that spills past a frame reads as stutter the FPS counter misses.
@cm_rgob_batch_size = 48

# Clears every classification intermediate. Called when classification arms and again when
# the sweep finishes, so saves carry only cm_rgob_cov and the zeroed counters.
cm_rgob_reset_classification = {
	clear_global_variable_map = cm_rgob_best
	clear_global_variable_map = cm_rgob_total
	clear_global_variable_map = cm_rgob_total_claim
	clear_global_variable_map = cm_rgob_row_claim
	clear_global_variable_map = cm_rgob_row_pm
	clear_global_variable_map = cm_rgob_row_share
	clear_global_variable_map = cm_rgob_row_next
	clear_global_variable_map = cm_rgob_good_head
	clear_global_variable_map = cm_rgob_pair_bt
	clear_global_variable_map = cm_rgob_pair_next
	clear_global_variable_map = cm_rgob_pm_head
	set_global_variable = { name = cm_rgob_row_count value = 0 }
	set_global_variable = { name = cm_rgob_pair_count value = 0 }
}

# Clears the batched sweep's work list and cursor. Called when the work list is rebuilt at
# lobby init, which is what cancels a sweep a save interrupted mid-tail, and again when the
# sweep finishes.
cm_rgob_reset_sweep = {
	clear_global_variable_map = cm_rgob_wl
	set_global_variable = { name = cm_rgob_wl_count value = 0 }
	set_global_variable = { name = cm_rgob_wl_idx value = 0 }
	if = {
		limit = { has_global_variable = cm_rgob_sweep }
		remove_global_variable = cm_rgob_sweep
	}
}

# Province scope. Coverage per building type: sum of each worth-using production method's
# raw-input shares whose good is an RGO in this province, max over the type's methods.
# Only nonzero coverages are stored. Both passes are driven off what this province actually
# has - the goods present and the methods those goods summed - rather than off the whole
# row and pair sets, which a province of three or four raw materials would almost entirely
# skip. The presence list is deduped because it is iterated, not just membership-tested:
# a repeat would walk its good's chain twice and double every share.
cm_rgob_recompute_province = {
	clear_variable_map = cm_rgob_cov
	save_temporary_scope_as = cm_rgob_prov
	set_local_variable = { name = cm_rgob_any value = 0 }
	every_location_in_province = {
		limit = { exists = raw_material }
		raw_material = { save_temporary_scope_as = cm_rgob_g }
		scope:cm_rgob_prov = {
			if = {
				limit = { NOT = { is_target_in_local_variable_list = { name = cm_rgob_present target = scope:cm_rgob_g } } }
				add_to_local_variable_list = { name = cm_rgob_present target = scope:cm_rgob_g }
				set_local_variable = { name = cm_rgob_any value = 1 }
			}
		}
	}
	if = {
		limit = { local_var:cm_rgob_any = 1 }
		# Per-method sums of the shares whose good is present.
		every_in_local_list = {
			variable = cm_rgob_present
			save_temporary_scope_as = cm_rgob_g
			if = {
				limit = { is_key_in_global_variable_map = { name = cm_rgob_good_head target = scope:cm_rgob_g } }
				set_local_variable = { name = cm_rgob_ri value = "global_variable_map(cm_rgob_good_head|scope:cm_rgob_g)" }
				while = {
					limit = { local_var:cm_rgob_ri > 0 }
					"global_variable_map(cm_rgob_row_pm|local_var:cm_rgob_ri)" = { save_temporary_scope_as = cm_rgob_p }
					set_local_variable = { name = cm_rgob_s value = 0 }
					if = {
						limit = { scope:cm_rgob_prov = { is_key_in_variable_map = { name = cm_rgob_sum target = scope:cm_rgob_p } } }
						set_local_variable = { name = cm_rgob_s value = "scope:cm_rgob_prov.variable_map(cm_rgob_sum|scope:cm_rgob_p)" }
					}
					set_local_variable = { name = cm_rgob_s2 value = { value = local_var:cm_rgob_s add = "global_variable_map(cm_rgob_row_share|local_var:cm_rgob_ri)" } }
					scope:cm_rgob_prov = {
						remove_from_variable_map = { name = cm_rgob_sum key = scope:cm_rgob_p }
						add_to_variable_map = { name = cm_rgob_sum key = scope:cm_rgob_p value = local_var:cm_rgob_s2 }
					}
					set_local_variable = { name = cm_rgob_rn value = "global_variable_map(cm_rgob_row_next|local_var:cm_rgob_ri)" }
					set_local_variable = { name = cm_rgob_ri value = local_var:cm_rgob_rn }
				}
			}
		}
		# Fold each summed method's pairs into the building type's running max.
		if = {
			limit = { has_variable_map = cm_rgob_sum }
			every_key_in_variable_map = {
				variable = cm_rgob_sum
				save_temporary_scope_as = cm_rgob_p
				if = {
					limit = { is_key_in_global_variable_map = { name = cm_rgob_pm_head target = scope:cm_rgob_p } }
					set_local_variable = { name = cm_rgob_c value = "scope:cm_rgob_prov.variable_map(cm_rgob_sum|scope:cm_rgob_p)" }
					set_local_variable = { name = cm_rgob_pi value = "global_variable_map(cm_rgob_pm_head|scope:cm_rgob_p)" }
					while = {
						limit = { local_var:cm_rgob_pi > 0 }
						"global_variable_map(cm_rgob_pair_bt|local_var:cm_rgob_pi)" = { save_temporary_scope_as = cm_rgob_b }
						set_local_variable = { name = cm_rgob_prev value = 0 }
						if = {
							limit = { scope:cm_rgob_prov = { is_key_in_variable_map = { name = cm_rgob_cov target = scope:cm_rgob_b } } }
							set_local_variable = { name = cm_rgob_prev value = "scope:cm_rgob_prov.variable_map(cm_rgob_cov|scope:cm_rgob_b)" }
						}
						if = {
							limit = { local_var:cm_rgob_c > local_var:cm_rgob_prev }
							scope:cm_rgob_prov = {
								remove_from_variable_map = { name = cm_rgob_cov key = scope:cm_rgob_b }
								add_to_variable_map = { name = cm_rgob_cov key = scope:cm_rgob_b value = local_var:cm_rgob_c }
							}
						}
						set_local_variable = { name = cm_rgob_pn value = "global_variable_map(cm_rgob_pair_next|local_var:cm_rgob_pi)" }
						set_local_variable = { name = cm_rgob_pi value = local_var:cm_rgob_pn }
					}
				}
			}
		}
		scope:cm_rgob_prov = {
			clear_variable_map = cm_rgob_sum
			clear_local_variable_list = cm_rgob_present
		}
	}
}

# Armed by the classification completion branch, because the rgob waves must finish first
# and so this cannot run from cm_on_init_lobby_globals like the trmm sweep. The work list
# was built at lobby init, so arming is two writes.
cm_rgob_arm_sweep = {
	if = {
		limit = { global_var:cm_rgob_wl_count > 0 }
		set_global_variable = { name = cm_rgob_wl_idx value = 0 }
		set_global_variable = { name = cm_rgob_sweep value = yes }
	}
}

# One driver tick of the coverage sweep. Guarded on the flag because the driver's state
# machine keeps ticking after its gate goes false, and because every client's Execute is
# broadcast in multiplayer. Mid-session raw material changes stay stale until the next lobby.
cm_rgob_sweep_batch = {
	if = {
		limit = { has_global_variable = cm_rgob_sweep }
		if = {
			limit = { global_var:cm_rgob_wl_idx < global_var:cm_rgob_wl_count }
			set_local_variable = { name = cm_rgob_whi value = { value = global_var:cm_rgob_wl_idx add = @cm_rgob_batch_size } }
			if = {
				limit = { local_var:cm_rgob_whi > global_var:cm_rgob_wl_count }
				set_local_variable = { name = cm_rgob_whi value = global_var:cm_rgob_wl_count }
			}
			set_local_variable = { name = cm_rgob_wi value = global_var:cm_rgob_wl_idx }
			while = {
				limit = { local_var:cm_rgob_wi < local_var:cm_rgob_whi }
				change_local_variable = { name = cm_rgob_wi add = 1 }
				# The work list is a lobby-time snapshot of province slices.
				if = {
					limit = { exists = "global_variable_map(cm_rgob_wl|local_var:cm_rgob_wi)" }
					"global_variable_map(cm_rgob_wl|local_var:cm_rgob_wi)" = {
						cm_rgob_recompute_province = yes
					}
				}
			}
			set_global_variable = { name = cm_rgob_wl_idx value = local_var:cm_rgob_whi }
		}
		else = {
			cm_rgob_reset_classification = yes
			cm_rgob_reset_sweep = yes
		}
	}
}
