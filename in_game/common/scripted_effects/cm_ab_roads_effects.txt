# Auto-Build Roads: builds the road network worth the most per gold, across three targets that
# run in a fixed order. Proximity first, then market access, then connecting locations to the
# capital. Each one plans a parent tree, walks it from the leaves up so every corridor
# carries the value of everything behind it, and the monthly drain buys the best corridors its budget
# share reaches. A lower target gets a turn only in a month the one above it ordered nothing.

# Orders that produced nothing before a corridor is set aside.
@cm_ab_rd_max_tries = 3
# How long a set-aside corridor stays out of the running.
@cm_ab_rd_bad_days = 3650
# Relaxation waves a spread may run. Each wave is one hop of the frontier, so this is a diameter
# bound; the engine aborts a while at 1000 iterations either way.
@cm_ab_rd_max_waves = 200

# The plan. Root is the country.
cm_ab_rd_plan = {
	if = {
		limit = {
			cm_ab_master_active = yes
			exists = var:cm_ab_roads_enabled
			has_advance = road_building # Unlocks gravel_road: in_game/common/advances/3_road_unlocks.txt:1
		}
		save_scope_as = cm_country
		save_temporary_scope_value_as = { name = cm_ab_rd_tier value = cm_ab_rd_tier_cost_value }
		# Each target records whether its pass ran at all, which is what tells a target that planned
		# nothing apart from one that never got to look. Targets 1 and 2 both wait on the river
		# harvest, so a plan forced before it lands leaves only target 3 with candidates.
		set_variable = { name = cm_ab_rd_ran_1 value = 0 }
		set_variable = { name = cm_ab_rd_ran_2 value = 0 }
		set_variable = { name = cm_ab_rd_ran_3 value = 0 }
		cm_ab_rd_plan_proximity = yes
		cm_ab_rd_plan_market_access = yes
		cm_ab_rd_plan_capital = yes
	}
}

# Target 1. Root is the country, scope:cm_country and scope:cm_ab_rd_tier set. Borrows the placement
# finder's multi-source spread, which is why it carries gates the other two targets do not need.
cm_ab_rd_plan_proximity = {
	if = {
		limit = {
			exists = var:cm_ab_roads_proximity
			cm_pf_river_data_is_current = yes
			NOT = { exists = var:cm_pf_mode_open }
			# cm_pf_n marks a primed run and finalize drops it while leaving cm_pf_all_done, so this
			# pair is "primed but never finished". Re-priming under one would destroy it.
			OR = {
				NOT = { has_variable = cm_pf_n }
				has_variable = cm_pf_all_done
			}
		}
		set_variable = { name = cm_ab_rd_ran_1 value = 1 }
		set_variable = { name = cm_ab_rd_plan_active value = 1 }
		cm_pf_borrow_setting = { S = roads V = 2 } # Max Roads: the network the plan is pricing
		cm_pf_borrow_setting = { S = naval V = 2 } # Max Presence
		cm_pf_borrow_setting = { S = method V = 1 } # Tax Base
		cm_pf_borrow_setting = { S = accuracy V = 1 } # Exhaustive: the phased runner scores candidates, and the plan scores none
		# The multi-source spread the plan reads runs only for the governor mode.
		set_variable = { name = cm_pf_view_g value = 1 }
		cm_pf_prep_core = yes
		cm_ab_rd_value_pass = yes
		cm_pf_clear_run_state = yes
		remove_variable = cm_pf_view_g
		remove_variable = cm_ab_rd_plan_active
		cm_pf_return_setting = { S = roads }
		cm_pf_return_setting = { S = naval }
		cm_pf_return_setting = { S = method }
		cm_pf_return_setting = { S = accuracy }
	}
	else = {
		# Which of the four it was, because the gate is the usual reason this target plans nothing and
		# a bare skip line costs a round to chase.
		if = {
			limit = { NOT = { exists = var:cm_ab_roads_proximity } }
			cm_dbg_log = { cat = roads action = cm_log_rd_skip_prox_off }
		}
		else_if = {
			limit = { cm_pf_river_data_is_current = no }
			cm_dbg_log = { cat = roads action = cm_log_rd_skip_prox_river }
		}
		else_if = {
			limit = { exists = var:cm_pf_mode_open }
			cm_dbg_log = { cat = roads action = cm_log_rd_skip_prox_mode }
		}
		else = {
			cm_dbg_log = { cat = roads action = cm_log_rd_skip_prox_run }
		}
	}
}

# Turns the spread into corridor scores. Ordered iterators sort high to low, so the negated
# predicted proximity visits every tile before the one it reaches the sources through, and one pass
# carries each tile's own unrealized tax base onto every corridor behind it.
# Root is the country, scope:cm_country set.
cm_ab_rd_value_pass = {
	every_owned_location = {
		set_variable = { name = cm_ab_rd_sub value = 0 }
		remove_variable = cm_ab_rd_par_1
		remove_variable = cm_ab_rd_score_1
	}
	ordered_owned_location = {
		order_by = cm_ab_rd_leaf_order
		max = cm_max_guess_at_number_of_locations
		check_range_bounds = no
		cm_ab_rd_value_location = yes
	}
	every_owned_location = { remove_variable = cm_ab_rd_sub }
}

# Root is the location, scope:cm_country set.
cm_ab_rd_value_location = {
	change_variable = { name = cm_ab_rd_sub add = cm_ab_rd_node_demand }
	if = {
		limit = {
			has_variable = cm_pf_prev
			var:cm_pf_prev = {
				is_land = yes
				owner ?= scope:cm_country
			}
		}
		set_local_variable = { name = cm_ab_rd_push value = var:cm_ab_rd_sub }
		var:cm_pf_prev = {
			change_variable = { name = cm_ab_rd_sub add = local_var:cm_ab_rd_push }
		}
		# A corridor already carrying the best road this country can lay needs nothing, but the tiles
		# behind it still do, which is why the total travels up either way. A corridor with nothing
		# behind it left to gain is not worth gold, so it never becomes a candidate.
		# A road only lowers the proximity cost across its own span, so a corridor with zero
		# proximity at both ends can never raise it at either.
		if = {
			limit = {
				var:cm_ab_rd_sub > 0
				OR = {
					cm_ab_location_has_proximity = yes
					var:cm_pf_prev = { cm_ab_location_has_proximity = yes }
				}
				NOT = { has_latest_road_to = var:cm_pf_prev }
			}
			save_temporary_scope_as = cm_ab_rd_from
			var:cm_pf_prev = { save_temporary_scope_as = cm_ab_rd_to }
			set_variable = { name = cm_ab_rd_par_1 value = scope:cm_ab_rd_to }
			set_variable = { name = cm_ab_rd_score_1 value = cm_ab_rd_score_value }
		}
	}
}

# Target 2. Root is the country, scope:cm_country and scope:cm_ab_rd_tier set.
cm_ab_rd_plan_market_access = {
	if = {
		limit = {
			exists = var:cm_ab_roads_market_access
			# The relaxation skips river-connected edges, which needs the harvested crossings.
			cm_pf_river_data_is_current = yes
		}
		set_variable = { name = cm_ab_rd_ran_2 value = 1 }
		# Country-scope, and the pass runs in location scope, so both are read once here.
		save_temporary_scope_value_as = { name = cm_ab_rd_mafac value = cm_ab_rd_ma_segment_factor }
		save_temporary_scope_value_as = { name = cm_ab_rd_magain value = cm_ab_rd_tier_ma_gain }
		cm_ab_rd_ma_pass = yes
		cm_ab_rd_ma_value_pass = yes
		every_owned_location = {
			remove_variable = cm_ab_rd_ma_l
			remove_variable = cm_ab_rd_ma_g
			clear_variable_map = cm_ab_rd_ed
		}
	}
}

# A location's transport loss to its market centre is 1 + local_market_access - market_access, and it
# is additive along the path, so the drop between a location and the neighbour it reaches the centre
# through is what that edge costs and a road takes the tier's share off it. Reading the loss live
# means every sea leg, port hop and river already in the route is carried without being modelled.
# Root is the country, scope:cm_country set.
cm_ab_rd_ma_pass = {
	every_owned_location = {
		remove_variable = cm_ab_rd_par_2
		remove_variable = cm_ab_rd_ma_g
		if = {
			limit = {
				is_land = yes
				exists = market
			}
			set_variable = { name = cm_ab_rd_ma_l value = cm_ab_rd_loss_value }
			cm_ab_rd_cache_edges = yes
		}
		else = { remove_variable = cm_ab_rd_ma_l }
	}
	# Second sweep, because picking a parent reads every neighbour's loss and the first sweep is what
	# sets them.
	every_owned_location = {
		limit = { has_variable = cm_ab_rd_ma_l }
		cm_ab_rd_ma_pick_parent = yes
	}
}

# The edge distances a relaxation re-reads every time a location re-enters the frontier. Caching them
# once turns a distance_to plus two movement-cost chains per attempt into one map read.
# Root is the location, scope:cm_country set.
cm_ab_rd_cache_edges = {
	clear_variable_map = cm_ab_rd_ed
	save_temporary_scope_as = cm_ab_rd_from
	every_neighbor_location = {
		limit = {
			is_land = yes
			owner ?= scope:cm_country
			market ?= scope:cm_ab_rd_from.market
		}
		save_temporary_scope_as = cm_ab_rd_to
		scope:cm_ab_rd_from = {
			# A river edge already carries market access better than any road laid on it, so a road
			# there buys nothing: MARKET_DOWNSTREAM_FACTOR 0.5 and MARKET_UPSTREAM_FACTOR 0.9
			# against the road's own (1 + market_access) floor of 0.6.
			# loading_screen/common/defines/00_defines.txt:1787-1788
			if = {
				limit = {
					NOT = {
						is_target_in_variable_list = { name = cm_pf_river_neighbors target = scope:cm_ab_rd_to }
					}
				}
				set_local_variable = { name = cm_ab_rd_ud value = cm_ab_rd_unit_distance_value }
				add_to_variable_map = {
					name = cm_ab_rd_ed
					key = scope:cm_ab_rd_to
					value = local_var:cm_ab_rd_ud
				}
			}
		}
	}
}

# Root is the location, scope:cm_country set.
# Root is the location, scope:cm_country, scope:cm_ab_rd_mafac and scope:cm_ab_rd_magain set.
cm_ab_rd_ma_pick_parent = {
	save_temporary_scope_as = cm_ab_rd_a
	set_local_variable = { name = cm_ab_rd_bestc value = 999 }
	set_local_variable = { name = cm_ab_rd_bestl value = 999 }
	every_key_in_variable_map = {
		variable = cm_ab_rd_ed
		save_temporary_scope_as = cm_ab_rd_b
		if = {
			limit = { scope:cm_ab_rd_b = { has_variable = cm_ab_rd_ma_l } }
			# A quoted map lookup resolves to nothing read straight into an expression, so capture it.
			scope:cm_ab_rd_a = {
				set_local_variable = {
					name = cm_ab_rd_ud
					value = "variable_map(cm_ab_rd_ed|scope:cm_ab_rd_b)"
				}
			}
			set_local_variable = {
				name = cm_ab_rd_cand
				value = {
					value = scope:cm_ab_rd_b.var:cm_ab_rd_ma_l
					add = {
						value = local_var:cm_ab_rd_ud
						multiply = scope:cm_ab_rd_mafac
					}
				}
			}
			if = {
				limit = { local_var:cm_ab_rd_cand < local_var:cm_ab_rd_bestc }
				set_local_variable = { name = cm_ab_rd_bestc value = local_var:cm_ab_rd_cand }
				set_local_variable = { name = cm_ab_rd_bestl value = scope:cm_ab_rd_b.var:cm_ab_rd_ma_l }
				# The iterator makes the key the scope, so the parent has to be written on the location
				# being solved rather than on the neighbour being weighed.
				scope:cm_ab_rd_a = {
					set_variable = { name = cm_ab_rd_par_2 value = scope:cm_ab_rd_b }
				}
			}
		}
	}
	# The modelled distance ranks the neighbours; the gain comes from the engine instead. A location
	# reaches its centre through the parent, so the drop in loss between the two is what that edge
	# really costs, or less where the parent guess is wrong, and a road takes the tier's share off it.
	# Measured 2026-08-06: the modelled segment ran 1.70x the engine's own drop, so a model asked for
	# the magnitude could never see the tenth a gravel road is worth.
	if = {
		limit = {
			has_variable = cm_ab_rd_par_2
			var:cm_ab_rd_ma_l > local_var:cm_ab_rd_bestl
		}
		set_variable = {
			name = cm_ab_rd_ma_g
			value = {
				value = var:cm_ab_rd_ma_l
				subtract = local_var:cm_ab_rd_bestl
				multiply = scope:cm_ab_rd_magain
			}
		}
	}
	else = { remove_variable = cm_ab_rd_par_2 }
}

# Temporary scaffolding while the market access model is being judged. For every location it takes
# the cached neighbour with the LOWEST predicted loss, which is the most favourable reading available,
# and totals that edge's modelled segment against the segment the engine's own losses imply. If the
# two totals disagree by more than a tenth the model can never see a road's own tenth and the
# relaxation is asking a modelled number to out-resolve its own error.
# Root is the country, scope:cm_country and scope:cm_ab_rd_mafac set.
# Root is the location, scope:cm_country and scope:cm_ab_rd_mafac set.
# Root is the location, scope:cm_country set.
cm_ab_rd_front_push = {
	if = {
		limit = { NOT = { has_variable = cm_ab_rd_inq } }
		set_variable = { name = cm_ab_rd_inq value = 1 }
		save_temporary_scope_as = cm_ab_rd_pushed
		scope:cm_country = {
			add_to_variable_list = { name = cm_ab_rd_front target = scope:cm_ab_rd_pushed }
		}
	}
}

# A wave-cap exit leaves pushed locations in the list, and a stale cm_ab_rd_inq would block their
# next push. Root is the country, scope:cm_country set.
cm_ab_rd_front_residue_clear = {
	every_in_list = {
		variable = cm_ab_rd_front
		remove_variable = cm_ab_rd_inq
	}
	clear_variable_list = cm_ab_rd_front
}

# Highest predicted loss is farthest from the centre, so ordering by it visits every tile before the
# one it routes through. Root is the country, scope:cm_country set.
cm_ab_rd_ma_value_pass = {
	every_owned_location = {
		set_variable = { name = cm_ab_rd_sub value = 0 }
		remove_variable = cm_ab_rd_score_2
	}
	ordered_owned_location = {
		order_by = cm_ab_rd_ma_order
		max = cm_max_guess_at_number_of_locations
		check_range_bounds = no

	}
	every_owned_location = { remove_variable = cm_ab_rd_sub }
}

# Root is the location, scope:cm_country set.
cm_ab_rd_ma_value_location = {
	# Plain tax base: every location behind this corridor gains the same absolute access the road takes
	# off it, so the corridor is worth its own gain times everything routing through it.
	change_variable = { name = cm_ab_rd_sub add = location_tax_base }
	if = {
		limit = { has_variable = cm_ab_rd_par_2 }
		set_local_variable = { name = cm_ab_rd_push value = var:cm_ab_rd_sub }
		var:cm_ab_rd_par_2 = {
			change_variable = { name = cm_ab_rd_sub add = local_var:cm_ab_rd_push }
		}
		if = {
			limit = {
				var:cm_ab_rd_sub > 0
				has_variable = cm_ab_rd_ma_g
				var:cm_ab_rd_ma_g > 0
				NOT = { has_latest_road_to = var:cm_ab_rd_par_2 }
			}
			save_temporary_scope_as = cm_ab_rd_from
			var:cm_ab_rd_par_2 = { save_temporary_scope_as = cm_ab_rd_to }
			set_variable = { name = cm_ab_rd_score_2 value = cm_ab_rd_ma_score_value }
		}
	}
}

# Target 3. Root is the country, scope:cm_country and scope:cm_ab_rd_tier set.
cm_ab_rd_plan_capital = {
	if = {
		limit = { exists = var:cm_ab_roads_capital }
		set_variable = { name = cm_ab_rd_ran_3 value = 1 }
		cm_ab_rd_cc_spread = yes
		cm_ab_rd_cc_value_pass = yes
		every_owned_location = { remove_variable = cm_ab_rd_cc_d }
	}
}

# A hop-count sweep outward from everything already connected. Hops rather than gold: each location
# improves exactly once so the wave count is the hop diameter rather than the location count, the
# per-edge cost read stays out of the inner loop, and the nearest neighbour toward the connected set
# is the parent wanted anyway. The capital is seeded directly rather than trusting it to answer
# has_road_to_capital about itself, or a country with no roads at all would seed nothing and this
# target would never do anything. Root is the country, scope:cm_country set.
cm_ab_rd_cc_spread = {
	clear_variable_list = cm_ab_rd_front
	every_owned_location = {
		remove_variable = cm_ab_rd_par_3
		remove_variable = cm_ab_rd_cc_d
	}
	if = {
		limit = { exists = capital }
		capital = {
			set_variable = { name = cm_ab_rd_cc_d value = 0 }
			cm_ab_rd_front_push = yes
		}
	}
	every_owned_location = {
		limit = {
			is_land = yes
			has_road_to_capital = yes
			NOT = { has_variable = cm_ab_rd_cc_d }
		}
		set_variable = { name = cm_ab_rd_cc_d value = 0 }
		cm_ab_rd_front_push = yes
	}
	set_local_variable = { name = cm_ab_rd_changed value = 1 }
	set_local_variable = { name = cm_ab_rd_wave value = 0 }
	while = {
		limit = {
			local_var:cm_ab_rd_changed > 0
			local_var:cm_ab_rd_wave < @cm_ab_rd_max_waves
		}
		change_local_variable = { name = cm_ab_rd_wave add = 1 }
		set_local_variable = { name = cm_ab_rd_changed value = 0 }
		clear_local_variable_list = cm_ab_rd_front_cur
		every_in_list = {
			variable = cm_ab_rd_front
			remove_variable = cm_ab_rd_inq
			add_to_local_variable_list = { name = cm_ab_rd_front_cur target = this }
		}
		clear_variable_list = cm_ab_rd_front
		every_in_local_list = {
			variable = cm_ab_rd_front_cur
			cm_ab_rd_cc_relax = yes
		}
	}
	cm_ab_rd_front_residue_clear = yes
}

# Root is the location, scope:cm_country set.
cm_ab_rd_cc_relax = {
	save_temporary_scope_as = cm_ab_rd_a
	set_local_variable = {
		name = cm_ab_rd_cand
		value = {
			value = var:cm_ab_rd_cc_d
			add = 1
		}
	}
	every_neighbor_location = {
		limit = {
			is_land = yes
			owner ?= scope:cm_country
			OR = {
				NOT = { has_variable = cm_ab_rd_cc_d }
				var:cm_ab_rd_cc_d > local_var:cm_ab_rd_cand
			}
		}
		set_variable = { name = cm_ab_rd_cc_d value = local_var:cm_ab_rd_cand }
		set_variable = { name = cm_ab_rd_par_3 value = scope:cm_ab_rd_a }
		cm_ab_rd_front_push = yes
		change_local_variable = { name = cm_ab_rd_changed add = 1 }
	}
}

# Roads drain. Root is the country, scope:cm_country set. A target gets a turn only in a month the
# one above it ordered nothing and did not stop on gold: a corridor the category cannot afford holds
# the pool for itself through cm_ab_cost_blocked, and a lower target spending it would defeat that.
# The test is re-made every month and never latches, so conquest or a new road tier puts proximity
# back in front the first month it has a corridor to lay.
cm_ab_pass_roads = {
	cm_ab_rd_check_orders = yes
	if = {
		limit = {
			has_advance = road_building # Unlocks gravel_road: in_game/common/advances/3_road_unlocks.txt:1
			modifier:overlord_blocked_from_building_roads = no
		}
		save_temporary_scope_value_as = { name = cm_ab_rd_tier value = cm_ab_rd_tier_cost_value }
		set_local_variable = { name = cm_ab_rd_ordered value = 0 }
		if = {
			limit = { exists = var:cm_ab_roads_proximity }
			cm_ab_rd_issue_pass = { TGT = 1 RING = always }
		}
		if = {
			limit = {
				exists = var:cm_ab_roads_market_access
				local_var:cm_ab_rd_ordered = 0
				local_var:cm_ab_cost_blocked = 0
			}
			cm_ab_rd_issue_pass = { TGT = 2 RING = always }
		}
		if = {
			limit = {
				exists = var:cm_ab_roads_capital
				local_var:cm_ab_rd_ordered = 0
				local_var:cm_ab_cost_blocked = 0
			}
			cm_ab_rd_issue_pass = { TGT = 3 RING = cm_ab_rd_cc_at_frontier }
		}
	}
}

# Nothing bounds how many corridors a cycle orders or how many may be in flight: the pool is the
# only limit, and the first corridor it cannot cover stops the pass through cm_ab_cost_blocked.
# $TGT$ = the target's number, $RING$ = an extra per-candidate trigger or always.
# Root is the country, scope:cm_country set.
cm_ab_rd_issue_pass = {
	ordered_owned_location = {
		order_by = { value = var:cm_ab_rd_score_$TGT$ }
		max = cm_max_guess_at_number_of_locations
		check_range_bounds = no
		limit = {
			has_variable = cm_ab_rd_score_$TGT$
			has_variable = cm_ab_rd_par_$TGT$
			$RING$ = yes
			# cm_ab_rd_bad condemns the corridor that kept producing nothing, not the location it
			# starts at, which may have a perfectly buildable corridor to another neighbour.
			trigger_if = {
				limit = { has_variable = cm_ab_rd_bad }
				OR = {
					NOT = { has_variable = cm_ab_rd_bad_par }
					NOT = { var:cm_ab_rd_bad_par = var:cm_ab_rd_par_$TGT$ }
				}
			}
			has_road_constructions = no
			modifier:road_building_blocked = no
			cm_ab_rd_meets_min_discount = yes
			cm_market_has_construction_goods_for_road = yes
			var:cm_ab_rd_par_$TGT$ = {
				owner ?= scope:cm_country
				modifier:road_building_blocked = no
			}
			NOT = { has_latest_road_to = var:cm_ab_rd_par_$TGT$ }
		}
		cm_ab_rd_try_build = { TGT = $TGT$ }
	}
}

# Root is the location, scope:cm_country set. Reads and sets the category's cost-block signal.
cm_ab_rd_try_build = {
	if = {
		limit = { local_var:cm_ab_cost_blocked = 0 }
		save_temporary_scope_as = cm_ab_rd_from
		var:cm_ab_rd_par_$TGT$ = { save_temporary_scope_as = cm_ab_rd_to }
		save_temporary_scope_value_as = { name = cm_cost value = cm_ab_rd_cost_value }
		if = {
			limit = { cm_ab_can_afford = { POOL = cm_ab_pool_roads } }
			# The parent is recorded with the order because a later plan rewrites cm_ab_rd_par_N
			# under an order still in flight, and check_orders would then judge it against a corridor
			# it was never for.
			set_variable = { name = cm_ab_rd_ord_par value = scope:cm_ab_rd_to }
			scope:cm_country = {
				cm_ab_rd_construct = yes
				change_variable = { name = cm_ab_pool_roads subtract = scope:cm_cost }
				# construct_road charges at once, so the gold the next corridor is tested against has
				# already lost this one. Booking it in cm_q_planned_spend as well would take it out
				# twice and halve the budget; that list is for the builds the queue window fires later.
				add_to_variable_list = { name = cm_ab_rd_building target = scope:cm_ab_rd_from }
			}
			save_temporary_scope_as = cmf_log_loc
			scope:cm_country = {
				cm_dbg_log_loc = {
					cat = roads
					action = cm_log_rd_ordered_$TGT$
					arg1 = cm_log_rd_a_cost
					value = scope:cm_cost
				}
			}
			change_local_variable = { name = cm_ab_rd_ordered add = 1 }
		}
		else = {
			# Waiting for the best corridor beats laying a worse one, the same stop Production takes.
			cm_ab_stop_on_cost = { COST = scope:cm_cost }
		}
	}
}

# Root is the country, expects scope:cm_ab_rd_from and scope:cm_ab_rd_to.
cm_ab_rd_construct = {
	if = {
		# railroad_advance unlocks railroad: in_game/common/advances/3_road_unlocks.txt:28
		limit = { has_advance = railroad_advance }
		construct_road = { road_type = road_type:railroad from = scope:cm_ab_rd_from to = scope:cm_ab_rd_to }
	}
	else_if = {
		# modern_road_advance unlocks modern_road: in_game/common/advances/3_road_unlocks.txt:20
		limit = { has_advance = modern_road_advance }
		construct_road = { road_type = road_type:modern_road from = scope:cm_ab_rd_from to = scope:cm_ab_rd_to }
	}
	else_if = {
		# paved_road_advance unlocks paved_road: in_game/common/advances/3_road_unlocks.txt:12
		limit = { has_advance = paved_road_advance }
		construct_road = { road_type = road_type:paved_road from = scope:cm_ab_rd_from to = scope:cm_ab_rd_to }
	}
	else = {
		construct_road = { road_type = road_type:gravel_road from = scope:cm_ab_rd_from to = scope:cm_ab_rd_to }
	}
}

# Rebuilds the in-flight list and judges the orders that left it. A road order reports neither cost
# nor result, so a construction that never appears is the only sign the engine refused one, and a
# corridor that keeps producing nothing is set aside for a while, which is what a crossing no road
# can span comes to. Root is the country, scope:cm_country set.
cm_ab_rd_check_orders = {
	if = {
		limit = { has_variable_list = cm_ab_rd_building }
		clear_variable_list = cm_ab_rd_keep
		every_in_list = {
			variable = cm_ab_rd_building
			if = {
				limit = { has_road_constructions = yes }
				remove_variable = cm_ab_rd_try
				save_temporary_scope_as = cm_ab_rd_live
				scope:cm_country = {
					add_to_variable_list = { name = cm_ab_rd_keep target = scope:cm_ab_rd_live }
				}
			}
			else_if = {
				limit = {
					has_variable = cm_ab_rd_ord_par
					has_latest_road_to = var:cm_ab_rd_ord_par
				}
				remove_variable = cm_ab_rd_try
				remove_variable = cm_ab_rd_ord_par
			}
			else = {
				if = {
					limit = { has_variable = cm_ab_rd_try }
					change_variable = { name = cm_ab_rd_try add = 1 }
				}
				else = {
					set_variable = { name = cm_ab_rd_try value = 1 }
				}
				if = {
					limit = { var:cm_ab_rd_try >= @cm_ab_rd_max_tries }
					# The corridor is condemned, not the location: a crossing no road can span says
					# nothing about this location's other neighbours, and a later target will want them.
					if = {
						limit = { has_variable = cm_ab_rd_ord_par }
						set_variable = { name = cm_ab_rd_bad_par value = var:cm_ab_rd_ord_par }
					}
					set_variable = { name = cm_ab_rd_bad value = 1 days = @cm_ab_rd_bad_days }
					remove_variable = cm_ab_rd_try
					remove_variable = cm_ab_rd_ord_par
				}
			}
		}
		# The survivors are the corridors still being built; everything else has left the list.
		clear_variable_list = cm_ab_rd_building
		every_in_list = {
			variable = cm_ab_rd_keep
			save_temporary_scope_as = cm_ab_rd_live
			scope:cm_country = {
				add_to_variable_list = { name = cm_ab_rd_building target = scope:cm_ab_rd_live }
			}
		}
		clear_variable_list = cm_ab_rd_keep
	}
}

# Console diagnostic: effect c:<TAG> = { cm_ab_rd_probe = yes }. Ungated like the other probes,
# since running it is the request. What it answers is which targets still hold candidates, which is
# the difference between a target that stayed quiet because the one above it was still working and
# one that stayed quiet because it had nothing to lay. Root is the country.
cm_ab_rd_probe = {
	save_scope_as = cm_country
	set_variable = { name = cm_ab_rd_p1 value = 0 }
	set_variable = { name = cm_ab_rd_p2 value = 0 }
	set_variable = { name = cm_ab_rd_p2p value = 0 }
	set_variable = { name = cm_ab_rd_p3 value = 0 }
	set_variable = { name = cm_ab_rd_p3r value = 0 }
	set_variable = { name = cm_ab_rd_pf value = 0 }
	set_variable = { name = cm_ab_rd_ps value = 0 }
	every_owned_location = {
		if = {
			limit = { has_variable = cm_ab_rd_score_1 }
			scope:cm_country = { change_variable = { name = cm_ab_rd_p1 add = 1 } }
		}
		if = {
			limit = { has_variable = cm_ab_rd_score_2 }
			scope:cm_country = { change_variable = { name = cm_ab_rd_p2 add = 1 } }
		}
		# The market access tree's parent count, which separates a relaxation that improved nothing
		# from one that improved plenty the store gate then rejected.
		if = {
			limit = { has_variable = cm_ab_rd_par_2 }
			scope:cm_country = { change_variable = { name = cm_ab_rd_p2p add = 1 } }
		}
		if = {
			limit = { has_variable = cm_ab_rd_score_3 }
			scope:cm_country = { change_variable = { name = cm_ab_rd_p3 add = 1 } }
			if = {
				limit = { cm_ab_rd_cc_at_frontier = yes }
				scope:cm_country = { change_variable = { name = cm_ab_rd_p3r add = 1 } }
			}
		}
		# A planned corridor the drain will pass over, which a count of 0 orders cannot tell apart
		# from an empty plan or a spent pool.
		if = {
			limit = {
				OR = {
					has_variable = cm_ab_rd_score_1
					has_variable = cm_ab_rd_score_2
					has_variable = cm_ab_rd_score_3
				}
				cm_market_has_construction_goods_for_road = no
			}
			scope:cm_country = { change_variable = { name = cm_ab_rd_ps add = 1 } }
		}
	}
	if = {
		limit = { has_variable_list = cm_ab_rd_building }
		every_in_list = {
			variable = cm_ab_rd_building
			scope:cm_country = { change_variable = { name = cm_ab_rd_pf add = 1 } }
		}
	}
	cmf_log_value = { action = cm_log_rd_p_prox value = var:cm_ab_rd_p1 }
	cmf_log_value = { action = cm_log_rd_p_ma value = var:cm_ab_rd_p2 }
	cmf_log_value = { action = cm_log_rd_p_ma_par value = var:cm_ab_rd_p2p }
	cmf_log_value = { action = cm_log_rd_p_cap value = var:cm_ab_rd_p3 }
	cmf_log_value = { action = cm_log_rd_p_cap_ring value = var:cm_ab_rd_p3r }
	cmf_log_value = { action = cm_log_rd_p_flight value = var:cm_ab_rd_pf }
	cmf_log_value = { action = cm_log_rd_p_short value = var:cm_ab_rd_ps }
	if = {
		limit = { has_variable = cm_ab_pool_roads }
		cmf_log_decimal_value = { action = cm_log_rd_p_pool value = var:cm_ab_pool_roads }
	}
	# Only speaks up when a target never got to look, which is the case a count of 0 cannot tell
	# apart from a target that looked and found nothing.
	# cm_ab_rd_clear_plan removes these, and an unset left side is three script errors per read.
	if = {
		limit = {
			trigger_if = {
				limit = { has_variable = cm_ab_rd_ran_1 }
				var:cm_ab_rd_ran_1 = 0
			}
			trigger_else = { always = yes }
		}
		cmf_log = { action = cm_log_rd_p_skip_prox }
	}
	if = {
		limit = {
			trigger_if = {
				limit = { has_variable = cm_ab_rd_ran_2 }
				var:cm_ab_rd_ran_2 = 0
			}
			trigger_else = { always = yes }
		}
		cmf_log = { action = cm_log_rd_p_skip_ma }
	}
	if = {
		limit = {
			trigger_if = {
				limit = { has_variable = cm_ab_rd_ran_3 }
				var:cm_ab_rd_ran_3 = 0
			}
			trigger_else = { always = yes }
		}
		cmf_log = { action = cm_log_rd_p_skip_cap }
	}
	remove_variable = cm_ab_rd_p1
	remove_variable = cm_ab_rd_p2
	remove_variable = cm_ab_rd_p2p
	remove_variable = cm_ab_rd_p3
	remove_variable = cm_ab_rd_p3r
	remove_variable = cm_ab_rd_pf
	remove_variable = cm_ab_rd_ps
}
