# Rebuild priority feature lookup tables from CMM list field data.
# Called on registration and on every setting change.
# Scope: country
cm_rebuild_priority_lookups = {
	if = {
		limit = { NOT = { has_variable_list = cmm_list_items_cm__priority_list } }
	}
	else = {

	# Ordered features list for monthly pulse iteration in priority order.
	cmm_build_list_ordered_values = { setting = cm__priority_list list_name = cm_priority_features_list }

	# Gate threshold by feature, numeric field slot 1, direct values.
	cmm_build_list_field_map = { setting = cm__priority_list field_slot = 1 map_name = cm_priority_min_profit multiply = 1 }

	# Min discount by feature, numeric field slot 2, CMM -33..33 -> -0.33..0.33.
	cmm_build_list_field_map = { setting = cm__priority_list field_slot = 2 map_name = cm_priority_min_discount multiply = 0.01 }

	# Enabled features, bool field slot 3.
	cmm_build_list_bool_list = { setting = cm__priority_list field_slot = 3 list_name = cm_priority_features_enabled }

	} # end guard
}

# Re-register the threshold field with the active metric's bounds and localization. The
# generated registration uses the widest bounds of any metric, since it re-runs on every load
# and clamps stored values.
# Called on registration and on metric change.
# Scope: country
cm_apply_expand_metric_bounds = {
	if = {
		limit = {
			OR = {
				NOT = { exists = var:cm_expand_metric }
				var:cm_expand_metric <= 2
			}
		}
		cmm_register_list_numeric_field = {
			mod_id = cm
			setting_id = priority_list
			field_id = min_profit
			default_value = 0
			min_value = 0
			max_value = 15
			step_value = 0.1
		}
	}
	else = {
		cmm_register_list_numeric_field = {
			mod_id = cm
			setting_id = priority_list
			field_id = min_profit
			default_value = 0
			min_value = 0
			max_value = 100
			step_value = 1
		}
	}
	cmm_set_list_field_format = { mod_id = cm setting_id = priority_list field_id = min_profit }
	# The metric-change path doesn't reset the loc keys.
	if = {
		limit = {
			exists = var:cm_expand_metric
			var:cm_expand_metric = 2
		}
		cmm_set_list_field_localization = {
			mod_id = cm
			setting_id = priority_list
			field_id = min_profit
			name = cm__priority_list__min_profit_profit_name
			root = cm__priority_list__min_profit
		}
	}
	else_if = {
		limit = {
			exists = var:cm_expand_metric
			var:cm_expand_metric >= 3
		}
		cmm_set_list_field_localization = {
			mod_id = cm
			setting_id = priority_list
			field_id = min_profit
			name = cm__priority_list__min_profit_roi_name
			root = cm__priority_list__min_profit_roi
		}
	}
	else = {
		cmm_set_list_field_localization = {
			mod_id = cm
			setting_id = priority_list
			field_id = min_profit
			name = cm__priority_list__min_profit_name
			root = cm__priority_list__min_profit
		}
	}
}

# Copy the live threshold column into metric $m$'s memory variables.
# Reads go through local variables; an inline variable_map lookup as an effect
# value resolves to empty. Unreadable entries are stored as 0.
# Scope: country
cm_save_expand_thresholds = {
	set_variable = { name = cm_expand_threshold_buildings_m$m$ value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = cmm target = flag:cm__priority_list_i1_f1 } }
		set_local_variable = { name = cm_tmp_field_key value = flag:cm__priority_list_i1_f1 }
		set_local_variable = { name = cm_tmp_read value = "variable_map(cmm|local_var:cm_tmp_field_key)" }
		if = {
			limit = {
				trigger_if = {
					limit = { exists = local_var:cm_tmp_read }
					local_var:cm_tmp_read >= 0
				}
				trigger_else = {
					always = no
				}
			}
			set_variable = { name = cm_expand_threshold_buildings_m$m$ value = local_var:cm_tmp_read }
		}
	}
	set_variable = { name = cm_expand_threshold_rgos_m$m$ value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = cmm target = flag:cm__priority_list_i2_f1 } }
		set_local_variable = { name = cm_tmp_field_key value = flag:cm__priority_list_i2_f1 }
		set_local_variable = { name = cm_tmp_read2 value = "variable_map(cmm|local_var:cm_tmp_field_key)" }
		if = {
			limit = {
				trigger_if = {
					limit = { exists = local_var:cm_tmp_read2 }
					local_var:cm_tmp_read2 >= 0
				}
				trigger_else = {
					always = no
				}
			}
			set_variable = { name = cm_expand_threshold_rgos_m$m$ value = local_var:cm_tmp_read2 }
		}
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = cm_tmp_field_key }
	}
}

# Copy metric $m$'s memory variables into the live threshold column. Missing
# entries load as 0.
# Scope: country
cm_load_expand_thresholds = {
	remove_from_variable_map = { name = cmm key = flag:cm__priority_list_i1_f1 }
	if = {
		limit = { exists = var:cm_expand_threshold_buildings_m$m$ }
		add_to_variable_map = { name = cmm key = flag:cm__priority_list_i1_f1 value = var:cm_expand_threshold_buildings_m$m$ }
	}
	else = {
		add_to_variable_map = { name = cmm key = flag:cm__priority_list_i1_f1 value = 0 }
	}
	remove_from_variable_map = { name = cmm key = flag:cm__priority_list_i2_f1 }
	if = {
		limit = { exists = var:cm_expand_threshold_rgos_m$m$ }
		add_to_variable_map = { name = cmm key = flag:cm__priority_list_i2_f1 value = var:cm_expand_threshold_rgos_m$m$ }
	}
	else = {
		add_to_variable_map = { name = cmm key = flag:cm__priority_list_i2_f1 value = 0 }
	}
}

# Copy metric $m$'s threshold memory from scope:old_country.
# Scope: country
cm_copy_expand_thresholds = {
	if = {
		limit = { scope:old_country = { exists = var:cm_expand_threshold_buildings_m$m$ } }
		set_variable = { name = cm_expand_threshold_buildings_m$m$ value = scope:old_country.var:cm_expand_threshold_buildings_m$m$ }
	}
	if = {
		limit = { scope:old_country = { exists = var:cm_expand_threshold_rgos_m$m$ } }
		set_variable = { name = cm_expand_threshold_rgos_m$m$ value = scope:old_country.var:cm_expand_threshold_rgos_m$m$ }
	}
}

# Restore 0 in the live threshold column where an entry is unreadable.
# Scope: country
cm_repair_expand_thresholds = {
	set_local_variable = { name = cm_tmp_field_key value = flag:cm__priority_list_i1_f1 }
	set_local_variable = { name = cm_tmp_read value = "variable_map(cmm|local_var:cm_tmp_field_key)" }
	if = {
		limit = {
			trigger_if = {
				limit = { exists = local_var:cm_tmp_read }
				NOT = { local_var:cm_tmp_read >= 0 }
			}
			trigger_else = {
				always = yes
			}
		}
		remove_from_variable_map = { name = cmm key = flag:cm__priority_list_i1_f1 }
		add_to_variable_map = { name = cmm key = flag:cm__priority_list_i1_f1 value = 0 }
	}
	set_local_variable = { name = cm_tmp_field_key value = flag:cm__priority_list_i2_f1 }
	set_local_variable = { name = cm_tmp_read2 value = "variable_map(cmm|local_var:cm_tmp_field_key)" }
	if = {
		limit = {
			trigger_if = {
				limit = { exists = local_var:cm_tmp_read2 }
				NOT = { local_var:cm_tmp_read2 >= 0 }
			}
			trigger_else = {
				always = yes
			}
		}
		remove_from_variable_map = { name = cmm key = flag:cm__priority_list_i2_f1 }
		add_to_variable_map = { name = cmm key = flag:cm__priority_list_i2_f1 value = 0 }
	}
}

# Per-metric threshold memory: each metric keeps its own per-feature column values.
# var:cm_expand_metric still holds the outgoing metric when this runs; the cmm map
# already holds the incoming one.
# Scope: country
cm_apply_expand_metric_change = {
	set_local_variable = { name = cm_tmp_incoming_metric value = "variable_map(cmm|flag:cm__expand_metric)" }
	set_local_variable = { name = cm_tmp_metric_changed value = 0 }
	if = {
		limit = { NOT = { local_var:cm_tmp_incoming_metric = var:cm_expand_metric } }
		set_local_variable = { name = cm_tmp_metric_changed value = 1 }
		switch = {
			trigger = var:cm_expand_metric
			1 = { cm_save_expand_thresholds = { m = 1 } }
			2 = { cm_save_expand_thresholds = { m = 2 } }
			3 = { cm_save_expand_thresholds = { m = 3 } }
			4 = { cm_save_expand_thresholds = { m = 4 } }
		}
	}
	cmm_sync_setting_alias = {
		setting = cm__expand_metric
		alias = cm_expand_metric
	}
	cm_apply_expand_metric_bounds = yes
	if = {
		limit = { local_var:cm_tmp_metric_changed = 1 }
		switch = {
			trigger = var:cm_expand_metric
			1 = { cm_load_expand_thresholds = { m = 1 } }
			2 = { cm_load_expand_thresholds = { m = 2 } }
			3 = { cm_load_expand_thresholds = { m = 3 } }
			4 = { cm_load_expand_thresholds = { m = 4 } }
		}
		cm_rebuild_priority_lookups = yes
	}
}

	} # end guard
}

# Projects one category's row into the per-category variables the budget engine and the
# settings-row visibility read. Turning a category off zeroes its pool and writes off its debt and
# its hold; the monthly refresh skips it while off.
# $cat$ = industry | collapsar | food | utilities | roads. Scope: country
cm_sa_sync_category = {
	# A macro never resolves inside a quoted lookup, so the key goes through a local.
	set_local_variable = { name = cm_sa_cat_key value = flag:cm_sa_cat_$cat$ }
	if = {
		limit = {
			has_variable_map = cm_sa_cat_budget
			is_key_in_variable_map = { name = cm_sa_cat_budget target = local_var:cm_sa_cat_key }
		}
		set_variable = { name = cm_sa_$cat$_budget value = "variable_map(cm_sa_cat_budget|local_var:cm_sa_cat_key)" }
	}
	if = {
		limit = { is_target_in_variable_list = { name = cm_sa_categories_enabled target = flag:cm_sa_cat_$cat$ } }
		set_variable = { name = cm_sa_$cat$_enabled value = 1 }
		cm_sa_clamp_pool = { cat = $cat$ }
	}
	else = {
		remove_variable = cm_sa_$cat$_enabled
		set_variable = { name = cm_sa_pool_$cat$ value = 0 }
		# A category that gets no further allotment has nothing left to settle a debt with, and
		# nothing left to spend what it was holding on.
		remove_variable = cm_sa_debt_$cat$
		remove_variable = cm_sa_resv_$cat$
	}
}

cm_toggle_priority_feature = {
	clear_variable_list = cm_priority_desired_enabled
	every_in_list = {
		variable = cm_priority_features_enabled
		save_scope_as = cm_desired_feature
		root = { add_to_variable_list = { name = cm_priority_desired_enabled target = scope:cm_desired_feature } }
	}
	if = {
		limit = { is_target_in_variable_list = { name = cm_priority_desired_enabled target = flag:$feature$ } }
		remove_list_variable = { name = cm_priority_desired_enabled target = flag:$feature$ }
	}
	else = {
		add_to_variable_list = { name = cm_priority_desired_enabled target = flag:$feature$ }
	}
	cmm_for_each_list_item = {
		setting = cm__priority_list
		effect = cm_apply_priority_desired_item
	}
	cm_rebuild_priority_lookups = yes
	clear_variable_list = cm_priority_desired_enabled
}

# Rebuild all lookup tables from CMM list field data.
# Called on registration and on every setting change.
# Scope: country
cm_rebuild_auto_build_lookups = {
	# Guard: skip if CMM list data doesn't exist yet; first registration may not have completed.
	if = {
		limit = { NOT = { has_variable_list = cmm_list_items_cm__auto_build_list } }
		# List not initialized yet - lookups will be built on next registration cycle
	}
	else = {

	# Ordered building types list for monthly pulse iteration in priority order.
	cmm_build_list_ordered_values = { setting = cm__auto_build_list list_name = cm_auto_build_building_types_list }

	# Enabled building types, bool field slot 1.
	cmm_build_list_bool_list = { setting = cm__auto_build_list field_slot = 1 list_name = cm_auto_build_building_types_enabled }

	# Subjects-enabled building types, bool field slot 2.
	cmm_build_list_bool_list = { setting = cm__auto_build_list field_slot = 2 list_name = cm_auto_build_building_types_in_subjects_enabled }

	# Min control by building type, numeric field slot 3, CMM 0-100 -> 0-1.
	cmm_build_list_field_map = { setting = cm__auto_build_list field_slot = 3 map_name = cm_min_control_by_building_type multiply = 0.01 }

	# Min discount by building type, numeric field slot 4, CMM -33..33 -> -0.33..0.33.

# Hide/show auto-build list rows based on whether the building's advance is researched.
# Scope: country
cm_update_auto_build_list_visibility = {
	if = {
		limit = { has_variable_list = cmm_list_items_cm__auto_build_list }
		cmm_for_each_list_item = {
			setting = cm__auto_build_list
			effect = cm_auto_build_list_update_visibility_item
		}
	}
}

# Callback for cmm_for_each_list_item. Shows or hides item $i$ based on advance status.
# $i$ = item position in the list. $setting$ = composite key. scope:cmm_list_current_item_value = building_type.
# Root: country
cm_auto_build_list_update_visibility_item = {
	if = {
		limit = {
			OR = {
				can_build_building = scope:cmm_list_current_item_value
				# An obsolete key means a newer tier of the row's chain is unlocked.
				building_type_is_obsolete = scope:cmm_list_current_item_value
			}
		}
		cmm_show_list_item = { mod_id = cm setting_id = auto_build_list item = $i$ }
	}
	else = {
		cmm_hide_list_item = { mod_id = cm setting_id = auto_build_list item = $i$ }
	}

	# Accept $setting$ param from cmm_for_each_list_item, unused here.
	if = {
		limit = { always = no }
		cmf_suppress = { v = $setting$_$i$ }
	}
}
