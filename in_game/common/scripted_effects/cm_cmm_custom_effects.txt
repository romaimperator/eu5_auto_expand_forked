# Custom CMM integration code not managed by the visual editor.
# Called after cm_register_cmf_mod (visual-editor-managed) completes.

# Post-registration: override default values and build lookup tables.
# Scope: country
cm_cmm_post_registration = {
	# Override min_control defaults: early_council_hall(1), minting_office(2), settlement_building(3) should start at 0
	# (they increase control / population growth, so best to build everywhere regardless of control)
	if = {
		limit = { NOT = { has_variable = cm_cmm_auto_build_list_initialized } }
		set_variable = { name = cm_cmm_auto_build_list_initialized value = 1 }
		add_to_variable_map = { name = cmm key = flag:cm__auto_build_list_i1_f3 value = 0 }
		add_to_variable_map = { name = cmm key = flag:cm__auto_build_list_i2_f3 value = 0 }
		add_to_variable_map = { name = cmm key = flag:cm__auto_build_list_i3_f3 value = 0 }
	}

	# Override priority list defaults: enable buildings (item 1) and RGOs (item 2) by default
	# Auto Build Capital Buildings (item 7) starts at +33% min_discount so capital buildings
	# are constructed immediately regardless of construction material prices.
	if = {
		limit = { NOT = { has_variable = cm_cmm_priority_list_initialized } }
		set_variable = { name = cm_cmm_priority_list_initialized value = 1 }
		add_to_variable_map = { name = cmm key = flag:cm__priority_list_i1_f3 value = 1 }
		add_to_variable_map = { name = cmm key = flag:cm__priority_list_i2_f3 value = 1 }
		add_to_variable_map = { name = cmm key = flag:cm__priority_list_i7_f2 value = 33 }
	}

	# Override urbanize ignore list defaults: Gold (item 1) and Silver (item 2) ignored by default.
	if = {
		limit = { NOT = { has_variable = cm_cmm_auto_urbanize_ignore_list_initialized } }
		set_variable = { name = cm_cmm_auto_urbanize_ignore_list_initialized value = 1 }
		add_to_variable_map = { name = cmm key = flag:cm__auto_urbanize_ignore_list_i1_f1 value = 1 }
		add_to_variable_map = { name = cmm key = flag:cm__auto_urbanize_ignore_list_i2_f1 value = 1 }
	}

	# Build lookup tables from CMM list fields
	cm_rebuild_priority_lookups = yes
	cm_rebuild_auto_build_lookups = yes
	cm_rebuild_auto_food_lookups = yes
	cm_rebuild_auto_urbanize_lookups = yes
	cm_rebuild_auto_town_rights_lookups = yes

	# Hide buildings whose advance hasn't been researched yet
	cm_update_auto_build_list_visibility = yes
	# Hide town rights whose potential doesn't pass for this country
	cm_update_auto_town_rights_list_visibility = yes
}

# Rebuild priority feature lookup tables from CMM list field data.
# Called on registration and on every setting change.
# Scope: country
cm_rebuild_priority_lookups = {
	if = {
		limit = { NOT = { has_variable_list = cmm_list_items_cm__priority_list } }
	}
	else = {

	# Ordered features list (for monthly pulse iteration in priority order)
	cmm_build_list_ordered_values = { setting = cm__priority_list list_name = cm_priority_features_list }

	# Min profit by feature (numeric field slot 1, direct gold values)
	cmm_build_list_field_map = { setting = cm__priority_list field_slot = 1 map_name = cm_priority_min_profit multiply = 1 }

	# Min discount by feature (numeric field slot 2, CMM -33..33 → -0.33..0.33)
	cmm_build_list_field_map = { setting = cm__priority_list field_slot = 2 map_name = cm_priority_min_discount multiply = 0.01 }

	# Enabled features (bool field slot 3)
	cmm_build_list_bool_list = { setting = cm__priority_list field_slot = 3 list_name = cm_priority_features_enabled }

	} # end guard
}

# Rebuild all lookup tables from CMM list field data.
# Called on registration and on every setting change.
# Scope: country
cm_rebuild_auto_build_lookups = {
	# Guard: skip if CMM list data doesn't exist yet (first registration may not have completed)
	if = {
		limit = { NOT = { has_variable_list = cmm_list_items_cm__auto_build_list } }
		# List not initialized yet — lookups will be built on next registration cycle
	}
	else = {

	# Ordered building types list (for monthly pulse iteration in priority order)
	cmm_build_list_ordered_values = { setting = cm__auto_build_list list_name = cm_auto_build_building_types_list }

	# Enabled building types (bool field slot 1)
	cmm_build_list_bool_list = { setting = cm__auto_build_list field_slot = 1 list_name = cm_auto_build_building_types_enabled }

	# Subjects-enabled building types (bool field slot 2)
	cmm_build_list_bool_list = { setting = cm__auto_build_list field_slot = 2 list_name = cm_auto_build_building_types_in_subjects_enabled }

	# Min control by building type (numeric field slot 3, CMM 0-100 → 0-1)
	cmm_build_list_field_map = { setting = cm__auto_build_list field_slot = 3 map_name = cm_min_control_by_building_type multiply = 0.01 }

	# Min discount by building type (numeric field slot 4, CMM -33..33 → -0.33..0.33)
	cmm_build_list_field_map = { setting = cm__auto_build_list field_slot = 4 map_name = cm_min_discount_by_building_type multiply = 0.01 }

	# Min RGOs by building type (numeric field slot 5, direct values)
	cmm_build_list_field_map = { setting = cm__auto_build_list field_slot = 5 map_name = cm_min_rgos_by_building_type multiply = 1 }

	} # end guard
}

# Hide/show auto-build list rows based on whether the building's advance is researched.
# Scope: country
cm_update_auto_build_list_visibility = {
	if = {
		limit = { has_variable_list = cmm_list_items_cm__auto_build_list }
		cmm_for_each_list_item = {
			setting = cm__auto_build_list
			effect = cm_auto_build_list_update_visibility_item
		}
	}
}

# Callback for cmm_for_each_list_item. Shows or hides item $i$ based on advance status.
# $i$ = item position in the list. $setting$ = composite key. scope:cmm_list_current_item_value = building_type.
# Root: country
cm_auto_build_list_update_visibility_item = {
	if = {
		limit = { can_build_building = scope:cmm_list_current_item_value }
		cmm_show_list_item = { mod_id = cm setting_id = auto_build_list item = $i$ }
	}
	else = {
		cmm_hide_list_item = { mod_id = cm setting_id = auto_build_list item = $i$ }
	}

	# Accept $setting$ param from cmm_for_each_list_item (unused here)
	if = {
		limit = { always = no }
		cmf_suppress = { v = $setting$_$i$ }
	}
}

# Hide/show auto town rights list rows based on whether the country can currently
# grant the right (basic advance researched, plus the right's potential block passes).
# Scope: country

		limit = { has_variable_list = cmm_list_items_cm__auto_town_rights_list }
		cmm_for_each_list_item = {
			setting = cm__auto_town_rights_list
			effect = cm_auto_town_rights_list_update_visibility_item
		}
	}
}

# Callback for cmm_for_each_list_item. Shows or hides item $i$ based on the right's
# country-level prerequisites.
# Root: country
cm_auto_town_rights_list_update_visibility_item = {
	if = {
		limit = {
			modifier:can_grant_town_rights = yes
			cm_auto_town_rights_potential_passes = yes
		}
		cmm_show_list_item = { mod_id = cm setting_id = auto_town_rights_list item = $i$ }
	}
	else = {
		cmm_hide_list_item = { mod_id = cm setting_id = auto_town_rights_list item = $i$ }
	}

	# Accept $setting$ param from cmm_for_each_list_item (unused here)
	if = {
		limit = { always = no }
		cmf_suppress = { v = $setting$_$i$ }
	}
}

# CMM-specific warning suppression for lookup tables built from CMM list fields.
_cm_cmm_suppress_warnings = {
	if = {
		limit = { always = no }
		# CMM-derived lookup maps (set at runtime by cmm_build_list_* utilities)
		cmf_suppress = { v = cm_min_control_by_building_type }
		cmf_suppress = { v = cm_min_discount_by_building_type }
		cmf_suppress = { v = cm_min_rgos_by_building_type }
		# Priority list lookup tables
		cmf_suppress = { v = cm_priority_features_list }
		cmf_suppress = { v = cm_priority_features_enabled }
		cmf_suppress = { v = cm_priority_min_profit }
		cmf_suppress = { v = cm_priority_min_discount }
		cmf_suppress = { v = cm_cmm_priority_list_initialized }
		cmf_suppress = { v = cmm_list_item_hidden }
		# Food auto-build (lookup lists)
		cmf_suppress = { v = cm_auto_food_rgo_types_list }
		cmf_suppress = { v = cm_auto_food_rgo_types_enabled }
		cmf_suppress = { v = cm_auto_food_rgo_types_river_only }
		cmf_suppress = { v = cm_auto_food_rgo_types_in_subjects_enabled }
		# Auto-urbanize ignore list
		cmf_suppress = { v = cm_auto_urbanize_ignored_rgo_goods }
		cmf_suppress = { v = cm_cmm_auto_urbanize_ignore_list_initialized }
		# Auto town rights (lookup lists)
		cmf_suppress = { v = cm_auto_town_rights_list }
		cmf_suppress = { v = cm_auto_town_rights_enabled }
	}
}
