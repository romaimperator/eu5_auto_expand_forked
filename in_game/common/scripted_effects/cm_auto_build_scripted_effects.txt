# Root is the country. Expects scope:drop_target_bt and scope:dragged_bt (saved in chain). Reorders cm_auto_build_building_types_list so dragged is before drop target.
cm_auto_build_reorder_list_effect = {
  set_local_variable = { name = i value = 0 }
  # debug_log = "Reordering list"
  every_in_list = {
    variable = cm_auto_build_building_types_list
    save_scope_as = current_bt
    # debug_log = "Adding building type [THIS.GetBuildingType.GetNameWithNoTooltip] to temp list"
    scope:country = { add_to_variable_list = { name = cm_auto_build_order_temp target = scope:current_bt } }
  }
  clear_variable_list = cm_auto_build_building_types_list
  set_local_variable = { name = done value = 0 }
  every_in_list = {
    variable = cm_auto_build_order_temp
    save_scope_as = current_bt
    if = { limit = { NOT = { this = scope:building_type } }
      if = { limit = { local_var:i = local_var:cm_auto_build_move_target_index }
        # debug_log = "Adding building type [THIS.GetBuildingType.GetNameWithNoTooltip] with moving building type [SCOPE.sBuildingType('building_type').GetNameWithNoTooltip]"
        scope:country = { add_to_variable_list = { name = cm_auto_build_building_types_list target = scope:building_type } }
        scope:country = { add_to_variable_list = { name = cm_auto_build_building_types_list target = scope:current_bt } }
        set_local_variable = { name = done value = 1 }
      } else = {
        # debug_log = "Adding building type [THIS.GetBuildingType.GetNameWithNoTooltip]"
        scope:country = { add_to_variable_list = { name = cm_auto_build_building_types_list target = scope:current_bt } }
      }
      change_local_variable = { name = i add = 1 }
    }
  }
  if = { limit = { local_var:done = 0 }
    # debug_log = "Adding building type [SCOPE.sBuildingType('building_type').GetNameWithNoTooltip] to end of list"
    scope:country = { add_to_variable_list = { name = cm_auto_build_building_types_list target = scope:building_type } }
  }
  clear_variable_list = cm_auto_build_order_temp
}

# Root is the country. Expects scope:building_type (the row to move). Moves it one position earlier in the list.
cm_auto_build_move_building_type_up_effect = {
  # debug_log = "Moving building type up"
  set_local_variable = { name = cm_auto_build_move_index value = 0 }
  set_local_variable = { name = cm_auto_build_move_done value = 0 }
  every_in_list = {
    variable = cm_auto_build_building_types_list
    save_scope_as = current_bt
    # debug_log = "Checking building type [THIS.GetBuildingType.GetNameWithNoTooltip] for matching [SCOPE.sBuildingType('building_type').GetNameWithNoTooltip]"
    if = { limit = { this = scope:building_type }
      scope:country = {
        if = { limit = { AND = { local_var:cm_auto_build_move_index >= 1 local_var:cm_auto_build_move_done = 0 } }
          set_local_variable = { name = cm_auto_build_move_target_index value = { value = local_var:cm_auto_build_move_index subtract = 1 } }
          cm_auto_build_reorder_list_effect = {}
          set_local_variable = { name = cm_auto_build_move_done value = 1 }
        }
      }
    }
    change_local_variable = { name = cm_auto_build_move_index add = 1 }
    save_scope_as = previous_bt
  }
}

# Root is the country. Expects scope:building_type (the row to move). Moves it one position later in the list.
cm_auto_build_move_building_type_down_effect = {
  # debug_log = "Moving building type down"
  set_local_variable = { name = cm_auto_build_move_index value = 0 }
  set_local_variable = { name = cm_auto_build_move_done value = 0 }
  every_in_list = {
    variable = cm_auto_build_building_types_list
    save_scope_as = current_bt
    # debug_log = "Checking building type [THIS.GetBuildingType.GetNameWithNoTooltip] for matching [SCOPE.sBuildingType('building_type').GetNameWithNoTooltip]"
    scope:country = {
      if = { limit = { AND = { scope:current_bt = scope:building_type local_var:cm_auto_build_move_done = 0 } }
        set_local_variable = { name = cm_auto_build_move_target_index value = { value = local_var:cm_auto_build_move_index add = 1 } }
        cm_auto_build_reorder_list_effect = {}
        set_local_variable = { name = cm_auto_build_move_done value = 1 }
      }
      change_local_variable = { name = cm_auto_build_move_index add = 1 }
    }
    save_scope_as = previous_bt
  }
}

# Root is the country. Expects scope:building_type.
# If scope:building_type is obsolete for this country, walks the upgrade chain
# and re-saves scope:building_type to the current non-obsolete version.
# Unrolled to max depth 4 (dock → dry_dock → shipyard → grand_shipyard → naval_base).
# Also collects the obsolete tiers it walks into cm_auto_build_chain_predecessors.
cm_auto_build_resolve_building_type = {
	clear_variable_list = cm_auto_build_chain_predecessors
	if = { limit = { building_type_is_obsolete = scope:building_type }
		add_to_variable_list = { name = cm_auto_build_chain_predecessors target = scope:building_type }
		scope:building_type = { cm_find_upgrade_building_type = yes }
		scope:return_scope = { save_scope_as = building_type }
		if = { limit = { building_type_is_obsolete = scope:building_type }
			add_to_variable_list = { name = cm_auto_build_chain_predecessors target = scope:building_type }
			scope:building_type = { cm_find_upgrade_building_type = yes }
			scope:return_scope = { save_scope_as = building_type }
			if = { limit = { building_type_is_obsolete = scope:building_type }
				add_to_variable_list = { name = cm_auto_build_chain_predecessors target = scope:building_type }
				scope:building_type = { cm_find_upgrade_building_type = yes }
				scope:return_scope = { save_scope_as = building_type }
				if = { limit = { building_type_is_obsolete = scope:building_type }
					add_to_variable_list = { name = cm_auto_build_chain_predecessors target = scope:building_type }
					scope:building_type = { cm_find_upgrade_building_type = yes }
					scope:return_scope = { save_scope_as = building_type }
				}
			}
		}
	}
}
