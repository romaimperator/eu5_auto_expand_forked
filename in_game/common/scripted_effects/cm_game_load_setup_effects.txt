# Populates the global list of building types for classification by the hidden window GUI, and arms
# that classification only when the save carries no completed one at the current version. Everything
# the passes record is a global map that saves and reloads with the game, so a load that reuses them
# skips five GUI passes over every building type and their nested demand and production-method
# datamodels. Replacement tiers are the one part that cannot be reused: GetNextReplacementBuilding
# returns only what the loading player has unlocked, so a reused classification arms the upgrade-map
# refresh in its place. The list itself is rebuilt every time: it is the datamodel both drivers bind
# to, and the driver that fires the refresh has to have every type under it.
# Called from on_game_start_after_lobby and on_game_load_after_lobby, once per lobby, not per player.
cm_populate_building_types_to_process = {
  clear_global_variable_list = cm_building_types_to_process
  set_global_variable = { name = cm_building_types_to_process_count value = 0 }
  every_building_type = {
    add_to_global_variable_list = { name = cm_building_types_to_process target = this }
    change_global_variable = { name = cm_building_types_to_process_count add = 1 }
  }
  if = {
    limit = { NOT = { cm_building_classification_is_current = yes } }
    # A partial classification's stamp must not survive the rebuild that replaces it.
    remove_global_variable = cm_bt_class_stamp
    clear_global_variable_map = cm_building_type_processed
    clear_global_variable_list = cm_base_goods
    # Written by the rgob collect wave, so it is rebuilt with the rest of the classification.
    clear_global_variable_map = cm_cu_core_in
    # Written by the rgob scan wave, and outlives the scan state the sweep clears.
    clear_global_variable_map = cm_bt_output
    cm_rgob_reset_classification = yes
    # Trigger building classification in the hidden window GUI
    set_global_variable = { name = cm_process_buildings value = yes }
  }
  else = {
    set_global_variable = { name = cm_refresh_upgrade_types value = yes }
  }
}

# Numeric ids for every location, in both directions. Location.GetKey returns the runtime id
# rather than the script key, so location:<GetKey> is rejected and a command has no other way to
# name one. Called from cm_on_init_lobby_globals, once per lobby.
cm_ln_populate_location_ids = {
  clear_global_variable_map = cm_ln_loc_id
  clear_global_variable_map = cm_ln_loc_of_id
  set_global_variable = { name = cm_ln_loc_n value = 0 }
  every_province_definition = {
    every_location_in_province_definition = {
      save_temporary_scope_as = cm_ln_loc
      change_global_variable = { name = cm_ln_loc_n add = 1 }
      add_to_global_variable_map = { name = cm_ln_loc_id key = scope:cm_ln_loc value = global_var:cm_ln_loc_n }
      add_to_global_variable_map = { name = cm_ln_loc_of_id key = global_var:cm_ln_loc_n value = scope:cm_ln_loc }
    }
  }
}

# One id per lucky nation for the observer console bridge, in both directions. Country.GetTag
# returns a display tag, which for the Great Yuan is YUA while script marks CHI, so a command built
# from it scopes to an unrelated country. Ids are assigned once and never reissued, so a cycle's
# rows stay resolvable across the monthly refresh that assigns them. Root is the country.
cm_ln_assign_country_id = {
  save_temporary_scope_as = cm_ln_ctry
  if = {
    limit = { NOT = { is_key_in_global_variable_map = { name = cm_ln_ctry_id target = scope:cm_ln_ctry } } }
    change_global_variable = { name = cm_ln_ctry_n add = 1 }
    add_to_global_variable_map = { name = cm_ln_ctry_id key = scope:cm_ln_ctry value = global_var:cm_ln_ctry_n }
    add_to_global_variable_map = { name = cm_ln_ctry_of_id key = global_var:cm_ln_ctry_n value = scope:cm_ln_ctry }
  }
}

# Called from cm_ln_on_lobby, ahead of the refresh that assigns the ids, once per lobby.
cm_ln_reset_country_ids = {
  clear_global_variable_map = cm_ln_ctry_id
  clear_global_variable_map = cm_ln_ctry_of_id
  set_global_variable = { name = cm_ln_ctry_n value = 0 }
}

# Logs once per lobby, once building-type classification, the R.G.O. Bonus coverage sweep and the
# river-adjacency harvest have all finished. Called from each one's completion branch, so the last
# to land is the one that logs. Root is the country.
cm_log_setup_complete_if_ready = {
  if = {
    limit = {
      global_var:cm_setup_complete_logged = 0
      NOT = { has_global_variable = cm_process_buildings }
      NOT = { has_global_variable = cm_rgob_sweep }
      cm_pf_river_data_is_current = yes
    }
    set_global_variable = { name = cm_setup_complete_logged value = 1 }
    cmf_log = { action = cm_log_setup_complete }
  }
}

# Populates the global list of river locations for the placement finder's river-adjacency precompute
# in the hidden window GUI. Called from cm_on_init_lobby_globals, once per lobby, not per player.
# River adjacency is map geometry and cm_pf_river_neighbors is saved, so the harvest runs only when
# the save carries no completed one at the current version: a fresh save, a bumped
# cm_pf_river_version_value, or a harvest a save-and-quit interrupted.
cm_pf_populate_river_locations = {
  if = {
    limit = { NOT = { cm_pf_river_data_is_current = yes } }
    # A partial harvest's stamp must not survive the rebuild that replaces it.
    remove_global_variable = cm_pf_river_stamp
    clear_global_variable_list = cm_pf_river_locations
    clear_global_variable_list = cm_pf_river_batch
    set_global_variable = { name = cm_pf_river_count value = 0 }
    set_global_variable = { name = cm_pf_river_idx value = 0 }
    every_province_definition = {
      every_location_in_province_definition = {
        # Cleared everywhere, not just on current river locations: a patch that takes a river away
        # would otherwise leave a stale list the path search still reads.
        clear_variable_list = cm_pf_river_neighbors
        if = {
          limit = { has_river = yes }
          add_to_global_variable_list = { name = cm_pf_river_locations target = this }
          change_global_variable = { name = cm_pf_river_count add = 1 }
        }
      }
    }
    # Trigger the river-adjacency precompute in the hidden window GUI.
    set_global_variable = { name = cm_pf_river_process value = yes }
  }
}

# Enumerates every province slice into the R.G.O. Bonus sweep work list so the batched
# recompute can cursor it instead of rescanning the map each tick. Called from
# cm_on_init_lobby_globals, which puts the enumeration in the load rather than 2.5s after it.
# Keyed by a counter into a map, not a list: list insertion cost grows with length, and a
# list cursor would force an O(n) rescan every tick. Stateless: always rebuilds from scratch.
cm_rgob_populate_work_list = {
  cm_rgob_reset_sweep = yes
  every_province_definition = {
    every_province_in_province_definition = {
      change_global_variable = { name = cm_rgob_wl_count add = 1 }
      add_to_global_variable_map = { name = cm_rgob_wl key = global_var:cm_rgob_wl_count value = this }
    }
  }
}
