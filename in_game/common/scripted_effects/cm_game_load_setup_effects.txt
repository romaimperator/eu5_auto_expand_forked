# Populates the global list of building types for classification by the hidden window GUI.
# Called from on_game_start_after_lobby and on_game_load_after_lobby, once per lobby, not per player.
# Stateless: always rebuilds from scratch so no version tracking or migration is needed.
cm_populate_building_types_to_process = {
  clear_global_variable_list = cm_building_types_to_process
  set_global_variable = { name = cm_building_types_to_process_count value = 0 }
  clear_global_variable_map = cm_building_type_processed
  every_building_type = {
    add_to_global_variable_list = { name = cm_building_types_to_process target = this }
    change_global_variable = { name = cm_building_types_to_process_count add = 1 }
  }
  # Trigger building classification in the hidden window GUI
  set_global_variable = { name = cm_process_buildings value = yes }
}

# Expects scopes: country
cm_reset_locations_lists = {
  scope:cm_country = {
    # Directly owned locations first, sorted by max control
    clear_variable_list = cm_all_locations_including_subjects_list
    ordered_owned_location = {
      order_by = max_control
      save_scope_as = cm_location
      max = cm_max_guess_at_number_of_locations
      check_range_bounds = no
      scope:cm_country = { add_to_variable_list = { name = cm_all_locations_including_subjects_list target = scope:cm_location } }
    }

    # Make unsorted list of locations of all subjects and below
    clear_variable_list = cm_all_locations_including_subjects_list_temp
    every_subject_or_below = {
      every_owned_location = {
        save_scope_as = cm_location
        scope:cm_country = { add_to_variable_list = { name = cm_all_locations_including_subjects_list_temp target = scope:cm_location } }
      }
    }

    # Sort and add the subject locations to the end of the list
    ordered_in_list = {
      variable = cm_all_locations_including_subjects_list_temp
      order_by = max_control
      max = cm_max_guess_at_number_of_locations
      check_range_bounds = no
      save_scope_as = cm_location
      scope:cm_country = { add_to_variable_list = { name = cm_all_locations_including_subjects_list target = scope:cm_location } }
    }
    clear_variable_list = cm_all_locations_including_subjects_list_temp
  }
}
