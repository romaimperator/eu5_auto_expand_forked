# Populates the global list of building types for classification by the hidden window GUI.
# Called from on_game_start_after_lobby and on_game_load_after_lobby, once per lobby, not per player.
# Stateless: always rebuilds from scratch so no version tracking or migration is needed.
cm_populate_building_types_to_process = {
  clear_global_variable_list = cm_building_types_to_process
  set_global_variable = { name = cm_building_types_to_process_count value = 0 }
  clear_global_variable_map = cm_building_type_processed
  clear_global_variable_list = cm_base_goods
  cm_rgob_reset_classification = yes
  every_building_type = {
    add_to_global_variable_list = { name = cm_building_types_to_process target = this }
    change_global_variable = { name = cm_building_types_to_process_count add = 1 }
  }
  # Trigger building classification in the hidden window GUI
  set_global_variable = { name = cm_process_buildings value = yes }
}

# Populates the global list of river locations for the placement finder's river-adjacency precompute
# in the hidden window GUI. Called from cm_on_init_lobby_globals, once per lobby, not per player.
# River adjacency is map geometry and cm_pf_river_neighbors is saved, so the harvest runs only when
# the save carries no completed one at the current version: a fresh save, a bumped
# cm_pf_river_version_value, or a harvest a save-and-quit interrupted.
cm_pf_populate_river_locations = {
  if = {
    limit = { NOT = { cm_pf_river_data_is_current = yes } }
    # A partial harvest's stamp must not survive the rebuild that replaces it.
    remove_global_variable = cm_pf_river_stamp
    clear_global_variable_list = cm_pf_river_locations
    clear_global_variable_list = cm_pf_river_batch
    set_global_variable = { name = cm_pf_river_count value = 0 }
    set_global_variable = { name = cm_pf_river_idx value = 0 }
    every_province_definition = {
      every_location_in_province_definition = {
        # Cleared everywhere, not just on current river locations: a patch that takes a river away
        # would otherwise leave a stale list the path search still reads.
        clear_variable_list = cm_pf_river_neighbors
        if = {
          limit = { has_river = yes }
          add_to_global_variable_list = { name = cm_pf_river_locations target = this }
          change_global_variable = { name = cm_pf_river_count add = 1 }
        }
      }
    }
    # Trigger the river-adjacency precompute in the hidden window GUI.
    set_global_variable = { name = cm_pf_river_process value = yes }
  }
}

# Enumerates every province slice into the R.G.O. Bonus sweep work list so the batched
# recompute can cursor it instead of rescanning the map each tick. Called from
# cm_on_init_lobby_globals, which puts the enumeration in the load rather than 2.5s after it.
# Keyed by a counter into a map, not a list: list insertion cost grows with length, and a
# list cursor would force an O(n) rescan every tick. Stateless: always rebuilds from scratch.
cm_rgob_populate_work_list = {
  cm_rgob_reset_sweep = yes
  every_province_definition = {
    every_province_in_province_definition = {
      change_global_variable = { name = cm_rgob_wl_count add = 1 }
      add_to_global_variable_map = { name = cm_rgob_wl key = global_var:cm_rgob_wl_count value = this }
    }
  }
}

# Expects scopes: country
cm_reset_locations_lists = {
  scope:cm_country = {
    # Directly owned locations first, sorted by max control
    clear_variable_list = cm_all_locations_including_subjects_list
    ordered_owned_location = {
      order_by = max_control
      save_scope_as = cm_location
      max = cm_max_guess_at_number_of_locations
      check_range_bounds = no
      scope:cm_country = { add_to_variable_list = { name = cm_all_locations_including_subjects_list target = scope:cm_location } }
    }

    # Make unsorted list of locations of all subjects and below
    clear_variable_list = cm_all_locations_including_subjects_list_temp
    every_subject_or_below = {
      every_owned_location = {
        save_scope_as = cm_location
        scope:cm_country = { add_to_variable_list = { name = cm_all_locations_including_subjects_list_temp target = scope:cm_location } }
      }
    }

    # Sort and add the subject locations to the end of the list
    ordered_in_list = {
      variable = cm_all_locations_including_subjects_list_temp
      order_by = max_control
      max = cm_max_guess_at_number_of_locations
      check_range_bounds = no
      save_scope_as = cm_location
      scope:cm_country = { add_to_variable_list = { name = cm_all_locations_including_subjects_list target = scope:cm_location } }
    }
    clear_variable_list = cm_all_locations_including_subjects_list_temp
  }
}
