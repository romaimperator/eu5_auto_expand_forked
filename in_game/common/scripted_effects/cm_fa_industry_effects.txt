# Full-automation Industry: stages expand, from-zero, obsolete-upgrade, and RGO candidates
# on ROI Tax Income, then the greedy country-wide drain approves the best of them under the
# industry pool. Staging applies no slot, under-construction, or affordability gates: a
# currently blocked good candidate must still score so it can anchor the bar and hold it
# against worse buildable ones.

# Root is the country, scope:cm_country set.
cm_fa_run_industry = {
	# From-zero candidate types this pulse: buildable, non-obsolete production types.
	clear_variable_list = cm_fa_ind_types
	every_in_global_list = {
		variable = cm_building_types_to_process
		limit = {
			cm_is_building_type_production_building = yes
			is_foreign = no
			cm_fa_cu_is_always_delete = no
			save_temporary_scope_as = cm_fa_try_type
			scope:cm_country = {
				can_build_building = scope:cm_fa_try_type
				NOT = { building_type_is_obsolete = scope:cm_fa_try_type }
			}
		}
		save_scope_as = cm_fa_list_type
		scope:cm_country = { add_to_variable_list = { name = cm_fa_ind_types target = scope:cm_fa_list_type } }
	}
	every_market_present_in_country = {
		limit = { cm_fa_market_in_slice = yes }
		# The market iterators carry no ownership filter of their own, and a shared market
		# holds other countries' locations.
		ordered_location_in_market = {
			order_by = max_control
			max = cm_max_guess_at_number_of_locations
			check_range_bounds = no
			limit = {
				owner ?= scope:cm_country
				NOT = { integration_level = conquered }
				cm_fa_location_has_proximity = yes
				cm_location_auto_food_workers_ok = yes
			}
			save_scope_as = cm_location
			cm_pop_track = yes
			# Standing buildings: expands and obsolete upgrades.
			ordered_buildings_in_location = {
				order_by = building_index
				max = 100
				check_range_bounds = no
				save_scope_as = cm_building
				building_type = { save_scope_as = cm_building_type }
				if = {
					limit = { cm_is_building_type_obsolete = yes }
					if = {
						limit = {
							scope:cm_building_type = { cm_is_building_type_production_building = yes }
							scope:cm_location = { cm_is_building_type_upgradeable_and_can_be_upgraded_by_owner = yes }
						}
						scope:cm_building_type = { cm_find_current_building_type = yes }
						cm_fa_stage_upgrade_candidate = yes
					}
				} else_if = {
					limit = {
						scope:cm_building_type = {
							cm_is_building_type_production_building = yes
							cm_fa_cu_is_always_delete = no
						}
						scope:cm_building = {
							is_at_max_level = no
							is_opened = yes
						}
						scope:cm_location = {
							cm_location_has_enough_pops_available = yes
							cm_is_location_under_building_limit = yes
							NOT = { is_target_in_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_building_type } }
						}
					}
					cm_fa_stage_build_candidate = { cat = 1 }
				}
			}
			# From-zero production buildings.
			scope:cm_country = {
				every_in_list = {
					variable = cm_fa_ind_types
					save_scope_as = cm_building_type
					if = {
						limit = {
							scope:cm_location = {
								NOT = { has_building = scope:cm_building_type }
								cm_location_allowed_to_build = { target = scope:cm_building_type }
								cm_location_has_enough_pops_available = yes
								cm_is_location_under_building_limit = yes
								NOT = { is_target_in_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_building_type } }
								NOT = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:cm_building_type } }
							}
						}
						cm_fa_stage_build_candidate = { cat = 1 }
					}
				}
			}
			# RGO expansion.
			if = {
				limit = {
					cm_rgo_basic_expand_eligibility = yes
					NOT = { cm_location_auto_food_supersedes_rgo = yes }
				}
				cm_fa_stage_rgo_candidate = yes
			}
		}
	}
}

# Stages a build candidate into the full-automation build queue and marks its category,
# first stager wins. $cat$ = 1 industry, 2 collapsar, 3 food, 4 utilities, 5 governors, 6 forts,
# 7 manpower.
# Expects scope:cm_country, scope:cm_location, scope:cm_building_type.
cm_fa_stage_build_candidate = {
	cm_stage_location_and_type_to_queue = { COUNTRY_LIST = cm_fa_b_locations LOCATION_LIST = cm_fa_b_types DONE_LIST = cm_fa_done_b_types }
	scope:cm_location = {
		if = {
			limit = { NOT = { is_key_in_variable_map = { name = cm_fa_b_cats target = scope:cm_building_type } } }
			add_to_variable_map = { name = cm_fa_b_cats key = scope:cm_building_type value = $cat$ }
		}
	}
}

# Expects scope:cm_country, scope:cm_location.
cm_fa_stage_rgo_candidate = {
	cm_stage_location_to_queue = { COUNTRY_LIST = cm_fa_r_locations DONE_VAR = cm_fa_done_rgo }
}

# Full-automation obsolete-upgrade staging, keyed by the standing obsolete type. Mirrors
# cm_stage_upgrade_candidate's scope dance and shared markers; the ROI gate replaces the
# min-gold and min-discount gates and runs at scoring. scope:cm_building_type holds the
# replacement while the gates run. Expects scope:cm_country, scope:cm_location,
# scope:cm_building, scope:cm_building_type as the standing obsolete type,
# scope:cm_current_building_type as the resolved replacement.
cm_fa_stage_upgrade_candidate = {
	cm_clear_finished_upgrade_in_progress = yes
	scope:cm_building_type = { save_scope_as = cm_fa_staged_type }
	scope:cm_current_building_type = { save_scope_as = cm_building_type }
	set_local_variable = { name = cm_fa_upg_ok value = 0 }
	if = {
		limit = {
			scope:cm_building = { cm_is_not_under_construction = yes }
			scope:cm_location = {
				cm_location_building_type_not_under_construction = yes
				cm_location_has_enough_pops_available = yes
				cm_is_location_under_building_limit = yes
			}
			NOT = { scope:cm_location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:cm_fa_staged_type } } }
			# Already staged in the player's upgrade queue this pulse
			NOT = { scope:cm_location = { is_target_in_variable_list = { name = cm_gui_upgrade_building_types target = scope:cm_fa_staged_type } } }
		}
		set_local_variable = { name = cm_fa_upg_ok value = 1 }
	}
	scope:cm_fa_staged_type = { save_scope_as = cm_building_type }
	if = {
		limit = { local_var:cm_fa_upg_ok = 1 }
		cm_stage_location_and_type_to_queue = { COUNTRY_LIST = cm_fa_u_locations LOCATION_LIST = cm_fa_u_types DONE_LIST = cm_fa_done_u_types }
		# Shared marker so no feature from-zero-builds the replacement this pulse.
		scope:cm_location = {
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_current_building_type } } }
				add_to_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_current_building_type }
			}
		}
	}
}

# score = -(cost / (adjusted income * 12)), floored at the -99999 sentinel; bigger =
# better, matching cm_q_compute_metric_score's ROI branch. Root is the item's location.
# Expects scope:cm_income (plus the amenity scopes when available) and scope:cm_q_score_cost.
# Result exported as scope:cm_fa_score_val. Callers gate on cm_fa_ind_roi_passes first.
cm_fa_compute_roi_score = {
	set_local_variable = { name = cm_fa_denom value = cm_bld_adjusted_income }
	change_local_variable = { name = cm_fa_denom multiply = 12 }
	set_local_variable = { name = cm_fa_sc value = scope:cm_q_score_cost }
	change_local_variable = { name = cm_fa_sc divide = local_var:cm_fa_denom }
	change_local_variable = { name = cm_fa_sc multiply = -1 }
	change_local_variable = { name = cm_fa_sc max = -99999 }
	save_temporary_scope_value_as = { name = cm_fa_score_val value = local_var:cm_fa_sc }
}

# Folds a scored candidate into its building type's rotation-remembered best payback. Called
# before the placement gate, so a candidate whose location goes to another building still sets
# its own type's bar. Root is the country. Expects scope:cm_fa_anchor_type,
# scope:cm_fa_anchor_loc and scope:cm_fa_score_val.
cm_fa_ind_fold_bt_anchor = {
	if = {
		limit = {
			trigger_if = {
				limit = {
					has_variable_map = cm_fa_ind_bt_anchor
					is_key_in_variable_map = { name = cm_fa_ind_bt_anchor target = scope:cm_fa_anchor_type }
				}
				"variable_map(cm_fa_ind_bt_anchor|scope:cm_fa_anchor_type)" < scope:cm_fa_score_val
			}
			trigger_else = { always = yes }
		}
		remove_from_variable_map = { name = cm_fa_ind_bt_anchor key = scope:cm_fa_anchor_type }
		add_to_variable_map = { name = cm_fa_ind_bt_anchor key = scope:cm_fa_anchor_type value = scope:cm_fa_score_val }
		remove_from_variable_map = { name = cm_fa_ind_bt_loc key = scope:cm_fa_anchor_type }
		add_to_variable_map = { name = cm_fa_ind_bt_loc key = scope:cm_fa_anchor_type value = scope:cm_fa_anchor_loc }
	}
}

# Clears what a market's last visit left behind, both the anchors its locations set and the best
# candidate it was remembered by. Called as the market enters a slice, so the fresh scores that
# follow are what the market is judged on.
# Root is the country, expects scope:cm_fa_slice_mkt.
cm_fa_ind_drop_market_memory = {
	remove_from_variable_map = { name = cm_fa_ind_mkt_best key = scope:cm_fa_slice_mkt }
	remove_from_variable_map = { name = cm_fa_ind_mkt_cost key = scope:cm_fa_slice_mkt }
	save_temporary_scope_as = cm_fa_anchor_owner
	# Collected first: removing a key inside the walk would mutate the map being walked.
	clear_variable_list = cm_fa_ind_drop
	if = {
		limit = { has_variable_map = cm_fa_ind_bt_loc }
		every_key_in_variable_map = {
			variable = cm_fa_ind_bt_loc
			save_scope_as = cm_fa_drop_bt
			# The iterator makes the key the scope, so the map read has to name its owner.
			"scope:cm_fa_anchor_owner.variable_map(cm_fa_ind_bt_loc|scope:cm_fa_drop_bt)" = { save_temporary_scope_as = cm_fa_drop_loc }
			if = {
				limit = { scope:cm_fa_drop_loc = { market = { this = scope:cm_fa_slice_mkt } } }
				scope:cm_fa_anchor_owner = { add_to_variable_list = { name = cm_fa_ind_drop target = scope:cm_fa_drop_bt } }
			}
		}
	}
	every_in_list = {
		variable = cm_fa_ind_drop
		save_scope_as = cm_fa_drop_bt
		scope:cm_fa_anchor_owner = {
			remove_from_variable_map = { name = cm_fa_ind_bt_anchor key = scope:cm_fa_drop_bt }
			remove_from_variable_map = { name = cm_fa_ind_bt_loc key = scope:cm_fa_drop_bt }
		}
	}
	clear_variable_list = cm_fa_ind_drop
}

# The best candidate still standing at this location and what it costs, as local_var:cm_fa_rec_s
# and local_var:cm_fa_rec_c; the score stays at the sentinel where nothing is left.
# Root is the location.
cm_fa_ind_loc_best = {
	save_temporary_scope_as = cm_fa_rec_loc
	set_local_variable = { name = cm_fa_rec_s value = -2000000 }
	set_local_variable = { name = cm_fa_rec_c value = 0 }
	if = {
		limit = {
			has_variable = cm_fa_b_best
			has_variable_map = cm_fa_b_scores
			var:cm_fa_b_best > local_var:cm_fa_rec_s
		}
		ordered_key_in_variable_map = {
			variable = cm_fa_b_scores
			order_by = {
				save_temporary_scope_as = cm_fa_try_type
				value = "scope:cm_fa_rec_loc.variable_map(cm_fa_b_scores|scope:cm_fa_try_type)"
			}
			position = 0
			check_range_bounds = no
			save_temporary_scope_as = cm_fa_rec_type
		}
		if = {
			limit = {
				has_variable_map = cm_fa_b_costs
				is_key_in_variable_map = { name = cm_fa_b_costs target = scope:cm_fa_rec_type }
			}
			set_local_variable = { name = cm_fa_rec_s value = var:cm_fa_b_best }
			set_local_variable = { name = cm_fa_rec_c value = "variable_map(cm_fa_b_costs|scope:cm_fa_rec_type)" }
		}
	}
	if = {
		limit = {
			has_variable = cm_fa_u_best
			has_variable_map = cm_fa_u_scores
			var:cm_fa_u_best > local_var:cm_fa_rec_s
		}
		ordered_key_in_variable_map = {
			variable = cm_fa_u_scores
			order_by = {
				save_temporary_scope_as = cm_fa_try_type
				value = "scope:cm_fa_rec_loc.variable_map(cm_fa_u_scores|scope:cm_fa_try_type)"
			}
			position = 0
			check_range_bounds = no
			save_temporary_scope_as = cm_fa_rec_type
		}
		if = {
			limit = {
				has_variable_map = cm_fa_u_costs
				is_key_in_variable_map = { name = cm_fa_u_costs target = scope:cm_fa_rec_type }
			}
			set_local_variable = { name = cm_fa_rec_s value = var:cm_fa_u_best }
			set_local_variable = { name = cm_fa_rec_c value = "variable_map(cm_fa_u_costs|scope:cm_fa_rec_type)" }
		}
	}
	if = {
		limit = {
			has_variable = cm_fa_r_score
			var:cm_fa_r_score > local_var:cm_fa_rec_s
		}
		set_local_variable = { name = cm_fa_rec_s value = var:cm_fa_r_score }
		# glorpui_rgo_expand_cost reads the location by name, not from the root.
		save_temporary_scope_as = cm_location
		set_local_variable = { name = cm_fa_rec_c value = glorpui_rgo_expand_cost }
	}
}

# Remembers the best candidate each slice market still held when the drain stopped, so a market
# the rotation has moved off still weighs against the ones it can act on. Anything the pick loop
# consumed is gone from the maps by now, which is what keeps a bought or refused candidate out.
# Root is the country.
cm_fa_ind_record_market_best = {
	save_temporary_scope_as = cm_fa_rec_ctry
	every_in_list = {
		variable = cm_fa_drain_cands
		cm_fa_ind_loc_best = yes
		if = {
			limit = { local_var:cm_fa_rec_s > -2000000 }
			market = { save_scope_as = cm_fa_rec_mkt }
			scope:cm_fa_rec_ctry = {
				if = {
					limit = {
						trigger_if = {
							limit = {
								has_variable_map = cm_fa_ind_mkt_best
								is_key_in_variable_map = { name = cm_fa_ind_mkt_best target = scope:cm_fa_rec_mkt }
							}
							"variable_map(cm_fa_ind_mkt_best|scope:cm_fa_rec_mkt)" < local_var:cm_fa_rec_s
						}
						trigger_else = { always = yes }
					}
					remove_from_variable_map = { name = cm_fa_ind_mkt_best key = scope:cm_fa_rec_mkt }
					add_to_variable_map = { name = cm_fa_ind_mkt_best key = scope:cm_fa_rec_mkt value = local_var:cm_fa_rec_s }
					remove_from_variable_map = { name = cm_fa_ind_mkt_cost key = scope:cm_fa_rec_mkt }
					add_to_variable_map = { name = cm_fa_ind_mkt_cost key = scope:cm_fa_rec_mkt value = local_var:cm_fa_rec_c }
				}
			}
		}
	}
}

# The best candidate remembered in a market this cycle cannot reach, as local_var:cm_fa_off_best
# and local_var:cm_fa_off_cost. Root is the country, expects scope:cm_country.
cm_fa_ind_off_slice_best = {
	save_temporary_scope_as = cm_fa_off_ctry
	set_local_variable = { name = cm_fa_off_best value = -2000000 }
	set_local_variable = { name = cm_fa_off_cost value = 0 }
	if = {
		limit = { has_variable_map = cm_fa_ind_mkt_best }
		every_key_in_variable_map = {
			variable = cm_fa_ind_mkt_best
			save_scope_as = cm_fa_off_mkt
			# The iterator makes the key the scope, so the map reads have to name their owner.
			set_local_variable = { name = cm_fa_off_v value = "scope:cm_fa_off_ctry.variable_map(cm_fa_ind_mkt_best|scope:cm_fa_off_mkt)" }
			if = {
				limit = {
					local_var:cm_fa_off_v > local_var:cm_fa_off_best
					NOT = { cm_fa_market_in_slice = yes }
					scope:cm_fa_off_ctry = { is_key_in_variable_map = { name = cm_fa_ind_mkt_cost target = scope:cm_fa_off_mkt } }
				}
				set_local_variable = { name = cm_fa_off_best value = local_var:cm_fa_off_v }
				set_local_variable = { name = cm_fa_off_cost value = "scope:cm_fa_off_ctry.variable_map(cm_fa_ind_mkt_cost|scope:cm_fa_off_mkt)" }
			}
		}
	}
}

# Margin factor = 100 / margin percent; scores are negative years, so a type's anchor times
# the factor is the bar for holding margin percent of that type's payback rate. 0 = off.
# Root is the country. Result in local_var:cm_fa_margin_f.
cm_fa_ind_margin_factor = {
	set_local_variable = { name = cm_fa_margin_f value = 0 }
	if = {
		limit = {
			exists = var:cm_fa_industry_margin
			var:cm_fa_industry_margin > 0
		}
		set_local_variable = { name = cm_fa_margin_f value = 100 }
		change_local_variable = { name = cm_fa_margin_f divide = var:cm_fa_industry_margin }
	}
}

# One building type's admission bar: its anchor scaled by the margin factor, on the negative
# years scale. Stays at the sentinel when the margin is off or the type has no anchor yet,
# which admits every score. Expects scope:cm_country, scope:cm_fa_anchor_type and
# local_var:cm_fa_margin_f. Result in local_var:cm_fa_bt_bar.
cm_fa_ind_bt_bar = {
	set_local_variable = { name = cm_fa_bt_bar value = -2000000 }
	if = {
		limit = {
			local_var:cm_fa_margin_f > 0
			scope:cm_country = {
				has_variable_map = cm_fa_ind_bt_anchor
				is_key_in_variable_map = { name = cm_fa_ind_bt_anchor target = scope:cm_fa_anchor_type }
			}
		}
		set_local_variable = { name = cm_fa_bt_bar value = "scope:cm_country.variable_map(cm_fa_ind_bt_anchor|scope:cm_fa_anchor_type)" }
		change_local_variable = { name = cm_fa_bt_bar multiply = local_var:cm_fa_margin_f }
	}
}

# Temporary diagnostic: one CMM log line per Industry approval. The income is recovered from
# the score and the cost, so it is the adjusted income the ROI was computed from. Root is the
# location. Expects scope:cm_country, scope:cm_building_type, scope:cm_cost, and
# scope:cm_fa_pick_score as the item's stored negated years-to-recoup.
cm_fa_ind_log_queued = {
	if = {
		limit = { always = yes }
		set_local_variable = { name = cm_fa_ilog_roi value = scope:cm_fa_pick_score }
		change_local_variable = { name = cm_fa_ilog_roi multiply = -1 }
		set_local_variable = { name = cm_fa_ilog_inc value = 0 }
		if = {
			limit = { local_var:cm_fa_ilog_roi > 0 }
			set_local_variable = { name = cm_fa_ilog_inc value = scope:cm_cost }
			change_local_variable = { name = cm_fa_ilog_inc divide = local_var:cm_fa_ilog_roi }
			change_local_variable = { name = cm_fa_ilog_inc divide = 12 }
		}
		save_scope_as = cmf_log_loc
		scope:cm_building_type = { save_scope_as = cmf_log_bt }
		scope:cm_country = {
			cm_dbg_log_bld = { CAT = industry
				action = cm_log_ind_queued
				arg1 = cm_log_ind_roi
				arg2 = cm_log_ind_income
				value = local_var:cm_fa_ilog_roi
				value2 = local_var:cm_fa_ilog_inc
			}
		}
	}
}

# Greedy country-wide drain: actionable builds and upgrades are approved best-first, each held
# to the margin share of the best payback rate recorded for its own building type; RGO
# expansions bypass the margin and drain by ROI. Payback is cost over income in gold, both
# already priced in the candidate's own market, so scores rank directly against each other
# across markets.
# Root is the country.
cm_fa_pass_industry = {
	cm_fa_ind_margin_factor = yes
	# The picks below run hundreds of times, so the candidate set is collected once here and
	# re-read from the list instead of re-scanning every owned location. Only the markets
	# staged this cycle: a market outside the slice holds no candidates.
	clear_variable_list = cm_fa_drain_cands
	every_market_present_in_country = {
		limit = { cm_fa_market_in_slice = yes }
		every_location_in_market = {
			limit = {
				owner ?= scope:cm_country
				OR = {
					has_variable = cm_fa_b_best
					has_variable = cm_fa_u_best
					has_variable = cm_fa_r_score
				}
			}
			save_scope_as = cm_fa_cand_loc
			scope:cm_country = { add_to_variable_list = { name = cm_fa_drain_cands target = scope:cm_fa_cand_loc } }
		}
	}
	cm_fa_ind_off_slice_best = yes
	cm_fa_ind_pick_loop = yes
	cm_fa_ind_record_market_best = yes
	clear_variable_list = cm_fa_drain_cands
}

# Approves the best candidate in the country repeatedly until every candidate map is spent or
# the pool cannot cover the next one; each pick tests its own building type's bar. The outer
# batch loop is what carries the pick count past a single while's 1000-iteration abort; it
# re-enters only while the inner loop's last iteration picked something.
# Root is the country. Expects cm_fa_drain_cands built and local_var:cm_fa_margin_f set.
cm_fa_ind_pick_loop = {
	set_local_variable = { name = cm_fa_more value = 1 }
	set_local_variable = { name = cm_fa_outer value = 0 }
	while = {
		limit = {
			local_var:cm_fa_more = 1
			local_var:cm_fa_cost_blocked = 0
			local_var:cm_fa_outer < 20
		}
		change_local_variable = { name = cm_fa_outer add = 1 }
		set_local_variable = { name = cm_fa_guard value = 0 }
		while = {
			limit = {
				local_var:cm_fa_more = 1
				local_var:cm_fa_cost_blocked = 0
				# 999 stays under the engine's 1000-iteration while abort.
				local_var:cm_fa_guard < 999
			}
			change_local_variable = { name = cm_fa_guard add = 1 }
			set_local_variable = { name = cm_fa_more value = 0 }
			# Best actionable candidate per domain.
			save_temporary_scope_value_as = { name = cm_fa_dom_b value = -2000000 }
			save_temporary_scope_value_as = { name = cm_fa_dom_u value = -2000000 }
			save_temporary_scope_value_as = { name = cm_fa_dom_r value = -2000000 }
			ordered_in_list = {
				variable = cm_fa_drain_cands
				limit = { has_variable = cm_fa_b_best }
				order_by = { value = var:cm_fa_b_best }
				position = 0
				check_range_bounds = no
				save_temporary_scope_as = cm_fa_pick_b_loc
				save_temporary_scope_value_as = { name = cm_fa_dom_b value = var:cm_fa_b_best }
			}
			ordered_in_list = {
				variable = cm_fa_drain_cands
				limit = { has_variable = cm_fa_u_best }
				order_by = { value = var:cm_fa_u_best }
				position = 0
				check_range_bounds = no
				save_temporary_scope_as = cm_fa_pick_u_loc
				save_temporary_scope_value_as = { name = cm_fa_dom_u value = var:cm_fa_u_best }
			}
			ordered_in_list = {
				variable = cm_fa_drain_cands
				limit = { has_variable = cm_fa_r_score }
				order_by = { value = var:cm_fa_r_score }
				position = 0
				check_range_bounds = no
				save_temporary_scope_as = cm_fa_pick_r_loc
				save_temporary_scope_value_as = { name = cm_fa_dom_r value = var:cm_fa_r_score }
			}
			set_local_variable = { name = cm_fa_win value = scope:cm_fa_dom_b }
			change_local_variable = { name = cm_fa_win max = scope:cm_fa_dom_u }
			change_local_variable = { name = cm_fa_win max = scope:cm_fa_dom_r }
			# The picks prune cm_fa_drain_cands, which is only safe because they run after
			# all three scans above have closed. Moving one inside a scan body would mutate
			# the list mid-iteration.
			if = {
				# A market the rotation has moved off holds a better candidate, so the pool waits
				# for it instead of buying down the order here.
				limit = {
					local_var:cm_fa_win > -2000000
					local_var:cm_fa_off_best > local_var:cm_fa_win
				}
				cm_fa_stop_on_cost = { COST = local_var:cm_fa_off_cost }
				set_local_variable = { name = cm_fa_hold_dbg value = { value = local_var:cm_fa_off_best multiply = -1 } }
				cm_dbg_log_decimal = { CAT = industry action = cm_log_fa_hold value = local_var:cm_fa_hold_dbg }
			}
			else_if = {
				# Ties go to the build domain, then upgrades.
				limit = {
					scope:cm_fa_dom_b >= { value = scope:cm_fa_dom_u }
					scope:cm_fa_dom_b >= { value = scope:cm_fa_dom_r }
					scope:cm_fa_dom_b > -2000000
				}
				scope:cm_fa_pick_b_loc = { cm_fa_pick_best_build_item = yes }
				set_local_variable = { name = cm_fa_more value = 1 }
			} else_if = {
				limit = {
					scope:cm_fa_dom_u >= { value = scope:cm_fa_dom_r }
					scope:cm_fa_dom_u > -2000000
				}
				scope:cm_fa_pick_u_loc = { cm_fa_pick_best_upgrade_item = yes }
				set_local_variable = { name = cm_fa_more value = 1 }
			} else_if = {
				# RGOs have no input good, so they can never be unprofitable and never steal a
				# building's opportunity; the margin never gates them, only ROI order.
				limit = { scope:cm_fa_dom_r > -2000000 }
				scope:cm_fa_pick_r_loc = { cm_fa_approve_rgo_item = yes }
				set_local_variable = { name = cm_fa_more value = 1 }
			}
		}
	}
}

# Approves the highest-scored full-automation build at this location, then consumes it;
# the key is removed even when the gates fail, so every pick shrinks a map. Root is the
# location.
cm_fa_pick_best_build_item = {
	save_temporary_scope_as = cm_fa_pick_loc
	ordered_key_in_variable_map = {
		variable = cm_fa_b_scores
		order_by = {
			save_temporary_scope_as = cm_fa_try_type
			value = "scope:cm_fa_pick_loc.variable_map(cm_fa_b_scores|scope:cm_fa_try_type)"
		}
		position = 0
		check_range_bounds = no
		save_temporary_scope_as = cm_fa_pick_type
	}
	scope:cm_fa_pick_type = {
		save_scope_as = cm_building_type
		save_temporary_scope_as = cm_fa_anchor_type
	}
	save_temporary_scope_value_as = { name = cm_cost value = "variable_map(cm_fa_b_costs|scope:cm_fa_pick_type)" }
	save_temporary_scope_value_as = { name = cm_fa_pick_score value = "variable_map(cm_fa_b_scores|scope:cm_fa_pick_type)" }
	cm_fa_ind_bt_bar = yes
	market = { save_temporary_scope_as = cm_market }
	if = {
		limit = {
			# No cost key means the pick came back empty or the engine blocked the build at scoring.
			has_variable_map = cm_fa_b_costs
			is_key_in_variable_map = { name = cm_fa_b_costs target = scope:cm_fa_pick_type }
			scope:cm_fa_pick_score >= { value = local_var:cm_fa_bt_bar }
			# Approved by another feature this cycle: consume silently.
			NOT = { is_target_in_variable_list = { name = cm_q_approved_build_types target = scope:cm_building_type } }
		}
		if = {
			limit = {
				cm_get_open_build_queue_slots > 0
				cm_market_has_construction_goods_for_building = yes
			}
			if = {
				limit = { cm_fa_can_afford = { POOL = cm_fa_pool_industry } }
				cm_approve_gui_build = yes
				cm_fa_ind_log_queued = yes
				scope:cm_country = { change_variable = { name = cm_fa_pool_industry subtract = scope:cm_cost } }
				# The engine converts this build into the standing obsolete predecessor's upgrade.
				if = {
					limit = {
						has_variable_map = cm_fa_b_conv
						is_key_in_variable_map = { name = cm_fa_b_conv target = scope:cm_fa_pick_type }
					}
					"variable_map(cm_fa_b_conv|scope:cm_fa_pick_type)" = { save_temporary_scope_as = cm_fa_conv_pred }
					if = {
						limit = { NOT = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:cm_fa_conv_pred } } }
						add_to_variable_list = { name = cm_upgrade_in_progress target = scope:cm_fa_conv_pred }
					}
					remove_from_variable_map = { name = cm_upgrade_in_progress_targets key = scope:cm_fa_conv_pred }
					add_to_variable_map = { name = cm_upgrade_in_progress_targets key = scope:cm_fa_conv_pred value = scope:cm_building_type }
				} else = {
					# A conversion must fire exactly once.
					cm_fa_fill_build_slots = { POOL = cm_fa_pool_industry }
					if = {
						# The batch stopped on the pool or the treasury alone: stop all further
						# Industry picks rather than let a worse candidate spend what is left.
						limit = { local_var:cm_q_copy_stop = 2 }
						cm_fa_stop_on_cost = { COST = scope:cm_cost }
					}
				}
			} else = {
				cm_fa_stop_on_cost = { COST = scope:cm_cost }
			}
		}
	}
	remove_from_variable_map = { name = cm_fa_b_scores key = scope:cm_fa_pick_type }
	remove_from_variable_map = { name = cm_fa_b_costs key = scope:cm_fa_pick_type }
	remove_from_variable_map = { name = cm_fa_b_conv key = scope:cm_fa_pick_type }
	remove_from_variable_map = { name = cm_fa_b_cats key = scope:cm_fa_pick_type }
	cm_fa_refresh_build_best = yes
	cm_fa_drain_prune_loc = yes
}

# Approves RGO levels at this location under the industry pool, mirroring the player RGO
# drain's slot, headroom, and laborer budget; consumes the score first so the pick loop
# always advances. Root is the location.
cm_fa_approve_rgo_item = {
	set_local_variable = { name = cm_fa_rgo_dbg value = 0 }
	if = {
		limit = { has_variable = cm_fa_r_score }
		set_local_variable = { name = cm_fa_rgo_dbg value = { value = var:cm_fa_r_score multiply = -1 } }
	}
	remove_variable = cm_fa_r_score
	save_temporary_scope_as = cm_location
	if = {
		limit = { NOT = { cm_rgo_basic_expand_eligibility = yes } }
		cm_fa_rgo_log = { ACTION = cm_log_rgo_elig }
	}
	else_if = {
		limit = { NOT = { cm_location_rgo_free_this_cycle = yes } }
		cm_fa_rgo_log = { ACTION = cm_log_rgo_served }
	}
	else_if = {
		limit = { cm_location_auto_food_supersedes_rgo = yes }
		cm_fa_rgo_log = { ACTION = cm_log_rgo_food }
	}
	else_if = {
		limit = { NOT = { cm_market_has_construction_goods_for_rgo = yes } }
		cm_fa_rgo_log = { ACTION = cm_log_rgo_goods }
	}
	else = {
		if = {
			limit = { cm_get_open_build_queue_slots > 0 }
			set_local_variable = { name = cm_fa_rgo_budget value = cm_get_open_build_queue_slots }
			set_local_variable = { name = cm_fa_rgo_headroom value = cm_get_max_rgo_workers_floor }
			change_local_variable = { name = cm_fa_rgo_headroom subtract = cm_modifier_rgo_level }
			change_local_variable = { name = cm_fa_rgo_headroom subtract = cm_get_rgos_under_construction }
			if = {
				limit = { local_var:cm_fa_rgo_headroom < local_var:cm_fa_rgo_budget }
				set_local_variable = { name = cm_fa_rgo_budget value = local_var:cm_fa_rgo_headroom }
			}
			set_local_variable = { name = cm_fa_rgo_slack value = cm_location_rgo_promotion_buffer_slack }
			set_local_variable = { name = cm_fa_rgo_done value = 0 }
			while = {
				limit = {
					local_var:cm_fa_rgo_budget > 0
					local_var:cm_fa_rgo_slack >= local_var:cm_fa_rgo_done
					cm_get_open_build_queue_slots > 0
					cm_get_max_rgo_workers_floor > { value = cm_modifier_rgo_level add = cm_get_rgos_under_construction }
					cm_fa_can_afford_rgo = { POOL = cm_fa_pool_industry }
				}
				set_local_variable = { name = cm_fa_rgo_cost value = glorpui_rgo_expand_cost }
				construct_rgo_upgrade = { }
				scope:cm_country = { change_variable = { name = cm_fa_pool_industry subtract = local_var:cm_fa_rgo_cost } }
				change_local_variable = { name = cm_fa_rgo_budget subtract = 1 }
				change_local_variable = { name = cm_fa_rgo_done add = 1 }
			}
			if = {
				# Same non-affordability conditions as the while above, affordability negated.
				limit = {
					local_var:cm_fa_rgo_budget > 0
					local_var:cm_fa_rgo_slack >= local_var:cm_fa_rgo_done
					cm_get_open_build_queue_slots > 0
					cm_get_max_rgo_workers_floor > { value = cm_modifier_rgo_level add = cm_get_rgos_under_construction }
					NOT = { cm_fa_can_afford_rgo = { POOL = cm_fa_pool_industry } }
				}
				cm_fa_stop_on_cost = { COST = glorpui_rgo_expand_cost }
				cm_fa_rgo_log = { ACTION = cm_log_rgo_gold }
			}
			if = {
				limit = { local_var:cm_fa_rgo_done > 0 }
				cm_q_mark_rgo_served = yes
				cm_fa_rgo_log = { ACTION = cm_log_rgo_ok }
			}
			else_if = {
				limit = { local_var:cm_fa_cost_blocked = 0 }
				cm_fa_rgo_log = { ACTION = cm_log_rgo_room }
			}
		}
		else = {
			cm_fa_rgo_log = { ACTION = cm_log_rgo_slots }
		}
	}
	cm_fa_drain_prune_loc = yes
}
