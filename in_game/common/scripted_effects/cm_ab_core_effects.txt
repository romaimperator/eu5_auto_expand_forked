# Auto-Build Core Goods: builds producers of masonry, lumber, sand, and glass
# to push their market prices down to the target price. Materials are staged and drained
# in the player's list order; within a material, RGO expansions come before producer
# buildings. Each market is sized in producer levels, enough to close the gap between what
# it produces of a material and what it demands, plus the overshoot; the best-producing
# locations take those levels first, several at a time, and levels already paid for but not
# yet producing come off the count. Only markets whose center the country owns are worked,
# so the price it collapses is one it holds. Core Goods approvals skip the market
# construction-goods gates: a shortage of those goods is exactly what the builds cure.

# Console: effect location:<key> = { cm_ab_core_probe = yes }
# Every number masonry staging reads at this location, each beside the engine reading it is
# derived from, so a condition answering differently from the game shows up as a mismatched
# pair rather than as a verdict to be taken on trust. Root is the location.
cm_ab_core_probe = {
	owner = { save_scope_as = cm_country }
	building_type:mason = { save_scope_as = cm_building_type }
	goods:masonry = { save_scope_as = cm_ab_good_t }

	}

	}
	cm_ab_core_pair = {
		ACTION = cm_log_core_p_cap

	}
	cm_ab_core_pair = {
		ACTION = cm_log_core_p_market

	cm_ab_core_pair = {
		ACTION = cm_log_core_p_want

	cm_ab_core_verdict = { COND = cm_ab_core_not_at_max ACTION = cm_log_core_p_v_max }

	cm_ab_core_verdict = { COND = cm_location_has_no_auto_build_chain_predecessor ACTION = cm_log_core_p_v_pred }
	cm_ab_core_verdict = { COND = cm_check_location_meets_min_market_access ACTION = cm_log_core_p_v_access }
	cm_ab_core_verdict = { COND = cm_ab_location_has_proximity ACTION = cm_log_core_p_v_prox }

# Walks this location's whole market the way staging does and reports what one cycle could
# absorb, plus the spread of the order key: an order key whose lowest and highest readings
# match ranks nothing, so the sweep takes locations in whatever order the engine hands them.
# Root is the location, expects scope:cm_country, scope:cm_building_type and scope:cm_ab_market.
cm_ab_core_market_survey = {
	set_local_variable = { name = cm_ab_core_n_owned value = 0 }
	set_local_variable = { name = cm_ab_core_n_limit value = 0 }
	set_local_variable = { name = cm_ab_core_n_full value = 0 }
	set_local_variable = { name = cm_ab_core_n_room value = 0 }
	set_local_variable = { name = cm_ab_core_ord_lo value = 999999 }
	set_local_variable = { name = cm_ab_core_ord_hi value = -999999 }
	set_local_variable = { name = cm_ab_core_n_infl value = 0 }
	scope:cm_ab_market = {
		every_location_in_market = {
			limit = { owner ?= scope:cm_country }
			save_scope_as = cm_location
			change_local_variable = { name = cm_ab_core_n_owned add = 1 }
			change_local_variable = { name = cm_ab_core_ord_lo min = cm_ab_core_efficiency_order }
			change_local_variable = { name = cm_ab_core_ord_hi max = cm_ab_core_efficiency_order }
			if = {
				limit = {
					cm_ab_core_unconquered = yes
					cm_ab_location_has_proximity = yes
					cm_ab_core_candidate_ok = yes
				}
				change_local_variable = { name = cm_ab_core_n_limit add = 1 }
				change_local_variable = { name = cm_ab_core_n_infl add = cm_ab_core_type_inflight }
				if = {
					limit = {
						cm_ab_core_placeable = yes
						cm_ab_core_not_at_max = yes
						cm_location_has_no_auto_build_chain_predecessor = yes
						cm_ab_core_pops_ok = yes
						cm_check_location_meets_min_market_access = yes
						cm_ab_core_type_room > 0
						cm_ab_core_not_claimed = yes
					}
					change_local_variable = { name = cm_ab_core_n_full add = 1 }
					change_local_variable = { name = cm_ab_core_n_room add = cm_ab_core_type_room }
				}
			}
		}
	}
	save_temporary_scope_as = cmf_log_loc
	scope:cm_building_type = { save_temporary_scope_as = cmf_log_bt }
	cm_ab_core_pair = {
		ACTION = cm_log_core_p_m_counts
		V1 = cm_ab_core_survey_owned
		V2 = cm_ab_core_survey_eligible
	}
	cm_ab_core_pair = {
		ACTION = cm_log_core_p_m_room
		V1 = cm_ab_core_survey_placeable
		V2 = cm_ab_core_survey_room
	}
	cm_ab_core_pair = {
		ACTION = cm_log_core_p_m_order
		V1 = cm_ab_core_survey_ord_lo
		V2 = cm_ab_core_survey_ord_hi
	}
	cm_ab_core_pair = {
		ACTION = cm_log_core_p_m_infl
		V1 = cm_ab_core_survey_inflight
		V2 = cm_ab_core_survey_left
	}
}

# Logs $V1$ and $V2$ on one line. Root is the location, expects scope:cm_country,
# scope:cmf_log_bt and scope:cmf_log_loc.
cm_ab_core_pair = {
	save_temporary_scope_value_as = { name = cm_ab_core_v1 value = $V1$ }
	save_temporary_scope_value_as = { name = cm_ab_core_v2 value = $V2$ }
	scope:cm_country = {
		cm_dbg_log_bld = {
			cat = core
			action = $ACTION$
			arg1 = cm_log_core_p_a_mod
			arg2 = cm_log_core_p_a_engine
			value = scope:cm_ab_core_v1
			value2 = scope:cm_ab_core_v2
		}
	}
}

# Logs $COND$ as 1 or 0. Root is the location, expects scope:cm_country.
cm_ab_core_verdict = {
	save_temporary_scope_value_as = {
		name = cm_ab_core_vd
		value = {
			value = 0
			if = {
				limit = { $COND$ = yes }
				value = 1
			}
		}
	}
	scope:cm_country = {
		cm_dbg_log_value = { cat = core action = $ACTION$ value = scope:cm_ab_core_vd }
	}
}

# Which category already holds this location's mark for the type Core Goods is about to
# claim, recorded before the claim so a refusal names the holder rather than just failing.
# Root is the location, expects scope:cm_country and scope:cm_building_type.
cm_ab_core_claim_outcome = {
	if = {
		limit = { exists = global_var:cm_dbg_core }
		if = {
			limit = { NOT = { cm_ab_core_cat_key_present = yes } }
			scope:cm_country = { cm_ab_core_count = { N = cm_ab_core_c_claim_ok V = 1 } }
		}
		else = {
			scope:cm_country = { cm_ab_core_count = { N = cm_ab_core_c_held_2 V = 1 } }
		}
		if = {
			limit = { cm_ab_core_production_also_wants = yes }
			scope:cm_country = { cm_ab_core_count = { N = cm_ab_core_c_held_1 V = 1 } }
		}
	}
}

# Root is the country.
cm_ab_core_count = {
	if = {
		limit = {
			exists = global_var:cm_dbg_core
			exists = var:$N$
		}
		change_variable = { name = $N$ add = $V$ }
	}
}

# Root is the country, scope:cm_country set.
cm_ab_run_core = {
	cm_ab_core_reset_counts = yes
	every_market_center_in_country = {
		limit = { cm_market_in_slice = yes }
		save_scope_as = cm_ab_market
		scope:cm_country = { cm_ab_stage_core_market = yes }
	}
	cm_ab_core_count_marks = yes
}

# Locations still carrying the category's own mark for mason once staging has finished, which
# is what the scorer and the drain both look for. Root is the country.
cm_ab_core_count_marks = {
	if = {
		limit = {
			exists = global_var:cm_dbg_core
			has_variable_list = cm_ab_b_locations
		}
		building_type:mason = { save_temporary_scope_as = cm_ab_core_mk }
		every_in_list = {
			variable = cm_ab_b_locations
			limit = {
				has_variable_map = cm_ab_b_cat_2
				is_key_in_variable_map = { name = cm_ab_b_cat_2 target = scope:cm_ab_core_mk }
			}
			scope:cm_country = { cm_ab_core_count = { N = cm_ab_core_c_mark_end V = 1 } }
		}
	}
}

# Stages one market's candidates, each material under its own level count.
# Root is the country, expects scope:cm_ab_market.
cm_ab_stage_core_market = {
	if = {
		limit = { has_variable_list = cm_ab_core_goods_enabled }
		every_in_list = {
			variable = cm_ab_core_goods_enabled
			save_scope_as = cm_ab_core_item
			scope:cm_country = { cm_ab_stage_core_good = yes }
		}
	}
}

# Sets local_var:cm_ab_core_want to the producer levels scope:cm_ab_core_item's
# shortfall needs in scope:cm_ab_market, and 0 once its goal price is reached. Leaves
# scope:cm_building_type on the material's lead producer, whose per-level output is what turns
# a goods shortfall into a level count. Root is the country, expects scope:cm_ab_market and
# scope:cm_ab_core_item.
cm_ab_core_size_good = {
	set_local_variable = { name = cm_ab_core_want value = 0 }
	scope:cm_ab_core_item = { save_temporary_scope_as = cm_ab_good_t }
	if = {
		limit = { NOT = { cm_ab_core_target_met = yes } }
		if = {
			limit = { scope:cm_ab_core_item = goods:masonry }
			building_type:mason = { save_scope_as = cm_building_type }
		}
		else_if = {
			limit = { scope:cm_ab_core_item = goods:lumber }
			building_type:lumber_mill = { save_scope_as = cm_building_type }
		}
		else_if = {
			limit = { scope:cm_ab_core_item = goods:sand }
			building_type:sand_pit = { save_scope_as = cm_building_type }
		}
		else_if = {
			limit = { scope:cm_ab_core_item = goods:glass }
			building_type:glass_guild = { save_scope_as = cm_building_type }
			cm_ab_resolve_building_type = yes
		}
		set_local_variable = { name = cm_ab_core_want value = scope:cm_ab_market.cm_ab_core_levels_wanted }
		if = {
			limit = { scope:cm_ab_market = { cm_ab_core_market_watched = yes } }
			save_temporary_scope_value_as = { name = cm_ab_core_short value = scope:cm_ab_market.cm_ab_core_shortfall }
			save_temporary_scope_value_as = { name = cm_ab_core_out value = "global_variable_map(cm_bt_output|scope:cm_building_type)" }
			save_temporary_scope_value_as = { name = cm_ab_core_need value = scope:cm_ab_market.cm_ab_core_supply_wanted }
			save_temporary_scope_value_as = { name = cm_ab_core_dem value = "scope:cm_ab_market.goods_demand_in_market(scope:cm_ab_good_t)" }
			scope:cm_building_type = { save_temporary_scope_as = cmf_log_bt }
			var:cm_ab_core_watch_loc = { save_temporary_scope_as = cmf_log_loc }
			cm_dbg_log_bld = {
				cat = core
				action = cm_log_core_size
				arg1 = cm_log_core_a_short
				arg2 = cm_log_core_a_out
				value = scope:cm_ab_core_short
				value2 = scope:cm_ab_core_out
			}
			cm_dbg_log_bld = {
				cat = core
				action = cm_log_core_supply
				arg1 = cm_log_core_a_need
				arg2 = cm_log_core_a_demand
				value = scope:cm_ab_core_need
				value2 = scope:cm_ab_core_dem
			}
			cm_dbg_log_value = {
				cat = core
				action = cm_log_core_want
				value = local_var:cm_ab_core_want
			}
		}
	}
	else_if = {
		limit = { scope:cm_ab_market = { cm_ab_core_market_watched = yes } }
		save_temporary_scope_value_as = { name = cm_ab_core_price value = "scope:cm_ab_market.market_price(scope:cm_ab_good_t)" }
		var:cm_ab_core_watch_loc = { save_temporary_scope_as = cmf_log_loc }
		cm_dbg_log_loc = {
			cat = core
			action = cm_log_core_target_met
			arg1 = cm_log_core_a_price
			value = scope:cm_ab_core_price
		}
	}
}

# Stages one material's candidates under its own level count, which is 0 once its goal price
# is reached. Root is the country, expects scope:cm_ab_market and scope:cm_ab_core_item.
cm_ab_stage_core_good = {
	cm_ab_core_size_good = yes
	set_local_variable = { name = cm_ab_good_room value = local_var:cm_ab_core_want }
	if = {
		limit = { local_var:cm_ab_good_room > 0 }
		if = {
			limit = { scope:cm_ab_core_item = goods:masonry }
			cm_ab_stage_core_building = { bt = mason }
		}
		else_if = {
			limit = { scope:cm_ab_core_item = goods:lumber }
			goods:lumber = { save_scope_as = cm_ab_good }
			cm_ab_stage_core_rgos = yes
			cm_ab_stage_core_building = { bt = lumber_mill }
		}
		else_if = {
			limit = { scope:cm_ab_core_item = goods:sand }
			goods:sand = { save_scope_as = cm_ab_good }
			cm_ab_stage_core_rgos = yes
			cm_ab_stage_core_building = { bt = sand_pit }
		}
		else_if = {
			limit = { scope:cm_ab_core_item = goods:glass }
			cm_ab_stage_core_building = { bt = glass_guild }
			cm_ab_stage_core_building = { bt = rural_glassmaker }
		}
	}
}

# Stages RGO expansions for scope:cm_ab_good across the country's locations in
# scope:cm_ab_market, in any order, while levels remain. The drain-only list bypasses GUI
# scoring: the cost is script-side. An RGO level counts as one producer level against the
# material's count, which is an approximation: an RGO level's output is location-specific and
# has no per-type constant to read. Root is the country.
cm_ab_stage_core_rgos = {
	# Iterating the market rather than the country matters here: this runs once per material
	# per market. The iterator carries no ownership filter of its own.
	scope:cm_ab_market = {
		every_location_in_market = {
			limit = {
				owner ?= scope:cm_country
				NOT = { integration_level = conquered }
				cm_ab_location_has_proximity = yes
				cm_ab_core_candidate_ok = yes
				raw_material ?= scope:cm_ab_good
				cm_is_rgo_not_max_level_already = yes
				cm_is_rgo_under_construction = no
				cm_ab_core_rgo_pops_ok = yes
				NOT = { exists = var:cm_ab_core_touched }
				# The player's RGO auto-expand staged it this cycle.
				NOT = { scope:cm_country = { is_target_in_variable_list = { name = cm_q_rgo_locations target = prev } } }
			}
			save_scope_as = cm_location
			if = {
				limit = {
					local_var:cm_ab_good_room > 0
					cm_ab_core_rgo_room > 0
				}
				set_local_variable = { name = cm_ab_lvl value = cm_ab_core_rgo_room }
				change_local_variable = { name = cm_ab_lvl min = local_var:cm_ab_good_room }
				scope:cm_country = {
					if = {
						limit = { NOT = { is_target_in_variable_list = { name = cm_ab_core_rgo_locations target = scope:cm_location } } }
						add_to_variable_list = { name = cm_ab_core_rgo_locations target = scope:cm_location }
					}
				}
				scope:cm_location = {
					set_variable = { name = cm_ab_core_touched value = yes }
					set_variable = { name = cm_ab_core_levels value = local_var:cm_ab_lvl }
				}
				change_local_variable = { name = cm_ab_good_room subtract = local_var:cm_ab_lvl }
			}
		}
	}
}

# Stages one producer type across scope:cm_ab_market while levels remain, best-producing
# location first. Chain bases resolve to the current tier (glass_guild through glass_mill);
# non-chain types pass through unchanged. Levels already paid for but not yet producing come
# off the count wherever they stand, staged this cycle or not.
# $bt$ = the producer's base type key. Root is the country.
cm_ab_stage_core_building = {
	building_type:$bt$ = { save_scope_as = cm_building_type }
	cm_ab_resolve_building_type = yes
	set_local_variable = { name = cm_ab_core_seen value = 0 }
	scope:cm_ab_market = {
		ordered_location_in_market = {
			order_by = cm_ab_core_efficiency_order
			max = cm_max_guess_at_number_of_locations
			check_range_bounds = no
			limit = {
				owner ?= scope:cm_country
				NOT = { integration_level = conquered }
				cm_ab_location_has_proximity = yes
				cm_ab_core_candidate_ok = yes
			}
			save_scope_as = cm_location
			if = {
				limit = { cm_ab_core_watched = yes }
				set_local_variable = { name = cm_ab_core_seen value = 1 }
				save_temporary_scope_value_as = { name = cm_ab_core_infl value = cm_ab_core_type_inflight }
				save_temporary_scope_as = cmf_log_loc
				scope:cm_country = {
					cm_dbg_log_loc = {
						cat = core
						action = cm_log_core_visit
						arg1 = cm_log_core_a_infl
						value = scope:cm_ab_core_infl
					}
					cm_dbg_log_value = {
						cat = core
						action = cm_log_core_room_in
						value = local_var:cm_ab_good_room
					}
				}
			}
			change_local_variable = { name = cm_ab_good_room subtract = cm_ab_core_type_inflight }
			if = {
				limit = {
					local_var:cm_ab_good_room > 0
					NOT = { exists = var:cm_ab_core_touched }
					cm_location_allowed_to_build = { target = scope:cm_building_type }
					cm_location_building_type_is_not_at_max_level = { target = scope:cm_building_type }
					# A standing older tier would turn this build into its upgrade.
					cm_location_has_no_auto_build_chain_predecessor = yes
					cm_ab_core_pops_ok = yes
					cm_check_location_meets_min_market_access = yes
					cm_ab_core_type_room > 0
					NOT = { is_target_in_variable_list = { name = cm_q_building_types target = scope:cm_building_type } }
					NOT = { is_target_in_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_building_type } }
				}
				set_local_variable = { name = cm_ab_lvl value = cm_ab_core_type_room }
				change_local_variable = { name = cm_ab_lvl min = local_var:cm_ab_good_room }
				cm_ab_core_claim_outcome = yes
				cm_ab_stage_build_candidate = { cat = 2 }
				scope:cm_location = {
					set_variable = { name = cm_ab_core_touched value = yes }
					set_variable = { name = cm_ab_core_levels value = local_var:cm_ab_lvl }
				}
				change_local_variable = { name = cm_ab_good_room subtract = local_var:cm_ab_lvl }
				scope:cm_country = {
					cm_ab_core_count = { N = cm_ab_core_c_stage_locs V = 1 }
					cm_ab_core_count = { N = cm_ab_core_c_stage_lvls V = local_var:cm_ab_lvl }
				}
				if = {
					limit = { cm_ab_core_watched = yes }
					scope:cm_country = {
						cm_dbg_log_value = {
							cat = core
							action = cm_log_core_staged
							value = local_var:cm_ab_lvl
						}
					}
				}
			}
			else_if = {
				limit = { cm_ab_core_watched = yes }
				cm_ab_core_skip_line = { COND = cm_ab_core_room_left ACTION = cm_log_core_no_room }
				cm_ab_core_skip_line = { COND = cm_ab_core_untouched ACTION = cm_log_core_touched }
				cm_ab_core_skip_line = { COND = cm_ab_core_pops_ok ACTION = cm_log_core_pops }
				cm_ab_core_skip_line = { COND = cm_ab_core_type_has_room ACTION = cm_log_core_type_room }
				cm_ab_core_skip_line = { COND = cm_check_location_meets_min_market_access ACTION = cm_log_core_access }
				cm_ab_core_skip_line = { COND = cm_location_has_no_auto_build_chain_predecessor ACTION = cm_log_core_predecessor }
				cm_ab_core_skip_line = { COND = cm_ab_core_not_at_max ACTION = cm_log_core_max_level }
				cm_ab_core_skip_line = { COND = cm_ab_core_placeable ACTION = cm_log_core_place }
				cm_ab_core_skip_line = { COND = cm_ab_core_not_claimed ACTION = cm_log_core_claimed }
			}
		}
	}
	if = {
		limit = {
			local_var:cm_ab_core_seen = 0
			scope:cm_ab_market = { cm_ab_core_market_watched = yes }
		}
		var:cm_ab_core_watch_loc = {
			cm_ab_core_skip_line = { COND = cm_ab_core_owned ACTION = cm_log_core_not_ours }
			cm_ab_core_skip_line = { COND = cm_ab_core_unconquered ACTION = cm_log_core_conquered }
			cm_ab_core_skip_line = { COND = cm_ab_location_has_proximity ACTION = cm_log_core_prox }
			cm_ab_core_skip_line = { COND = cm_ab_core_candidate_ok ACTION = cm_log_core_candidate }
		}
	}
}

# Expands this location's RGO up to the levels staging asked for. A target reached while
# queued drops the item. Root is the location, scope:cm_location set.
# Expects scope:cm_country.
cm_ab_core_drain_rgo = {
	raw_material = { save_temporary_scope_as = cm_ab_good_t }
	market = { save_temporary_scope_as = cm_ab_market }
	if = {
		limit = { NOT = { cm_ab_core_target_met = yes } }
		if = {
			limit = {
				exists = var:cm_ab_core_levels
				cm_is_rgo_not_max_level_already = yes
				# Also the dedup vs the player's RGO drain and Production, which ran earlier
				# this pass and construct immediately.
				cm_is_rgo_under_construction = no
				cm_location_rgo_free_this_cycle = yes
				cm_ab_core_rgo_pops_ok = yes
			}
			if = {
				limit = {
					local_var:cm_ab_cost_blocked = 0
					cm_ab_can_afford_rgo = { POOL = cm_ab_pool_core }
				}
				set_local_variable = { name = cm_ab_core_rgo_guard value = 0 }
				while = {
					limit = {
						local_var:cm_ab_core_rgo_guard < var:cm_ab_core_levels
						cm_get_max_rgo_workers_floor > { value = cm_modifier_rgo_level add = cm_get_rgos_under_construction }
						cm_ab_can_afford_rgo = { POOL = cm_ab_pool_core }
					}
					change_local_variable = { name = cm_ab_core_rgo_guard add = 1 }
					set_local_variable = { name = cm_ab_rgo_cost value = glorpui_rgo_expand_cost }
					construct_rgo_upgrade = { }
					scope:cm_country = { change_variable = { name = cm_ab_pool_core subtract = local_var:cm_ab_rgo_cost } }
				}
				if = {
					limit = { local_var:cm_ab_core_rgo_guard > 0 }
					cm_q_mark_rgo_served = yes
				}
			}
			else_if = {
				# The level is wanted, so the cost is the only thing that failed and the category
				# stops here.
				limit = { local_var:cm_ab_cost_blocked = 0 }
				cm_ab_stop_on_cost = { COST = glorpui_rgo_expand_cost }
			}
		}
	}
}

# Drains one producer type's staged entries across the queue. Root is the country.
# Expects scope:cm_building_type as the resolved tier. $good$ = the type's produced good,
# for the target re-check.
cm_ab_core_drain_type = {
	if = {
		limit = {
			local_var:cm_ab_cost_blocked = 0
			has_variable_list = cm_ab_b_locations
		}
		every_in_list = {
			variable = cm_ab_b_locations
			save_temporary_scope_as = cm_ab_drain_loc
			scope:cm_ab_drain_loc = {
				scope:cm_country = { cm_ab_core_count = { N = cm_ab_core_c_walk V = 1 } }
				if = {
					limit = { cm_ab_core_claimed_by_cat = yes }
					scope:cm_country = { cm_ab_core_count = { N = cm_ab_core_c_cat2 V = 1 } }
				}
				if = {
					limit = { cm_ab_core_has_cost_key = yes }
					scope:cm_country = { cm_ab_core_count = { N = cm_ab_core_c_cost V = 1 } }
				}
				if = {
					limit = {
						has_variable_map = cm_ab_b_cat_2
						is_key_in_variable_map = { name = cm_ab_b_cat_2 target = scope:cm_building_type }
						has_variable_map = cm_ab_b_costs
						is_key_in_variable_map = { name = cm_ab_b_costs target = scope:cm_building_type }
					}
					goods:$good$ = { save_temporary_scope_as = cm_ab_good_t }
					market = { save_temporary_scope_as = cm_ab_market }
					if = {
						limit = { cm_ab_core_target_met = yes }
						# Opportunity gone: drop the entry.
						remove_from_variable_map = { name = cm_ab_b_cat_2 key = scope:cm_building_type }
					} else = {
						scope:cm_country = { cm_ab_core_count = { N = cm_ab_core_c_drain_hit V = 1 } }
						cm_ab_core_approve_build = yes
					}
				}
				else_if = {
					limit = { cm_ab_core_watched = yes }
					cm_ab_core_skip_line = { COND = cm_ab_core_claimed_by_cat ACTION = cm_log_core_drain_cat }
					cm_ab_core_skip_line = { COND = cm_ab_core_has_cost_key ACTION = cm_log_core_drain_cost }
				}
			}
		}
	}
	else_if = {
		limit = {
			NOT = { local_var:cm_ab_cost_blocked = 0 }
			exists = var:cm_ab_core_watch_loc
		}
		cm_dbg_log = { cat = core action = cm_log_core_drain_blocked }
	}
}

# Constructs the levels staging asked for at this location, under the Core Goods pool, and
# consumes the item's queue keys. Modelled on cm_ab_try_approve_build, but building here
# instead of approving into the queue window's fire lists, which hold each type once and so
# cap a location at eight levels a cycle. Every Core Goods producer is priced in gold alone (no
# price block on mason, lumber_mill, sand_pit, the four glass tiers or rural_glassmaker), so
# construct_building charges the whole cost, and it books neither cm_q_planned_spend nor
# cm_q_slots_used because the gold is already gone and num_civil_constructions already counts
# the slot. Root is the location. Expects scope:cm_country and scope:cm_building_type.
cm_ab_core_approve_build = {
	if = {
		limit = { local_var:cm_ab_cost_blocked = 0 }
		save_temporary_scope_value_as = { name = cm_cost value = "variable_map(cm_ab_b_costs|scope:cm_building_type)" }
		# Written beside the cost, so a cost key present means a profit key present.
		save_temporary_scope_value_as = { name = cm_profit value = "variable_map(cm_ab_b_profit|scope:cm_building_type)" }
		if = {
			# Approved by another feature this cycle: consume silently.
			limit = { NOT = { is_target_in_variable_list = { name = cm_q_approved_build_types target = scope:cm_building_type } } }
			if = {
				limit = {
					exists = var:cm_ab_core_levels
					cm_ab_core_profitable_or_subsidized = yes
					cm_ab_can_afford = { POOL = cm_ab_pool_core }
				}
				set_local_variable = { name = cm_ab_core_bld_guard value = 0 }
				# The batch's own count against the engine cap.
				set_local_variable = { name = cm_ab_core_bld_levels value = 0 }
				if = {
					limit = {
						any_buildings_in_location = {
							owner = scope:cm_country
							building_type = scope:cm_building_type
						}
					}
					"building(scope:cm_building_type|scope:cm_country)" = {
						change_local_variable = { name = cm_ab_core_bld_levels add = building_level }
						change_local_variable = { name = cm_ab_core_bld_levels add = building_levels_under_construction }
					}
				}
				while = {
					limit = {
						local_var:cm_ab_core_bld_guard < var:cm_ab_core_levels
						local_var:cm_ab_core_bld_levels < cm_ab_core_type_cap
						cm_ab_can_afford = { POOL = cm_ab_pool_core }
					}
					change_local_variable = { name = cm_ab_core_bld_guard add = 1 }
					change_local_variable = { name = cm_ab_core_bld_levels add = 1 }
					cm_q_construct_build_now = yes
					cm_commit_market_demand = yes
					scope:cm_country = { change_variable = { name = cm_ab_pool_core subtract = scope:cm_cost } }
				}
				scope:cm_country = { cm_ab_core_count = { N = cm_ab_core_c_built V = local_var:cm_ab_core_bld_guard } }
				if = {
					limit = { cm_ab_core_watched = yes }
					save_temporary_scope_as = cmf_log_loc
					scope:cm_country = {
						cm_dbg_log_loc = {
							cat = core
							action = cm_log_core_built
							arg1 = cm_log_core_a_pool
							value = var:cm_ab_pool_core
						}
						cm_dbg_log_value = {
							cat = core
							action = cm_log_core_built_n
							value = local_var:cm_ab_core_bld_guard
						}
					}
				}
			}
			else_if = {
				# The level is wanted and profitable, so the cost is the only thing that failed.
				limit = {
					exists = var:cm_ab_core_levels
					cm_ab_core_profitable_or_subsidized = yes
				}
				cm_ab_stop_on_cost = { COST = scope:cm_cost }
			}
			else_if = {
				limit = {
					exists = var:cm_ab_core_levels
					cm_ab_core_watched = yes
				}
				save_temporary_scope_value_as = { name = cm_ab_core_prof value = cm_bld_profit_value }
				save_temporary_scope_as = cmf_log_loc
				scope:cm_country = {
					cm_dbg_log_loc = {
						cat = core
						action = cm_log_core_profit
						arg1 = cm_log_core_a_profit
						value = scope:cm_ab_core_prof
					}
				}
			}
		}
		remove_from_variable_map = { name = cm_ab_b_cat_2 key = scope:cm_building_type }
	}
}
