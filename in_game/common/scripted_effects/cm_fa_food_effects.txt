# Full-automation Food: samples each market's food supply against demand monthly, once a
# year enables auto-food on the best food-potential locations in markets whose median month
# still fell short, and stages the system-picked locations through the food option engine
# under the food pool. System picks never turn off except when the location stops being a
# rural settlement or leaves the country.

# Samples each market's monthly supply/demand ratio into a 7-bucket histogram (the maps
# cm_fa_food_h1..h7 keyed by market), from which the yearly evaluation reads the median.
# A histogram gives a spike-robust central value that EU5's lack of numeric arrays makes
# hard to compute exactly: the transient 100x-demand bug (and any demand-near-zero month)
# lands in an extreme bucket and never moves the median, and the target is applied at
# evaluation time so it stays tunable. Buckets rebuild each year.
# Root is the country, scope:cm_country set.
cm_fa_food_sample_markets = {
	if = {
		limit = { exists = var:cm_fa_food_enabled }
		every_market_present_in_country = {
			save_scope_as = cm_fa_market
			set_local_variable = { name = cm_fa_supply value = 0 }
			set_local_variable = { name = cm_fa_demand value = 0 }
			# All owners: the food market is shared. food_consumption is a negative
			# contribution, so subtracting it accumulates the demand magnitude.
			every_location_in_market = {
				change_local_variable = { name = cm_fa_supply add = food_production }
				change_local_variable = { name = cm_fa_demand subtract = food_consumption }
			}
			if = {
				limit = { local_var:cm_fa_demand > 0 }
				set_local_variable = { name = cm_fa_ratio value = local_var:cm_fa_supply }
				change_local_variable = { name = cm_fa_ratio divide = local_var:cm_fa_demand }
			} else = {
				# No demand reads as plentiful; the top bucket never lowers the median.
				set_local_variable = { name = cm_fa_ratio value = 999 }
			}
			scope:cm_country = { cm_fa_food_hist_add = yes }
		}
	}
}

# Increments this month's histogram bucket for scope:cm_fa_market by the ratio in
# local_var:cm_fa_ratio. The bucket boundaries (ratio 0.75/1.0/1.25/1.5/2.0/3.0) are the
# same everywhere the median is read. add_to_variable_map overwrites, so each bucket reads
# its count, bumps it, and writes it back; the read is inlined per bucket because a quoted
# variable_map() map name cannot be a macro. Root is the country, expects scope:cm_fa_market.
cm_fa_food_hist_add = {
	if = {
		limit = { local_var:cm_fa_ratio < 0.75 }
		set_local_variable = { name = cm_fa_hc value = 0 }
		if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h1 target = scope:cm_fa_market } }
			set_local_variable = { name = cm_fa_hc value = "variable_map(cm_fa_food_h1|scope:cm_fa_market)" } }
		change_local_variable = { name = cm_fa_hc add = 1 }
		add_to_variable_map = { name = cm_fa_food_h1 key = scope:cm_fa_market value = local_var:cm_fa_hc }
	} else_if = {
		limit = { local_var:cm_fa_ratio < 1.0 }
		set_local_variable = { name = cm_fa_hc value = 0 }
		if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h2 target = scope:cm_fa_market } }
			set_local_variable = { name = cm_fa_hc value = "variable_map(cm_fa_food_h2|scope:cm_fa_market)" } }
		change_local_variable = { name = cm_fa_hc add = 1 }
		add_to_variable_map = { name = cm_fa_food_h2 key = scope:cm_fa_market value = local_var:cm_fa_hc }
	} else_if = {
		limit = { local_var:cm_fa_ratio < 1.25 }
		set_local_variable = { name = cm_fa_hc value = 0 }
		if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h3 target = scope:cm_fa_market } }
			set_local_variable = { name = cm_fa_hc value = "variable_map(cm_fa_food_h3|scope:cm_fa_market)" } }
		change_local_variable = { name = cm_fa_hc add = 1 }
		add_to_variable_map = { name = cm_fa_food_h3 key = scope:cm_fa_market value = local_var:cm_fa_hc }
	} else_if = {
		limit = { local_var:cm_fa_ratio < 1.5 }
		set_local_variable = { name = cm_fa_hc value = 0 }
		if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h4 target = scope:cm_fa_market } }
			set_local_variable = { name = cm_fa_hc value = "variable_map(cm_fa_food_h4|scope:cm_fa_market)" } }
		change_local_variable = { name = cm_fa_hc add = 1 }
		add_to_variable_map = { name = cm_fa_food_h4 key = scope:cm_fa_market value = local_var:cm_fa_hc }
	} else_if = {
		limit = { local_var:cm_fa_ratio < 2.0 }
		set_local_variable = { name = cm_fa_hc value = 0 }
		if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h5 target = scope:cm_fa_market } }
			set_local_variable = { name = cm_fa_hc value = "variable_map(cm_fa_food_h5|scope:cm_fa_market)" } }
		change_local_variable = { name = cm_fa_hc add = 1 }
		add_to_variable_map = { name = cm_fa_food_h5 key = scope:cm_fa_market value = local_var:cm_fa_hc }
	} else_if = {
		limit = { local_var:cm_fa_ratio < 3.0 }
		set_local_variable = { name = cm_fa_hc value = 0 }
		if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h6 target = scope:cm_fa_market } }
			set_local_variable = { name = cm_fa_hc value = "variable_map(cm_fa_food_h6|scope:cm_fa_market)" } }
		change_local_variable = { name = cm_fa_hc add = 1 }
		add_to_variable_map = { name = cm_fa_food_h6 key = scope:cm_fa_market value = local_var:cm_fa_hc }
	} else = {
		set_local_variable = { name = cm_fa_hc value = 0 }
		if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h7 target = scope:cm_fa_market } }
			set_local_variable = { name = cm_fa_hc value = "variable_map(cm_fa_food_h7|scope:cm_fa_market)" } }
		change_local_variable = { name = cm_fa_hc add = 1 }
		add_to_variable_map = { name = cm_fa_food_h7 key = scope:cm_fa_market value = local_var:cm_fa_hc }
	}
}

# Yearly per-market check: where the median food month stayed below the fa_food_target
# surplus ratio, enable auto-food on the top food-potential locations until two active
# locations can still expand, the player's own included. The median bucket is logged for
# every market. Marking only; construction runs through the monthly staging.
# Root is the country.
cm_fa_food_yearly_evaluation = {
	if = {
		limit = {
			cm_fa_master_active = yes
			exists = var:cm_fa_food_enabled
		}
		save_scope_as = cm_country
		set_local_variable = { name = cm_fa_food_thresh value = cm_fa_food_target_ratio }
		cm_fa_food_cleanup = yes
		every_market_present_in_country = {
			save_scope_as = cm_fa_market
			scope:cm_country = { cm_fa_food_compute_median = yes }
			if = {
				limit = { local_var:cm_fa_med_below = 1 }
				set_local_variable = { name = cm_fa_food_n value = 0 }
				every_location_in_market = {
					limit = {
						owner ?= scope:cm_country
						NOT = { integration_level = conquered }
						cm_fa_food_location_effectively_active = yes
						cm_fa_food_location_counts_as_expandable = yes
					}
					change_local_variable = { name = cm_fa_food_n add = 1 }
				}
				if = {
					limit = { local_var:cm_fa_food_n < 2 }
					ordered_location_in_market = {
						order_by = cm_location_food_potential
						max = cm_max_guess_at_number_of_locations
						check_range_bounds = no
						limit = {
							owner ?= scope:cm_country
							NOT = { integration_level = conquered }
							cm_fa_location_has_proximity = yes
							location_rank = location_rank:rural_settlement
							cm_fa_location_has_food_rgo = yes
							NOT = { cm_fa_food_location_effectively_active = yes }
							# The player said no here.
							NOT = { exists = var:cm_auto_food_location_excluded }
							cm_check_location_meets_min_market_access = yes
							cm_location_food_potential >= cm_fa_food_min_potential_value
						}
						save_scope_as = cm_fa_cand
						# The iterator snapshots its limit; the count advances live.
						if = {
							limit = { local_var:cm_fa_food_n < 2 }
							scope:cm_fa_cand = { set_variable = { name = cm_fa_food_marked value = yes } }
							scope:cm_country = { add_to_variable_list = { name = cm_fa_food_locations target = scope:cm_fa_cand } }
							change_local_variable = { name = cm_fa_food_n add = 1 }
						}
					}
				}
			}
		}
		clear_variable_map = cm_fa_food_h1
		clear_variable_map = cm_fa_food_h2
		clear_variable_map = cm_fa_food_h3
		clear_variable_map = cm_fa_food_h4
		clear_variable_map = cm_fa_food_h5
		clear_variable_map = cm_fa_food_h6
		clear_variable_map = cm_fa_food_h7
	}
}

# Reads scope:cm_fa_market's histogram, finds the median bucket (the first whose running
# count reaches half the samples), and sets local_var:cm_fa_med_below to 1 when the median
# bucket's lower bound is under local_var:cm_fa_food_thresh. The lower bound biases a
# mid-bucket target toward building food, the safe direction. Root is the country.
cm_fa_food_compute_median = {
	set_local_variable = { name = cm_fa_c1 value = 0 }
	if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h1 target = scope:cm_fa_market } }
		set_local_variable = { name = cm_fa_c1 value = "variable_map(cm_fa_food_h1|scope:cm_fa_market)" } }
	set_local_variable = { name = cm_fa_c2 value = 0 }
	if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h2 target = scope:cm_fa_market } }
		set_local_variable = { name = cm_fa_c2 value = "variable_map(cm_fa_food_h2|scope:cm_fa_market)" } }
	set_local_variable = { name = cm_fa_c3 value = 0 }
	if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h3 target = scope:cm_fa_market } }
		set_local_variable = { name = cm_fa_c3 value = "variable_map(cm_fa_food_h3|scope:cm_fa_market)" } }
	set_local_variable = { name = cm_fa_c4 value = 0 }
	if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h4 target = scope:cm_fa_market } }
		set_local_variable = { name = cm_fa_c4 value = "variable_map(cm_fa_food_h4|scope:cm_fa_market)" } }
	set_local_variable = { name = cm_fa_c5 value = 0 }
	if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h5 target = scope:cm_fa_market } }
		set_local_variable = { name = cm_fa_c5 value = "variable_map(cm_fa_food_h5|scope:cm_fa_market)" } }
	set_local_variable = { name = cm_fa_c6 value = 0 }
	if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h6 target = scope:cm_fa_market } }
		set_local_variable = { name = cm_fa_c6 value = "variable_map(cm_fa_food_h6|scope:cm_fa_market)" } }
	set_local_variable = { name = cm_fa_c7 value = 0 }
	if = { limit = { is_key_in_variable_map = { name = cm_fa_food_h7 target = scope:cm_fa_market } }
		set_local_variable = { name = cm_fa_c7 value = "variable_map(cm_fa_food_h7|scope:cm_fa_market)" } }
	set_local_variable = { name = cm_fa_tot value = local_var:cm_fa_c1 }
	change_local_variable = { name = cm_fa_tot add = local_var:cm_fa_c2 }
	change_local_variable = { name = cm_fa_tot add = local_var:cm_fa_c3 }
	change_local_variable = { name = cm_fa_tot add = local_var:cm_fa_c4 }
	change_local_variable = { name = cm_fa_tot add = local_var:cm_fa_c5 }
	change_local_variable = { name = cm_fa_tot add = local_var:cm_fa_c6 }
	change_local_variable = { name = cm_fa_tot add = local_var:cm_fa_c7 }
	set_local_variable = { name = cm_fa_med_below value = 0 }
	set_local_variable = { name = cm_fa_med_bucket value = 0 }
	if = {
		limit = { local_var:cm_fa_tot > 0 }
		set_local_variable = { name = cm_fa_half value = local_var:cm_fa_tot }
		change_local_variable = { name = cm_fa_half multiply = 0.5 }
		set_local_variable = { name = cm_fa_cum value = 0 }
		cm_fa_food_median_step = { N = 1 LB = 0 }
		cm_fa_food_median_step = { N = 2 LB = 0.75 }
		cm_fa_food_median_step = { N = 3 LB = 1.0 }
		cm_fa_food_median_step = { N = 4 LB = 1.25 }
		cm_fa_food_median_step = { N = 5 LB = 1.5 }
		cm_fa_food_median_step = { N = 6 LB = 2.0 }
		cm_fa_food_median_step = { N = 7 LB = 3.0 }
		if = {
			limit = { local_var:cm_fa_med_lb < local_var:cm_fa_food_thresh }
			set_local_variable = { name = cm_fa_med_below value = 1 }
		}
	}
}

# One cumulative step of the median walk: adds bucket $N$'s count, and if the running count
# first reaches the halfway mark here, records this as the median bucket with lower bound
# $LB$. Root is the country.
cm_fa_food_median_step = {
	change_local_variable = { name = cm_fa_cum add = local_var:cm_fa_c$N$ }
	if = {
		limit = {
			local_var:cm_fa_med_bucket = 0
			local_var:cm_fa_cum >= local_var:cm_fa_half
		}
		set_local_variable = { name = cm_fa_med_bucket value = $N$ }
		set_local_variable = { name = cm_fa_med_lb value = $LB$ }
	}
}

# Drops entries that stopped being valid: lost ownership (the marker stays, inert without
# list membership) or left rural rank. Never removes for any other reason.
# Root is the country.
cm_fa_food_cleanup = {
	save_scope_as = cm_fa_chk_country
	every_in_list = {
		variable = cm_fa_food_locations
		save_scope_as = cm_fa_chk
		if = {
			limit = { scope:cm_fa_chk = { NOT = { owner = scope:cm_fa_chk_country } } }
			scope:cm_fa_chk_country = { remove_list_variable = { name = cm_fa_food_locations target = scope:cm_fa_chk } }
		} else_if = {
			limit = { scope:cm_fa_chk = { NOT = { location_rank = location_rank:rural_settlement } } }
			scope:cm_fa_chk = { remove_variable = cm_fa_food_marked }
			scope:cm_fa_chk_country = { remove_list_variable = { name = cm_fa_food_locations target = scope:cm_fa_chk } }
		}
	}
}

# Stages every system-picked location through the full-automation food flow under the food
# pool, independent of the player's own auto-food feature: a location the player flow also
# covers is deduped at staging (the earlier-running player queue wins), so it is never
# built twice. Root is the country, scope:cm_country set.
cm_fa_run_food = {
	cm_fa_food_cleanup = yes
	every_in_list = {
		variable = cm_fa_food_locations
		save_scope_as = cm_location
		if = {
			limit = {
				scope:cm_location = {
					owner = scope:cm_country
					NOT = { integration_level = conquered }
					location_rank = location_rank:rural_settlement
				}
			}
			scope:cm_location = { raw_material = { save_scope_as = cm_food_good } }
			scope:cm_location = { cm_fa_food_stage_in_location = yes }
		}
	}
}

# Full-automation mirror of cm_food_auto_build_in_location: the same marginal comparison
# over the five food options, with the gold and discount gates dropped; affordability
# runs only at the drain.
# Root is the location. Expects scopes: country, location, food_good.
cm_fa_food_stage_in_location = {
	if = { limit = { cm_get_open_build_queue_slots > 0 }
	if = { limit = { cm_check_location_meets_min_market_access = yes }
	if = { limit = { NOT = { cm_food_location_pops_still_filling = yes } }

		set_local_variable = { name = cm_food_best value = 0 }
		set_local_variable = { name = cm_food_action value = 0 }

		cm_food_consider_option = { CAN = cm_food_can_build_rgo MARGINAL = cm_food_marginal_rgo ACTION = 1 }
		cm_food_consider_option = { CAN = cm_food_can_build_farming_village MARGINAL = cm_food_marginal_farming_village ACTION = 2 }
		cm_food_consider_option = { CAN = cm_food_can_build_windmill MARGINAL = cm_food_rank_windmill ACTION = 3 }
		cm_food_consider_option = { CAN = cm_food_can_build_irrigation MARGINAL = cm_food_rank_irrigation ACTION = 4 }
		cm_food_consider_option = { CAN = cm_food_can_build_fishing_village MARGINAL = cm_food_marginal_fishing_village ACTION = 5 }

		if = {
			limit = { local_var:cm_food_action = 1 }
			cm_fa_stage_food_rgo = yes
		}
		cm_fa_food_stage_building_option = { ACTION = 2 BT = building_type:farming_village }
		cm_fa_food_stage_building_option = { ACTION = 3 BT = building_type:windmill }
		cm_fa_food_stage_building_option = { ACTION = 4 BT = building_type:irrigation_systems }
		cm_fa_food_stage_building_option = { ACTION = 5 BT = building_type:fishing_village }

	} # end pops check
	} # end market access
	} # end build queue
}

# Root is the location. Skips a winner some player queue already stages here this pulse.
cm_fa_food_stage_building_option = {
	if = {
		limit = { local_var:cm_food_action = $ACTION$ }
		$BT$ = { save_scope_as = cm_building_type }
		if = {
			limit = {
				scope:cm_location = {
					NOT = { is_target_in_variable_list = { name = cm_q_food_building_types target = scope:cm_building_type } }
					NOT = { is_target_in_variable_list = { name = cm_q_building_types target = scope:cm_building_type } }
					NOT = { is_target_in_variable_list = { name = cm_q_ungated_building_types target = scope:cm_building_type } }
				}
			}
			cm_fa_stage_build_candidate = { cat = 3 }
		}
	}
}

# Adds the location to the drain-only RGO list; no GUI scoring, so the staged counter is
# untouched. Root is the location. Expects scope:cm_country, scope:cm_location.
cm_fa_stage_food_rgo = {
	scope:cm_country = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = cm_fa_food_rgo_locations target = scope:cm_location } } }
			add_to_variable_list = { name = cm_fa_food_rgo_locations target = scope:cm_location }
		}
	}
}

# Food drain: staged food buildings in list order, then RGO levels, one level per location
# per cycle. Root is the country.
cm_fa_pass_food = {
	if = {
		limit = {
			local_var:cm_fa_cost_blocked = 0
			has_variable_list = cm_fa_b_locations
		}
		every_in_list = {
			variable = cm_fa_b_locations
			save_temporary_scope_as = cm_fa_drain_loc
			scope:cm_fa_drain_loc = {
				every_in_list = {
					variable = cm_fa_b_types
					save_scope_as = cm_building_type
					scope:cm_fa_drain_loc = {
						cm_fa_try_approve_build = { cat = 3 POOL = cm_fa_pool_food GOODS_GATE = cm_market_has_construction_goods_for_building SLOTS_GATE = cm_fa_location_has_open_build_slot }
					}
				}
			}
		}
	}
	if = {
		limit = {
			local_var:cm_fa_cost_blocked = 0
			has_variable_list = cm_fa_food_rgo_locations
		}
		every_in_list = {
			variable = cm_fa_food_rgo_locations
			save_temporary_scope_as = cm_location
			scope:cm_location = {
				if = {
					limit = {
						local_var:cm_fa_cost_blocked = 0
						cm_get_open_build_queue_slots > 0
						cm_is_rgo_not_max_level_already = yes
						cm_is_rgo_under_construction = no
						cm_location_rgo_promotion_buffer_slack >= 0
						cm_market_has_construction_goods_for_rgo = yes
						cm_fa_can_afford_rgo = { POOL = cm_fa_pool_food }
					}
					set_local_variable = { name = cm_fa_food_rgo_guard value = 0 }
					while = {
						limit = {
							# 99 stays under the engine's 1000-iteration while abort; the construction
							# room below is what ends the batch.
							local_var:cm_fa_food_rgo_guard < 99
							cm_get_open_build_queue_slots > 0
							cm_get_max_rgo_workers_floor > { value = cm_modifier_rgo_level add = cm_get_rgos_under_construction }
							cm_fa_can_afford_rgo = { POOL = cm_fa_pool_food }
						}
						change_local_variable = { name = cm_fa_food_rgo_guard add = 1 }
						set_local_variable = { name = cm_fa_rgo_cost value = glorpui_rgo_expand_cost }
						construct_rgo_upgrade = { }
						scope:cm_country = { change_variable = { name = cm_fa_pool_food subtract = local_var:cm_fa_rgo_cost } }
					}
				}
				else_if = {
					# The room is there and the level is wanted, so the cost is the only thing that
					# failed and the category stops here.
					limit = {
						local_var:cm_fa_cost_blocked = 0
						cm_get_open_build_queue_slots > 0
						cm_is_rgo_not_max_level_already = yes
						cm_is_rgo_under_construction = no
						cm_location_rgo_promotion_buffer_slack >= 0
						cm_market_has_construction_goods_for_rgo = yes
					}
					set_local_variable = { name = cm_fa_cost_blocked value = 1 }
				}
			}
		}
	}
}
