# Full-automation Price Collapsar: builds producers of masonry, lumber, sand, and glass
# to push their market prices down to the target price. Materials are staged and drained
# in the player's list order; within a material, RGO expansions come before producer
# buildings. Each market is sized in producer levels, enough to close the gap between what
# it produces of a material and what it demands, plus the overshoot; the best-producing
# locations take those levels first, several at a time, and levels already paid for but not
# yet producing come off the count. Only markets whose center the country owns are worked,
# so the price it collapses is one it holds. Collapsar approvals skip the market
# construction-goods gates: a shortage of those goods is exactly what the builds cure.

# Root is the country, scope:cm_country set.
cm_fa_run_collapsar = {
	every_market_center_in_country = {
		limit = { cm_fa_market_in_slice = yes }
		save_scope_as = cm_fa_market
		scope:cm_country = { cm_fa_stage_collapsar_market = yes }
	}
}

# Stages one market's candidates, each material under its own level count.
# Root is the country, expects scope:cm_fa_market.
cm_fa_stage_collapsar_market = {
	if = {
		limit = { has_variable_list = cm_fa_collapsar_goods_enabled }
		every_in_list = {
			variable = cm_fa_collapsar_goods_enabled
			save_scope_as = cm_fa_collapsar_item
			scope:cm_country = { cm_fa_stage_collapsar_good = yes }
		}
	}
}

# Console diagnostic: the capital market's masonry numbers, run on demand rather than from the
# cycle so it costs the 200-entry log nothing during play.
# effect c:<TAG> = { cm_fa_collapsar_probe_masonry = yes }
# target_price fails by returning a number rather than by erroring, so reading it out is the only
# way to know it works. Amounts and prices are logged x100 because the log prints whole numbers.
# Root is the country.
cm_fa_collapsar_probe_masonry = {
	save_scope_as = cm_country
	capital.market = { save_scope_as = cm_fa_market }
	goods:masonry = { save_temporary_scope_as = cm_fa_good_t }
	building_type:mason = { save_scope_as = cm_building_type }
	set_local_variable = { name = cm_fa_dbg value = { value = "scope:cm_fa_market.goods_supply_in_market(scope:cm_fa_good_t)" multiply = 100 } }
	cmf_log_value = { action = cm_log_cs_supply value = local_var:cm_fa_dbg }
	set_local_variable = { name = cm_fa_dbg value = { value = scope:cm_fa_market.produced_in_market:masonry multiply = 100 } }
	cmf_log_value = { action = cm_log_cs_produced value = local_var:cm_fa_dbg }
	set_local_variable = { name = cm_fa_dbg value = { value = "scope:cm_fa_market.goods_demand_in_market(scope:cm_fa_good_t)" multiply = 100 } }
	cmf_log_value = { action = cm_log_cs_demand value = local_var:cm_fa_dbg }
	set_local_variable = { name = cm_fa_dbg value = { value = "scope:cm_fa_market.market_price(scope:cm_fa_good_t)" multiply = 100 } }
	cmf_log_value = { action = cm_log_cs_price value = local_var:cm_fa_dbg }
	set_local_variable = { name = cm_fa_dbg value = { value = "scope:cm_fa_market.target_price(scope:cm_fa_good_t)" multiply = 100 } }
	cmf_log_value = { action = cm_log_cs_target value = local_var:cm_fa_dbg }
	set_local_variable = { name = cm_fa_dbg value = { value = cm_fa_collapsar_goal_price multiply = 100 } }
	cmf_log_value = { action = cm_log_cs_goal value = local_var:cm_fa_dbg }
	set_local_variable = { name = cm_fa_dbg value = scope:cm_fa_market.cm_fa_collapsar_levels_wanted }
	cmf_log_value = { action = cm_log_cs_wanted value = local_var:cm_fa_dbg }
	set_local_variable = { name = cm_fa_dbg value = capital.cm_fa_collapsar_type_room }
	cmf_log_value = { action = cm_log_cs_room value = local_var:cm_fa_dbg }
}

# Sets local_var:cm_fa_collapsar_want to the producer levels scope:cm_fa_collapsar_item's
# shortfall needs in scope:cm_fa_market, and 0 once its goal price is reached. Leaves
# scope:cm_building_type on the material's lead producer, whose per-level output is what turns
# a goods shortfall into a level count. Root is the country, expects scope:cm_fa_market and
# scope:cm_fa_collapsar_item.
cm_fa_collapsar_size_good = {
	set_local_variable = { name = cm_fa_collapsar_want value = 0 }
	scope:cm_fa_collapsar_item = { save_temporary_scope_as = cm_fa_good_t }
	if = {
		limit = { NOT = { cm_fa_collapsar_target_met = yes } }
		if = {
			limit = { scope:cm_fa_collapsar_item = goods:masonry }
			building_type:mason = { save_scope_as = cm_building_type }
		}
		else_if = {
			limit = { scope:cm_fa_collapsar_item = goods:lumber }
			building_type:lumber_mill = { save_scope_as = cm_building_type }
		}
		else_if = {
			limit = { scope:cm_fa_collapsar_item = goods:sand }
			building_type:sand_pit = { save_scope_as = cm_building_type }
		}
		else_if = {
			limit = { scope:cm_fa_collapsar_item = goods:glass }
			building_type:glass_guild = { save_scope_as = cm_building_type }
			cm_auto_build_resolve_building_type = yes
		}
		set_local_variable = { name = cm_fa_collapsar_want value = scope:cm_fa_market.cm_fa_collapsar_levels_wanted }
	}
}

# Stages one material's candidates under its own level count, which is 0 once its goal price
# is reached. Root is the country, expects scope:cm_fa_market and scope:cm_fa_collapsar_item.
cm_fa_stage_collapsar_good = {
	cm_fa_collapsar_size_good = yes
	set_local_variable = { name = cm_fa_good_room value = local_var:cm_fa_collapsar_want }
	if = {
		limit = { local_var:cm_fa_good_room > 0 }
		if = {
			limit = { scope:cm_fa_collapsar_item = goods:masonry }
			cm_fa_stage_collapsar_building = { bt = mason }
		}
		else_if = {
			limit = { scope:cm_fa_collapsar_item = goods:lumber }
			goods:lumber = { save_scope_as = cm_fa_good }
			cm_fa_stage_collapsar_rgos = yes
			cm_fa_stage_collapsar_building = { bt = lumber_mill }
		}
		else_if = {
			limit = { scope:cm_fa_collapsar_item = goods:sand }
			goods:sand = { save_scope_as = cm_fa_good }
			cm_fa_stage_collapsar_rgos = yes
			cm_fa_stage_collapsar_building = { bt = sand_pit }
		}
		else_if = {
			limit = { scope:cm_fa_collapsar_item = goods:glass }
			cm_fa_stage_collapsar_building = { bt = glass_guild }
			cm_fa_stage_collapsar_building = { bt = rural_glassmaker }
		}
	}
}

# Stages RGO expansions for scope:cm_fa_good across the country's locations in
# scope:cm_fa_market, in any order, while levels remain. The drain-only list bypasses GUI
# scoring: the cost is script-side. An RGO level counts as one producer level against the
# material's count, which is an approximation: an RGO level's output is location-specific and
# has no per-type constant to read. Root is the country.
cm_fa_stage_collapsar_rgos = {
	# Iterating the market rather than the country matters here: this runs once per material
	# per market. The iterator carries no ownership filter of its own.
	scope:cm_fa_market = {
		every_location_in_market = {
			limit = {
				owner ?= scope:cm_country
				NOT = { integration_level = conquered }
				cm_fa_location_has_proximity = yes
				cm_fa_collapsar_candidate_ok = yes
				raw_material ?= scope:cm_fa_good
				cm_is_rgo_not_max_level_already = yes
				cm_is_rgo_under_construction = no
				cm_fa_collapsar_rgo_pops_ok = yes
				NOT = { exists = var:cm_fa_collapsar_touched }
				# The player's RGO auto-expand staged it this cycle.
				NOT = { scope:cm_country = { is_target_in_variable_list = { name = cm_q_rgo_locations target = prev } } }
			}
			save_scope_as = cm_location
			if = {
				limit = {
					local_var:cm_fa_good_room > 0
					cm_fa_collapsar_rgo_room > 0
				}
				set_local_variable = { name = cm_fa_lvl value = cm_fa_collapsar_rgo_room }
				change_local_variable = { name = cm_fa_lvl min = local_var:cm_fa_good_room }
				scope:cm_country = {
					if = {
						limit = { NOT = { is_target_in_variable_list = { name = cm_fa_collapsar_rgo_locations target = scope:cm_location } } }
						add_to_variable_list = { name = cm_fa_collapsar_rgo_locations target = scope:cm_location }
					}
				}
				scope:cm_location = {
					set_variable = { name = cm_fa_collapsar_touched value = yes }
					set_variable = { name = cm_fa_collapsar_levels value = local_var:cm_fa_lvl }
				}
				change_local_variable = { name = cm_fa_good_room subtract = local_var:cm_fa_lvl }
			}
		}
	}
}

# Stages one producer type across scope:cm_fa_market while levels remain, best-producing
# location first. Chain bases resolve to the current tier (glass_guild through glass_mill);
# non-chain types pass through unchanged. Levels already paid for but not yet producing come
# off the count wherever they stand, staged this cycle or not.
# $bt$ = the producer's base type key. Root is the country.
cm_fa_stage_collapsar_building = {
	building_type:$bt$ = { save_scope_as = cm_building_type }
	cm_auto_build_resolve_building_type = yes
	scope:cm_fa_market = {
		ordered_location_in_market = {
			order_by = cm_fa_collapsar_efficiency_order
			limit = {
				owner ?= scope:cm_country
				NOT = { integration_level = conquered }
				cm_fa_location_has_proximity = yes
				cm_fa_collapsar_candidate_ok = yes
			}
			save_scope_as = cm_location
			change_local_variable = { name = cm_fa_good_room subtract = cm_fa_collapsar_type_inflight }
			if = {
				limit = {
					local_var:cm_fa_good_room > 0
					NOT = { exists = var:cm_fa_collapsar_touched }
					cm_location_allowed_to_build = { target = scope:cm_building_type }
					cm_location_building_type_is_not_at_max_level = { target = scope:cm_building_type }
					# A standing older tier would turn this build into its upgrade.
					cm_location_has_no_auto_build_chain_predecessor = yes
					cm_fa_collapsar_pops_ok = yes
					cm_check_location_meets_min_market_access = yes
					cm_fa_collapsar_type_room > 0
					NOT = { is_target_in_variable_list = { name = cm_q_building_types target = scope:cm_building_type } }
					NOT = { is_target_in_variable_list = { name = cm_q_ungated_building_types target = scope:cm_building_type } }
					NOT = { is_target_in_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_building_type } }
				}
				set_local_variable = { name = cm_fa_lvl value = cm_fa_collapsar_type_room }
				change_local_variable = { name = cm_fa_lvl min = local_var:cm_fa_good_room }
				cm_fa_stage_build_candidate = { cat = 2 }
				scope:cm_location = {
					set_variable = { name = cm_fa_collapsar_touched value = yes }
					set_variable = { name = cm_fa_collapsar_levels value = local_var:cm_fa_lvl }
				}
				change_local_variable = { name = cm_fa_good_room subtract = local_var:cm_fa_lvl }
			}
		}
	}
}

# Collapsar drain, in the player's material order; within a material, RGO levels first,
# then producer buildings, the glass chain's current tier before rural_glassmaker.
# Root is the country.
cm_fa_pass_collapsar = {
	if = {
		limit = { has_variable_list = cm_fa_collapsar_goods_enabled }
		every_in_list = {
			variable = cm_fa_collapsar_goods_enabled
			save_scope_as = cm_fa_collapsar_item
			# An iterator limit cannot see what its own body sets, so the stop is tested inside.
			if = {
				limit = { local_var:cm_fa_cost_blocked = 0 }
				scope:cm_country = { cm_fa_drain_collapsar_good = yes }
			}
		}
	}
}

# Drains one material's staged entries. Root is the country, expects
# scope:cm_fa_collapsar_item.
cm_fa_drain_collapsar_good = {
	if = {
		limit = { scope:cm_fa_collapsar_item = goods:masonry }
		building_type:mason = { save_scope_as = cm_building_type }
		cm_fa_collapsar_drain_type = { good = masonry }
	}
	else_if = {
		limit = { scope:cm_fa_collapsar_item = goods:lumber }
		cm_fa_collapsar_drain_rgos = { good = lumber }
		building_type:lumber_mill = { save_scope_as = cm_building_type }
		cm_fa_collapsar_drain_type = { good = lumber }
	}
	else_if = {
		limit = { scope:cm_fa_collapsar_item = goods:sand }
		cm_fa_collapsar_drain_rgos = { good = sand }
		building_type:sand_pit = { save_scope_as = cm_building_type }
		cm_fa_collapsar_drain_type = { good = sand }
	}
	else_if = {
		limit = { scope:cm_fa_collapsar_item = goods:glass }
		building_type:glass_guild = { save_scope_as = cm_building_type }
		cm_auto_build_resolve_building_type = yes
		cm_fa_collapsar_drain_type = { good = glass }
		building_type:rural_glassmaker = { save_scope_as = cm_building_type }
		cm_fa_collapsar_drain_type = { good = glass }
	}
}

# Drains one good's staged RGO expansions in list order. $good$ = the RGO good key.
# Root is the country.
cm_fa_collapsar_drain_rgos = {
	if = {
		limit = {
			local_var:cm_fa_cost_blocked = 0
			has_variable_list = cm_fa_collapsar_rgo_locations
		}
		every_in_list = {
			variable = cm_fa_collapsar_rgo_locations
			limit = { raw_material ?= goods:$good$ }
			save_temporary_scope_as = cm_location
			scope:cm_location = { cm_fa_collapsar_drain_rgo = yes }
		}
	}
}

# Expands this location's RGO up to the levels staging asked for. A target reached while
# queued drops the item. Root is the location, scope:cm_location set.
# Expects scope:cm_country.
cm_fa_collapsar_drain_rgo = {
	raw_material = { save_temporary_scope_as = cm_fa_good_t }
	market = { save_temporary_scope_as = cm_fa_market }
	if = {
		limit = { NOT = { cm_fa_collapsar_target_met = yes } }
		if = {
			limit = {
				exists = var:cm_fa_collapsar_levels
				cm_is_rgo_not_max_level_already = yes
				# Also the dedup vs the player's RGO drain and Industry, which ran earlier
				# this pass and construct immediately.
				cm_is_rgo_under_construction = no
				cm_fa_collapsar_rgo_pops_ok = yes
			}
			if = {
				limit = {
					local_var:cm_fa_cost_blocked = 0
					cm_fa_can_afford_rgo = { POOL = cm_fa_pool_collapsar }
				}
				set_local_variable = { name = cm_fa_collapsar_rgo_guard value = 0 }
				while = {
					limit = {
						local_var:cm_fa_collapsar_rgo_guard < var:cm_fa_collapsar_levels
						cm_get_max_rgo_workers_floor > { value = cm_modifier_rgo_level add = cm_get_rgos_under_construction }
						cm_fa_can_afford_rgo = { POOL = cm_fa_pool_collapsar }
					}
					change_local_variable = { name = cm_fa_collapsar_rgo_guard add = 1 }
					set_local_variable = { name = cm_fa_rgo_cost value = glorpui_rgo_expand_cost }
					construct_rgo_upgrade = { }
					scope:cm_country = { change_variable = { name = cm_fa_pool_collapsar subtract = local_var:cm_fa_rgo_cost } }
				}
			}
			else_if = {
				# The level is wanted, so the cost is the only thing that failed and the category
				# stops here.
				limit = { local_var:cm_fa_cost_blocked = 0 }
				set_local_variable = { name = cm_fa_cost_blocked value = 1 }
			}
		}
	}
}

# Drains one producer type's staged entries across the queue. Root is the country.
# Expects scope:cm_building_type as the resolved tier. $good$ = the type's produced good,
# for the target re-check.
cm_fa_collapsar_drain_type = {
	if = {
		limit = {
			local_var:cm_fa_cost_blocked = 0
			has_variable_list = cm_fa_b_locations
		}
		every_in_list = {
			variable = cm_fa_b_locations
			save_temporary_scope_as = cm_fa_drain_loc
			scope:cm_fa_drain_loc = {
				if = {
					limit = {
						trigger_if = {
							limit = {
								has_variable_map = cm_fa_b_cats
								is_key_in_variable_map = { name = cm_fa_b_cats target = scope:cm_building_type }
							}
							"variable_map(cm_fa_b_cats|scope:cm_building_type)" = 2
						}
						trigger_else = { always = no }
						has_variable_map = cm_fa_b_costs
						is_key_in_variable_map = { name = cm_fa_b_costs target = scope:cm_building_type }
					}
					goods:$good$ = { save_temporary_scope_as = cm_fa_good_t }
					market = { save_temporary_scope_as = cm_fa_market }
					if = {
						limit = { cm_fa_collapsar_target_met = yes }
						# Opportunity gone: drop the entry.
						remove_from_variable_map = { name = cm_fa_b_costs key = scope:cm_building_type }
						remove_from_variable_map = { name = cm_fa_b_cats key = scope:cm_building_type }
					} else = {
						cm_fa_collapsar_approve_build = yes
					}
				}
			}
		}
	}
}

# Constructs the levels staging asked for at this location, under the collapsar pool, and
# consumes the item's queue keys. Modelled on cm_fa_try_approve_build, but building here
# instead of approving into the queue window's fire lists, which hold each type once and so
# cap a location at eight levels a cycle. Every collapsar producer is priced in gold alone (no
# price block on mason, lumber_mill, sand_pit, the four glass tiers or rural_glassmaker), so
# construct_building charges the whole cost, and it books neither cm_q_planned_spend nor
# cm_q_slots_used because the gold is already gone and num_civil_constructions already counts
# the slot. Root is the location. Expects scope:cm_country and scope:cm_building_type.
cm_fa_collapsar_approve_build = {
	if = {
		limit = { local_var:cm_fa_cost_blocked = 0 }
		save_temporary_scope_value_as = { name = cm_cost value = "variable_map(cm_fa_b_costs|scope:cm_building_type)" }
		if = {
			# Approved by another feature this cycle: consume silently.
			limit = { NOT = { is_target_in_variable_list = { name = cm_q_approved_build_types target = scope:cm_building_type } } }
			if = {
				limit = {
					exists = var:cm_fa_collapsar_levels
					cm_fa_can_afford = { POOL = cm_fa_pool_collapsar }
				}
				set_local_variable = { name = cm_fa_collapsar_bld_guard value = 0 }
				# The batch's own count against the engine cap.
				set_local_variable = { name = cm_fa_collapsar_bld_levels value = 0 }
				if = {
					limit = {
						any_buildings_in_location = {
							owner = scope:cm_country
							building_type = scope:cm_building_type
						}
					}
					"building(scope:cm_building_type|scope:cm_country)" = {
						change_local_variable = { name = cm_fa_collapsar_bld_levels add = building_level }
						change_local_variable = { name = cm_fa_collapsar_bld_levels add = building_levels_under_construction }
					}
				}
				while = {
					limit = {
						local_var:cm_fa_collapsar_bld_guard < var:cm_fa_collapsar_levels
						OR = {
							building_type_max_level = { building_type = scope:cm_building_type value <= 0 }
							building_type_max_level = { building_type = scope:cm_building_type value > local_var:cm_fa_collapsar_bld_levels }
						}
						cm_fa_can_afford = { POOL = cm_fa_pool_collapsar }
					}
					change_local_variable = { name = cm_fa_collapsar_bld_guard add = 1 }
					change_local_variable = { name = cm_fa_collapsar_bld_levels add = 1 }
					cm_q_construct_build_now = yes
					cm_commit_market_demand = yes
					scope:cm_country = { change_variable = { name = cm_fa_pool_collapsar subtract = scope:cm_cost } }
				}
			}
			else_if = {
				# The level is wanted, so the cost is the only thing that failed.
				limit = { exists = var:cm_fa_collapsar_levels }
				set_local_variable = { name = cm_fa_cost_blocked value = 1 }
			}
		}
		remove_from_variable_map = { name = cm_fa_b_costs key = scope:cm_building_type }
		remove_from_variable_map = { name = cm_fa_b_cats key = scope:cm_building_type }
	}
}
