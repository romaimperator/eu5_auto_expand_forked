# The lucky-nation observer bridge. A scripted GUI Execute is a synced scripted_gui_command and is
# dropped on a client with no country of its own, so with no player country cm_ln_queue_window.gui
# reaches script through the console instead. Console effect blocks do not run in multiplayer or
# ironman, so the bridge is single-player only and its drivers are gated accordingly.
#
# The console runs one command at a time and refuses any command issued while another is running,
# so every item appends its call to a per-country GUI buffer and the sweep's terminal state submits
# the whole buffer as one command.
#
# A console command is parsed when it runs, costing a fixed 3,376 log lines whatever it names plus
# a term for very large named trees. Both bite: a command per item wrote 11.9 million lines across
# two cycles, and naming the approval pass added about a million per command on top. So the
# per-item effects below store their arguments and nothing else, cm_ln_c_process does the scoring
# once per cycle, and cm_ln_c_sync_check reaches the approval pass through an on_action, whose leaf
# is resolved from the loaded database instead of being re-instantiated.
#
# Every effect here is entered from that command, so root is the country. Values cross as integers
# scaled by 1000, because GUI has no CFixedPoint to string function.

cm_ln_c_rows_clear = {
	set_global_variable = { name = cm_ln_r_n value = 0 }
	clear_global_variable_map = cm_ln_r_kind
	clear_global_variable_map = cm_ln_r_ctry
	clear_global_variable_map = cm_ln_r_l
	clear_global_variable_map = cm_ln_r_b
	clear_global_variable_map = cm_ln_r_nb
	clear_global_variable_map = cm_ln_r_c
	clear_global_variable_map = cm_ln_r_i
	clear_global_variable_map = cm_ln_r_p
	clear_global_variable_map = cm_ln_r_t
	clear_global_variable_map = cm_ln_r_r
}

# Args: Y => cm_ln_ctry_id, L => cm_ln_loc_id, B => cm_ln_bt_id, C => cost, I => income,
# P => profit, T => location total income, R => location RGO profit per level.
# The country crosses as an id like the other objects, because the tag the command is scoped to is
# a display tag and names a different country for the Great Yuan.
cm_ln_c_row_b = {
	change_global_variable = { name = cm_ln_r_n add = 1 }
	add_to_global_variable_map = { name = cm_ln_r_kind key = global_var:cm_ln_r_n value = 1 }
	add_to_global_variable_map = { name = cm_ln_r_ctry key = global_var:cm_ln_r_n value = $Y$ }
	add_to_global_variable_map = { name = cm_ln_r_l key = global_var:cm_ln_r_n value = $L$ }
	add_to_global_variable_map = { name = cm_ln_r_b key = global_var:cm_ln_r_n value = $B$ }
	add_to_global_variable_map = { name = cm_ln_r_c key = global_var:cm_ln_r_n value = $C$ }
	add_to_global_variable_map = { name = cm_ln_r_i key = global_var:cm_ln_r_n value = $I$ }
	add_to_global_variable_map = { name = cm_ln_r_p key = global_var:cm_ln_r_n value = $P$ }
	add_to_global_variable_map = { name = cm_ln_r_t key = global_var:cm_ln_r_n value = $T$ }
	add_to_global_variable_map = { name = cm_ln_r_r key = global_var:cm_ln_r_n value = $R$ }
}

# Reads one stored number, scaled back down. Expects scope:cm_ln_k. One effect per map because a
# $PARAM$ inside a quoted lookup never substitutes, and the local is read in a trigger as well
# because a read inside the lookup does not count as a use.
cm_ln_c_read_cost = {
	set_local_variable = { name = cm_ln_c_v value = "global_variable_map(cm_ln_r_c|scope:cm_ln_k)" }
	if = {
		limit = { local_var:cm_ln_c_v = local_var:cm_ln_c_v }
		change_local_variable = { name = cm_ln_c_v divide = 1000 }
		save_temporary_scope_value_as = { name = cm_cost value = local_var:cm_ln_c_v }
	}
}

# Expects scope:cm_ln_k. The sweep queues every country's rows before the one command that scores
# them, so a pass walks several countries' rows and cm_ln_r_ctry is what tells them apart.
# cm_ln_c_own reports whether THIS row resolved a country, which a nonzero id does not: a country
# annexed between the sweep and the process command leaves the lookup empty, and the temporary scope
# whichever row resolved before it left behind is still standing.
cm_ln_c_row_common = {
	set_local_variable = { name = cm_ln_c_y value = "global_variable_map(cm_ln_r_ctry|scope:cm_ln_k)" }
	set_local_variable = { name = cm_ln_c_own value = 0 }
	if = {
		limit = { local_var:cm_ln_c_y > 0 }
		"global_variable_map(cm_ln_ctry_of_id|local_var:cm_ln_c_y)" = {
			save_temporary_scope_as = cm_country
			set_local_variable = { name = cm_ln_c_own value = 1 }
		}
	}
	set_local_variable = { name = cm_ln_c_l value = "global_variable_map(cm_ln_r_l|scope:cm_ln_k)" }
}

# Scores every row the sweep delivered, then clears them. One command names this per cycle, so the
# scoring tree is instantiated once instead of once per item.
cm_ln_c_process = {
	if = {
		limit = { has_global_variable_map = cm_ln_r_kind }
		every_key_in_global_variable_map = {
			variable = cm_ln_r_kind
			save_scope_as = cm_ln_k
			cm_ln_c_row_common = yes
			if = {
				limit = {
					exists = global_var:cm_dbg_core
					local_var:cm_ln_c_own = 1
				}
				scope:cm_country = { cm_ab_core_count = { N = cm_ab_core_c_rows V = 1 } }
				if = {
					limit = { "global_variable_map(cm_ln_r_kind|scope:cm_ln_k)" = 1 }
					scope:cm_country = { cm_ab_core_count = { N = cm_ab_core_c_rows_b V = 1 } }
				}
			}
			if = {
				limit = {
					local_var:cm_ln_c_own = 1
					local_var:cm_ln_c_l > 0
				}
				"global_variable_map(cm_ln_loc_of_id|local_var:cm_ln_c_l)" = {
					save_temporary_scope_as = cm_indirect_loc
					save_temporary_scope_value_as = { name = cm_cur_market_access value = market_access }
					# RGO rows carry no building type, and an absent key leaves the local holding
					# the previous row's value.
					set_local_variable = { name = cm_ln_c_b value = 0 }
					if = {
						limit = { is_key_in_global_variable_map = { name = cm_ln_r_b target = scope:cm_ln_k } }
						set_local_variable = { name = cm_ln_c_b value = "global_variable_map(cm_ln_r_b|scope:cm_ln_k)" }
					}
					if = {
						limit = { local_var:cm_ln_c_b > 0 }
						"global_variable_map(cm_ln_bt_of_id|local_var:cm_ln_c_b)" = { save_temporary_scope_as = cm_building_type }
					}
					# 1 build, 2 upgrade, 3 RGO, 4 urbanize.
					if = {
						limit = { "global_variable_map(cm_ln_r_kind|scope:cm_ln_k)" = 1 }
						cm_ln_c_read_cost = yes
						cm_ln_c_read_income = yes
						cm_ln_c_read_profit = yes
						cm_ln_c_read_total_income = yes
						cm_ln_c_read_rgo_profit = yes
						# scope:cm_obsolete_building is left unset: resolving it needs a per-location
						# building sweep the played path gets free from the engine. The body skips its
						# conversion-clash branch, so a build the engine turns into an upgrade is not
						# recorded and can be staged again next cycle.
						cm_ab_score_build_apply = yes
					}
					else_if = {
						limit = { "global_variable_map(cm_ln_r_kind|scope:cm_ln_k)" = 2 }
						if = {
							limit = { exists = scope:cm_building_type }
							scope:cm_building_type = { save_temporary_scope_as = cm_old_building_type }
						}
						# The replacement crosses as its own id rather than being resolved from
						# cm_building_type_to_upgrade_type. Staging decides obsolescence with the live engine
						# trigger, so a chain unlocked after classification is obsolete while absent from that
						# map, and the scope would keep the previous row's type.
						set_local_variable = { name = cm_ln_c_nb value = 0 }
						if = {
							limit = { is_key_in_global_variable_map = { name = cm_ln_r_nb target = scope:cm_ln_k } }
							set_local_variable = { name = cm_ln_c_nb value = "global_variable_map(cm_ln_r_nb|scope:cm_ln_k)" }
						}
						if = {
							limit = { local_var:cm_ln_c_nb > 0 }
							"global_variable_map(cm_ln_bt_of_id|local_var:cm_ln_c_nb)" = {
								save_temporary_scope_as = cm_new_building_type
								save_temporary_scope_as = cm_building_type
							}
						}
						cm_ln_c_read_cost = yes
						cm_ln_c_read_income = yes
						cm_ln_c_read_total_income = yes
						cm_ln_c_read_rgo_profit = yes
						cm_ab_score_upgrade_apply = yes
					}
					else_if = {
						limit = { "global_variable_map(cm_ln_r_kind|scope:cm_ln_k)" = 3 }
						save_temporary_scope_as = cm_location
						cm_ln_c_read_income = yes
						cm_ab_score_rgo_apply = yes
					}
					else = {
						cm_ln_c_read_cost = yes
						cm_q_urbanize_apply = yes
					}
				}
			}
		}
		cm_ln_c_rows_clear = yes
	}
}

# Ends the sweep's one arm, replacing the dropped cm_ln_scan_disarm Execute. Without it the gate
# stays true and _show never gets another rising edge, so the sweep fires once and stops.
cm_ln_c_scan_disarm = {
	if = {
		limit = { has_global_variable = cm_ln_scan_armed }
		remove_global_variable = cm_ln_scan_armed
	}
}

# Args: Y => cm_ln_ctry_id.
cm_ln_c_sync_check = {
	set_local_variable = { name = cm_ln_c_y value = $Y$ }
	if = {
		limit = { local_var:cm_ln_c_y > 0 }
		"global_variable_map(cm_ln_ctry_of_id|local_var:cm_ln_c_y)" = {
			trigger_event_silently = { on_action = cm_ln_c_on_sync_check }
		}
	}
}

# Args: Y => cm_ln_ctry_id.
cm_ln_c_cu_end = {
	set_local_variable = { name = cm_ln_c_y value = $Y$ }
	if = {
		limit = { local_var:cm_ln_c_y > 0 }
		"global_variable_map(cm_ln_ctry_of_id|local_var:cm_ln_c_y)" = {
			save_scope_as = cm_ln_country
			cm_ln_cu_end_apply = yes
		}
	}
}
