# Auto-Build budget engine: monthly per-category pools set from the treasury above the reserve,
# and the per-category dispatch that runs each drain. Every category gets its own share, then
# whatever none of them spent is offered again from the top of the player's category order, so
# the highest category that still wants gold takes it. A category buys its best pick whenever the
# treasury covers it, and what it spent past its own allotment becomes a debt it settles out of
# the allotments that follow. Pools and debts are country variables and persist in saves.

# Sets each enabled category's pool to its share of split = max(0, gold - reserve). A debt holds
# no gold, so it takes nothing off the top. Root is the country.
cm_ab_monthly_budget_refresh = {
	if = {
		# A reorganization holds the treasury until every governor its plan calls for stands, so no
		# category gets a pool this month. Urban Rights spends government power and Building Cleanup
		# spends nothing, so both keep running.
		limit = { exists = var:cm_gov_hold }
		cm_ab_zero_pools = yes
	}
	else = {
		set_local_variable = { name = cm_ab_split value = gold }
		change_local_variable = { name = cm_ab_split subtract = cm_get_minimum_gold }
		# Held allotments come off the top, so no other category is handed gold one of them is
		# saving for.
		change_local_variable = { name = cm_ab_split subtract = cm_ab_reserved_total }
		change_local_variable = { name = cm_ab_split max = 0 }
		cm_ab_set_category_pool = { cat = production }
		cm_ab_set_category_pool = { cat = core }
		cm_ab_set_category_pool = { cat = food }
		cm_ab_set_category_pool = { cat = custom }
		cm_ab_set_category_pool = { cat = roads }
		cm_ab_set_category_pool = { cat = military }
	}
}

# pool = pct * split, plus whatever the category held back last cycle.
# $cat$ = production | core | food | custom | roads | military. Root is the country.
cm_ab_set_category_pool = {
	if = {
		limit = { exists = var:cm_ab_$cat$_enabled }
		set_local_variable = { name = cm_ab_inc value = local_var:cm_ab_split }
		change_local_variable = { name = cm_ab_inc multiply = var:cm_ab_$cat$_budget }
		change_local_variable = { name = cm_ab_inc multiply = 0.01 }
		if = {
			# The hold is handed back whole and re-made by the drain if the pick is still out of
			# reach, so it grows month on month until the pick is bought.
			limit = { exists = var:cm_ab_resv_$cat$ }
			change_local_variable = { name = cm_ab_inc add = var:cm_ab_resv_$cat$ }
			remove_variable = cm_ab_resv_$cat$
		}
		set_variable = { name = cm_ab_pool_$cat$ value = local_var:cm_ab_inc }
	}
}

# pool = min(pool, pct * the live treasury above the reserve). Called on budget-slider change.
# $cat$ = category id. Root is the country.
cm_ab_clamp_pool = {
	if = {
		limit = { has_variable = cm_ab_pool_$cat$ }
		set_local_variable = { name = cm_ab_cap value = gold }
		change_local_variable = { name = cm_ab_cap subtract = cm_get_minimum_gold }
		change_local_variable = { name = cm_ab_cap max = 0 }
		change_local_variable = { name = cm_ab_cap multiply = var:cm_ab_$cat$_budget }
		change_local_variable = { name = cm_ab_cap multiply = 0.01 }
		change_variable = { name = cm_ab_pool_$cat$ min = local_var:cm_ab_cap }
	}
}

# Root is the country.
# The slice is not cleared here: it bounds the player's own priority features too, so turning
# Auto-Build off must not restart their rotation.
cm_ab_clear_budget_state = {
	cm_ab_zero_pools = yes
	cm_ab_clear_debts = yes
	cm_ab_clear_reservations = yes
	cm_ab_prod_clear_memory = yes
	cm_ab_rd_clear_plan = yes
	remove_variable = cm_ab_surplus_last
}

# Root is the country.
cm_ab_zero_pools = {
	set_variable = { name = cm_ab_pool_production value = 0 }
	set_variable = { name = cm_ab_pool_core value = 0 }
	set_variable = { name = cm_ab_pool_food value = 0 }
	set_variable = { name = cm_ab_pool_custom value = 0 }
	set_variable = { name = cm_ab_pool_roads value = 0 }
	set_variable = { name = cm_ab_pool_military value = 0 }
}

# Writes off what every category owes. Root is the country.
cm_ab_clear_debts = {
	remove_variable = cm_ab_debt_production
	remove_variable = cm_ab_debt_core
	remove_variable = cm_ab_debt_food
	remove_variable = cm_ab_debt_custom
	remove_variable = cm_ab_debt_roads
	remove_variable = cm_ab_debt_military
}

# Releases every held allotment back into the split. Root is the country.
cm_ab_clear_reservations = {
	remove_variable = cm_ab_resv_production
	remove_variable = cm_ab_resv_core
	remove_variable = cm_ab_resv_food
	remove_variable = cm_ab_resv_custom
	remove_variable = cm_ab_resv_roads
	remove_variable = cm_ab_resv_military
}

	set_local_variable = { name = cm_ab_dbg_total value = 0 }
	cm_ab_log_balance = { cat = food ACTION = cm_log_ab_bal_food }
	cm_ab_log_balance = { cat = core ACTION = cm_log_ab_bal_core }
	cm_ab_log_balance = { cat = roads ACTION = cm_log_ab_bal_roads }
	cm_ab_log_balance = { cat = production ACTION = cm_log_ab_bal_production }
	cm_ab_log_balance = { cat = custom ACTION = cm_log_ab_bal_custom }
	cm_ab_log_balance = { cat = military ACTION = cm_log_ab_bal_military }
	cmf_log_decimal_value = { action = cm_log_ab_bal_total value = local_var:cm_ab_dbg_total }
	# What the re-offer could not place: gold every category declined, which no balance shows.
	set_local_variable = { name = cm_ab_dbg value = 0 }
	if = {
		limit = { exists = var:cm_ab_surplus_last }
		set_local_variable = { name = cm_ab_dbg value = var:cm_ab_surplus_last }
	}
	cmf_log_decimal_value = { action = cm_log_ab_surplus value = local_var:cm_ab_dbg }
	if = {
		limit = { exists = var:cm_ab_production_min_roi }
		cmf_log_value = { action = cm_log_ab_prod_cap value = var:cm_ab_production_min_roi }
	}
	save_temporary_scope_as = cm_country
	cm_ab_prod_off_slice_best = yes
	if = {
		limit = { local_var:cm_ab_off_best > -2000000 }
		change_local_variable = { name = cm_ab_off_best multiply = -1 }
	}
	else = {
		set_local_variable = { name = cm_ab_off_best value = 0 }
	}
	cmf_log_decimal_value = { action = cm_log_ab_off_best value = local_var:cm_ab_off_best }
	cmf_log_decimal_value = { action = cm_log_ab_off_cost value = local_var:cm_ab_off_cost }
}

# One category's balance, positive where it is holding an allotment and negative where it owes;
# a category can never be doing both, since owing is what stops it before it can hold.
# $cat$ = category id, $ACTION$ = its log key. Root is the country, accumulates the probe's
# local cm_ab_dbg_total.
cm_ab_log_balance = {
	set_local_variable = { name = cm_ab_dbg value = 0 }
	if = {
		limit = { exists = var:cm_ab_resv_$cat$ }
		set_local_variable = { name = cm_ab_dbg value = var:cm_ab_resv_$cat$ }
	}
	if = {
		limit = { exists = var:cm_ab_debt_$cat$ }
		change_local_variable = { name = cm_ab_dbg subtract = var:cm_ab_debt_$cat$ }
	}
	change_local_variable = { name = cm_ab_dbg_total add = local_var:cm_ab_dbg }
	cmf_log_decimal_value = { action = $ACTION$ value = local_var:cm_ab_dbg }
}

# $v$ = variable name. Root is the new country, expects scope:old_country.
cm_ab_copy_budget_var = {

		limit = { scope:old_country = { exists = var:$v$ } }
		set_variable = { name = $v$ value = scope:old_country.var:$v$ }
	}
}

# Approves one staged build at this location under its category's pool,
# then consumes its keys; a key missing from cm_ab_b_costs means the engine blocked the
# build at scoring and the item is skipped. Callers iterate their own category's entries.
# $cat$ = category code, $POOL$ = pool variable, $GOODS_GATE$ = the market
# construction-goods gate trigger, $SLOTS_GATE$ = the open-construction-slot gate trigger.
# Root is the location. Expects scope:cm_country, scope:cm_building_type.
cm_ab_try_approve_build = {
	if = {
		limit = {
			# A category that stopped on cost approves nothing else this cycle.
			# cm_ab_no_hold lifts both halves for the fort line.
			OR = {
				local_var:cm_ab_cost_blocked = 0
				local_var:cm_ab_no_hold = 1
			}
			has_variable_map = cm_ab_b_cat_$cat$
			is_key_in_variable_map = { name = cm_ab_b_cat_$cat$ target = scope:cm_building_type }
			has_variable_map = cm_ab_b_costs
			is_key_in_variable_map = { name = cm_ab_b_costs target = scope:cm_building_type }
		}
		save_temporary_scope_value_as = { name = cm_cost value = "variable_map(cm_ab_b_costs|scope:cm_building_type)" }
		market = { save_temporary_scope_as = cm_market }
		if = {
			# Approved by another feature this cycle: consume silently.
			limit = { NOT = { is_target_in_variable_list = { name = cm_q_approved_build_types target = scope:cm_building_type } } }
			if = {
				limit = {
					$SLOTS_GATE$ = yes
					$GOODS_GATE$ = yes
					cm_ab_can_afford = { POOL = $POOL$ }
				}
				cm_approve_gui_build = yes
				scope:cm_country = { change_variable = { name = $POOL$ subtract = scope:cm_cost } }
				if = {
					limit = { cm_ab_type_is_single_level = no }
					cm_ab_fill_build_slots = { POOL = $POOL$ }
				}
			}
			else_if = {
				# The two gates pass here, so the cost is the only thing that failed.
				limit = {
					$SLOTS_GATE$ = yes
					$GOODS_GATE$ = yes
					local_var:cm_ab_no_hold = 0
				}
				cm_ab_stop_on_cost = { COST = scope:cm_cost }
			}
		}
		remove_from_variable_map = { name = cm_ab_b_cat_$cat$ key = scope:cm_building_type }
	}
}

# Auto-Build entry to cm_q_fill_build_slots: the batch draws its category pool down in a
# scratch variable so one affordability trigger serves every category, then writes the balance
# back. $POOL$ = the category pool variable name. Root is the location.
# Expects scope:cm_country, scope:cm_building_type, scope:cm_cost.
cm_ab_fill_build_slots = {
	scope:cm_country = { set_variable = { name = cm_ab_copy_pool value = var:$POOL$ } }
	cm_q_fill_build_slots = { AFFORD = cm_ab_copy_can_afford CLAIM = cm_ab_claim_build_copy }
	scope:cm_country = {
		set_variable = { name = $POOL$ value = var:cm_ab_copy_pool }
		remove_variable = cm_ab_copy_pool
	}
}

# Per-level claim for an Auto-Build batch: the shared claim plus the pool draw.
# Root is the location. Expects scope:cm_country, scope:cm_cost.
cm_ab_claim_build_copy = {
	cm_q_claim_build_copy = yes
	scope:cm_country = { change_variable = { name = cm_ab_copy_pool subtract = scope:cm_cost } }
}

# Runs one category's drain under its pool. $WRAP$ = cm_ab_drain_one for the share walk and
# cm_ab_drain_surplus for the re-offer. Root is the country, expects scope:cm_ab_cat.
cm_ab_drain_category = {
	switch = {
		trigger = scope:cm_ab_cat
		flag:cm_ab_cat_production = { $WRAP$ = { CAT = production PASS = cm_ab_pass_production } }
		flag:cm_ab_cat_core = { $WRAP$ = { CAT = core PASS = cm_ab_pass_core } }
		flag:cm_ab_cat_food = { $WRAP$ = { CAT = food PASS = cm_ab_pass_food } }
		flag:cm_ab_cat_custom = { $WRAP$ = { CAT = custom PASS = cm_ab_pass_custom } }
		flag:cm_ab_cat_roads = { $WRAP$ = { CAT = roads PASS = cm_ab_pass_roads } }
		# Military runs a ledger it can owe against only once its manpower chain has a tier it can
		# build. Until then the category is the fort line alone, which its own share covers, so it
		# takes no part in the re-offer.
		flag:cm_ab_cat_military = {
			if = {
				limit = { exists = var:cm_ln_bk_active }
				$WRAP$ = { CAT = military PASS = cm_ab_pass_military }
			}
			else_if = {
				limit = { local_var:cm_ab_reoffer = 0 }
				cm_ab_pass_military = yes
			}
		}
	}
}

# Clears the two signals cm_ab_drain_one leaves for the re-offer, so neither is read unset.
# $CAT$ = category id. Root is the country.
cm_ab_init_drain_signals = {
	set_local_variable = { name = cm_ab_hungry_$CAT$ value = 0 }
	set_local_variable = { name = cm_ab_own_$CAT$ value = 0 }
}

# The share walk: settles what the category owes out of its own allotment, then spends what is
# left of it, or stands the category down for the cycle while it still owes. Each category answers
# only for its own stop, so the signal is cleared per category rather than once for the loop.
# $CAT$ = category id, $PASS$ = its drain effect. Root is the country.
cm_ab_drain_one = {
	set_local_variable = { name = cm_ab_cost_blocked value = 0 }
	set_local_variable = { name = cm_ab_hold_cost value = 0 }
	if = {
		limit = { NOT = { has_variable = cm_ab_pool_$CAT$ } }
		set_variable = { name = cm_ab_pool_$CAT$ value = 0 }
	}
	set_local_variable = { name = cm_ab_paid value = 0 }
	if = {
		limit = { exists = var:cm_ab_debt_$CAT$ }
		set_local_variable = { name = cm_ab_paid value = var:cm_ab_pool_$CAT$ }
		change_local_variable = { name = cm_ab_paid min = var:cm_ab_debt_$CAT$ }
		change_variable = { name = cm_ab_debt_$CAT$ subtract = local_var:cm_ab_paid }
		change_variable = { name = cm_ab_pool_$CAT$ subtract = local_var:cm_ab_paid }
		if = {
			limit = { var:cm_ab_debt_$CAT$ <= 0 }
			remove_variable = cm_ab_debt_$CAT$
		}
	}
	# A debt is gold last month already spent, so the allotment that settles it is not spent now
	# and goes to the leftover either way.
	change_variable = { name = cm_ab_surplus add = local_var:cm_ab_paid }
	if = {
		# Still owing, so it buys nothing and is not offered the leftover.
		limit = { NOT = { exists = var:cm_ab_debt_$CAT$ } }
		# Caps the hold, and is read again by the re-offer, so it is named per category.
		set_local_variable = { name = cm_ab_own_$CAT$ value = var:cm_ab_pool_$CAT$ }
		set_local_variable = { name = cm_ab_ledger value = 1 }
		$PASS$ = yes
		set_local_variable = { name = cm_ab_ledger value = 0 }
		cm_ab_settle_pool = { CAT = $CAT$ }
	}
	set_variable = { name = cm_ab_pool_$CAT$ value = 0 }
}

# The re-offer: hands the leftover to a category that stopped on a pick it could not pay for.
# The walk is in list order, so the highest such category takes as much of it as it can use.
# Its own hold is handed back first, the leftover being able to cover the pick outright.
# $CAT$ = category id, $PASS$ = its drain effect. Root is the country.
cm_ab_drain_surplus = {
	if = {
		limit = {
			local_var:cm_ab_hungry_$CAT$ = 1
			var:cm_ab_surplus > 0
		}
		set_local_variable = { name = cm_ab_cost_blocked value = 0 }
		set_local_variable = { name = cm_ab_hold_cost value = 0 }
		set_variable = { name = cm_ab_pool_$CAT$ value = var:cm_ab_surplus }
		set_variable = { name = cm_ab_surplus value = 0 }
		if = {
			limit = { exists = var:cm_ab_resv_$CAT$ }
			change_variable = { name = cm_ab_pool_$CAT$ add = var:cm_ab_resv_$CAT$ }
			remove_variable = cm_ab_resv_$CAT$
		}
		set_local_variable = { name = cm_ab_ledger value = 1 }
		$PASS$ = yes
		set_local_variable = { name = cm_ab_ledger value = 0 }
		cm_ab_settle_pool = { CAT = $CAT$ }
		set_variable = { name = cm_ab_pool_$CAT$ value = 0 }
	}
}

# What the pool holds once a drain has run: an overspend becomes a debt, a pick it could not
# reach holds an allotment sized to that pick, and the rest goes to the leftover.
# $CAT$ = category id. Root is the country.
cm_ab_settle_pool = {
	if = {
		# It bought past its allotment, and the overspend is what it now owes.
		limit = { var:cm_ab_pool_$CAT$ < 0 }
		set_variable = { name = cm_ab_debt_$CAT$ value = 0 }
		change_variable = { name = cm_ab_debt_$CAT$ subtract = var:cm_ab_pool_$CAT$ }
	}
	else = {
		if = {
			limit = { local_var:cm_ab_cost_blocked = 1 }
			set_local_variable = { name = cm_ab_hungry_$CAT$ value = 1 }
		}
		if = {
			# The hold stays capped at its own allotment: leftover gold can be spent this month
			# but never banked, or a low category would hoard what the whole order is handed.
			limit = {
				local_var:cm_ab_cost_blocked = 1
				local_var:cm_ab_hold_cost > 0
				local_var:cm_ab_own_$CAT$ > 0
				var:cm_ab_pool_$CAT$ > 0
			}
			set_local_variable = { name = cm_ab_keep value = var:cm_ab_pool_$CAT$ }
			change_local_variable = { name = cm_ab_keep min = local_var:cm_ab_hold_cost }
			change_local_variable = { name = cm_ab_keep min = local_var:cm_ab_own_$CAT$ }
			set_variable = { name = cm_ab_resv_$CAT$ value = local_var:cm_ab_keep }
			change_variable = { name = cm_ab_pool_$CAT$ subtract = local_var:cm_ab_keep }
		}
		change_variable = { name = cm_ab_surplus add = var:cm_ab_pool_$CAT$ }
	}
}

# Stands the category down on a pick it could not pay for, recording what that pick costs so
# cm_ab_settle_pool holds an allotment sized to it. $COST$ = the pick's cost.
cm_ab_stop_on_cost = {
	set_local_variable = { name = cm_ab_cost_blocked value = 1 }
	set_local_variable = { name = cm_ab_hold_cost value = $COST$ }
}

# Runs one category's staging. Root is the country, expects scope:cm_ab_cat.
cm_ab_stage_category = {
	switch = {
		trigger = scope:cm_ab_cat
		flag:cm_ab_cat_production = { cm_ab_run_production = yes }
		flag:cm_ab_cat_core = { cm_ab_run_core = yes }
		flag:cm_ab_cat_food = { cm_ab_run_food = yes }
		flag:cm_ab_cat_custom = { cm_ab_run_custom = yes }
		flag:cm_ab_cat_military = { cm_ln_mil_run = yes }
	}
}
