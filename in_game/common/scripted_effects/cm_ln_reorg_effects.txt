# Lucky Nations capital and governor reorganization: every twenty-five years, every five years
# while a governor slot stands empty, once income passes the starting threshold, or whenever a new
# governor slot opens, a lucky nation re-picks its capital and plans a full set of governor sites,
# then urbanizes the ones below city rank, builds roads out to them and places the governors. It
# supersedes the full-automation Governors category for these countries.
#
# The plan is built one search per month so no single tick carries a whole placement finder
# run, and each search treats the sites already committed to as proximity sources, so site
# two is chosen knowing site one will exist.
#
# Both searches hold their pick un-committed for a couple of months and re-run before acting on
# it, because a location can score far above the field for a single month and be ranked poorly
# again afterwards. A site that does not survive the re-check is dropped for the rest of the run
# and the next best takes its place.

# 25 years, the reorganization cadence.
@cm_ln_cooldown_days = 9125
# 5 years, the shorter cadence for a country holding a governor slot it has not filled.
@cm_ln_gov_cooldown_days = 1825
# Months a reorganization may run before it is abandoned and replanned from scratch. Under the
# twenty-five-year cadence, so a stuck run cannot outlive the period it belongs to.
@cm_ln_reorg_max_months = 300
# A capital scoring below this share of the best location is worth moving. Scores are
# normalised to 0-1000 against the best, and the gap is hysteresis against churn.
@cm_ln_cap_keep = 900
# Monthly income that opens a reorganization before the cadence date is reached.
@cm_ln_income_start = 100
# A capital scoring below this share of the best is worth the opening move.
@cm_ln_open_cap_keep = 700
# Months a pick waits before it is scored again.
@cm_ln_probe_months = 2
# Share of its first score a pick must still reach on the re-check to be kept.
@cm_ln_probe_keep = 700
# Months with no progress the opening move may wait on its preconditions before it is given up, so
# a country whose search never becomes possible does not stay shut out of construction. A month the
# river harvest advanced through does not count, so the wait is only spent on a harvest that has
# stopped, which is what an observer run leaves behind.
@cm_ln_open_max_wait = 2

# Root is the country, scope:cm_country set.
cm_ln_reorg_tick = {
	if = {
		limit = { NOT = { exists = var:cm_ln_phase } }
		set_variable = { name = cm_ln_phase value = 0 }
	}
	if = {
		limit = { var:cm_ln_phase = 0 }
		cm_ln_reorg_try_start = yes
	}
	if = {
		limit = { var:cm_ln_phase > 0 }
		change_variable = { name = cm_ln_age add = 1 }
	}
	remove_variable = cm_ln_hold
	# The urbanize and road orders serve both the placement and the relocation phase, and they run
	# whatever the slot and credit counts are, so they sit ahead of the switch rather than inside
	# either step. Placement claims gold after them.
	if = {
		limit = {
			var:cm_ln_phase >= 5
			var:cm_ln_phase <= 6
		}
		set_variable = { name = cm_ln_tick_spent value = 0 }
		cm_ln_prune_pending = yes
		cm_ln_apply_plan_drops = yes
		cm_ln_road_pass = yes
		cm_ln_rank_pass = yes
	}
	switch = {
		trigger = var:cm_ln_phase
		1 = { cm_ln_reorg_capital_search = yes }
		2 = { cm_ln_reorg_capital_move = yes }
		3 = { cm_ln_reorg_plan_step = yes }
		4 = { cm_ln_reorg_classify = yes }
		5 = { cm_ln_reorg_place_step = yes }
		6 = { cm_ln_reorg_move_step = yes }
		7 = { cm_ln_reorg_confirm_step = yes }
		8 = { cm_ln_reorg_teardown = yes }
	}
	remove_variable = cm_ln_tick_spent
	if = {
		limit = {
			var:cm_ln_phase > 0
			var:cm_ln_age >= @cm_ln_reorg_max_months
		}
		cm_ln_reorg_teardown = yes
	}
	# Nothing else may spend a coin while a planned governor is still to be ordered, and nothing is
	# owed while one is merely building, which is why this reads cm_ln_pending rather than the phase
	# alone. It sits after the switch so a site the confirm step just handed back is covered this
	# month, and the whole tick runs ahead of the budget refresh and the construction cycle.
	if = {
		limit = {
			var:cm_ln_phase >= 5
			var:cm_ln_phase <= 6
			has_variable_list = cm_ln_pending
			variable_list_size = { name = cm_ln_pending value >= 1 }
		}
		set_variable = { name = cm_ln_hold value = 1 }
	}
}

# Root is the country, scope:cm_country set.
cm_ln_reorg_try_start = {
	if = {
		limit = {
			is_multiplayer_session = no
			at_war = no
			in_civil_war = no
			exists = capital
			cm_pf_river_data_is_current = yes
			NOT = { has_global_variable = cm_ln_token }
			cm_ln_no_human_search_in_flight = yes
			OR = {
				AND = {
					OR = {
						# The twenty-fifth year of the grand campaign, which starts 1337.4.1
						# (START_DATE, loading_screen/common/defines/00_defines.txt:2).
						current_date >= 1362.1.1
						monthly_income_total >= @cm_ln_income_start
					}
					OR = {
						NOT = { exists = var:cm_ln_cooldown }
						AND = {
							NOT = { exists = var:cm_ln_gov_cooldown }
							cm_ln_gov_slot_open = yes
						}
					}
				}
				cm_ln_gov_slots_grew = yes
			}
		}
		cm_ln_reorg_begin = yes
	}
}

# Root is the country, scope:cm_country set.
cm_ln_reorg_begin = {
	set_global_variable = { name = cm_ln_token value = 1 }
	set_variable = { name = cm_ln_cooldown value = 1 days = @cm_ln_cooldown_days }
	set_variable = { name = cm_ln_gov_cooldown value = 1 days = @cm_ln_gov_cooldown_days }
	set_variable = { name = cm_ln_age value = 0 }
	set_variable = { name = cm_ln_free_credits value = 0 }
	set_variable = { name = cm_ln_plan_n value = 0 }
	remove_variable = cm_ln_plan_exhausted
	clear_variable_list = cm_ln_plan
	clear_variable_list = cm_ln_plan_drop
	clear_variable_list = cm_ln_pending
	clear_variable_list = cm_ln_moves
	clear_variable_list = cm_ln_pre_seats
	cm_ln_clear_road_state = yes
	cm_ln_probe_end = yes
	cm_ln_probe_clear_rejects = yes
	# Governors is superseded while a reorganization owns the placements.
	remove_variable = cm_fa_gov_armed
	remove_variable = cm_fa_gov_loc
	remove_variable = cm_fa_gov_naval
	every_owned_location = {
		limit = { has_variable = cm_ln_plan_ord }
		remove_variable = cm_ln_plan_ord
		remove_variable = cm_ln_plan_naval
		remove_variable = cm_ln_rank_wait
		remove_variable = cm_ln_place_tries
		remove_variable = cm_ln_paid
	}
	# The capital move's remove_if can destroy seats outright, and a seat lost that way is
	# ours to replace at no cost.
	every_owned_location = {
		limit = { cm_location_has_governor = yes }
		save_scope_as = cm_ln_seat
		scope:cm_country = { add_to_variable_list = { name = cm_ln_pre_seats target = scope:cm_ln_seat } }
	}
	cm_ln_pf_settings_on = yes
	set_variable = { name = cm_ln_phase value = 1 }
}

# The finder reads these as ordinary country variables, and a country the player tag-switched
# out of carries their values, so all four are written rather than left to their defaults.
# Root is the country.
cm_ln_pf_settings_on = {
	set_variable = { name = cm_placement_finder_accuracy value = 5 } # Fastest
	set_variable = { name = cm_placement_finder_naval value = 1 } # Current presence
	set_variable = { name = cm_placement_finder_roads value = 2 } # Max Roads
	set_variable = { name = cm_placement_finder_method value = 1 } # Tax Base
}

# Root is the country.
cm_ln_pf_settings_off = {
	remove_variable = cm_placement_finder_accuracy
	remove_variable = cm_placement_finder_naval
	remove_variable = cm_placement_finder_roads
	remove_variable = cm_placement_finder_method
}

# --- The pick probation ---
#
# One slot serves the capital search and the governor plan, which never hold a pick at the
# same time.

# Root is the country. Expects scope:cm_ln_probe_pick.
cm_ln_probe_start = {
	set_variable = { name = cm_ln_probe_site value = scope:cm_ln_probe_pick }
	set_variable = { name = cm_ln_probe_score value = scope:cm_ln_probe_pick.var:$SCORE$ }
	set_variable = { name = cm_ln_probe_wait value = @cm_ln_probe_months }
}

# The pick's fresh score out of the 1000 its first one was normalised to. A site the run left
# unscored and one that changed hands, which keeps the earlier run's score, both read as zero.
# Root is the country, scope:cm_country set. Leaves scope:cm_ln_probe_l and scope:cm_ln_probe_now.
cm_ln_probe_share = {
	var:cm_ln_probe_site = { save_temporary_scope_as = cm_ln_probe_l }
	if = {
		limit = {
			scope:cm_ln_probe_l = {
				owner ?= scope:cm_country
				has_variable = $SCORE$
			}
		}
		save_temporary_scope_value_as = {
			name = cm_ln_probe_now
			value = {
				value = scope:cm_ln_probe_l.var:$SCORE$
				multiply = 1000 # Scores are 0 to 1000 against the run's best: cm_pf_finalize.
				divide = {
					value = var:cm_ln_probe_score
					min = 0.001
				}
			}
		}
	}
	else = {
		save_temporary_scope_value_as = { name = cm_ln_probe_now value = 0 }
	}
}

# Root is the country.
cm_ln_probe_end = {
	remove_variable = cm_ln_probe_site
	remove_variable = cm_ln_probe_score
	remove_variable = cm_ln_probe_wait
}

# Root is the country, scope:cm_country set. Expects scope:cm_ln_probe_l and scope:cm_ln_probe_now.
cm_ln_probe_drop = {
	if = {
		limit = { scope:cm_ln_probe_l = { owner ?= scope:cm_country } }
		scope:cm_ln_probe_l = { set_variable = { name = cm_ln_probe_reject value = 1 } }
	}
	cmf_log_value = { action = cm_ln_log_probe_dropped value = scope:cm_ln_probe_now }
	cm_ln_probe_end = yes
}

# Root is the country.
cm_ln_probe_clear_rejects = {
	every_owned_location = {
		limit = { has_variable = cm_ln_probe_reject }
		remove_variable = cm_ln_probe_reject
	}
}

# Root is the country, scope:cm_country set.
cm_ln_reorg_capital_search = {
	if = {
		limit = { exists = var:cm_ln_probe_site }
		if = {
			limit = { var:cm_ln_probe_wait > 0 }
			change_variable = { name = cm_ln_probe_wait subtract = 1 }
		}
		if = {
			limit = {
				var:cm_ln_probe_wait <= 0
				cm_ln_no_human_search_in_flight = yes
				cm_pf_river_data_is_current = yes
			}
			cm_ln_reorg_cap_search_run = yes
			cm_ln_reorg_cap_probe_check = yes
		}
	}
	else_if = {
		limit = {
			cm_ln_no_human_search_in_flight = yes
			cm_pf_river_data_is_current = yes
		}
		cm_ln_reorg_cap_search_run = yes
		cm_ln_reorg_cap_probe_open = yes
	}
}

# Root is the country, scope:cm_country set.
cm_ln_reorg_cap_search_run = {
	set_variable = { name = cm_pf_view_c value = 1 }
	remove_variable = cm_pf_view_g
	cm_pf_prep_core = yes
	cm_pf_run_tiered_sync = yes
	remove_variable = cm_pf_view_c
}

# Root is the country, scope:cm_country set.
cm_ln_reorg_cap_probe_open = {
	set_local_variable = { name = cm_ln_picked value = 0 }
	ordered_owned_location = {
		order_by = { value = var:cm_cf_score }
		position = 0
		check_range_bounds = no
		limit = {
			has_variable = cm_cf_score
			NOT = { integration_level = conquered }
			NOT = { has_variable = cm_ln_probe_reject }
		}
		save_scope_as = cm_ln_probe_pick
		set_local_variable = { name = cm_ln_picked value = 1 }
		scope:cm_country = { cm_ln_probe_start = { SCORE = cm_cf_score } }
	}
	if = {
		limit = { local_var:cm_ln_picked = 0 }
		set_variable = { name = cm_ln_phase value = 2 }
	}
}

# Root is the country, scope:cm_country set.
cm_ln_reorg_cap_probe_check = {
	cm_ln_probe_share = { SCORE = cm_cf_score }
	if = {
		limit = { scope:cm_ln_probe_now >= @cm_ln_probe_keep }
		set_variable = { name = cm_ln_cap_target value = var:cm_ln_probe_site }
		cmf_log_value = { action = cm_ln_log_probe_kept value = scope:cm_ln_probe_now }
		cm_ln_probe_end = yes
		set_variable = { name = cm_ln_phase value = 2 }
	}
	else = {
		cm_ln_probe_drop = yes
		cm_ln_reorg_cap_probe_open = yes
	}
}

# The move ends the month: set_capital shifts every location's proximity, the country's
# topography impact and has_road_to_capital everywhere, and the governor plan reads all three.
# Root is the country, scope:cm_country set.
cm_ln_reorg_capital_move = {
	if = {
		limit = {
			exists = var:cm_ln_cap_target
			var:cm_ln_cap_target = { owner = scope:cm_country }
			NOT = { var:cm_ln_cap_target = capital }
			capital = {
				trigger_if = {
					limit = { has_variable = cm_cf_score }
					var:cm_cf_score < @cm_ln_cap_keep
				}
				trigger_else = { always = yes }
			}
		}
		set_capital = var:cm_ln_cap_target
		capital = { cm_ln_capital_start_town_upgrade = yes }
		cmf_log = { action = cm_ln_log_capital_moved }
	}
	remove_variable = cm_ln_cap_target
	# A capital reject says nothing about the same site as a governor spot.
	cm_ln_probe_clear_rejects = yes
	set_variable = { name = cm_ln_land_left value = cm_ln_gov_land_slots }
	set_variable = { name = cm_ln_naval_left value = cm_ln_gov_naval_slots }
	set_variable = { name = cm_ln_phase value = 3 }
}

# The urbanize pass gates on population, the RGO ignore list and the food reservation, so a
# rural capital can sit at rank indefinitely. Root is the location. Expects scope:cm_country.
cm_ln_capital_start_town_upgrade = {
	if = {
		limit = {
			location_rank = location_rank:rural_settlement
			NOT = { has_building = building_type:cm_upgrade_rural_to_town_building }
			NOT = { any_buildings_in_location = { building_type = building_type:cm_upgrade_rural_to_town_building building_levels_under_construction > 0 } }
			cm_location_allowed_to_build = { target = building_type:cm_upgrade_rural_to_town_building }
		}
		construct_building = { building_type = building_type:cm_upgrade_rural_to_town_building }
	}
}

# One governor search a month. A pick on probation has taken no slot yet, so the re-check run
# still scores it as a candidate. Root is the country, scope:cm_country set.
cm_ln_reorg_plan_step = {
	if = {
		limit = { exists = var:cm_ln_probe_site }
		if = {
			limit = { var:cm_ln_probe_wait > 0 }
			change_variable = { name = cm_ln_probe_wait subtract = 1 }
		}
		if = {
			limit = {
				var:cm_ln_probe_wait <= 0
				cm_ln_no_human_search_in_flight = yes
				cm_pf_river_data_is_current = yes
			}
			cm_ln_reorg_gov_search_run = yes
			cm_ln_reorg_gov_probe_check = yes
		}
	}
	else_if = {
		limit = {
			OR = {
				var:cm_ln_land_left > 0
				var:cm_ln_naval_left > 0
			}
			cm_ln_no_human_search_in_flight = yes
			cm_pf_river_data_is_current = yes
		}
		cm_ln_reorg_gov_search_run = yes
		cm_ln_reorg_gov_probe_open = yes
	}
	else_if = {
		limit = {
			var:cm_ln_land_left <= 0
			var:cm_ln_naval_left <= 0
		}
		set_variable = { name = cm_ln_phase value = 4 }
	}
}

# Root is the country, scope:cm_country set.
cm_ln_reorg_gov_search_run = {
	set_variable = { name = cm_pf_view_g value = 1 }
	remove_variable = cm_pf_view_c
	set_variable = { name = cm_ln_replan value = 1 }
	set_variable = { name = cm_ln_gov_valid_only value = 1 }
	cm_pf_prep_core = yes
	cm_pf_run_tiered_sync = yes
	remove_variable = cm_pf_view_g
	remove_variable = cm_ln_replan
	remove_variable = cm_ln_gov_valid_only
}

# Root is the country, scope:cm_country set.
cm_ln_reorg_gov_probe_open = {
	set_local_variable = { name = cm_ln_picked value = 0 }
	ordered_owned_location = {
		order_by = { value = var:cm_gf_score }
		position = 0
		check_range_bounds = no
		limit = {
			has_variable = cm_gf_score
			NOT = { has_variable = cm_ln_probe_reject }
		}
		save_scope_as = cm_ln_probe_pick
		set_local_variable = { name = cm_ln_picked value = 1 }
		scope:cm_country = { cm_ln_probe_start = { SCORE = cm_gf_score } }
	}
	# Nothing left the search would take, so the plan is as full as the map allows and the confirm
	# step is free to end the run with the remaining room unfilled.
	if = {
		limit = { local_var:cm_ln_picked = 0 }
		set_variable = { name = cm_ln_plan_exhausted value = 1 }
		set_variable = { name = cm_ln_phase value = 4 }
	}
}

# Root is the country, scope:cm_country set.
cm_ln_reorg_gov_probe_check = {
	cm_ln_probe_share = { SCORE = cm_gf_score }
	if = {
		limit = { scope:cm_ln_probe_now >= @cm_ln_probe_keep }
		scope:cm_ln_probe_l = { save_scope_as = cm_ln_pick }
		cmf_log_value = { action = cm_ln_log_probe_kept value = scope:cm_ln_probe_now }
		cm_ln_probe_end = yes
		cm_ln_reorg_record_pick = yes
	}
	else = {
		cm_ln_probe_drop = yes
		cm_ln_reorg_gov_probe_open = yes
	}
}

# Root is the country, scope:cm_country and scope:cm_ln_pick set.
cm_ln_reorg_record_pick = {
	add_to_variable_list = { name = cm_ln_plan target = scope:cm_ln_pick }
	change_variable = { name = cm_ln_plan_n add = 1 }
	set_local_variable = { name = cm_ln_ord value = 1000 }
	change_local_variable = { name = cm_ln_ord subtract = var:cm_ln_plan_n }
	scope:cm_ln_pick = { set_variable = { name = cm_ln_plan_ord value = local_var:cm_ln_ord } }
	if = {
		limit = {
			var:cm_ln_naval_left > 0
			scope:cm_ln_pick = { cm_ln_site_is_naval = yes }
		}
		scope:cm_ln_pick = { set_variable = { name = cm_ln_plan_naval value = 1 } }
		change_variable = { name = cm_ln_naval_left subtract = 1 }
	}
	else = {
		change_variable = { name = cm_ln_land_left subtract = 1 }
	}
}

# Root is the country, scope:cm_country set.
cm_ln_reorg_classify = {
	remove_global_variable = cm_ln_token
	cm_ln_strip_finder_outputs = yes
	cm_ln_pf_settings_off = yes
	clear_variable_list = cm_ln_pending
	clear_variable_list = cm_ln_moves
	every_in_list = {
		variable = cm_ln_plan
		save_scope_as = cm_ln_p
		if = {
			limit = {
				scope:cm_ln_p = {
					owner ?= scope:cm_country
					cm_location_has_governor = no
				}
			}
			scope:cm_country = { add_to_variable_list = { name = cm_ln_pending target = scope:cm_ln_p } }
		}
	}
	every_owned_location = {
		limit = {
			cm_location_has_governor = yes
			NOT = { has_variable = cm_ln_plan_ord }
		}
		save_scope_as = cm_ln_seat
		scope:cm_country = { add_to_variable_list = { name = cm_ln_moves target = scope:cm_ln_seat } }
	}
	every_in_list = {
		variable = cm_ln_pre_seats
		save_scope_as = cm_ln_seat
		if = {
			limit = { scope:cm_ln_seat = { cm_location_has_governor = no } }
			scope:cm_country = { change_variable = { name = cm_ln_free_credits add = 1 } }
		}
	}
	clear_variable_list = cm_ln_pre_seats
	set_variable = { name = cm_ln_phase value = 5 }
}

# Root is the country, scope:cm_country set.
cm_ln_reorg_place_step = {
	if = {
		limit = {
			has_variable_list = cm_ln_pending
			variable_list_size = { name = cm_ln_pending value >= 1 }
			cm_ln_gov_slot_open = yes
		}
		cm_ln_place_next = { FREE = no }
	}
	else = {
		set_variable = { name = cm_ln_phase value = 6 }
	}
}

# Holds the run open until every planned governor physically stands. A site whose governor is still
# building is left alone and owes nothing, so the treasury is free that month. A site that has no
# governor and nothing building goes back to the placement phase with the treasury behind it again,
# and one that has swallowed @cm_ln_place_max_tries orders, or has changed hands, is given up so a
# fresh search can put the slot somewhere buildable. Root is the country, scope:cm_country set.
cm_ln_reorg_confirm_step = {
	set_local_variable = { name = cm_ln_cf_building value = 0 }
	every_in_list = {
		variable = cm_ln_plan
		save_scope_as = cm_ln_p
		if = {
			limit = { scope:cm_ln_p = { NOT = { owner ?= scope:cm_country } } }
			scope:cm_country = {
				cmf_log_value = { action = cm_ln_log_gov_site_given_up value = 2 }
				add_to_variable_list = { name = cm_ln_plan_drop target = scope:cm_ln_p }
			}
		}
		else_if = {
			# Delivered, so the order it was paid for owes the country nothing.
			limit = { scope:cm_ln_p = { cm_location_has_governor = yes } }
			scope:cm_ln_p = { remove_variable = cm_ln_paid }
		}
		else_if = {
			limit = { scope:cm_ln_p = { cm_ln_site_gov_under_construction = yes } }
			set_local_variable = { name = cm_ln_cf_building value = 1 }
		}
		else = {
			# The order went in and left nothing behind, which is not a shape anything in the plan
			# accounts for, so it is logged as the failure it is rather than only retried.
			scope:cm_country = {
				cmf_log = { action = cm_ln_log_gov_place_no_show }
				cm_ln_bank_paid_credit = { SITE = scope:cm_ln_p }
				cm_ln_place_retry_or_drop = { SITE = scope:cm_ln_p }
			}
		}
	}
	cm_ln_apply_plan_drops = yes
	if = {
		limit = {
			has_variable_list = cm_ln_pending
			variable_list_size = { name = cm_ln_pending value >= 1 }
		}
		set_variable = { name = cm_ln_phase value = 5 }
	}
	else_if = {
		limit = {
			local_var:cm_ln_cf_building = 0
			cm_ln_gov_slot_open = yes
			NOT = { exists = var:cm_ln_plan_exhausted }
			NOT = { has_global_variable = cm_ln_token }
			cm_ln_no_human_search_in_flight = yes
			cm_pf_river_data_is_current = yes
		}
		# A slot with no plan site behind it, so the search runs again for the room that is left.
		set_global_variable = { name = cm_ln_token value = 1 }
		cm_ln_pf_settings_on = yes
		set_variable = { name = cm_ln_land_left value = cm_ln_gov_land_free }
		set_variable = { name = cm_ln_naval_left value = cm_ln_gov_naval_free }
		set_variable = { name = cm_ln_phase value = 3 }
	}
	else_if = {
		limit = {
			local_var:cm_ln_cf_building = 0
			OR = {
				cm_ln_gov_slot_open = no
				exists = var:cm_ln_plan_exhausted
			}
		}
		set_variable = { name = cm_ln_phase value = 8 }
	}
}

# --- The opening capital move ---
#
# One capital move at the start of the game, before the country builds anything, so it never
# invests in a capital it is about to leave. This is the only move that costs anything; the
# reorganization's own moves above are free.

# Root is the country, scope:cm_country set.
cm_ln_open_capital_tick = {
	if = {
		limit = { NOT = { exists = var:cm_ln_open_done } }
		if = {
			limit = { NOT = { exists = var:cm_ln_open_wait } }
			set_variable = { name = cm_ln_open_wait value = 0 }
		}
		if = {
			limit = {
				is_multiplayer_session = no
				exists = capital
				cm_pf_river_data_is_current = yes
				NOT = { has_global_variable = cm_ln_token }
				cm_ln_no_human_search_in_flight = yes
			}
			cm_ln_open_capital_run = yes
		}
		else_if = {
			limit = { cm_ln_open_river_advancing = yes }
			set_variable = { name = cm_ln_open_wait value = 0 }
			set_variable = { name = cm_ln_open_idx value = global_var:cm_pf_river_idx }
		}
		else = {
			change_variable = { name = cm_ln_open_wait add = 1 }
			if = {
				limit = { var:cm_ln_open_wait >= @cm_ln_open_max_wait }
				set_variable = { name = cm_ln_open_done value = 1 }
				remove_variable = cm_ln_open_idx
				cmf_log = { action = cm_ln_log_open_gave_up }
			}
		}
	}
}

# The search and the move both land in this tick, since no governor plan is waiting on the
# proximity the move shifts. Root is the country, scope:cm_country set.
cm_ln_open_capital_run = {
	set_global_variable = { name = cm_ln_token value = 1 }
	cm_ln_pf_settings_open = yes
	set_variable = { name = cm_pf_view_c value = 1 }
	remove_variable = cm_pf_view_g
	cm_pf_prep_core = yes
	cm_pf_run_tiered_sync = yes
	remove_variable = cm_pf_view_c
	remove_variable = cm_ln_cap_target
	ordered_owned_location = {
		order_by = { value = var:cm_cf_score }
		position = 0
		check_range_bounds = no
		limit = {
			has_variable = cm_cf_score
			NOT = { integration_level = conquered }
		}
		save_scope_as = cm_ln_cap_pick
		scope:cm_country = { set_variable = { name = cm_ln_cap_target value = scope:cm_ln_cap_pick } }
	}
	if = {
		limit = { NOT = { exists = var:cm_ln_cap_target } }
		cmf_log = { action = cm_ln_log_open_no_target }
	}
	else_if = {
		limit = {
			NOT = { var:cm_ln_cap_target = capital }
			NOT = {
				capital = {
					has_variable = cm_cf_score
					var:cm_cf_score >= @cm_ln_open_cap_keep
				}
			}
		}
		set_capital = var:cm_ln_cap_target
		cm_ln_open_charge_move = yes
		capital = { cm_ln_capital_start_town_upgrade = yes }
		cmf_log = { action = cm_ln_log_capital_moved }
	}
	remove_variable = cm_ln_cap_target
	cm_ln_strip_finder_outputs = yes
	cm_ln_pf_settings_off = yes
	remove_global_variable = cm_ln_token
	remove_variable = cm_ln_open_wait
	remove_variable = cm_ln_open_idx
	set_variable = { name = cm_ln_open_done value = 1 }
}

# A share of the vanilla capital_movement price, which is scaled_gold 12 (120% of economic base)
# plus stability 20: in_game/common/prices/00_hardcoded.txt:213. A player reaches 66% off the
# gold and 33% off the stability without much effort, so the charge is what one would really
# pay. The gold half stops at the treasury, so a country that cannot cover it pays what it has.
# Root is the country.
cm_ln_open_charge_move = {
	add_gold = {
		value = cm_ln_open_capital_charge
		multiply = -1
	}
	add_stability = -13
}

# Scores on the roads that actually stand, which the country has no advance to improve this
# early. Root is the country.
cm_ln_pf_settings_open = {
	set_variable = { name = cm_placement_finder_accuracy value = 5 } # Fastest
	set_variable = { name = cm_placement_finder_naval value = 1 } # Current presence
	set_variable = { name = cm_placement_finder_roads value = 1 } # Current
	set_variable = { name = cm_placement_finder_method value = 1 } # Tax Base
}

# --- The opening governor placement ---
#
# Every governor slot the country starts with, one a month from the month after the opening capital
# move so the search reads the proximity and the roads the new capital gives. Each site is picked by
# its own search, which sees the sites already committed to as proximity sources, so a second
# governor is never planted beside the first. It takes the best location a governor can stand on as
# it is, so the opening carries none of the reorganization's urbanize and road steps, and the charge
# stops at the treasury so the placement lands whatever the country can pay.

# Root is the country, scope:cm_country set.
cm_ln_open_gov_tick = {
	if = {
		limit = {
			NOT = { exists = var:cm_ln_open_gov_done }
			is_multiplayer_session = no
			exists = capital
			cm_pf_river_data_is_current = yes
			NOT = { has_global_variable = cm_ln_token }
			cm_ln_no_human_search_in_flight = yes
		}
		# Bounded by the room the country opens with rather than by the live slot count, so a
		# placement the engine takes no slot for cannot re-run the search every month forever.
		if = {
			limit = { NOT = { exists = var:cm_ln_open_gov_left } }
			set_variable = { name = cm_ln_open_gov_left value = cm_ln_gov_land_free }
			change_variable = { name = cm_ln_open_gov_left add = cm_ln_gov_naval_free }
			if = {
				limit = { var:cm_ln_open_gov_left <= 0 }
				cmf_log = { action = cm_ln_log_open_gov_none }
			}
		}
		if = {
			limit = {
				var:cm_ln_open_gov_left > 0
				cm_ln_gov_slot_open = yes
			}
			change_variable = { name = cm_ln_open_gov_left subtract = 1 }
			cm_ln_open_gov_run = yes
		}
		# construct_building counts against the slot immediately, so the last placement of the
		# opening closes it in its own tick rather than a month later.
		if = {
			limit = {
				OR = {
					var:cm_ln_open_gov_left <= 0
					cm_ln_gov_slot_open = no
				}
			}
			cm_ln_open_gov_finish = yes
		}
	}
}

# Root is the country.
cm_ln_open_gov_finish = {
	set_variable = { name = cm_ln_open_gov_done value = 1 }
	remove_variable = cm_ln_open_gov_left
	clear_variable_list = cm_ln_open_gov_sites
}

# The search and the placement both land in this tick. Root is the country, scope:cm_country set.
cm_ln_open_gov_run = {

	cm_ln_open_gov_filter_on = yes
	set_variable = { name = cm_pf_view_g value = 1 }
	remove_variable = cm_pf_view_c
	cm_pf_prep_core = yes
	cm_pf_run_tiered_sync = yes
	remove_variable = cm_pf_view_g

	cm_ln_open_gov_filter_off = yes
	cm_ln_strip_finder_outputs = yes
	cm_ln_pf_settings_off = yes
	remove_global_variable = cm_ln_token
}

# One placement, so each counter carries at most the one slot its kind has room for.
# Root is the country.
cm_ln_open_gov_filter_on = {
	set_variable = { name = cm_ln_gov_valid_only value = 1 }
	set_variable = { name = cm_ln_land_left value = 0 }
	if = {
		limit = { cm_fa_gov_has_land_room = yes }
		set_variable = { name = cm_ln_land_left value = 1 }
	}
	set_variable = { name = cm_ln_naval_left value = 0 }
	if = {
		limit = { cm_fa_gov_has_naval_room = yes }
		set_variable = { name = cm_ln_naval_left value = 1 }
	}
}

# Root is the country.
cm_ln_open_gov_filter_off = {
	remove_variable = cm_ln_gov_valid_only
	remove_variable = cm_ln_land_left
	remove_variable = cm_ln_naval_left
}

# Root is the country, scope:cm_country set.
cm_ln_open_gov_place = {
	ordered_owned_location = {
		order_by = { value = var:cm_gf_score }
		position = 0
		check_range_bounds = no
		limit = {
			has_variable = cm_gf_score
			cm_fa_gov_location_placeable = yes
			NOT = { scope:cm_country = { is_target_in_variable_list = { name = cm_ln_open_gov_sites target = prev } } }
		}
		save_scope_as = cm_ln_open_gov_site
	}
	if = {
		limit = { exists = scope:cm_ln_open_gov_site }
		cm_ln_open_gov_resolve_type = yes
		scope:cm_ln_open_gov_site = { save_temporary_scope_as = cm_location }
		# The charge stops at the treasury, so the placement lands even where the country
		# cannot cover it.
		add_gold = {
			value = cm_ln_open_gov_charge
			multiply = -1
		}
		scope:cm_ln_open_gov_site = {
			construct_building = {
				building_type = scope:cm_building_type
				cost_multiplier = 0
				cost_multiplier_reason = cm_ln_open_gov_placement
				owner = scope:cm_country
				payer = scope:cm_country
			}
		}
		# The governor is months from standing, so the next search reads this site as a source
		# from here rather than from the live proximity.
		add_to_variable_list = { name = cm_ln_open_gov_sites target = scope:cm_ln_open_gov_site }
		cmf_log = { action = cm_ln_log_open_gov_placed }
	}
	else = {
		# Nothing the search would take, so the slots that are left stay empty for the
		# reorganization to fill.
		set_variable = { name = cm_ln_open_gov_left value = 0 }
		cmf_log = { action = cm_ln_log_open_gov_none }
	}
}

# Root is the country, scope:cm_country set. Expects scope:cm_ln_open_gov_site. Sets
# scope:cm_building_type.
cm_ln_open_gov_resolve_type = {
	if = {
		limit = {
			cm_fa_gov_has_naval_room = yes
			scope:cm_ln_open_gov_site = { cm_location_allowed_to_build = { target = building_type:naval_governor } }
		}
		building_type:naval_governor = { save_scope_as = cm_building_type }
	}
	else = {
		building_type:local_governor = { save_scope_as = cm_building_type }
	}
}
