# Effect expects building_type scope. Sets the scope return_scope to the upgrade building type, or leaves it unset if not upgradeable.
# Upgrade mappings are populated by the hidden window GUI via BuildingType.GetNextReplacementBuilding.
cm_find_upgrade_building_type = {
	if = { limit = { is_key_in_global_variable_map = { name = cm_building_type_to_upgrade_type target = this } }
		"global_variable_map(cm_building_type_to_upgrade_type|this)" = { save_scope_as = cm_return_scope }
	} else = {
		if = { limit = { debug_only = yes } debug_log = "No upgrade building type found" }
	}
}

# Root is the building type. Expects scope:cm_country.
# Saves the first non-obsolete type along the cm_building_type_to_upgrade_type chain
# as scope:cm_current_building_type, walking at most 4 hops. The result is the root
# type itself when the root is not obsolete or when no map entry exists.
cm_find_current_building_type = {
	save_scope_as = cm_current_building_type
	if = {
		limit = {
			scope:cm_country = { building_type_is_obsolete = scope:cm_current_building_type }
			is_key_in_global_variable_map = { name = cm_building_type_to_upgrade_type target = scope:cm_current_building_type }
		}
		"global_variable_map(cm_building_type_to_upgrade_type|scope:cm_current_building_type)" = { save_scope_as = cm_current_building_type }
	}
	if = {
		limit = {
			scope:cm_country = { building_type_is_obsolete = scope:cm_current_building_type }
			is_key_in_global_variable_map = { name = cm_building_type_to_upgrade_type target = scope:cm_current_building_type }
		}
		"global_variable_map(cm_building_type_to_upgrade_type|scope:cm_current_building_type)" = { save_scope_as = cm_current_building_type }
	}
	if = {
		limit = {
			scope:cm_country = { building_type_is_obsolete = scope:cm_current_building_type }
			is_key_in_global_variable_map = { name = cm_building_type_to_upgrade_type target = scope:cm_current_building_type }
		}
		"global_variable_map(cm_building_type_to_upgrade_type|scope:cm_current_building_type)" = { save_scope_as = cm_current_building_type }
	}
	if = {
		limit = {
			scope:cm_country = { building_type_is_obsolete = scope:cm_current_building_type }
			is_key_in_global_variable_map = { name = cm_building_type_to_upgrade_type target = scope:cm_current_building_type }
		}
		"global_variable_map(cm_building_type_to_upgrade_type|scope:cm_current_building_type)" = { save_scope_as = cm_current_building_type }
	}
}
