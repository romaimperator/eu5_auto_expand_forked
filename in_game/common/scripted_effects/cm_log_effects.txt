# Every log line this mod writes goes through one of these, gated on its category's toggle in the
# Debug tab. The action log is a single 200-entry pane shared with every other mod and nothing
# evicts, so one talkative feature fills it before anything else has said a word.
#
# $CAT$ is the category's alias suffix and $ACTION$ the localization key. The toggles are global
# settings, so the alias is a global variable and a lucky nation reads the same one the player set.
# Categories, registered in cm_cmm_effects.txt: setup, cycle, industry, budget, governors, military,
# government.

# Root is the country.
cm_dbg_log = {
	if = {
		limit = { exists = global_var:cm_dbg_$CAT$ }
		cmf_log = { action = $ACTION$ }
	}
}

# Root is the country. $VALUE$ renders as a whole number.
cm_dbg_log_value = {
	if = {
		limit = { exists = global_var:cm_dbg_$CAT$ }
		cmf_log_value = { action = $ACTION$ value = $VALUE$ }
	}
}

# Root is the country. $VALUE$ renders with two decimals.
cm_dbg_log_decimal = {
	if = {
		limit = { exists = global_var:cm_dbg_$CAT$ }
		cmf_log_decimal_value = { action = $ACTION$ value = $VALUE$ }
	}
}

# Root is the country. Expects scope:cmf_log_loc. $ARG1$ labels $VALUE$.
cm_dbg_log_loc = {
	if = {
		limit = { exists = global_var:cm_dbg_$CAT$ }
		cmf_log_with_location = { action = $ACTION$ arg1 = $ARG1$ value = $VALUE$ }
	}
}

# Root is the country. Expects scope:cmf_log_bt and scope:cmf_log_loc.
cm_dbg_log_bld = {
	if = {
		limit = { exists = global_var:cm_dbg_$CAT$ }
		cmf_log_with_building_location = {
			action = $ACTION$
			arg1 = $ARG1$
			arg2 = $ARG2$
			value = $VALUE$
			value2 = $VALUE2$
		}
	}
}
