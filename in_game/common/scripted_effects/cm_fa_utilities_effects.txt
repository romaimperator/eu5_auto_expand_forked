# Full-automation Utilities: the capital-only set (from cm_run_auto_build_capital), the
# auto_build_list roster (cm_cmm_effects.txt), mercury patios, the vanilla market-center
# buildings (game in_game/common/building_types/market_buildings.txt, the is_market_center-gated
# set), bailiffs and the unique local_max_control buildings, the marketplace chain, the
# disease-resistance buildings, and granaries last. Runs on default thresholds and never reads
# or writes the player's auto-build settings or toggles.
# Every location sweep here is confined to the cycle's market slice
# (cm_fa_slice_effects.txt), except the capital set and the max-control buildings. Bailiffs
# survey once every four years into a pick list that stages monthly until it empties.

# Four years, the bailiff survey cadence.

# Stages one utility type at this location when it fits; the capital and market-center
# sets have no obsolete chains. The not-at-max-level check keeps multi-level types like
# chancery topping up. Root is the location. Expects scope:cm_country, scope:cm_location;
# $bt$ = the type key.
cm_fa_stage_utility_type_here = {
	building_type:$bt$ = { save_scope_as = cm_building_type }
	if = {
		limit = {
			cm_location_building_type_is_not_at_max_level = { target = scope:cm_building_type }
			cm_location_allowed_to_build = { target = scope:cm_building_type }
			cm_location_has_enough_pops_available = yes
			cm_check_location_meets_min_market_access = yes
			cm_get_open_build_queue_slots > 0
			NOT = { is_target_in_variable_list = { name = cm_q_ungated_building_types target = scope:cm_building_type } }
		}
		cm_fa_stage_build_candidate = { cat = 4 }
	}
}

# Stages one utility type across owned locations; chain bases resolve to the current tier,
# and a location holding an older tier is left to the upgrade features. Tops up to max level,
# since nothing else in full automation grows these types: Industry only touches production
# buildings. Root is the country. $bt$ = the raw type key.
cm_fa_stage_utility_everywhere = {
	building_type:$bt$ = { save_scope_as = cm_building_type }
	cm_auto_build_resolve_building_type = yes
	ordered_owned_location = {
		order_by = max_control
		max = cm_max_guess_at_number_of_locations
		check_range_bounds = no
		limit = {
			cm_location_market_in_fa_slice = yes
			NOT = { integration_level = conquered }
			cm_fa_location_has_proximity = yes
		}
		save_scope_as = cm_location
		if = {
			limit = {
				cm_fa_utilities_type_has_room = yes
				cm_location_allowed_to_build = { target = scope:cm_building_type }
				cm_location_has_no_auto_build_chain_predecessor = yes
				cm_location_has_enough_pops_available = yes
				cm_fa_utilities_rgo_req_met = yes
				cm_check_location_meets_min_market_access = yes
				cm_get_open_build_queue_slots > 0
				NOT = { is_target_in_variable_list = { name = cm_q_ungated_building_types target = scope:cm_building_type } }
			}
			cm_fa_stage_build_candidate = { cat = 4 }
		}
	}
}

# The four-yearly bailiff survey, wealthiest locations first: local_max_control earns tax
# base in proportion to the location's wealth. The country-wide num_bailiffs modifier caps
# how many can stand (game in_game/common/building_types/rural_buildings.txt, bailiff
# allow), so the survey stops at the remaining room and the cap goes to the best locations.
# That cap is also why this sweep is not sliced: a permanent cap handed to one slice's best
# would never reach the country's. The cap fills slowly enough that re-ranking every owned
# location every month changes nothing, hence the four-year cadence on the call.
# Root is the country.
cm_fa_pick_bailiffs = {
	clear_variable_list = cm_fa_bailiff_picks
	building_type:bailiff = { save_scope_as = cm_building_type }
	set_local_variable = { name = cm_fa_bailiff_room value = modifier:num_bailiffs }
	change_local_variable = { name = cm_fa_bailiff_room subtract = total_building_levels_including_construction:bailiff }
	ordered_owned_location = {
		order_by = cm_location_raw_wealth
		max = cm_max_guess_at_number_of_locations
		check_range_bounds = no
		save_scope_as = cm_location
		if = {
			limit = {
				local_var:cm_fa_bailiff_room >= 1
				cm_fa_max_control_site_ok = yes
			}
			scope:cm_country = { add_to_variable_list = { name = cm_fa_bailiff_picks target = scope:cm_location } }
			change_local_variable = { name = cm_fa_bailiff_room subtract = 1 }
		}
	}
}

# Stages the surveyed picks the country can act on this month, wealthiest first. A pick the
# Utilities pool cannot cover is left in the list and staged again next month, so a survey
# landing on a poor month is not lost; the room is re-read each month because bailiffs
# finished since the survey have taken part of the cap. Root is the country.
cm_fa_stage_bailiffs = {
	if = {
		limit = {
			has_variable_list = cm_fa_bailiff_picks
			variable_list_size = { name = cm_fa_bailiff_picks value >= 1 }
		}
		building_type:bailiff = { save_scope_as = cm_building_type }
		cm_fa_prune_bailiff_picks = yes
		set_local_variable = { name = cm_fa_bailiff_room value = modifier:num_bailiffs }
		change_local_variable = { name = cm_fa_bailiff_room subtract = total_building_levels_including_construction:bailiff }
		ordered_in_list = {
			variable = cm_fa_bailiff_picks
			order_by = cm_location_raw_wealth
			max = cm_max_guess_at_number_of_locations
			check_range_bounds = no
			save_scope_as = cm_location
			if = {
				limit = {
					local_var:cm_fa_bailiff_room >= 1
					cm_fa_max_control_site_ok = yes
				}
				cm_fa_stage_build_candidate = { cat = 4 }
				change_local_variable = { name = cm_fa_bailiff_room subtract = 1 }
			}
		}
	}
}

# Drops picks a bailiff will never reach, so the list shrinks to empty over the four years:
# lost to another owner, already carrying one (has_building holds from the moment the first
# level is queued), or urbanized out of reach - bailiff lists only rural_settlement among
# the four rank fields (game in_game/common/building_types/rural_buildings.txt:10), so a
# rank upgrade retires the site for good. Root is the country, expects scope:cm_country,
# scope:cm_building_type.
cm_fa_prune_bailiff_picks = {
	every_in_list = {
		variable = cm_fa_bailiff_picks
		save_scope_as = cm_fa_bailiff_pick
		if = {
			limit = {
				OR = {
					NOT = { owner ?= scope:cm_country }
					has_building = scope:cm_building_type
					NOT = { location_rank = location_rank:rural_settlement }
				}
			}
			scope:cm_country = { remove_list_variable = { name = cm_fa_bailiff_picks target = scope:cm_fa_bailiff_pick } }
		}
	}
}

# Stages one unique local_max_control type across owned locations, wealthiest first, on the
# bailiff's terms: tax base over control is what a control building earns, and neither the
# proximity gate nor the market slice applies. No country-wide cap limits these, so they take
# no survey and a location newly meeting the type's own rules stages the next month. One level
# per location. Root is the country. $bt$ = the type key.
cm_fa_stage_max_control_type = {
	building_type:$bt$ = { save_scope_as = cm_building_type }
	if = {
		limit = { cm_fa_country_can_build_type = yes }
		ordered_owned_location = {
			order_by = cm_location_raw_wealth
			max = cm_max_guess_at_number_of_locations
			check_range_bounds = no
			save_scope_as = cm_location
			if = {
				limit = { cm_fa_max_control_site_ok = yes }
				cm_fa_stage_build_candidate = { cat = 4 }
			}
		}
	}
}

# Stages the marketplace chain's current tier across owned locations, lowest market access

		}
		save_scope_as = cm_location
		if = {
			limit = {
				cm_location_building_type_is_not_at_max_level = { target = scope:cm_building_type }
				cm_location_allowed_to_build = { target = scope:cm_building_type }
				cm_location_has_no_auto_build_chain_predecessor = yes
				cm_location_has_enough_pops_available = yes

				NOT = { is_target_in_variable_list = { name = cm_q_ungated_building_types target = scope:cm_building_type } }

# One max-control type, wealthiest locations first and past the location's open slots, so a
# bailiff and a unique type both reach the queue in the pass that staged them.
# Root is the country. $bt$ = the type key.
cm_fa_drain_max_control_type = {
	building_type:$bt$ = { save_scope_as = cm_building_type }
	if = {
		limit = {
			local_var:cm_fa_cost_blocked = 0
			has_variable_list = cm_fa_b_locations
		}
		ordered_in_list = {
			variable = cm_fa_b_locations
			order_by = cm_location_raw_wealth
			max = cm_max_guess_at_number_of_locations

				cm_fa_try_approve_build = { cat = 4 POOL = cm_fa_pool_utilities GOODS_GATE = cm_market_has_construction_goods_for_building SLOTS_GATE = always }
			}
		}
