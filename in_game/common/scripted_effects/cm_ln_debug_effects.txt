# Temporary instrumentation for the lucky-nation Industry drain: per-cycle counters written
# while a cycle scores and drains, reported to the CMM log when RGO candidates arrived and
# none was constructed. Every call site is gated on cm_ln_is_active, so the player path is
# untouched.

# $N$ = counter suffix. Root is the country.
cm_ln_dbg_bump = {
	if = {
		limit = { NOT = { has_variable = cm_ln_dbg_$N$ } }
		set_variable = { name = cm_ln_dbg_$N$ value = 0 }
	}
	change_variable = { name = cm_ln_dbg_$N$ add = 1 }
}

# Root is the country.
cm_ln_dbg_reset = {
	set_variable = { name = cm_ln_dbg_r_rows value = 0 }
	set_variable = { name = cm_ln_dbg_r_elig value = 0 }
	set_variable = { name = cm_ln_dbg_r_goods value = 0 }
	set_variable = { name = cm_ln_dbg_r_roi value = 0 }
	set_variable = { name = cm_ln_dbg_r_ok value = 0 }
	set_variable = { name = cm_ln_dbg_mkts value = 0 }
	set_variable = { name = cm_ln_dbg_cands value = 0 }
	set_variable = { name = cm_ln_dbg_pick_b value = 0 }
	set_variable = { name = cm_ln_dbg_pick_u value = 0 }
	set_variable = { name = cm_ln_dbg_pick_r value = 0 }
	set_variable = { name = cm_ln_dbg_built_r value = 0 }
	set_variable = { name = cm_ln_dbg_blk value = 0 }
	set_variable = { name = cm_ln_dbg_r_blk value = 0 }
	remove_variable = cm_ln_dbg_r_cost
	remove_variable = cm_ln_dbg_r_inc
	remove_variable = cm_ln_dbg_rj_cost
	remove_variable = cm_ln_dbg_rj_inc
}

# First blocking reason only, so the code names the gate that actually stopped the item.
# $C$ = reason code. Root is the country.
cm_ln_dbg_r_block = {
	if = {
		limit = { var:cm_ln_dbg_r_blk = 0 }
		set_variable = { name = cm_ln_dbg_r_blk value = $C$ }
	}
}

# $C$ = trip-site code. Root is the country.
cm_ln_dbg_blk_trip = {
	if = {
		limit = { var:cm_ln_dbg_blk = 0 }
		set_variable = { name = cm_ln_dbg_blk value = $C$ }
	}
}

# Splits cm_fa_score_rgo_apply's conjunction so the failing half is countable, and keeps the
# best-scoring RGO's own cost and income for comparison against the tooltip.
# Root is the location, expects scope:cm_country and scope:cm_income.
cm_ln_dbg_rgo_score_probe = {
	scope:cm_country = { cm_ln_dbg_bump = { N = r_rows } }
	if = {
		limit = { NOT = { cm_rgo_basic_expand_eligibility = yes } }
		scope:cm_country = { cm_ln_dbg_bump = { N = r_elig } }
	}
	else_if = {
		limit = { NOT = { cm_market_has_construction_goods_for_rgo = yes } }
		scope:cm_country = { cm_ln_dbg_bump = { N = r_goods } }
	}
	else_if = {
		limit = { NOT = { cm_fa_ind_roi_passes = { COST = glorpui_rgo_expand_cost } } }
		scope:cm_country = { cm_ln_dbg_bump = { N = r_roi } }
		save_temporary_scope_value_as = { name = cm_ln_dbg_c value = glorpui_rgo_expand_cost }
		scope:cm_country = {
			if = {
				limit = { NOT = { has_variable = cm_ln_dbg_rj_inc } }
				set_variable = { name = cm_ln_dbg_rj_inc value = scope:cm_income }
				set_variable = { name = cm_ln_dbg_rj_cost value = scope:cm_ln_dbg_c }
			}
			else_if = {
				limit = { var:cm_ln_dbg_rj_inc < scope:cm_income }
				set_variable = { name = cm_ln_dbg_rj_inc value = scope:cm_income }
				set_variable = { name = cm_ln_dbg_rj_cost value = scope:cm_ln_dbg_c }
			}
		}
	}
	else = {
		scope:cm_country = { cm_ln_dbg_bump = { N = r_ok } }
		save_temporary_scope_value_as = { name = cm_ln_dbg_c value = glorpui_rgo_expand_cost }
		scope:cm_country = {
			if = {
				limit = { NOT = { has_variable = cm_ln_dbg_r_inc } }
				set_variable = { name = cm_ln_dbg_r_inc value = scope:cm_income }
				set_variable = { name = cm_ln_dbg_r_cost value = scope:cm_ln_dbg_c }
			}
			else_if = {
				limit = { var:cm_ln_dbg_r_inc < scope:cm_income }
				set_variable = { name = cm_ln_dbg_r_inc value = scope:cm_income }
				set_variable = { name = cm_ln_dbg_r_cost value = scope:cm_ln_dbg_c }
			}
		}
	}
}

# Reason codes: 1 eligibility, 2 auto-food, 3 market goods, 4 slots, 5 rgo level headroom,
# 6 pool or gold. Root is the location, expects scope:cm_country.
cm_ln_dbg_rgo_approve_probe = {
	if = {
		limit = { NOT = { cm_rgo_basic_expand_eligibility = yes } }
		scope:cm_country = { cm_ln_dbg_r_block = { C = 1 } }
	}
	else_if = {
		limit = { cm_location_auto_food_supersedes_rgo = yes }
		scope:cm_country = { cm_ln_dbg_r_block = { C = 2 } }
	}
	else_if = {
		limit = { NOT = { cm_market_has_construction_goods_for_rgo = yes } }
		scope:cm_country = { cm_ln_dbg_r_block = { C = 3 } }
	}
	else_if = {
		limit = { NOT = { cm_get_open_build_queue_slots > 0 } }
		scope:cm_country = { cm_ln_dbg_r_block = { C = 4 } }
	}
	else_if = {
		limit = { NOT = { cm_get_max_rgo_workers_floor > { value = cm_modifier_rgo_level add = cm_get_rgos_under_construction } } }
		scope:cm_country = { cm_ln_dbg_r_block = { C = 5 } }
	}
	else_if = {
		limit = { NOT = { cm_fa_can_afford_rgo = { POOL = cm_fa_pool_industry } } }
		scope:cm_country = { cm_ln_dbg_r_block = { C = 6 } }
	}
}

# The counter reads sit behind their own if, so a player country never reaches a variable
# only a lucky nation carries. Root is the country.
cm_ln_dbg_report = {
	if = {
		limit = { cm_ln_is_active = yes }
		if = {
			limit = {
				var:cm_ln_dbg_r_rows > 0
				var:cm_ln_dbg_built_r = 0
			}
			cm_ln_dbg_report_lines = yes
		}
	}
}
