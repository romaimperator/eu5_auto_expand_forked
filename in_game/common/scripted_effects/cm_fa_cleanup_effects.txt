# Full-automation Building Cleanup: once a year, closes production buildings that stopped
# earning, destroys them where the location has no building levels to spare, reopens the
# ones whose market recovered, and clears non-food production out of shorthanded auto-food
# locations. Closing and reopening are GUI-only (ToggleBuilding), so both queue into
# cm_fa_cu_toggle and fire from cm_construct_queue_window.gui; destroying is the script
# effect destroy_building and runs inline.

# Root is the country.
cm_fa_cu_yearly_pass = {
	if = {
		limit = {
			cm_fa_master_active = yes
			exists = var:cm_fa_cleanup_enabled
		}
		save_scope_as = cm_country
		clear_variable_list = cm_fa_cu_toggle
		cm_fa_cu_purge_pass = yes
		if = {
			limit = {
				exists = var:cm_fa_cleanup_auto_food
				# Without the food roster every building reads as non-food.
				has_variable_list = cm_auto_food_rgo_types_list
			}
			cm_fa_cu_auto_food_pass = yes
		}
		cm_fa_cu_validate_tracking = yes
		every_market_present_in_country = {
			save_scope_as = cm_fa_market
			scope:cm_country = {
				cm_fa_cu_survey_market = yes
				cm_fa_cu_close_pass = yes
				cm_fa_cu_reopen_pass = yes
			}
		}
		cm_fa_cu_delete_closed_pass = yes
		if = {
			limit = { variable_list_size = { name = cm_fa_cu_toggle value >= 1 } }
			set_variable = { name = cm_fa_cu_fire value = 1 }
		}
	}
}

# Destroys the always-delete types outright, no profit, subsidy, cap, or share check. Runs
# before everything else so the removals never reach the survey counts.
# Root is the country, scope:cm_country set.
cm_fa_cu_purge_pass = {
	every_owned_location = {
		limit = { NOT = { integration_level = conquered } }
		save_scope_as = cm_location
		every_buildings_in_location = {
			limit = {
				owner = scope:cm_country
				building_type = { cm_fa_cu_is_always_delete = yes }
			}
			save_scope_as = cm_building
			scope:cm_location = { destroy_building = scope:cm_building }
			scope:cm_country = { cmf_log = { action = cm_fa_log_cu_purged } }
		}
	}
}

# Clears the way for food production where auto-food is running and the pops are already
# spoken for. Runs beside the purge, ahead of the survey counts, and ahead of
# cm_fa_cu_validate_tracking so a destroyed building's closed-tracking entry is dropped there.
# Root is the country, scope:cm_country set.
cm_fa_cu_auto_food_pass = {
	every_owned_location = {
		limit = {
			NOT = { integration_level = conquered }
			cm_fa_food_location_effectively_active = yes
			cm_location_has_worker_deficit = yes
		}
		save_scope_as = cm_location
		every_buildings_in_location = {
			limit = {
				owner = scope:cm_country
				# Do not tear down what the treasury is still paying to build.
				cm_is_not_under_construction = yes
				cm_fa_cu_af_is_removable = yes
			}
			save_scope_as = cm_building
			if = {
				# One level keeps the location's market access and food capacity.
				limit = { building_type = building_type:market_village }
				if = {
					limit = { building_level > 1 }
					scope:cm_location = {
						change_building_level_in_location = {
							building = building_type:market_village
							value = { value = 1 subtract = scope:cm_building.building_level }
							owner = scope:cm_country
						}
					}
					scope:cm_country = { cmf_log = { action = cm_fa_log_cu_af_trimmed } }
				}
			}
			else = {
				scope:cm_location = { destroy_building = scope:cm_building }
				scope:cm_country = { cmf_log = { action = cm_fa_log_cu_af_removed } }
			}
		}
	}
}

# Flags a building type whose majority input is one of the four construction materials.
# Fired from cm_rgob_collect_entry per goods demand entry of a worth-using method, so it
# reuses that wave's per-method input total. Expects scope:cm_rgob_bt, scope:cm_rgob_pm,
# scope:cm_rgob_good, scope:cm_rgob_amount.
cm_fa_cu_mark_core_input = {
	if = {
		limit = {
			scope:cm_rgob_amount > 0
			is_key_in_global_variable_map = { name = cm_rgob_total target = scope:cm_rgob_pm }
			scope:cm_rgob_good = {
				OR = {
					this = goods:masonry
					this = goods:lumber
					this = goods:sand
					this = goods:glass
				}
			}
		}
		set_local_variable = { name = cm_fa_cu_tot value = "global_variable_map(cm_rgob_total|scope:cm_rgob_pm)" }
		if = {
			limit = { local_var:cm_fa_cu_tot > 0 }
			set_local_variable = { name = cm_fa_cu_share value = { value = scope:cm_rgob_amount divide = local_var:cm_fa_cu_tot } }
			if = {
				limit = { local_var:cm_fa_cu_share > 0.5 }
				add_to_global_variable_map = { name = cm_fa_cu_core_in key = scope:cm_rgob_bt value = 1 }
			}
		}
	}
}

# Drops tracking for buildings that are gone, no longer ours, or open again, so a building
# reopened by hand goes back to the player and a close that never landed becomes a fresh
# candidate. Root is the country, scope:cm_country set.
cm_fa_cu_validate_tracking = {
	every_owned_location = {
		limit = { has_variable_list = cm_fa_cu_closed_types }
		save_scope_as = cm_location
		every_in_list = {
			variable = cm_fa_cu_closed_types
			save_scope_as = cm_building_type
			if = {
				limit = {
					scope:cm_location = {
						NOT = {
							any_buildings_in_location = {
								owner = scope:cm_country
								building_type = scope:cm_building_type
								is_opened = no
							}
						}
					}
				}
				scope:cm_location = { remove_list_variable = { name = cm_fa_cu_closed_types target = scope:cm_building_type } }
			}
		}
	}
}

# Counts one market's production buildings per type: how many are open, whether any open one
# still earns, and how many this system has closed. Root is the country, expects
# scope:cm_fa_market.
cm_fa_cu_survey_market = {
	clear_variable_map = cm_fa_cu_open
	clear_variable_map = cm_fa_cu_prof_open
	clear_variable_map = cm_fa_cu_closed_n
	clear_variable_map = cm_fa_cu_acted
	clear_variable_map = cm_fa_cu_reop
	every_owned_location = {
		limit = {
			market = scope:cm_fa_market
			NOT = { integration_level = conquered }
		}
		save_scope_as = cm_location
		every_buildings_in_location = {
			limit = {
				owner = scope:cm_country
				building_type = { cm_is_building_type_production_building = yes }
			}
			save_scope_as = cm_building
			building_type = { save_scope_as = cm_building_type }
			if = {
				limit = { scope:cm_building = { is_opened = yes } }
				scope:cm_country = {
					cm_fa_cu_bump_open = yes
					if = {
						limit = { scope:cm_building = { cm_fa_cu_is_unprofitable = no } }
						add_to_variable_map = { name = cm_fa_cu_prof_open key = scope:cm_building_type value = 1 }
					}
				}
			} else_if = {
				limit = { scope:cm_building = { cm_fa_cu_is_tracked_closed = yes } }
				scope:cm_country = { cm_fa_cu_bump_closed_n = yes }
			}
		}
	}
}

# Closes one market's unprofitable production buildings least profitable first, destroying
# instead where the location is near its cap, until each type's half share is spent. A market
# village is cut back to one level instead, never closed or destroyed.
# Root is the country, expects scope:cm_fa_market.
cm_fa_cu_close_pass = {
	ordered_owned_building = {
		order_by = cm_fa_cu_neg_profit
		max = cm_max_guess_at_number_of_buildings
		check_range_bounds = no
		limit = {
			is_opened = yes
			is_subsidized = no
			# A building exists and reads no profit from the moment its first level is
			# queued, so without this the pass closes what it is still paying to build.
			cm_is_not_under_construction = yes
			cm_fa_cu_is_unprofitable = yes
			building_type = { cm_is_building_type_production_building = yes }
			# One level keeps the location's market access.
			trigger_if = {
				limit = { building_type = building_type:market_village }
				building_level > 1
			}
			location = {
				owner = scope:cm_country
				market = scope:cm_fa_market
				NOT = { integration_level = conquered }
			}
		}
		save_scope_as = cm_building
		building_type = { save_scope_as = cm_building_type }
		location = { save_scope_as = cm_location }
		# The iterator snapshots its limit; the share spent advances live.
		if = {
			limit = { scope:cm_country = { cm_fa_cu_group_has_room = yes } }
			scope:cm_country = { cm_fa_cu_bump_acted = yes }
			if = {
				limit = { building_type = building_type:market_village }
				scope:cm_location = {
					change_building_level_in_location = {
						building = building_type:market_village
						value = { value = 1 subtract = scope:cm_building.building_level }
						owner = scope:cm_country
					}
				}
				scope:cm_country = { cmf_log = { action = cm_fa_log_cu_trimmed } }
			}
			else_if = {
				limit = { scope:cm_location = { cm_fa_cu_location_near_cap = yes } }
				scope:cm_location = { destroy_building = scope:cm_building }
				scope:cm_country = { cmf_log = { action = cm_fa_log_cu_deleted } }
			}
			else = {
				scope:cm_country = {
					add_to_variable_list = { name = cm_fa_cu_toggle target = scope:cm_building }
					cmf_log = { action = cm_fa_log_cu_closed }
				}
				cm_fa_cu_track_closed = yes
			}
		}
	}
}

# Reopens the closed buildings of one market's recovered types, best potential profit first.
# Root is the country, expects scope:cm_fa_market.
cm_fa_cu_reopen_pass = {
	ordered_owned_building = {
		order_by = building_potential_profit
		max = cm_max_guess_at_number_of_buildings
		check_range_bounds = no
		limit = {
			cm_fa_cu_is_tracked_closed = yes
			location = {
				owner = scope:cm_country
				market = scope:cm_fa_market
			}
		}
		save_scope_as = cm_building
		building_type = { save_scope_as = cm_building_type }
		location = { save_scope_as = cm_location }
		if = {
			limit = { scope:cm_country = { cm_fa_cu_group_reopen_room = yes } }
			scope:cm_country = {
				cm_fa_cu_bump_reop = yes
				add_to_variable_list = { name = cm_fa_cu_toggle target = scope:cm_building }
				cmf_log = { action = cm_fa_log_cu_reopened }
			}
			cm_fa_cu_untrack = yes
		}
	}
}

# Destroys the buildings this system closed wherever the location has since run out of
# building levels. Runs after the reopens, so a recovered type is restored rather than lost.
# Root is the country, scope:cm_country set.
cm_fa_cu_delete_closed_pass = {
	every_owned_location = {
		limit = {
			has_variable_list = cm_fa_cu_closed_types
			cm_fa_cu_location_near_cap = yes
		}
		save_scope_as = cm_location
		every_buildings_in_location = {
			limit = {
				owner = scope:cm_country
				cm_fa_cu_is_tracked_closed = yes
			}
			save_scope_as = cm_building
			building_type = { save_scope_as = cm_building_type }
			scope:cm_location = { destroy_building = scope:cm_building }
			cm_fa_cu_untrack = yes
			scope:cm_country = { cmf_log = { action = cm_fa_log_cu_deleted } }
		}
	}
}

# Expects scope:cm_location, scope:cm_building_type.
cm_fa_cu_track_closed = {
	scope:cm_location = {
		add_to_variable_list = { name = cm_fa_cu_closed_types target = scope:cm_building_type }
	}
}

# Expects scope:cm_location, scope:cm_building_type.
cm_fa_cu_untrack = {
	scope:cm_location = {
		remove_list_variable = { name = cm_fa_cu_closed_types target = scope:cm_building_type }
	}
}

# add_to_variable_map overwrites, so each counter reads its entry, bumps it, and writes it
# back; the read is inlined per map because a quoted variable_map() map name cannot be a
# macro. Root is the country, expects scope:cm_building_type.
cm_fa_cu_bump_open = {
	set_local_variable = { name = cm_fa_cu_n value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = cm_fa_cu_open target = scope:cm_building_type } }
		set_local_variable = { name = cm_fa_cu_n value = "variable_map(cm_fa_cu_open|scope:cm_building_type)" }
	}
	change_local_variable = { name = cm_fa_cu_n add = 1 }
	add_to_variable_map = { name = cm_fa_cu_open key = scope:cm_building_type value = local_var:cm_fa_cu_n }
}
