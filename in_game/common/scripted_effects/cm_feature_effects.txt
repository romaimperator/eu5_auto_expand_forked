# Feature effects for the unified priority dispatcher.
# Each effect is called by the dispatcher in cm_on_action.txt when
# the corresponding feature is enabled in the priority list.
# All effects expect scope:country to be set.

# Per-cycle construction queue. The monthly dispatcher enqueues candidates in
# priority-feature then max_control order: a country list of locations, each carrying
# a location list of building types. The hidden window drains it in that order, grabs
# the engine cost/income GUI bindings, and constructs.
# Expects scope:country, scope:location, scope:building_type.
cm_stage_candidate = {
	scope:location = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = cm_q_building_types target = scope:building_type } } }
			add_to_variable_list = { name = cm_q_building_types target = scope:building_type }
		}
	}
	scope:country = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = cm_q_locations target = scope:location } } }
			add_to_variable_list = { name = cm_q_locations target = scope:location }
		}
	}
}

# ─── Auto Expand Buildings ───
# Expands multi-level buildings that are profitable and have pops available.
# Flags obsolete buildings for GUI-driven upgrade processing.
cm_run_auto_expand_buildings = {
	if = {
		limit = { has_variable_list = cm_mass_auto_expand_building_types }
		# ── Mass mode: iterate all existing buildings at all owned locations ──
		ordered_owned_location = {
			order_by = max_control
			max = cm_max_guess_at_number_of_locations
			check_range_bounds = no
			limit = { NOT = { integration_level = conquered } }
			save_scope_as = location
			ordered_buildings_in_location = {
				order_by = building_index
				max = 100
				check_range_bounds = no
				save_scope_as = building
				building_type = { save_scope_as = building_type }
				if = {
					limit = {
						# Building type is mass-toggled for this country
						scope:country = { is_target_in_variable_list = { name = cm_mass_auto_expand_building_types target = scope:building_type } }
						# Not excluded by the player at this location
						NOT = { scope:location = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:building_type } } }
					}
					if = {
						limit = { cm_is_building_type_obsolete = no }
						if = {
							limit = {
								scope:building = {
									cm_building_can_auto_expand_other_than_checking_for_upgrades = yes
									is_at_max_level = no
									cm_is_not_under_construction = yes
								}
							}
							cm_stage_candidate = yes
						}
					} else_if = {
						limit = {
							scope:location = { cm_is_building_type_upgradeable_and_can_be_upgraded_by_owner = yes }
						}
						cm_clear_finished_upgrade_in_progress = yes
						if = {
							limit = {
								scope:building = {
									cm_building_can_auto_expand_for_upgrade = yes
									cm_is_not_under_construction = yes
								}
								cm_auto_expand_check_minimum_gold_reserve = yes
								NOT = { scope:location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:building_type } } }
							}
							scope:location = {
								if = {
									limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_building_types target = scope:building_type } } }
									add_to_variable_list = { name = cm_gui_upgrade_building_types target = scope:building_type }
									add_to_variable_list = { name = cm_q_upgrade_gated_types target = scope:building_type }
								}
							}
							scope:country = {
								if = {
									limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_locations target = scope:location } } }
									add_to_variable_list = { name = cm_gui_upgrade_locations target = scope:location }
								}
							}
						}
					}
				}
			}
			# ── Mass mode: construct missing buildings from zero levels ──
			scope:country = {
				every_in_list = {
					variable = cm_mass_auto_expand_building_types
					save_scope_as = building_type
					if = {
						limit = {
							NOT = { scope:location = { has_building = scope:building_type } }
							NOT = { scope:location = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:building_type } } }
						}
						# Clean up in-progress upgrade tracking if the old building type is gone (upgrade completed)
						if = {
							limit = { scope:location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:building_type } } }
							scope:location = { remove_list_variable = { name = cm_upgrade_in_progress target = scope:building_type } }
						}
						if = {
							limit = { cm_is_building_type_obsolete = no }
							if = {
								limit = { cm_location_can_auto_expand_from_zero_levels = yes }
								cm_stage_candidate = yes
							}
						} else_if = {
							limit = {
								scope:location = { cm_is_building_type_upgradeable_and_can_be_upgraded_by_owner = yes }
								}
							cm_clear_finished_upgrade_in_progress = yes
							if = {
								limit = {
									cm_location_can_auto_expand_from_zero_levels_for_upgrade = yes
									NOT = { scope:location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:building_type } } }
								}
								scope:location = {
									if = {
										limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_building_types target = scope:building_type } } }
										add_to_variable_list = { name = cm_gui_upgrade_building_types target = scope:building_type }
										add_to_variable_list = { name = cm_q_upgrade_gated_types target = scope:building_type }
									}
								}
								scope:country = {
									if = {
										limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_locations target = scope:location } } }
										add_to_variable_list = { name = cm_gui_upgrade_locations target = scope:location }
									}
								}
							}
						}
					}
				}
			}
		}
	}
	# ── Normal mode: iterate per-location registered building types ──
	# Runs unconditionally — per-location registrations are processed alongside mass mode.
	ordered_owned_location = {
		order_by = max_control
		max = cm_max_guess_at_number_of_locations
		check_range_bounds = no
		limit = { NOT = { integration_level = conquered } has_variable_list = cm_auto_expand_registered_building_types }
		save_scope_as = location
		every_in_list = {
			variable = cm_auto_expand_registered_building_types
			save_scope_as = building_type
			if = {
				limit = { scope:location = { has_building = scope:building_type } }
				scope:location = { "building(scope:building_type|scope:country)" = { save_scope_as = building } }
				if = {
					limit = { cm_is_building_type_obsolete = no }
					if = {
						limit = {
							scope:building = {
								cm_building_can_auto_expand_other_than_checking_for_upgrades = yes
								is_at_max_level = no
								cm_is_not_under_construction = yes
							}
						}
						cm_stage_candidate = yes
					}
				} else_if = {
					limit = {
						scope:location = { cm_is_building_type_upgradeable_and_can_be_upgraded_by_owner = yes }
					}
					cm_clear_finished_upgrade_in_progress = yes
					if = {
						limit = {
							scope:building = {
								cm_building_can_auto_expand_for_upgrade = yes
								cm_is_not_under_construction = yes
							}
							cm_auto_expand_check_minimum_gold_reserve = yes
							NOT = { scope:location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:building_type } } }
						}
						scope:location = {
							if = {
								limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_building_types target = scope:building_type } } }
								add_to_variable_list = { name = cm_gui_upgrade_building_types target = scope:building_type }
								add_to_variable_list = { name = cm_q_upgrade_gated_types target = scope:building_type }
							}
						}
						scope:country = {
							if = {
								limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_locations target = scope:location } } }
								add_to_variable_list = { name = cm_gui_upgrade_locations target = scope:location }
							}
						}
					}
				}
			} else = {
				# Building has no levels — not yet queued for construction
				# Clean up in-progress upgrade tracking if the old building type is gone (upgrade completed)
				if = {
					limit = { scope:location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:building_type } } }
					scope:location = { remove_list_variable = { name = cm_upgrade_in_progress target = scope:building_type } }
				}
				if = {
					limit = { cm_is_building_type_obsolete = no }
					if = {
						limit = { cm_location_can_auto_expand_from_zero_levels = yes }
						cm_stage_candidate = yes
					}
				} else_if = {
					limit = { scope:location = { cm_is_building_type_upgradeable_and_can_be_upgraded_by_owner = yes } }
					cm_clear_finished_upgrade_in_progress = yes
					if = {
						limit = {
							cm_location_can_auto_expand_from_zero_levels_for_upgrade = yes
							NOT = { scope:location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:building_type } } }
						}
						scope:location = {
							if = {
								limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_building_types target = scope:building_type } } }
								add_to_variable_list = { name = cm_gui_upgrade_building_types target = scope:building_type }
								add_to_variable_list = { name = cm_q_upgrade_gated_types target = scope:building_type }
							}
						}
						scope:country = {
							if = {
								limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_locations target = scope:location } } }
								add_to_variable_list = { name = cm_gui_upgrade_locations target = scope:location }
							}
						}
					}
				}
			}
		}
	}
}

				limit = { cm_should_rgo_auto_expand = yes }

				limit = { cm_should_rgo_auto_expand = yes }
				cm_stage_rgo_candidate = yes
			}
		}
	}
}

# Clears cm_upgrade_in_progress for scope:building_type at scope:location when the
# previously queued upgrade construction has finished or was cancelled. This allows
# the next monthly pulse to queue another upgrade for the same building type.
# Expects: scope:building_type, scope:location, scope:country
cm_clear_finished_upgrade_in_progress = {
	if = {
		limit = {
			scope:location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:building_type } }
		}
		if = {
			limit = { cm_does_building_type_count_for_fort_building_cost_modifier = yes }
			# A fort upgrade never exposes its new tier to script while building, so the target-based
			# check below would clear the tracking and re-stage it every pulse. For forts, clear on the
			# source tier disappearing instead, which is the reliable finished signal.
			if = {
				limit = { scope:location = { NOT = { has_building = scope:building_type } } }
				scope:location = { remove_list_variable = { name = cm_upgrade_in_progress target = scope:building_type } }
				scope:location = { remove_from_variable_map = { name = cm_upgrade_in_progress_targets key = scope:building_type } }
			}
		} else_if = {
			limit = {
				scope:location = {
					has_variable_map = cm_upgrade_in_progress_targets
					is_key_in_variable_map = { name = cm_upgrade_in_progress_targets target = scope:building_type }
				}
			}
			scope:location = { "variable_map(cm_upgrade_in_progress_targets|scope:building_type)" = { save_temporary_scope_as = cm_upgrade_target } }
			if = {
				limit = {
					trigger_if = {
						limit = { scope:location = { has_building = scope:cm_upgrade_target } }
						# Upgrade target exists but has no levels under construction — construction finished
						scope:location = { "building(scope:cm_upgrade_target|scope:country)" = { building_levels_under_construction = 0 } }
					}
					trigger_else = {
						# Upgrade target doesn't exist — construction was cancelled
						always = yes
					}
				}
				scope:location = { remove_list_variable = { name = cm_upgrade_in_progress target = scope:building_type } }
				scope:location = { remove_from_variable_map = { name = cm_upgrade_in_progress_targets key = scope:building_type } }
			}
		} else = {
			# No recorded target - stale entry
			scope:location = { remove_list_variable = { name = cm_upgrade_in_progress target = scope:building_type } }
		}
	}
}

# ─── Auto Build ───
# Auto-constructs one-per-location buildings in priority order.
cm_run_auto_build = {
	cm_reset_locations_lists = { }
	every_in_list = {
		variable = cm_auto_build_building_types_list
		save_scope_as = building_type
		scope:country = {
			if = {
				limit = { is_target_in_variable_list = { name = cm_auto_build_building_types_enabled target = scope:building_type } }
				# Save in_subjects flag before resolution overwrites scope:building_type
				if = { limit = { is_target_in_variable_list = { name = cm_auto_build_building_types_in_subjects_enabled target = scope:building_type } }
					set_local_variable = { name = cm_auto_build_include_subjects value = 1 }
				} else = {
					set_local_variable = { name = cm_auto_build_include_subjects value = 0 }
				}
				# Resolve obsolete building type to current non-obsolete upgrade (e.g. dock → dry_dock)
				cm_auto_build_resolve_building_type = yes
				if = {
					limit = { local_var:cm_auto_build_include_subjects = 0 }
					ordered_owned_location = {
						order_by = max_control
						max = cm_max_guess_at_number_of_locations
						check_range_bounds = no
						limit = { NOT = { integration_level = conquered } }
						save_scope_as = location
						if = {
							limit = { AND = { cm_get_open_build_queue_slots > 0 cm_location_can_auto_build = yes } }
							cm_stage_ungated_candidate = yes
						}
					}
				} else = {
					every_in_list = {
						variable = cm_all_locations_including_subjects_list
						save_scope_as = location
						if = {
							limit = { AND = { NOT = { integration_level = conquered } cm_get_open_build_queue_slots > 0 cm_location_can_auto_build = yes } }
							cm_stage_ungated_candidate = yes
						}
					}
				}
			}
		}
	}
	cm_reset_locations_lists = { }
}

		}
	}
}

				cm_try_construct_capital_building = { building_type_text = supreme_court }
				cm_try_construct_capital_building = { building_type_text = chancery }
				cm_try_construct_capital_building = { building_type_text = republican_assembly }
				cm_try_construct_capital_building = { building_type_text = royal_garden }

# Expects scopes: country, capital_location.
# Tries to construct a single capital-only building if all conditions pass.
cm_try_construct_capital_building = {
	building_type:$building_type_text$ = { save_scope_as = building_type }
	if = {
		limit = {
			scope:capital_location = { cm_capital_location_can_auto_build = yes }
		}
		scope:capital_location = { save_scope_as = location }
		cm_stage_ungated_candidate = yes
	}
}

# ─── Auto Town Rights ───
# Grants enabled town rights to eligible locations in priority order.
# Granting costs 5 government_power per use (modified by grant_town_rights_cost_modifier).
cm_run_auto_town_rights = {
	if = {
		limit = {
			has_variable_list = cm_auto_town_rights_list
			modifier:can_grant_town_rights = yes
		}
		every_in_list = {
			variable = cm_auto_town_rights_list
			save_scope_as = town_right
			scope:country = {
				if = {
					limit = {
						is_target_in_variable_list = { name = cm_auto_town_rights_enabled target = scope:town_right }
						government_power >= 5
					}
					ordered_owned_location = {
						order_by = max_control
						max = cm_max_guess_at_number_of_locations
						check_range_bounds = no
						limit = {
							NOT = { integration_level = conquered }
							has_max_town_rights = no
							NOT = { has_town_rights = scope:town_right }
							scope:country = { government_power >= 5 }
						}
						save_scope_as = target
						if = {
							limit = { scope:country = { cm_can_grant_specific_town_right_at_location = yes } }
							grant_town_rights = scope:town_right
						}
					}
				}
			}
		}
	}
}
