# Feature effects for the unified priority dispatcher.
# Each effect is called by the dispatcher in cm_on_action.txt when
# the corresponding feature is enabled in the priority list.
# All effects expect scope:cm_country to be set.

# One country's construction cycle: clears last cycle's queues, calls every enabled priority
# feature and full-automation category in the country's own order, and arms the construct
# window when anything staged. Driven by cm_unified_auto_expand for the player and by
# cm_ln_monthly_cycle for a lucky nation. Root is the country.
cm_run_construction_cycle = {
	save_scope_as = cm_country
	if = { limit = { exists = var:cm_should_construct }
		# A cycle can straddle one month tick at high speed; a second straight skip means it is stuck.
		change_variable = { name = cm_q_skipped_pulses add = 1 }
		if = { limit = { var:cm_q_skipped_pulses >= 2 }
			cm_q_reset_cycle_state = yes
			set_variable = { name = cm_q_skipped_pulses value = 0 }
			# A lucky nation shares the player's 200-entry log, so its pacing lines stay out of it.
			if = { limit = { cm_ln_is_active = no }
				cmf_log = { action = cm_log_cycle_force_reset }
			}
		} else = {
			if = { limit = { cm_ln_is_active = no }
				cmf_log = { action = cm_log_cycle_skipped }
			}
		}
	} else = {
		set_variable = { name = cm_q_skipped_pulses value = 0 }
		# Start a fresh construction queue each cycle, drained once by the queue window.
		cm_clear_queue_pair = { COUNTRY_LIST = cm_q_locations LOCATION_LIST = cm_q_building_types }
		clear_variable_list = cm_q_rgo_locations
		cm_clear_queue_pair = { COUNTRY_LIST = cm_q_ungated_locations LOCATION_LIST = cm_q_ungated_building_types }
		cm_clear_queue_pair = { COUNTRY_LIST = cm_q_urbanize_locations LOCATION_LIST = cm_q_urbanize_building_types }
		cm_clear_queue_pair = { COUNTRY_LIST = cm_q_food_locations LOCATION_LIST = cm_q_food_building_types }
		clear_variable_list = cm_q_food_rgo_locations
		clear_variable_list = cm_q_rank_locations
		cm_clear_queue_pair = { COUNTRY_LIST = cm_fa_b_locations LOCATION_LIST = cm_fa_b_types }
		cm_clear_queue_pair = { COUNTRY_LIST = cm_fa_u_locations LOCATION_LIST = cm_fa_u_types }
		clear_variable_list = cm_fa_r_locations
		clear_variable_list = cm_fa_food_rgo_locations
		clear_variable_list = cm_fa_collapsar_rgo_locations
		cm_q_clear_foreign_queue = yes
		# The upgrade queue carries two extra per-location lists alongside the drained one
		every_in_list = {
			variable = cm_gui_upgrade_locations
			save_scope_as = cm_q_up_loc
			scope:cm_q_up_loc = {
				clear_variable_list = cm_gui_upgrade_building_types
				clear_variable_list = cm_q_upgrade_gated_types
				clear_variable_list = cm_q_upgrade_target_types
			}
		}
		clear_variable_list = cm_gui_upgrade_locations
		cm_clear_queue_pair = { COUNTRY_LIST = cm_q_approved_build_locations LOCATION_LIST = cm_q_approved_build_types }
		cm_clear_queue_pair = { COUNTRY_LIST = cm_q_approved_upgrade_locations LOCATION_LIST = cm_q_approved_upgrade_types }
		clear_variable_list = cm_q_approved_rank_locations
		set_variable = { name = cm_q_planned_spend value = 0 }
		set_variable = { name = cm_q_staged value = 0 }
		set_variable = { name = cm_q_processed value = 0 }
		set_variable = { name = cm_q_progress_last value = 0 }
		set_variable = { name = cm_q_stall_rounds value = 0 }
		cm_q_reset_market_committed_for_cycle = yes
		if = { limit = { cm_should_pause_auto_expand = no }
			if = { limit = { has_variable_list = cm_priority_features_list }
				every_in_list = {
					variable = cm_priority_features_list
					save_scope_as = cm_feature
					if = { limit = { scope:cm_country = { is_target_in_variable_list = { name = cm_priority_features_enabled target = scope:cm_feature } } }
						switch = {
							trigger = scope:cm_feature
							flag:cm_feature_auto_expand_buildings = { scope:cm_country = { cm_run_auto_expand_buildings = yes } }
							flag:cm_feature_auto_expand_rgos = { scope:cm_country = { cm_run_auto_expand_rgos = yes } }
							flag:cm_feature_upgrade_all_buildings = { scope:cm_country = { cm_run_upgrade_all_buildings = yes } }
							flag:cm_feature_auto_build = { scope:cm_country = { cm_run_auto_build = yes } }
							flag:cm_feature_auto_food = { scope:cm_country = { cm_run_auto_food = yes } }
							flag:cm_feature_auto_urbanize = { scope:cm_country = { cm_run_auto_urbanize = yes } }
							flag:cm_feature_auto_build_capital = { scope:cm_country = { cm_run_auto_build_capital = yes } }
						}
					}
				}
			}
		}
		# Full-automation staging after the priority features, so the player's own
		# features always stage and claim first. Categories stage in the player's
		# order, so the higher one claims a candidate both of them want.
		if = {
			limit = {
				cm_fa_master_active = yes
				has_variable_list = cm_fa_categories_enabled
			}
			every_in_list = {
				variable = cm_fa_categories_enabled
				save_scope_as = cm_fa_cat
				scope:cm_country = { cm_fa_stage_category = yes }
			}
		}
		# Show the construction-queue window to construct this month's staged items.
		# The full-automation RGO lists bypass the staged counter (no GUI scoring), so
		# they arm the window on their own.
		if = {
			limit = {
				OR = {
					var:cm_q_staged >= 1
					AND = {
						has_variable_list = cm_fa_food_rgo_locations
						variable_list_size = { name = cm_fa_food_rgo_locations value >= 1 }
					}
					AND = {
						has_variable_list = cm_fa_collapsar_rgo_locations
						variable_list_size = { name = cm_fa_collapsar_rgo_locations value >= 1 }
					}
				}
			}
			set_variable = { name = cm_should_construct value = yes }
		}
	}
}

# Add-if-absent queue staging shared by the cm_stage_* effects below: appends
# scope:cm_location to the country's $COUNTRY_LIST$ and scope:cm_building_type to the
# location's $LOCATION_LIST$. $DONE_LIST$ is the queue's per-location processed-marker list.
# Expects scope:cm_country, scope:cm_location, scope:cm_building_type.
cm_stage_location_and_type_to_queue = {
	scope:cm_country = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = $COUNTRY_LIST$ target = scope:cm_location } } }
			add_to_variable_list = { name = $COUNTRY_LIST$ target = scope:cm_location }
			scope:cm_location = { clear_variable_list = $DONE_LIST$ }
		}
	}
	scope:cm_location = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = $LOCATION_LIST$ target = scope:cm_building_type } } }
			add_to_variable_list = { name = $LOCATION_LIST$ target = scope:cm_building_type }
			scope:cm_country = { change_variable = { name = cm_q_staged add = 1 } }
		}
	}
}

# Location-only variant for queues without a per-location type list.
# $DONE_VAR$ is the queue's per-location processed-marker variable.
# Expects scope:cm_country, scope:cm_location.
cm_stage_location_to_queue = {
	scope:cm_country = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = $COUNTRY_LIST$ target = scope:cm_location } } }
			add_to_variable_list = { name = $COUNTRY_LIST$ target = scope:cm_location }
			change_variable = { name = cm_q_staged add = 1 }
			scope:cm_location = { remove_variable = $DONE_VAR$ }
		}
	}
}

# Empties each queued location's $LOCATION_LIST$, then the country's $COUNTRY_LIST$.
# Root is the country.
cm_clear_queue_pair = {
	every_in_list = {
		variable = $COUNTRY_LIST$
		save_scope_as = cm_q_clear_loc
		scope:cm_q_clear_loc = { clear_variable_list = $LOCATION_LIST$ }
	}
	clear_variable_list = $COUNTRY_LIST$
}

# outliving their cycle, and clearing the per-location scoring state keeps an aborted
# cycle's scores out of the next approval pass. Root is the country.

	cm_clear_queue_pair = { COUNTRY_LIST = cm_q_approved_upgrade_locations LOCATION_LIST = cm_q_approved_upgrade_types }
	clear_variable_list = cm_q_approved_rank_locations

		variable = cm_q_locations
		save_scope_as = cm_q_reset_loc
		scope:cm_q_reset_loc = { cm_q_loc_reset_build_scoring = yes }
	}
	every_in_list = {
		variable = cm_gui_upgrade_locations
		save_scope_as = cm_q_reset_loc

	}
	every_in_list = {
		variable = cm_q_food_rgo_locations
		save_scope_as = cm_q_reset_loc
		scope:cm_q_reset_loc = { cm_q_loc_reset_food_rgo_scoring = yes }
	}
	every_in_list = {
		variable = cm_q_rank_locations
		save_scope_as = cm_q_reset_loc
		scope:cm_q_reset_loc = { cm_q_loc_reset_rank_scoring = yes }

		variable = cm_fa_b_locations

	every_in_list = {

# Root is the country. The foreign queue keeps no per-location state, so its whole teardown
# is four country-side structures.
cm_q_clear_foreign_queue = {
	clear_variable_list = cm_q_fb_locations
	clear_variable_list = cm_q_fb_types
	clear_variable_map = cm_q_fb_pair
	clear_variable_map = cm_q_fb_costs
}

# Per-cycle construction queue: a country list of locations, each carrying a location list
# of building types. The queue window scores each entry with the engine cost/income GUI
# bindings, then cm_q_approval_pass approves best metric first.
# Expects scope:cm_country, scope:cm_location, scope:cm_building_type.
cm_stage_candidate = {
	cm_stage_location_and_type_to_queue = { COUNTRY_LIST = cm_q_locations LOCATION_LIST = cm_q_building_types DONE_LIST = cm_q_done_build_types }
}

# Per-cycle foreign construction queue. Every part of it is stored on the acting country: a
# foreign location's variables are shared save data that another player's cycle also writes.
# The two lists are the queue window's outer and inner datamodels and cm_q_fb_pair is the
# (location, type) membership its cross product filters on, so one foreign type per location
# per cycle; a second type at the same location waits for the next one.
# Expects scope:cm_country, scope:cm_location, scope:cm_building_type.
cm_stage_foreign_candidate = {
	scope:cm_country = {
		if = {
			limit = {
				NOT = {
					AND = {
						has_variable_map = cm_q_fb_pair
						is_key_in_variable_map = { name = cm_q_fb_pair target = scope:cm_location }
					}
				}
			}
			add_to_variable_map = { name = cm_q_fb_pair key = scope:cm_location value = scope:cm_building_type }
			add_to_variable_list = { name = cm_q_fb_locations target = scope:cm_location }
			add_to_variable_list = { name = cm_q_fb_types target = scope:cm_building_type }
			change_variable = { name = cm_q_staged add = 1 }
		}
	}
}

# Gated upgrade candidate for the unified queue, keyed by the standing obsolete type.
# The queue window resolves the replacement tier and its engine values at drain time.
# Eligibility reads the standing building for employment and goods; the discount and
# pops gates read the replacement type because the replacement is what gets constructed.
# scope:cm_building_type holds the replacement between the stash and the restore below.
# Expects scope:cm_country, scope:cm_location, scope:cm_building, scope:cm_building_type as the
# standing obsolete type, scope:cm_current_building_type as the resolved replacement.
cm_stage_upgrade_candidate = {
	cm_clear_finished_upgrade_in_progress = yes
	scope:cm_building_type = { save_scope_as = cm_ae_staged_type }
	scope:cm_current_building_type = { save_scope_as = cm_building_type }
	if = {
		limit = {
			scope:cm_building = {
				cm_building_can_auto_expand_for_upgrade = yes
				cm_is_not_under_construction = yes
			}
			scope:cm_location = { cm_location_building_type_not_under_construction = yes }
			cm_auto_expand_check_minimum_gold_reserve = yes
			NOT = { scope:cm_location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:cm_ae_staged_type } } }
		}
		scope:cm_location = {
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_building_types target = scope:cm_ae_staged_type } } }
				add_to_variable_list = { name = cm_gui_upgrade_building_types target = scope:cm_ae_staged_type }
				add_to_variable_list = { name = cm_q_upgrade_gated_types target = scope:cm_ae_staged_type }
				scope:cm_country = { change_variable = { name = cm_q_staged add = 1 } }
			}
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_current_building_type } } }
				add_to_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_current_building_type }
			}
		}
		scope:cm_country = {
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_locations target = scope:cm_location } } }
				add_to_variable_list = { name = cm_gui_upgrade_locations target = scope:cm_location }
				scope:cm_location = { clear_variable_list = cm_q_done_upgrade_types }
			}
		}
	}
	scope:cm_ae_staged_type = { save_scope_as = cm_building_type }
}

# From-zero auto-expand path for a building type with no levels at the location. Expects
# scope:cm_country, scope:cm_location, scope:cm_building_type.
cm_stage_from_zero_candidate = {
	if = {
		limit = { scope:cm_location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:cm_building_type } } }
		scope:cm_location = { remove_list_variable = { name = cm_upgrade_in_progress target = scope:cm_building_type } }
	}
	if = {
		limit = { cm_is_building_type_obsolete = no }
		if = {
			limit = {
				cm_location_can_auto_expand_from_zero_levels = yes
				NOT = { scope:cm_location = { is_target_in_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_building_type } } }
			}
			cm_stage_candidate = yes
		}
	}
}

# Staging decision for a building that already has levels at the location. Expects scope:cm_country,
# scope:cm_location, scope:cm_building, and scope:cm_building_type as the standing type.
cm_stage_expand_or_upgrade_candidate = {
	if = {
		limit = { cm_is_building_type_obsolete = no }
		if = {
			limit = {
				scope:cm_building = {
					cm_building_can_auto_expand_other_than_checking_for_upgrades = yes
					is_at_max_level = no
				}
				# A staged upgrade already constructs this type here this pulse
				NOT = { scope:cm_location = { is_target_in_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_building_type } } }
				scope:cm_location = { cm_get_open_build_queue_slots > 0 }
			}
			cm_stage_candidate = yes
		}
	} else_if = {
		limit = {
			scope:cm_location = { cm_is_building_type_upgradeable_and_can_be_upgraded_by_owner = yes }
		}
		scope:cm_building_type = { cm_find_current_building_type = yes }
		cm_stage_upgrade_candidate = yes
	}
}

# Drain-time approval: the queue window fires the engine build action once per
# approved entry. The script construct_building effect does not charge
# special-currency price components.
# Root is the location. Expects scope:cm_country, scope:cm_building_type, scope:cm_cost.
cm_approve_gui_build = {
	if = {
		limit = { NOT = { is_target_in_variable_list = { name = cm_q_approved_build_types target = scope:cm_building_type } } }
		cm_q_approve_build_copy = yes
	}
}

# Base approved entry; extra levels of the same type go to the rank lists.
# Root is the location. Expects scope:cm_country, scope:cm_building_type, scope:cm_cost.
cm_q_approve_build_copy = {
	add_to_variable_list = { name = cm_q_approved_build_types target = scope:cm_building_type }
	save_temporary_scope_as = cm_approved_location
	scope:cm_country = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = cm_q_approved_build_locations target = scope:cm_approved_location } } }
			add_to_variable_list = { name = cm_q_approved_build_locations target = scope:cm_approved_location }
			scope:cm_approved_location = { clear_variable_list = cm_q_seen_build_types }
		}
	}
	cm_q_claim_build_copy = yes
	cm_ln_construct_approved_build = yes
}

# One extra level of an approved build. A variable list holds each target once, so the
# player's batch sends every level to a rank list of its own; a lucky nation has no fire pass
# and constructs it here. $GUARD$ is the caller's copy counter.
# Root is the location. Expects scope:cm_country, scope:cm_building_type.
cm_q_add_build_rank_copy = {
	if = {
		limit = { scope:cm_country = { cm_ln_is_active = yes } }
		cm_ln_construct_build = yes
	} else = {
		if = {
			limit = { local_var:$GUARD$ = 1 }
			add_to_variable_list = { name = cm_q_approved_build_x2 target = scope:cm_building_type }
		} else_if = {
			limit = { local_var:$GUARD$ = 2 }
			add_to_variable_list = { name = cm_q_approved_build_x3 target = scope:cm_building_type }
		} else_if = {
			limit = { local_var:$GUARD$ = 3 }
			add_to_variable_list = { name = cm_q_approved_build_x4 target = scope:cm_building_type }
		} else_if = {
			limit = { local_var:$GUARD$ = 4 }
			add_to_variable_list = { name = cm_q_approved_build_x5 target = scope:cm_building_type }
		} else_if = {
			limit = { local_var:$GUARD$ = 5 }

# Force-disables scope:cm_building_type at the root location without toggling. Root = location,

cm_disable_auto_expand_for_building_type = {
	if = { limit = { scope:cm_building_type = { is_foreign = yes } }

# Toggles auto-expand for scope:cm_building_type at the root location.
# Optional scope:cm_set_off forces the direction instead of toggling.
# A foreign type additionally expects scope:cm_country.
cm_apply_auto_expand_toggle = {
	if = { limit = { scope:cm_building_type = { is_foreign = yes } }
		cm_fb_apply_auto_expand_toggle = yes
	} else = {
		if = { limit = { owner = { is_target_in_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_building_type } } }
			# Mass mode: operate on exclusion list with inverted logic.
			if = { limit = { NOT = { exists = scope:cm_set_off } }
				if = { limit = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_building_type } }
					remove_list_variable = { name = cm_auto_expand_excluded_building_types target = scope:cm_building_type }
				} else = {
					add_to_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_building_type }
				}
			} else = {
				if = { limit = { scope:cm_set_off = yes }
					cm_disable_auto_expand_for_building_type = yes
				} else = {
					cm_enable_auto_expand_for_building_type = yes
				}
			}
		} else = {
			# Normal mode: operate on registration list
			if = { limit = { NOT = { exists = scope:cm_set_off } }
				if = { limit = { is_target_in_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_building_type } }
					remove_list_variable = { name = cm_auto_expand_registered_building_types target = scope:cm_building_type }
				} else = {
					add_to_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_building_type }
				}
			} else = {
				if = { limit = { scope:cm_set_off = yes }
					cm_disable_auto_expand_for_building_type = yes
				} else = {
					cm_enable_auto_expand_for_building_type = yes
				}
			}
		}
	}
}

# Ctrl-click bulk-toggle driver. Root = location (the clicked location). Expects
# scope:cm_building_type and scope:cm_country. Mirrors the province's overall state, like the
# existing shift-click filtered toggle: if every eligible same-owner location in the province
# already has this building type active, turns it off everywhere; otherwise turns it on
# everywhere, including locations without this building yet.
# scope:cm_fb_host selects which locations the sweep covers, which for a foreign type is the
# host's land rather than scope:cm_country's.
cm_toggle_auto_expand_for_building_type_in_province = {

	owner = { save_scope_as = cm_fb_host }

						NOT = { integration_level = conquered }
						NOT = { cm_location_is_auto_expand_active_for_building_type = yes }

				}

						owner = scope:cm_fb_host
						NOT = { integration_level = conquered }
					}

			} else = {

					limit = {
						owner = scope:cm_fb_host
						NOT = { integration_level = conquered }
					}
					cm_disable_auto_expand_for_building_type = yes
				}
			}
		}
	}
}

# Moves auto-expand registrations from researched obsolete types to their current
# replacement types. Unresolvable types stay registered and the obsolete fallback
# branches keep upgrading them. Expects scope:cm_country.
cm_transfer_obsolete_auto_expand_registrations = {
	ordered_owned_location = {
		order_by = max_control
		max = cm_max_guess_at_number_of_locations
		check_range_bounds = no
		limit = { has_variable_list = cm_auto_expand_registered_building_types }
		save_scope_as = cm_ae_transfer_location
		every_in_list = {
			variable = cm_auto_expand_registered_building_types
			save_scope_as = cm_ae_old_type
			if = {
				limit = { scope:cm_country = { building_type_is_obsolete = scope:cm_ae_old_type } }
				scope:cm_ae_old_type = { cm_find_current_building_type = yes }
				if = {
					limit = { NOT = { scope:cm_country = { building_type_is_obsolete = scope:cm_current_building_type } } }
					scope:cm_ae_transfer_location = {
						remove_list_variable = { name = cm_auto_expand_registered_building_types target = scope:cm_ae_old_type }
						# Mass coverage of the replacement supersedes a per-location entry
						if = {
							limit = {
								NOT = { scope:cm_country = { is_target_in_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_current_building_type } } }
								NOT = { is_target_in_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_current_building_type } }
							}
							add_to_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_current_building_type }
						}
					}
				}
			}
		}
	}
	if = {
		limit = { has_variable_list = cm_mass_auto_expand_building_types }
		every_in_list = {
			variable = cm_mass_auto_expand_building_types
			save_scope_as = cm_ae_old_type
			if = {
				limit = { scope:cm_country = { building_type_is_obsolete = scope:cm_ae_old_type } }
				scope:cm_ae_old_type = { cm_find_current_building_type = yes }
				if = {
					limit = { NOT = { scope:cm_country = { building_type_is_obsolete = scope:cm_current_building_type } } }
					scope:cm_country = {
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_current_building_type } } }
							add_to_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_current_building_type }
						}
						ordered_owned_location = {
							order_by = max_control
							max = cm_max_guess_at_number_of_locations
							check_range_bounds = no
							if = {
								limit = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_ae_old_type } }
								remove_list_variable = { name = cm_auto_expand_excluded_building_types target = scope:cm_ae_old_type }
								if = {
									limit = { NOT = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_current_building_type } } }
									add_to_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_current_building_type }
								}
							}
							if = {
								limit = { is_target_in_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_current_building_type } }
								remove_list_variable = { name = cm_auto_expand_registered_building_types target = scope:cm_current_building_type }
							}
						}
						remove_list_variable = { name = cm_mass_auto_expand_building_types target = scope:cm_ae_old_type }
					}
				}
			}
		}
	}
}

cm_run_auto_expand_buildings = {
	cm_transfer_obsolete_auto_expand_registrations = yes
	if = {
		limit = { has_variable_list = cm_mass_auto_expand_building_types }
		# -- Mass mode: iterate all existing buildings at all owned locations --
		ordered_owned_location = {
			order_by = max_control
			max = cm_max_guess_at_number_of_locations
			check_range_bounds = no
			limit = { NOT = { integration_level = conquered } }
			save_scope_as = cm_location
			ordered_buildings_in_location = {
				order_by = building_index
				max = 100
				check_range_bounds = no
				save_scope_as = cm_building
				building_type = { save_scope_as = cm_building_type }
				# Standing obsolete buildings match the lists through their replacement type
				scope:cm_building_type = { cm_find_current_building_type = yes }
				if = {
					limit = {
						# The mass list holds foreign types too; cm_run_auto_expand_foreign owns those,
						# and this loop would otherwise stage another country's building at our location.
						scope:cm_current_building_type = { is_foreign = no }
						# Building type is mass-toggled for this country
						scope:cm_country = { is_target_in_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_current_building_type } }
						# Not excluded by the player at this location
						NOT = { scope:cm_location = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_current_building_type } } }
					}
					cm_stage_expand_or_upgrade_candidate = yes
				}
			}
			# -- Mass mode: construct missing buildings from zero levels --
			scope:cm_country = {
				every_in_list = {
					variable = cm_mass_auto_expand_building_types
					limit = { is_foreign = no }
					save_scope_as = cm_building_type
					if = {
						limit = {
							NOT = { scope:cm_location = { has_building = scope:cm_building_type } }
							NOT = { scope:cm_location = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_building_type } } }
						}
						cm_stage_from_zero_candidate = yes
					}
				}
			}
		}
	}
	# -- Normal mode: iterate per-location registered building types --
	# Runs unconditionally - per-location registrations are processed alongside mass mode.
	ordered_owned_location = {
		order_by = max_control
		max = cm_max_guess_at_number_of_locations
		check_range_bounds = no
		limit = { NOT = { integration_level = conquered } has_variable_list = cm_auto_expand_registered_building_types }
		save_scope_as = cm_location
		# Upgrades for standing obsolete buildings whose replacement is registered
		ordered_buildings_in_location = {
			order_by = building_index
			max = 100
			check_range_bounds = no
			save_scope_as = cm_building
			building_type = { save_scope_as = cm_building_type }
			if = {
				limit = { cm_is_building_type_obsolete = yes }
				scope:cm_building_type = { cm_find_current_building_type = yes }
				if = {
					limit = {
						scope:cm_location = { is_target_in_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_current_building_type } }
						scope:cm_location = { cm_is_building_type_upgradeable_and_can_be_upgraded_by_owner = yes }
					}
					cm_stage_upgrade_candidate = yes
				}
			}
		}
		every_in_list = {
			variable = cm_auto_expand_registered_building_types
			save_scope_as = cm_building_type
			if = {
				limit = { scope:cm_location = { has_building = scope:cm_building_type } }
				scope:cm_location = { "building(scope:cm_building_type|scope:cm_country)" = { save_scope_as = cm_building } }
				cm_stage_expand_or_upgrade_candidate = yes
			} else = {
				cm_stage_from_zero_candidate = yes
			}
		}
	}
	cm_run_auto_expand_foreign = yes
}

# Clears cm_upgrade_in_progress for scope:cm_building_type at scope:cm_location when the
# previously queued upgrade construction has finished or was cancelled. This allows
# the next monthly pulse to queue another upgrade for the same building type.
# Expects: scope:cm_building_type, scope:cm_location, scope:cm_country
cm_clear_finished_upgrade_in_progress = {
	if = {
		limit = {
			scope:cm_location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:cm_building_type } }
		}
		if = {
			limit = { cm_does_building_type_count_for_fort_building_cost_modifier = yes }
			# A fort upgrade never exposes its new tier to script while building, so the target-based
			# check below would clear the tracking and re-stage it every pulse. For forts, clear on the
			# source tier disappearing instead, which is the reliable finished signal.
			if = {
				limit = { scope:cm_location = { NOT = { has_building = scope:cm_building_type } } }
				scope:cm_location = { remove_list_variable = { name = cm_upgrade_in_progress target = scope:cm_building_type } }
				scope:cm_location = { remove_from_variable_map = { name = cm_upgrade_in_progress_targets key = scope:cm_building_type } }
			}
		} else_if = {
			limit = {
				scope:cm_location = {
					has_variable_map = cm_upgrade_in_progress_targets
					is_key_in_variable_map = { name = cm_upgrade_in_progress_targets target = scope:cm_building_type }
				}
			}
			scope:cm_location = { "variable_map(cm_upgrade_in_progress_targets|scope:cm_building_type)" = { save_temporary_scope_as = cm_upgrade_target } }
			if = {
				limit = {
					trigger_if = {
						limit = { scope:cm_location = { has_building = scope:cm_upgrade_target } }
						# Upgrade target exists but has no levels under construction - construction finished
						scope:cm_location = { "building(scope:cm_upgrade_target|scope:cm_country)" = { building_levels_under_construction = 0 } }
					}
					trigger_else = {
						# Upgrade target doesn't exist - construction was cancelled
						always = yes
					}
				}
				scope:cm_location = { remove_list_variable = { name = cm_upgrade_in_progress target = scope:cm_building_type } }
				scope:cm_location = { remove_from_variable_map = { name = cm_upgrade_in_progress_targets key = scope:cm_building_type } }
			}
		} else = {
			# No recorded target - stale entry
			scope:cm_location = { remove_list_variable = { name = cm_upgrade_in_progress target = scope:cm_building_type } }
		}
	}
}

# --- Upgrade All Buildings ---
# Upgrades every obsolete building in all owned locations.
cm_run_upgrade_all_buildings = {
	ordered_owned_location = {
		order_by = max_control
		max = cm_max_guess_at_number_of_locations
		check_range_bounds = no
		limit = { NOT = { integration_level = conquered } }
		save_scope_as = cm_location
		ordered_buildings_in_location = {
			order_by = building_index
			max = 100
			check_range_bounds = no
			save_scope_as = cm_building
			building_type = { save_scope_as = cm_building_type }
			cm_clear_finished_upgrade_in_progress = yes
			if = {
				limit = {
					cm_is_building_obsolete = yes
					location = { cm_is_building_type_upgradeable_and_can_be_upgraded_by_owner = yes }
					cm_is_not_under_construction = yes
					cm_auto_expand_check_minimum_gold_reserve = yes
					cm_upgrade_all_meets_minimum_discount = yes
					NOT = { scope:cm_location = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:cm_building_type } } }
				}
				scope:cm_building_type = { cm_find_current_building_type = yes }
				if = {
					# An in-flight upgrade carries its construction levels on the replacement type.
					limit = {
						scope:cm_location = {
							NOT = {
								any_buildings_in_location = {
									building_type = scope:cm_current_building_type
									building_levels_under_construction > 0
								}
							}
						}
					}
					scope:cm_location = { remove_list_variable = { name = cm_q_upgrade_gated_types target = scope:cm_building_type } }
					# Record the replacement for the per-pulse construct dedupe
					scope:cm_location = {
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_current_building_type } } }
							add_to_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_current_building_type }
						}
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_building_types target = scope:cm_building_type } } }
							add_to_variable_list = { name = cm_gui_upgrade_building_types target = scope:cm_building_type }
							scope:cm_country = { change_variable = { name = cm_q_staged add = 1 } }
						}
					}
					scope:cm_country = {
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_locations target = scope:cm_location } } }
							add_to_variable_list = { name = cm_gui_upgrade_locations target = scope:cm_location }
							scope:cm_location = { clear_variable_list = cm_q_done_upgrade_types }
						}
					}
				}
			}
		}
	}
}

# --- Auto Build ---
# Auto-constructs one-per-location buildings in priority order.

					set_local_variable = { name = cm_auto_build_include_subjects value = 0 }
				}

							limit = { AND = { cm_get_open_build_queue_slots > 0 cm_location_can_auto_build = yes } }
							cm_stage_ungated_candidate = yes
						}
					}
				} else = {
					every_in_list = {
						variable = cm_all_locations_including_subjects_list
						save_scope_as = cm_location
						if = {
							limit = { AND = { NOT = { integration_level = conquered } cm_get_open_build_queue_slots > 0 cm_location_can_auto_build = yes } }
							cm_stage_ungated_candidate = yes
						}
					}
				}
			}
		}
	}
}

		limit = { has_variable_list = cm_auto_food_rgo_types_list }
		every_in_list = {
			variable = cm_auto_food_rgo_types_list
			save_scope_as = cm_food_good
			scope:cm_country = {
				cm_food_auto_build_for_rgo_type = yes
			}
		}
	}
}

# --- Auto Urbanize ---
# Promotes location ranks: towns to cities, and rural settlements to towns.
# Each upgrade type is gated by its own sub-toggle on the Urbanize tab.
cm_run_auto_urbanize = {
	if = {
		limit = { exists = var:cm_urbanize_town_to_city }
		ordered_owned_location = {
			order_by = max_control
			max = cm_max_guess_at_number_of_locations
			check_range_bounds = no
			save_scope_as = cm_location
			limit = { NOT = { integration_level = conquered } cm_location_can_mass_upgrade_to_city = yes }
			cm_stage_urbanize_candidate = { bt = cm_upgrade_town_to_city_building }
		}
	}
	if = {
		limit = { exists = var:cm_urbanize_rural_to_town }
		ordered_owned_location = {
			order_by = max_control
			max = cm_max_guess_at_number_of_locations
			check_range_bounds = no
			save_scope_as = cm_location
			limit = { NOT = { integration_level = conquered } cm_location_can_mass_upgrade_to_town = yes }
			cm_stage_urbanize_candidate = { bt = cm_upgrade_rural_to_town_building }
		}
	}
	if = {
		limit = {
			exists = var:cm_urbanize_city_to_megalopolis
			is_subject = no
		}
		ordered_owned_location = {
			order_by = max_control
			max = cm_max_guess_at_number_of_locations
			check_range_bounds = no
			save_scope_as = cm_location
			limit = {
				location_rank = location_rank:city
				NOT = { integration_level = conquered }
				modifier:cannot_upgrade_location = no
				population >= cm_megalopolis_stage_min_population
			}
			cm_stage_rank_candidate = yes
		}
	}
}

# --- Auto Build Capital ---
# Auto-constructs capital-only buildings in the country's capital location.
cm_run_auto_build_capital = {
	if = {
		limit = { exists = capital }
		capital = {
			save_scope_as = cm_capital_location
			if = {
				limit = { NOT = { integration_level = conquered } }
				cm_try_construct_capital_building = { building_type_text = royal_court }
				cm_try_construct_capital_building = { building_type_text = house_of_parliament }
				cm_try_construct_capital_building = { building_type_text = order_headquarters }
				cm_try_construct_capital_building = { building_type_text = admiralty }
				cm_try_construct_capital_building = { building_type_text = war_college }
				cm_try_construct_capital_building = { building_type_text = tax_assessor }
				cm_try_construct_capital_building = { building_type_text = supreme_court }
				cm_try_construct_capital_building = { building_type_text = chancery }
				cm_try_construct_capital_building = { building_type_text = republican_assembly }
				cm_try_construct_capital_building = { building_type_text = royal_garden }
				cm_try_construct_capital_building = { building_type_text = pirate_stronghold }
			}
		}
	}
}

# Expects scopes: country, capital_location.
# Tries to construct a single capital-only building if all conditions pass.
cm_try_construct_capital_building = {
	building_type:$building_type_text$ = { save_scope_as = cm_building_type }
	if = {
		limit = {
			scope:cm_capital_location = { cm_capital_location_can_auto_build = yes }
		}
		scope:cm_capital_location = { save_scope_as = cm_location }
		cm_stage_ungated_candidate = yes
	}
}

# --- Auto Town Rights ---
# Grants enabled town rights to eligible locations in priority order.
# Raw grant_town_rights skips the price.
cm_run_auto_town_rights = {
	if = {
		limit = {
			has_variable_list = cm_auto_town_rights_list
			modifier:can_grant_town_rights = yes
		}
		every_in_list = {
			variable = cm_auto_town_rights_list
			save_scope_as = cm_town_right
			scope:cm_country = {
				if = {
					limit = {
						is_target_in_variable_list = { name = cm_auto_town_rights_enabled target = scope:cm_town_right }
						cm_auto_town_rights_potential_passes = yes
						cm_town_rights_grant_affordable = yes
					}
					ordered_owned_location = {
						order_by = max_control
						max = cm_max_guess_at_number_of_locations
						check_range_bounds = no
						limit = {
							NOT = { integration_level = conquered }
							has_max_town_rights = no
							NOT = { has_town_rights = scope:cm_town_right }
						}
						save_scope_as = target
						# ordered_owned_location snapshots its limit before the body runs.
						if = {
							limit = {
								scope:cm_country = {
									cm_town_rights_grant_affordable = yes
									cm_can_grant_specific_town_right_at_location = yes
								}
							}
							grant_town_rights = scope:cm_town_right
							if = {
								limit = { NOT = { cm_town_rights_grants_are_free = yes } }
								scope:cm_country = { pay_price = price:grant_town_rights }
							}
						}
					}
				}
			}
		}
	}
}
