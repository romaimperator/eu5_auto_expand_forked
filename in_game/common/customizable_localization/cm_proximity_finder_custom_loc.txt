# Governor/capital finder tooltip status line: names rank 2/3 or a local maximum. Appended to
# the scored tooltip via [ROOT.GetLocation.Custom('cm_pf_gf_status' / 'cm_pf_cf_status')].
cm_pf_gf_status = {
	type = location
	text = {
		trigger = {
			has_variable = cm_gf_top
			var:cm_gf_top = 2
		}
		localization_key = cm_pf_status_second
	}
	text = {
		trigger = {
			has_variable = cm_gf_top
			var:cm_gf_top = 3
		}
		localization_key = cm_pf_status_third
	}
	text = {
		# Skip capital/governor tiles: they show their own stripes, not the gray local-max one.
		trigger = {
			has_variable = cm_gf_localmax
			NOT = { modifier:local_proximity_source > 0 }
		}
		localization_key = cm_pf_status_localmax
	}
}

cm_pf_cf_status = {
	type = location
	text = {
		trigger = {
			has_variable = cm_cf_top
			var:cm_cf_top = 2
		}
		localization_key = cm_pf_status_second
	}
	text = {
		trigger = {
			has_variable = cm_cf_top
			var:cm_cf_top = 3
		}
		localization_key = cm_pf_status_third
	}
	text = {
		# Skip source tiles: they show the magenta stripe, not the gray local-max one.
		trigger = {
			has_variable = cm_cf_localmax
			NOT = { modifier:local_proximity_source > 0 }
		}
		localization_key = cm_pf_status_localmax
	}
}

# Debug label for the predicted-proximity line in CM_PF_DEBUG_BLOCK, matching the roads setting
# the predicted value was computed under.
cm_pf_pred_label = {
	type = location
	text = {
		trigger = {
			owner = { cm_pf_roads_is_current = yes }
		}
		localization_key = cm_pf_pred_label_current
	}
	text = {
		localization_key = cm_pf_pred_label_max
	}
}

# Gates the entire --debug-- block behind the diagnostic setting: with it off, the debug-mode
# tooltip matches the plain one.
cm_pf_debug_block = {
	type = location
	text = {
		trigger = {
			owner = { exists = var:cm_placement_finder_debug_path }
		}
		localization_key = cm_pf_debug_block_text
	}
	text = {
		localization_key = cm_pf_blank
		fallback = yes
	}
}

# Debug path from the capital, shown at the bottom of the debug block. Custom loc cannot iterate,
# so the path is a fixed set of slots read off the hovered location's cm_pf_prev chain: slot k shows
# the node k-1 hops up the chain, gated on cm_pf_dep (its hop count from the capital). cm_pf_path_body
# concatenates them highest-depth-first so the capital leads; unfilled slots emit nothing.
cm_pf_path_block = {
	type = location
	text = {
		trigger = { has_variable = cm_pf_prev }
		localization_key = cm_pf_path_body
	}
	text = {
		localization_key = cm_pf_blank
		fallback = yes
	}
}
