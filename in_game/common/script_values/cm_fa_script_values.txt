# Full-automation script values.

# Root is the country. Returns the full-automation gold reserve setting.
cm_fa_get_reserve = {
	if = {
		limit = { exists = var:cm_fa_reserve }
		value = var:cm_fa_reserve
	}
}

# Root is the country. The food surplus target as a supply/demand ratio: the fa_food_target
# percent setting over 100 (125 -> 1.25). A market whose best food month stayed below this
# gets new auto-food locations.
cm_fa_food_target_ratio = {
	if = {
		limit = { exists = var:cm_fa_food_target }
		value = var:cm_fa_food_target
		divide = 100
	}
	else = {
		value = 1.25
	}
}

# The next RGO level's cost plus the full-automation reserve and this drain's committed
# spend. Root is the location, expects scopes: country, location.
cm_fa_rgo_gold_threshold = {
	value = glorpui_rgo_expand_cost
	scope:cm_country = {
		add = cm_fa_get_reserve
		add = var:cm_q_planned_spend
	}
}

# Ordered iterators sort high to low, so negating takes the lowest access first.
# Root is the location.
cm_fa_market_access_inverted = {
	value = 0
	subtract = market_access
}

# max(ceil(10% of the market's locations), 6): the cap on this market's locations with
# collapsar constructions still being absorbed. Root is the market.
cm_fa_collapsar_market_cap = {
	value = market_location_count
	multiply = 0.1
	ceiling = yes
	min = 6
}

# Root is the country. The urban-rights score window as a fraction: the
# fa_urban_rights_window percent setting over 100 (50 -> 0.5).
cm_fa_ur_window_ratio = {
	if = {
		limit = { exists = var:cm_fa_urban_rights_window }
		value = var:cm_fa_urban_rights_window
		divide = 100
	}
	else = {
		value = 0.5
	}
}

# Root is the country. Government power needed to grant without dropping below the
# full-automation urban-rights reserve.
cm_fa_ur_min_gov_power = {
	if = {
		limit = { exists = var:cm_fa_urban_rights_reserve }
		value = var:cm_fa_urban_rights_reserve
	}
	add = cm_grant_town_rights_cost
}

# The best fit score among the royal specialization rights still grantable here, 0 when
# none is. Root is the location, expects scope:cm_country.
# Rights: in_game/common/town_rights/01_discovery.txt
cm_fa_ur_best_available = {
	value = 0
	if = {
		limit = { cm_trmm_can_grant_right = { RIGHT = royal_tooling_rights } }
		min = cm_trmm_rightv_tooling
	}
	if = {
		limit = { cm_trmm_can_grant_right = { RIGHT = royal_jewelry_rights } }
		min = cm_trmm_rightv_jewelry
	}
	if = {
		limit = { cm_trmm_can_grant_right = { RIGHT = royal_naval_rights } }
		min = cm_trmm_rightv_naval
	}
	if = {
		limit = { cm_trmm_can_grant_right = { RIGHT = royal_textile_rights } }
		min = cm_trmm_rightv_textile
	}
	if = {
		limit = { cm_trmm_can_grant_right = { RIGHT = royal_weaponry_rights } }
		min = cm_trmm_rightv_weaponry
	}
	if = {
		limit = { cm_trmm_can_grant_right = { RIGHT = royal_book_rights } }
		min = cm_trmm_rightv_book
	}
	if = {
		limit = { cm_trmm_can_grant_right = { RIGHT = royal_artisan_rights } }
		min = cm_trmm_rightv_artisan
	}
	if = {
		limit = { cm_trmm_can_grant_right = { RIGHT = royal_brewing_rights } }
		min = cm_trmm_rightv_brewing
	}
	if = {
		limit = { cm_trmm_can_grant_right = { RIGHT = royal_masonry_rights } }
		min = cm_trmm_rightv_masonry
	}
}

# The baked best-fit score, safe on locations the map mode never scored.
# Root is the location.
cm_fa_ur_order = {
	value = 0
	if = {
		limit = { exists = var:cm_trmm_best_mil }
		add = var:cm_trmm_best_mil
	}
}
