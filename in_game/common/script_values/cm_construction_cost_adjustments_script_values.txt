# Root is the building_type. Expects scopes: location.
# Wrapper that ensures scope:cm_building_type is set, then delegates to the full calculation.
cm_get_construction_cost_adjustment_for_building_type = {
	if = { limit = { NOT = { exists = scope:cm_building_type } } save_temporary_scope_as = cm_building_type }
	value = cm_get_construction_cost_modifier_for_building_type
}

# Cost modifier calculation using per-good construction demand maps.
# These maps are populated by the hidden window GUI for all building types.
# The 28 construction goods below are a stable game constant.
#
# Algorithm: for each construction good, clamp the raw price adjustment to the engine's
# [MIN_PRICE_IMPACT, MAX_PRICE_IMPACT] = [-0.33, 3.0] range, then weight by demand share,
# demand_for_good / total_demand. Weights sum to 1.0, so the result stays in [-0.33, 3.0].
#
# Expects scopes: location, building_type
cm_get_construction_cost_modifier_for_building_type = {

  if = { limit = { AND = { has_global_variable_map = cm_construction_demand_books is_key_in_global_variable_map = { name = cm_construction_demand_books target = scope:cm_building_type } } }
    add = {
      goods:books = { save_temporary_scope_as = cm_good }
      value = glorpui_construction_good_adjustment
      min = "define:NMarket|MIN_PRICE_IMPACT"
      max = "define:NMarket|MAX_PRICE_IMPACT"
      multiply = "global_variable_map(cm_construction_demand_books|scope:cm_building_type)"
      divide = cm_get_construction_demand_total
    }
  }
  if = { limit = { AND = { has_global_variable_map = cm_construction_demand_clay is_key_in_global_variable_map = { name = cm_construction_demand_clay target = scope:cm_building_type } } }
