# Per-market construction-goods shortage margins for the auto-construction gate.

# Effective shortage factor: 75% of the engine's stall define (1.5x supply at the default 2.0),
# so the mod stops queuing with headroom below the real construction stall.
cm_get_market_shortage_factor = {
  value = "define:NMarket|MARKET_CONSTRUCTION_NEEDS_BLOCK_FACTOR"
  multiply = 0.75
}

# The shortage factor as a percentage, for the tooltip "exceeds N% of supply" text.
cm_get_market_shortage_factor_pct = {
  value = cm_get_market_shortage_factor
  multiply = 100
}

# Market scope. Shortage margin for one good: >= 0 means it is short for this build in this market.
# Expects scope:cm_market (committed-map key) and scope:cm_building_type (own demand).
cm_market_glass_build_overage = {
  value = "goods_demand_in_market(goods:glass)"
  if = { limit = { AND = { has_global_variable_map = cm_q_market_committed_glass is_key_in_global_variable_map = { name = cm_q_market_committed_glass target = scope:cm_market } } }
    add = "global_variable_map(cm_q_market_committed_glass|scope:cm_market)"
  }
  if = { limit = { AND = { has_global_variable_map = cm_construction_demand_glass is_key_in_global_variable_map = { name = cm_construction_demand_glass target = scope:cm_building_type } } }
    add = "global_variable_map(cm_construction_demand_glass|scope:cm_building_type)"
  }
  subtract = {
    value = "goods_supply_in_market(goods:glass)"
    multiply = cm_get_market_shortage_factor
  }
}

# --- Per-good shortage tooltip breakdown ---
# Location-scoped display values for the per-good tooltip rows. The *_disp_market/_limit/_committed
# values expect scope:cm_location; *_disp_own/_projected/_block also expect scope:cm_building_type.
# *_disp_block returns 1 when the build demands the good and it is short (row visibility).
# cm_rgo_*_disp_block are the snapshot (no committed/own) RGO-path equivalents.

cm_market_glass_disp_market = {
  value = "scope:cm_location.market.goods_demand_in_market(goods:glass)"
}

cm_market_glass_disp_limit = {
  value = "scope:cm_location.market.goods_supply_in_market(goods:glass)"
  multiply = cm_get_market_shortage_factor
}

cm_market_glass_disp_committed = {
  value = 0
  if = { limit = { AND = { has_global_variable_map = cm_q_market_committed_glass is_key_in_global_variable_map = { name = cm_q_market_committed_glass target = scope:cm_location.market } } }
    add = "global_variable_map(cm_q_market_committed_glass|scope:cm_location.market)"
  }
}

cm_market_glass_disp_own = {
  value = 0
  if = { limit = { AND = { has_global_variable_map = cm_construction_demand_glass is_key_in_global_variable_map = { name = cm_construction_demand_glass target = scope:cm_building_type } } }
    add = "global_variable_map(cm_construction_demand_glass|scope:cm_building_type)"
  }
}

cm_market_glass_disp_projected = {
  value = cm_market_glass_disp_market
  add = cm_market_glass_disp_committed
  add = cm_market_glass_disp_own
}

cm_market_glass_disp_block = {
  value = 0
  if = {
    limit = {
      AND = { has_global_variable_map = cm_construction_demand_glass is_key_in_global_variable_map = { name = cm_construction_demand_glass target = scope:cm_building_type } }
      cm_market_glass_disp_projected >= cm_market_glass_disp_limit
    }
    value = 1
  }
}
