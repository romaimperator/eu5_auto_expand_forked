# Expects scopes: building_type
cm_base_building_type_cost = {
	value = scope:building_type.building_base_cost_in_gold
}

# Expects scopes: location, building_type, country
cm_adjustment_to_base_building_cost = {
	scope:location = {
		value = 0
		scope:country = { subtract = modifier:global_build_buildings_efficiency }
		if = { limit = { cm_does_building_type_count_for_fort_building_cost_modifier = yes }
			scope:country = { subtract = modifier:global_fort_build_buildings_efficiency }
		}
		if = { limit = { cm_does_building_type_count_for_port_building_cost_modifier = yes }
			scope:country = { subtract = modifier:global_port_build_buildings_efficiency }
			subtract = modifier:local_port_build_buildings_efficiency
		}
		if = { limit = { OR = { location_rank = location_rank:town location_rank = location_rank:city } }
			scope:country = { subtract = modifier:global_urban_build_buildings_efficiency }
		} else_if = { limit = { location_rank = location_rank:rural_settlement }
			scope:country = { subtract = modifier:global_rural_build_buildings_efficiency }
		}
		subtract = modifier:local_build_buildings_efficiency
		subtract = modifier:local_build_new_buildings_efficiency
		if = { limit = { AND = { exists = scope:upgrade_building_type cm_building_levels_left_to_upgrade > 0 } }
			add = define:NEconomy|UPGRADE_PRICE_MODIFIER
		}
		scope:building_type = { add = cm_get_increase_per_level_cost_for_building_type } # mainly affects the construction center
	}
}

# Root is the location, also expects scopes: building_type, country, and upgrade_building_type
cm_building_levels_left_to_upgrade = {
	value = 0
	"building(scope:building_type|scope:country)" = { add = building_level }
	every_buildings_in_location = {
		limit = { building_type = scope:upgrade_building_type }
		subtract = building_levels_under_construction
	}
}

# Root is the location, also expects scope:building_type, scope:country
cm_adjusted_building_type_cost = {
	if = { limit = { NOT = { exists = scope:location } }
		save_temporary_scope_as = location
	}
	if = { limit = { exists = scope:upgrade_building_type }
		scope:upgrade_building_type = { value = building_base_cost_in_gold }
	} else = {
		scope:building_type = { value = building_base_cost_in_gold }
	}
	multiply = {
		value = 1
		add = cm_adjustment_to_base_building_cost
	}
	multiply = {
		value = 1
		scope:building_type = { add = cm_get_construction_cost_adjustment_for_building_type }
	}
	multiply = {
		value = 1
		add = scope:country.inflation
	}
}

	scope:location = {
		value = cm_adjusted_building_type_cost
	}
}

cm_expected_gold_available_for_building_type_for_auto_expand = {

# Country-scoped. Returns the unified gold reserve setting.
cm_get_minimum_gold = {
	if = { limit = { exists = var:cm_minimum_gold }
		value = var:cm_minimum_gold
	} else = { value = 50 }
}

# Root is the building. Per-level potential profit adjusted for location control.
cm_building_potential_profit_per_level_after_control = {
	value = building_potential_profit
	if = { limit = { building_level >= 1 }
		divide = building_level
	}
	multiply = location.local_control
}

# Expects scopes: good, location
cm_get_construction_good_adjustment = {
	if = { limit = { "scope:location.market.market_price(scope:good)" > "default_price(scope:good)" }
		value = "scope:location.market.market_price(scope:good)"
		divide = "default_price(scope:good)"
		subtract = 1
		divide = 2
	} else_if = { limit = { "scope:location.market.market_price(scope:good)" < "default_price(scope:good)" }
		value = "scope:location.market.market_price(scope:good)"
		divide = "default_price(scope:good)"
		subtract = 1
		multiply = {
			value = 1
			divide = 9
			add = 1
		}
	} else = {
		value = 0
	}
}
