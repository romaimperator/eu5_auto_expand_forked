# 100 gold is the base cost of all types of RGO defined in in_game/common/prices/00_hardcoded.txt
cm_get_defined_base_cost_for_rgo = 100

# Expects scopes: country, location
cm_expected_gold_available_for_rgo_auto_expand = {
	value = cm_get_adjusted_construction_cost_for_rgo
	if = { limit = { exists = scope:country.var:cm_minimum_gold }
		add = scope:country.var:cm_minimum_gold
	}
}

# Expects scopes: country, location. Sum of all multiplicative-style adjustments to the base RGO cost.
cm_get_adjustment_to_base_cost_for_rgo = {
	value = 0
	add = {
		# Per-good adjustment: (default_price - 2) / 4 encodes default_price * GOODS_RGO_PRICE_SCALE(0.25) - GOODS_RGO_BASE_COST(0.5),
		# both from loading_screen/common/defines/00_defines.txt (2 = 0.5/0.25, 4 = 1/0.25). Expensive goods cost more, cheap goods less.
		value = "default_price(scope:location.raw_material)"
		subtract = 2
		divide = 4
	}
	if = { limit = { scope:location.raw_material = { goods_method = mining } }
		add = scope:country.modifier:expand_rgo_mining_cost_modifier
	}
	if = { limit = { scope:location.raw_material = { goods_method = farming } }
		add = scope:country.modifier:expand_rgo_farming_cost_modifier
	}
	if = { limit = { scope:location.raw_material = { goods_method = hunting } }
		add = scope:country.modifier:expand_rgo_hunting_cost_modifier
	}
	if = { limit = { scope:location.raw_material = { goods_method = gathering } }
		add = scope:country.modifier:expand_rgo_gathering_cost_modifier
	}
	if = { limit = { scope:location.raw_material = { goods_method = forestry } }
		add = scope:country.modifier:expand_rgo_forestry_cost_modifier
	}
	add = scope:country.inflation
}

# Expects scopes: country, location
cm_get_adjusted_construction_cost_for_rgo = {
	value = cm_get_defined_base_cost_for_rgo
	multiply = {
		value = 1
		add = cm_get_adjustment_to_base_cost_for_rgo
	}
	multiply = {
		value = 1
		add = cm_get_construction_cost_adjustment_for_rgo
	}
}

cm_get_max_rgo_workers = {
	value = max_rgo_workers
}

cm_get_max_rgo_workers_floor = {
	value = max_rgo_workers
	floor = yes
}

cm_get_rgo_workers = {
	value = rgo_workers
}

cm_get_num_civil_constructions = {
	value = num_civil_constructions
}

cm_get_buildings_under_construction = {
	value = 0
	every_buildings_in_location = {
		add = building_levels_under_construction
	}
}

cm_get_rgos_under_construction = {
	value = num_civil_constructions
	subtract = cm_get_buildings_under_construction
}

cm_get_max_constructions_at_same_time = {
	value = modifier:max_constructions_at_same_time
}

# Root scope is the location
cm_get_open_build_queue_slots = {
	value = cm_get_max_constructions_at_same_time
	subtract = num_civil_constructions
}

# Root scope is the location
cm_modifier_rgo_level = {
  value = modifier:cm_rgo_level
}

# Root is the location. Estimates gold profit per RGO level.
# Uses total raw material output / current levels × market price as a proxy.
cm_rgo_estimated_profit_per_level = {
	raw_material = { save_temporary_scope_as = rgo_good }
	value = raw_material_output
	if = { limit = { cm_modifier_rgo_level >= 1 }
		divide = cm_modifier_rgo_level
	}
	multiply = "market.market_price(scope:rgo_good)"
}
