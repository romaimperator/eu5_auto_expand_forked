# Root is the country. The cleanup profit floor in gold: the cleanup_min_profit setting
# over 100 (5 -> 0.05). Stored in hundredths so the slider stays integral.
cm_cu_profit_floor = {
	if = {
		limit = { exists = var:cm_cu_min_profit }
		value = var:cm_cu_min_profit
		divide = 100
	}
	else = {
		value = 0.05
	}
}

# Root is the country. The floor a construction-material consumer must clear: double the
# normal one, never under 0.1. In a value block min is a floor, so this is the higher of the two.
cm_cu_core_floor = {
	value = cm_cu_profit_floor
	multiply = 2
	min = 0.1
}

# Root is the building, expects scope:cm_country.
cm_cu_effective_floor = {
	value = 0
	if = {
		limit = { cm_cu_core_goods_applies = yes }
		scope:cm_country = { add = cm_cu_core_floor }
	}
	else = {
		scope:cm_country = { add = cm_cu_profit_floor }
	}
}

# Ordered iterators sort high to low, so negating takes the least profitable first.
# Root is the building.
cm_cu_neg_profit = {
	value = 0
	subtract = building_profit
}

# Half the market's open buildings of this type, rounded down.
# Root is the country, expects scope:cm_building_type.
cm_cu_close_cap = {
	value = 0
	if = {
		limit = {
			has_variable_map = cm_cu_open
			is_key_in_variable_map = { name = cm_cu_open target = scope:cm_building_type }
		}
		value = "variable_map(cm_cu_open|scope:cm_building_type)"
		divide = 2
		floor = yes
	}
}

# Half the market's closed buildings of this type, rounded up.
# Root is the country, expects scope:cm_building_type.
cm_cu_reopen_cap = {
	value = 0
	if = {
		limit = {
			has_variable_map = cm_cu_closed_n
			is_key_in_variable_map = { name = cm_cu_closed_n target = scope:cm_building_type }
		}
		value = "variable_map(cm_cu_closed_n|scope:cm_building_type)"
		divide = 2
		ceiling = yes
	}
}
