# Base cost of all RGO types from in_game/common/prices/00_hardcoded.txt.
glorpui_rgo_defined_base_cost = 100

# Expects scopes: location, good. Cost impact of the good's market price vs its default price.
glorpui_construction_good_adjustment = {
	if = { limit = { "scope:cm_location.market.market_price(scope:cm_good)" > "default_price(scope:cm_good)" }
		value = "scope:cm_location.market.market_price(scope:cm_good)"
		divide = "default_price(scope:cm_good)"
		subtract = 1
		divide = 2
	}
	else_if = { limit = { "scope:cm_location.market.market_price(scope:cm_good)" < "default_price(scope:cm_good)" }
		value = "scope:cm_location.market.market_price(scope:cm_good)"
		divide = "default_price(scope:cm_good)"
		subtract = 1
		multiply = {
			value = 1
			divide = 9
			add = 1
		}
	}
	else = {
		value = 0
	}
}

# Expects scopes: country, location. Sum of all multiplicative-style adjustments to the base RGO cost.
glorpui_rgo_base_cost_adjustment = {
	value = 0
	add = {
		value = "default_price(scope:cm_location.raw_material)"
		multiply = "define:NEconomy|GOODS_RGO_PRICE_SCALE"
		subtract = "define:NEconomy|GOODS_RGO_BASE_COST"
	}
	if = { limit = { scope:cm_location.raw_material = { goods_method = mining } }
		add = scope:cm_country.modifier:expand_rgo_mining_cost_modifier
	}
	if = { limit = { scope:cm_location.raw_material = { goods_method = farming } }
		add = scope:cm_country.modifier:expand_rgo_farming_cost_modifier
	}
	if = { limit = { scope:cm_location.raw_material = { goods_method = hunting } }
		add = scope:cm_country.modifier:expand_rgo_hunting_cost_modifier
	}
	if = { limit = { scope:cm_location.raw_material = { goods_method = gathering } }
		add = scope:cm_country.modifier:expand_rgo_gathering_cost_modifier
	}
	if = { limit = { scope:cm_location.raw_material = { goods_method = forestry } }
		add = scope:cm_country.modifier:expand_rgo_forestry_cost_modifier
	}
	add = scope:cm_country.inflation
}

# Expects scopes: country, location. Cost impact of the construction good's market price.
glorpui_rgo_construction_cost_adjustment = {
	value = 0
	if = {
		limit = {
			scope:cm_location.raw_material = {
				OR = {
					goods_method = mining
					goods_method = farming
					goods_method = hunting
					goods_method = gathering
				}
			}
		}
		add = {
			goods:lumber = { save_temporary_scope_as = cm_good }
			value = glorpui_construction_good_adjustment
			min = "define:NMarket|MIN_PRICE_IMPACT"
			max = "define:NMarket|MAX_PRICE_IMPACT"
		}
	}
	else_if = { limit = { scope:cm_location.raw_material = { goods_method = forestry } }
		add = {
			goods:masonry = { save_temporary_scope_as = cm_good }
			value = glorpui_construction_good_adjustment
			min = "define:NMarket|MIN_PRICE_IMPACT"
			max = "define:NMarket|MAX_PRICE_IMPACT"
		}
	}
}

# Expects scopes: country, location. Gold cost of one RGO expansion level.
glorpui_rgo_expand_cost = {
	value = glorpui_rgo_defined_base_cost
	multiply = {
		value = 1
		add = glorpui_rgo_base_cost_adjustment
	}
	multiply = {
		value = 1
		add = glorpui_rgo_construction_cost_adjustment
	}
}
