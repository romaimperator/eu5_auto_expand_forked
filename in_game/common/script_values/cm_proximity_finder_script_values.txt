# The per-candidate search that consumes them runs in cm_proximity_finder_effects.txt.

# Location scope. Current proximity to the nearest source.
cm_pf_current_proximity = {
	value = proximity
}

# Location scope. Proximity a new governor could still add here.
cm_pf_gf_seed_value = {

# cm_pf_road_reduction_value, cm_pf_cap_adj_value, and cm_pf_capb_extra_value are the
# vanilla-derived values in cm_proximity_finder_generated_values.txt (generated by
# tools/cm_generate_proximity_finder.py).

# Country scope. Land jump cost. Optimal roads subtract the best-unlocked-road reduction (assumed
# everywhere); Current roads keep the base cost here and apply the real per-edge reduction in priming.
cm_pf_land_cost_value = {
	value = modifier:land_cost_on_distance_from_capital # Base 40: in_game/common/auto_modifiers/country.txt:146
	if = {
		limit = { var:cm_pf_roads_current = 0 }
		subtract = cm_pf_road_reduction_value
	}
	min = 1 # Floor keeps stacked modifiers from producing free jumps.
}

# Location scope. Expects scope:cm_pf_nb (the neighbor). Land-cost reduction from the road actually
# built on this edge, for Current-roads scoring: 20 (land base 40 minus road base 20, verified in
# game) plus the road type's proximity field (in_game/common/road_types/00_generic.txt).
cm_pf_edge_road_reduction_value = {
	value = 0
	if = {
		limit = { has_road_of_type_to = { target = scope:cm_pf_nb type = road_type:railroad } }
		add = 35 # 20 + railroad proximity 15
	}
	else_if = {
		limit = { has_road_of_type_to = { target = scope:cm_pf_nb type = road_type:modern_road } }
		add = 30 # 20 + modern_road proximity 10
	}
	else_if = {
		limit = { has_road_of_type_to = { target = scope:cm_pf_nb type = road_type:paved_road } }
		add = 25 # 20 + paved_road proximity 5
	}
	else_if = {
		limit = { has_road_of_type_to = { target = scope:cm_pf_nb type = road_type:gravel_road } }
		add = 20 # 20, gravel_road has no proximity field
	}
}

# Country scope. River jump cost; never worse than the land channel.
cm_pf_river_cost_value = {
	value = modifier:land_cost_going_upstream # Base 12: in_game/common/auto_modifiers/country.txt:148
	add = modifier:land_cost_going_downstream # Base 12: in_game/common/auto_modifiers/country.txt:149
	divide = 2 # Flow direction per jump is not script-readable; average the two.
	max = cm_pf_land_cost_value
	min = 1
}

# Country scope. Sea jump cost, with maximum maritime presence assumed everywhere.
cm_pf_sea_cost_value = {

	min = 1

# Country scope. Proximity efficiency for land and river jumps.
cm_pf_eff_land_value = {
	value = modifier:global_distance_from_capital_speed_propagation

}

# Location scope. Expects scope:cm_pf_c. Cost of leaving this location per channel: the raw
# channel cost over 1 plus the proximity efficiency, then over 1 plus the terrain impact
# (terrain never appears in any readable modifier, so it is mirrored manually). _c variants
# evaluate a capital candidate (current capital's topo bonus removed), _cb variants
# additionally grant the candidate's own topography bonus.
cm_pf_cl_g_value = {
	value = scope:cm_pf_c.var:cm_pf_land_cost
	divide = {
		value = 1
		add = scope:cm_pf_c.var:cm_pf_eff_land
		add = modifier:local_distance_from_capital_speed_propagation
		min = 0.1 # Floor keeps stacked maluses from flipping the sign.
	}
	divide = {
		value = 1
		add = cm_location_proximity_modifier
		min = 0.1
	}
}

# Current-roads scoring divides each edge's channel numerator by, mirroring the cm_pf_cl_g/_c/_cb
# denominators; only the numerator (min road/river) varies per edge, the divisor stays per-tile.
cm_pf_div_g_value = {
	value = 1
	multiply = {
		value = 1

		min = 0.1

# Location scope, ports. Expects scope:cm_pf_c and scope:cm_pf_sz (the port's sea zone).
# Cost of the hop between this port and its sea zone, either direction; the efficiency
# divisor reads the sea side, and no terrain applies.
cm_pf_cp_value = {
	value = scope:cm_pf_c.var:cm_pf_port_cost
	multiply = {
		value = 1
		add = modifier:local_port_cost_distance_impact # Harbor discount, -0.5 at full suitability: main_menu/common/static_modifiers/location.txt:139
		min = 0
	}
	divide = {
		value = 1
		add = scope:cm_pf_c.var:cm_pf_eff
		add = scope:cm_pf_sz.modifier:local_distance_from_capital_speed_propagation
		min = 0.1
	}
}
