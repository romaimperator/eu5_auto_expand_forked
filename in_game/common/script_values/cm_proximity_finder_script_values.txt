# Script values for the governor and capital placement finder mapmodes.
# The per-candidate search that consumes them runs in cm_proximity_finder_effects.txt.

# Location scope. Current proximity to the nearest source.
cm_pf_current_proximity = {
	value = proximity
}

# Location scope. Proximity a new governor could still add here, against the stored basis
# proximity cm_pf_pb.
cm_pf_gf_seed_value = {
	value = 80 # Governor source value: in_game/common/building_types/capital_buildings.txt:681
	subtract = var:cm_pf_pb
	min = 0
}

# Location scope. Proximity source already here from anything other than the capital or an
# existing governor; a new governor stacks on top of it, raising its source value past 80 by
# this much.
cm_pf_gf_stack_excess_value = {
	value = modifier:local_proximity_source
	if = {
		limit = { is_capital = yes }
		subtract = 100 # Capital source value: main_menu/common/static_modifiers/location.txt:330
	}
	if = {
		limit = { has_building = building_type:local_governor }
		subtract = 80 # Governor source value: in_game/common/building_types/capital_buildings.txt:681
	}
	if = {
		limit = { has_building = building_type:naval_governor }
		subtract = 80 # Naval governor source value: in_game/common/building_types/capital_buildings.txt:840
	}
	min = 0
	max = 20 # Sources stack to a 100 cap (main_menu/common/static_modifiers/location.txt:330); the governor's 80 leaves 20.
}

# Country scope. Land jump cost. Optimal roads subtract the best-unlocked-road reduction (assumed
# everywhere); Current roads keep the base cost here and apply the real per-edge reduction in priming.
cm_pf_land_cost_value = {
	value = modifier:land_cost_on_distance_from_capital # Base 40: in_game/common/auto_modifiers/country.txt:146
	if = {
		limit = { var:cm_pf_roads_current = 0 }
		subtract = cm_pf_road_reduction_value
	}
	min = 1 # Floor keeps stacked modifiers from producing free jumps.
}

# Location scope. Expects scope:cm_pf_nb (the neighbor). Land-cost reduction from the road actually
# built on this edge, for Current-roads scoring: 20 (land base 40 minus road base 20, verified in
# game) plus the road type's proximity field (in_game/common/road_types/00_generic.txt).
cm_pf_edge_road_reduction_value = {
	value = 0
	if = {
		limit = { has_road_of_type_to = { target = scope:cm_pf_nb type = road_type:railroad } }
		add = 35 # 20 + railroad proximity 15
	}
	else_if = {
		limit = { has_road_of_type_to = { target = scope:cm_pf_nb type = road_type:modern_road } }
		add = 30 # 20 + modern_road proximity 10
	}
	else_if = {
		limit = { has_road_of_type_to = { target = scope:cm_pf_nb type = road_type:paved_road } }
		add = 25 # 20 + paved_road proximity 5
	}
	else_if = {
		limit = { has_road_of_type_to = { target = scope:cm_pf_nb type = road_type:gravel_road } }
		add = 20 # 20, gravel_road has no proximity field
	}
}

# Country scope. Sum of harbor_suitability from the best harbor buildings the country has unlocked.
# The naval line obsoletes tier by tier, so only the best-unlocked tier counts; wharf and
# protected_harbor are parallel and stack. Advances: in_game/common/advances/1_building_unlocks.txt.
cm_pf_unlocked_harbor_buildings_value = {
	value = 0.1 # wharf, no advance: in_game/common/building_types/town_buildings.txt:399
	if = {
		limit = { has_advance = naval_base_advance }
		add = 0.3 # naval_base: in_game/common/building_types/port_buildings.txt:247
	}
	else_if = {
		limit = { has_advance = grand_shipyard_advance }
		add = 0.25 # grand_shipyard: in_game/common/building_types/port_buildings.txt:196
	}
	else_if = {
		limit = { has_advance = shipyard_advance }
		add = 0.2 # shipyard: in_game/common/building_types/port_buildings.txt:145
	}
	else_if = {
		limit = { has_advance = dry_dock_advance }
		add = 0.15 # dry_dock: in_game/common/building_types/port_buildings.txt:94
	}
	else_if = {
		limit = { has_advance = dock_advance }
		add = 0.1 # dock: in_game/common/building_types/port_buildings.txt:42
	}
	if = {
		limit = { has_advance = protected_harbor_advance }
		add = 0.1 # protected_harbor: in_game/common/building_types/port_buildings.txt:292
	}
}

cm_pf_eff_land_value = {

					location_rank = location_rank:city
					location_rank = location_rank:megalopolis

				NOT = { has_building = building_type:supreme_court }

		# supreme_court capital_country_modifier: in_game/common/building_types/capital_buildings.txt:330-376.
		# supreme_court_advance: in_game/common/advances/1_building_unlocks.txt:1123-1129.
		add = 0.1
	}
}

# Location scope. Expects scope:cm_pf_c. The local land-infrastructure speed-propagation bonus:
# the live modifier, plus (Best Possible only, roads option 3) each vanilla building that grants
# it, when the country has unlocked it, this location is eligible, and it has not been built yet.
# Current and Max Roads leave cm_pf_roads_best_possible at 0, a transparent passthrough of the
# bare modifier there, identical to today's behavior. Buildings stack additively (not an
# obsoleting ladder like naval's harbor tiers), matching "assume everything is built".
cm_pf_land_infra_value = {
	value = modifier:local_distance_from_capital_speed_propagation
	if = {
		limit = { scope:cm_pf_c.var:cm_pf_roads_best_possible = 1 }
		if = {
			limit = {
				has_river = yes
				NOT = { has_building = building_type:bridge_infrastructure }
			}
			# bridge_infrastructure, no advance gate, every location rank:
			# in_game/common/building_types/common_buildings.txt:41-75. num_roads > 0 waived: Best
			# Possible already assumes a road on every connection.
			add = 0.1
		}
		if = {
			limit = {
				OR = { location_rank = location_rank:city location_rank = location_rank:megalopolis }
				OR = { is_coastal = yes  has_river = yes  is_adjacent_to_lake = yes }
				development >= 20
				scope:cm_pf_c = { has_advance = pound_lock_canals_advance }
				NOT = { has_building = building_type:pound_lock_canal_infrastructure }
			}
			# pound_lock_canal_infrastructure, city/megalopolis only:
			# in_game/common/building_types/common_buildings.txt:77-113.
			# pound_lock_canals_advance: in_game/common/advances/0_age_of_renaissance.txt:226-233.
			add = 0.1
		}
		if = {
			limit = {
				is_capital = no
				has_river = yes
				OR = {
					region = region:russian_region
					region = region:ural_region
					region = region:west_siberia_region
					region = region:east_siberia_region
				}
				scope:cm_pf_c = {
					has_advance = ostrog_advance
					culture = { has_culture_group = culture_group:russian_group }
				}
				NOT = { has_building = building_type:ostrog }
			}
			# ostrog, Russian-culture-group only: in_game/common/building_types/unique_buildings.txt:1403-1437.
			# ostrog_advance: in_game/common/advances/culture_russian.txt:85-91.
			add = 0.1
		}
	}
}

# Location scope. Expects scope:cm_pf_c. Per-tile cost divisor (efficiency times terrain) that
# Current-roads scoring divides each edge's channel numerator by, mirroring the cm_pf_cl_g/_c/_cb
# denominators; only the numerator (min road/river) varies per edge. cm_pf_div_g/_c/_cb are the
# default divisor (topography + vegetation, this tile's full penalty), used for edges with no
# qualifying road/river; the _nv ("no vegetation") siblings below are used instead for edges
# cm_pf_prime_land_location marks in cm_pf_ev. Both add the two terrain components explicitly
# rather than through cm_location_proximity_modifier, since that value's road/river cancellation
# is a location-aggregate approximation the finder must not inherit - it already has precise
# per-edge data (see cm_pf_relax_from_land_g/_cap).
cm_pf_div_g_value = {
	value = 1
	multiply = {
		value = 1
		add = scope:cm_pf_c.var:cm_pf_eff_land
		add = cm_pf_land_infra_value
		min = 0.1
	}
	multiply = {
		value = 1
		add = cm_location_topography_proximity_modifier
		add = cm_location_vegetation_proximity_modifier
		min = 0.1
	}
}

		min = 0.1
	}
