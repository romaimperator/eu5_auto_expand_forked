# Flat stand-in for the development static modifier's local_build_buildings_efficiency:
# 0.002 per dev (main_menu/common/static_modifiers/location.txt:319) times 14 dev.
cm_dev_build_efficiency_approximation = 0.028

# Country scope. Scale factors for the cm_*_upgrade_price_parity auto modifiers.
cm_town_upgrade_parity_factor = {
	value = 1
	add = modifier:town_upgrade_cost_modifier
	multiply = {
		value = 1
		add = modifier:global_build_buildings_efficiency
		add = modifier:global_rural_build_buildings_efficiency
		add = cm_dev_build_efficiency_approximation
	}
	subtract = 1
}
