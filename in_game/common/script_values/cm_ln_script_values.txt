# Lucky Nations values.

# Country scope. A lucky nation keeps nothing back for its first year, then holds its tax base.
# Re-seeded monthly, so it tracks the tax base as the country grows.
cm_ln_reserve_value = {
	value = 0
	if = {
		# One year past the grand campaign start of 1337.4.1 (START_DATE,
		# loading_screen/common/defines/00_defines.txt:2).
		limit = { current_date >= 1338.4.1 }
		add = country_tax_base
	}
}

# Country scope. The gold half of cm_ln_open_charge_move, capped at the treasury so the
# opening capital move cannot overdraw it.
cm_ln_open_capital_charge = {
	value = country_economical_base
	multiply = 0.4
	max = gold
	min = 0
}

# Country scope. What the opening governor placement charges, capped at the treasury so a country
# that cannot cover it pays what it has. Expects scope:cm_ln_open_gov_site, scope:cm_building_type
# and scope:cm_location.
cm_ln_open_gov_charge = {
	value = scope:cm_ln_open_gov_site.cm_ln_build_cost_value
	max = gold
	min = 0
}

# Country scope. cm_ln_road_step lays gravel whatever the country has researched, so its tier
# term is the gravel price rather than cm_fa_rd_tier_cost_value's best unlocked road.
cm_ln_road_tier_cost_value = {
	value = 10 # Mirrors vanilla build_gravel_road price: in_game/common/prices/00_hardcoded.txt:11
	multiply = { value = 1 add = modifier:build_gravel_road_cost_modifier }
	multiply = { value = 1 add = inflation }
	min = 0
}

# Location scope. Estimated gold for one level of scope:cm_building_type here, the same shape
# the auto-food thresholds price their buildings with. Expects scope:cm_country,
# scope:cm_building_type and scope:cm_location.
cm_ln_build_cost_value = {
	scope:cm_building_type = { value = building_base_cost_in_gold }
	multiply = { value = 1 add = cm_food_build_cost_modifier }
	multiply = { value = 1 scope:cm_building_type = { add = cm_get_construction_cost_adjustment_for_building_type } }
	multiply = { value = 1 add = scope:cm_country.inflation }
	min = 0
}

# Country scope. What the month's earlier reorganization orders already committed. construct_building
# does not take its gold out of the treasury in time for the next order's check, so every threshold
# below carries this term.
cm_ln_tick_spent_value = {
	if = {
		limit = { exists = var:cm_ln_tick_spent }
		value = var:cm_ln_tick_spent
	}
}

# Country scope. Gold a reorganization needs before it orders its road.
# Expects scope:cm_fa_rd_from, scope:cm_fa_rd_to and scope:cm_fa_rd_tier.
cm_ln_road_gold_threshold = {
	value = scope:cm_fa_rd_from.cm_fa_rd_cost_value
	add = cm_ln_tick_spent_value
	add = cm_fa_get_reserve
}

# Country scope. Gold a reorganization needs before it pays for the governor at
# scope:cm_ln_target. Expects scope:cm_country, scope:cm_building_type and scope:cm_location.
cm_ln_placement_gold_threshold = {
	value = scope:cm_ln_target.cm_ln_build_cost_value
	add = cm_ln_tick_spent_value
	add = cm_fa_get_reserve
}

# Country scope. Gold a reorganization needs before it founds the town or city at
# scope:cm_ln_rank_target. Expects scope:cm_country, scope:cm_building_type and scope:cm_location.
cm_ln_rank_gold_threshold = {
	value = scope:cm_ln_rank_target.cm_ln_build_cost_value
	add = cm_ln_tick_spent_value
	add = cm_fa_get_reserve
}

# Location scope. The proximity source value a reorganization's spread assumes: the live
# sources minus the country's own governor seats, which are re-scored as candidates, plus the
# sites the plan has already committed to.
cm_ln_gov_replan_source_value = {
	value = modifier:local_proximity_source
	if = {
		limit = { has_building = building_type:local_governor }
		subtract = 80 # Governor source value: in_game/common/building_types/capital_buildings.txt:681
	}
	if = {
		limit = { has_building = building_type:naval_governor }
		subtract = 80 # Governor source value: in_game/common/building_types/capital_buildings.txt:840
	}
	min = 0
	if = {
		limit = { has_variable = cm_ln_plan_ord }
		add = 80 # Governor source value: in_game/common/building_types/capital_buildings.txt:681
	}
	max = 100 # Stacked-sources cap: main_menu/common/static_modifiers/location.txt:330
}

# Country scope. Land governors share their pool with lieutenancies.
# Vanilla allow: in_game/common/building_types/capital_buildings.txt:611
cm_ln_gov_land_slots = {
	value = modifier:num_local_governors
	subtract = total_building_levels_including_construction:lieutenancy
	min = 0
}

# Country scope. Vanilla allow: in_game/common/building_types/capital_buildings.txt:792
cm_ln_gov_naval_slots = {
	value = modifier:num_naval_governors
	min = 0
}

# Country scope. The room a replan has left to fill: the same counts cm_fa_gov_has_land_room and
# cm_fa_gov_has_naval_room compare, as a number rather than a verdict.
cm_ln_gov_land_free = {
	value = cm_ln_gov_land_slots
	subtract = total_building_levels_including_construction:local_governor
	min = 0
}

# Country scope.
cm_ln_gov_naval_free = {
	value = cm_ln_gov_naval_slots
	subtract = total_building_levels_including_construction:naval_governor
	min = 0
}

# Location scope. The fort drain walks the shared full-automation build list, most of whose
# entries belong to other categories and carry no rank.
cm_ln_ft_prio_value = {
	value = 0
	if = {
		limit = { has_variable = cm_ln_ft_prio }
		add = var:cm_ln_ft_prio
	}
}

# Country scope.
cm_ln_gov_slot_total = {
	value = cm_ln_gov_land_slots
	add = cm_ln_gov_naval_slots
}

# Country scope. Diplomacy slider position in tenths, for the auto modifier that cancels what the
# slider pays out. diplomatic_maintenance is a trigger rather than a value, so the position can
# only be read by comparison. Rounded down, so the cancel can never exceed the gain it mirrors.
cm_ln_dip_slider = {
	value = 0
	if = { limit = { diplomatic_maintenance >= 1 } add = 1 }
	else_if = { limit = { diplomatic_maintenance >= 0.9 } add = 0.9 }
	else_if = { limit = { diplomatic_maintenance >= 0.8 } add = 0.8 }
	else_if = { limit = { diplomatic_maintenance >= 0.7 } add = 0.7 }
	else_if = { limit = { diplomatic_maintenance >= 0.6 } add = 0.6 }
	else_if = { limit = { diplomatic_maintenance >= 0.5 } add = 0.5 }
	else_if = { limit = { diplomatic_maintenance >= 0.4 } add = 0.4 }
	else_if = { limit = { diplomatic_maintenance >= 0.3 } add = 0.3 }
	else_if = { limit = { diplomatic_maintenance >= 0.2 } add = 0.2 }
	else_if = { limit = { diplomatic_maintenance >= 0.1 } add = 0.1 }
}
