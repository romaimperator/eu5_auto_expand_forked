# Per-level cost multipliers mirroring increase_per_level_cost from in_game/common/building_types/
# (construction_center/fondaco/hanseatic_kontor/hanseatic_town_hall in town_buildings.txt + unique_buildings.txt;
# seljuk_mint + hsa_burgtor in unique_buildings.txt). Only a few buildings have it; maintain manually if the game adds more.
# Root is the building_type
# Expects scopes: location
cm_get_increase_per_level_cost_for_building_type = {
	if = { limit = { this = building_type:construction_center } value = 0.5 }
	else_if = { limit = { this = building_type:fondaco } value = 0.5 }
	else_if = { limit = { this = building_type:hanseatic_kontor } value = 0.5 }
	else_if = { limit = { this = building_type:hanseatic_town_hall } value = 0.5 }
	else_if = { limit = { this = building_type:seljuk_mint } value = 0.2 }
	else_if = { limit = { this = building_type:hsa_burgtor } value = 1 }
	else = { value = 0 }
  scope:location = { multiply = { value = 1 add = "location_building_level(prev)" } }
}
