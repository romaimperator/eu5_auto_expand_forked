# Root is the country. The food surplus target as a supply/demand ratio: the ab_food_target
# percent setting over 100 (125 -> 1.25). A market whose best food month stayed below this
# gets new auto-food locations.

	if = {
		limit = { exists = var:cm_ab_food_target }
		value = var:cm_ab_food_target
		divide = 100
	}
	else = {
		value = 1.25
	}

# The natural log of the current age's price_stability, the proportion of base price the target
# price scales toward at overwhelming supply, and whose reciprocal it scales toward at overwhelming
# demand (game_concept_target_price_desc,
# game/main_menu/localization/english/game_concepts_l_english.yml:1350). Held as a log because the
# exponent the model below wants is a log ratio and script has no log operator, and because the
# series further down only converges near 1, which price_stability never is. The values it is the
# log of: age_1_traditions 0.1 (game/in_game/common/age/00_default.txt:3), age_2_renaissance 0.1
# (:52), age_3_discovery 0.08 (:110), age_4_reformation 0.06 (:169), age_5_absolutism 0.04 (:232),
# age_6_revolutions 0.02 (:308). Four decimals because the script value parser rejects a literal
# carrying six, logging jomini_scriptvalue.h:498 "Badly read script value" and leaving the value
# unset; vanilla's own script values never go past four.
cm_ab_core_ln_price_stability = {
	value = -2.3026
	if = {
		limit = { current_age = age_3_discovery }
		value = -2.5257
	}
	else_if = {
		limit = { current_age = age_4_reformation }
		value = -2.8134
	}
	else_if = {
		limit = { current_age = age_5_absolutism }
		value = -3.2189
	}
	else_if = {
		limit = { current_age = age_6_revolutions }
		value = -3.9120
	}
}

# The goal price as a proportion of base, written as (M - 1) / (M + 1) for the series below. M is
# (100 + target) / 100, so the whole thing reduces to target / (200 + target).
# Expects scope:cm_country.
cm_ab_core_goal_u = {
	value = scope:cm_country.var:cm_ab_core_target
	divide = {
		value = 200
		add = scope:cm_country.var:cm_ab_core_target
	}
}

cm_ab_core_goal_u2 = {
	value = cm_ab_core_goal_u
	multiply = cm_ab_core_goal_u
}

# The natural log of the goal price as a proportion of base, by the series
# ln(M) = 2u + 2u^3/3 + 2u^5/5 in Horner form. The Price Target spans -33 to +33 percent, so u never
# leaves +-0.198 and the fifth-order truncation is good to a millionth.
# Expects scope:cm_country.
cm_ab_core_ln_goal = {
	value = cm_ab_core_goal_u2
	multiply = 0.4
	add = 0.6667
	multiply = cm_ab_core_goal_u2
	add = 2
	multiply = cm_ab_core_goal_u
}

# How much of the supply-versus-demand scale the engine spends moving the price, measured off six
# markets of one save on 2026-08-02 by solving each one's own supply, demand and target price for
# it: Jinjiang 0.701, Ozato 0.693 and Hangzhou 0.709 with demand over supply, Kyoto 0.753, Akita
# 0.745 and Chengdu 0.745 with supply over demand. The split by direction is wider than the two
# decimals those target prices are displayed to, so it is a real asymmetry rather than rounding.
# Which side applies follows the Price Target's own sign, since a goal under base price is reached
# with supply over demand and a goal over it with demand over supply.
@cm_ab_core_scale_supply = 0.745
@cm_ab_core_scale_demand = 0.7

cm_ab_core_balance_scale = {
	value = @cm_ab_core_scale_supply
	if = {
		limit = { scope:cm_country.var:cm_ab_core_target > 0 }
		value = @cm_ab_core_scale_demand
	}
}

# Where the goal price sits on the engine's supply-versus-demand scale: the exponent x for which
# price_stability raised to x is the goal price as a proportion of base. Zero at the base price,
# positive below it, negative above. Expects scope:cm_country.
cm_ab_core_price_exponent = {
	value = cm_ab_core_ln_goal
	divide = cm_ab_core_ln_price_stability
	divide = cm_ab_core_balance_scale
}

# Added to supply and to demand before the price model takes their ratio
# (SUPPLY_AND_DEMAND_STABILITY_OFFSET_CONSTANT,
# game/loading_screen/common/defines/00_defines.txt:1907).
@cm_ab_core_price_offset = 1

# The local production of scope:cm_ab_good_t that puts this market at the goal price, taking the
# market's own production and demand as the whole of what forms the price. The engine reads the
# scale as (supply - demand) / (supply + demand + 2), so this inverts that for supply at the
# exponent the goal sits on. Root is the market, expects scope:cm_ab_good_t and scope:cm_country.
cm_ab_core_supply_wanted = {
	value = "goods_demand_in_market(scope:cm_ab_good_t)"
	multiply = {
		value = 1
		add = cm_ab_core_price_exponent
	}
	add = {
		value = @cm_ab_core_price_offset
		multiply = 2
		multiply = cm_ab_core_price_exponent
	}
	divide = {
		value = 1
		subtract = cm_ab_core_price_exponent
	}
}

# How much of scope:cm_ab_good_t this market has to add to its own production to reach the goal
# price. Imports meet the physical requirement in full but are discounted when the price forms
# (BURGHER_TRADE_IMPACT_ON_SUPPLY_SCALE 0.1 and TRADE_IMPACT_ON_SUPPLY_SCALE 0.75,
# game/loading_screen/common/defines/00_defines.txt:1760-1761), so the shortfall is measured
# against what is produced here. Root is the market.
# produced_in_market takes a literal goods key, never a scope.
cm_ab_core_shortfall = {
	value = cm_ab_core_supply_wanted
	if = {
		limit = { scope:cm_ab_good_t = goods:masonry }
		subtract = produced_in_market:masonry
	}
	else_if = {
		limit = { scope:cm_ab_good_t = goods:lumber }
		subtract = produced_in_market:lumber
	}
	else_if = {
		limit = { scope:cm_ab_good_t = goods:sand }
		subtract = produced_in_market:sand
	}
	else_if = {
		limit = { scope:cm_ab_good_t = goods:glass }
		subtract = produced_in_market:glass
	}
	min = 0
}

				scope:cm_building_type = building_type:sand_pit

# How slow this tile is to cross: its topography movement cost plus its vegetation movement cost,
# 2 on flat open ground. Root is the location.

	value = 1 # Mirrors vanilla flatland movement_cost, shared by every other land topography: in_game/common/topography/00_default.txt:7
	if = { limit = { topography = mountains } value = 2 } # Mirrors vanilla mountains movement_cost: in_game/common/topography/00_default.txt:21
	else_if = { limit = { topography = hills } value = 1.5 } # Mirrors vanilla hills movement_cost: in_game/common/topography/00_default.txt:53
	else_if = { limit = { topography = wetlands } value = 1.5 } # Mirrors vanilla wetlands movement_cost: in_game/common/topography/00_default.txt:103
	else_if = { limit = { topography = plateau } value = 1.25 } # Mirrors vanilla plateau movement_cost: in_game/common/topography/00_default.txt:81
	if = { limit = { vegetation = jungle } add = 2 } # Mirrors vanilla jungle movement_cost: in_game/common/vegetation/00_default.txt:124
	else_if = { limit = { vegetation = forest } add = 1.5 } # Mirrors vanilla forest movement_cost: in_game/common/vegetation/00_default.txt:100
	else_if = { limit = { vegetation = woods } add = 1.25 } # Mirrors vanilla woods movement_cost: in_game/common/vegetation/00_default.txt:78
	else_if = { limit = { vegetation = desert } add = 1.1 } # Mirrors vanilla desert movement_cost: in_game/common/vegetation/00_default.txt:3
	else_if = { limit = { vegetation = farmland } add = 1.1 } # Mirrors vanilla farmland movement_cost: in_game/common/vegetation/00_default.txt:57
	else = { add = 1 } # Mirrors vanilla grasslands and sparse movement_cost: in_game/common/vegetation/00_default.txt:43
}

# The unit distance vanilla prices a road by: how far the two tiles are apart, weighted by how slow
# they are to cross. The engine's own figure is GetRoadDistanceTo, which is GUI-only and returns
# int32, a type MakeScopeValue nulls on, so the shape here is the one the OGAS Optimized workshop
# mod arrived at, and its two divisors are ITS fitted constants rather than vanilla values.
# The 1.65 is calibrated against the game's own Build Gravel Road tooltip, measured 2026-07-27 on
# three corridors out of Valladolid whose real distance multipliers were 12.72, 8.46 and 11.07
# against 7.71, 5.09 and 6.74 from the terms above: ratios 1.650, 1.659 and 1.641. Both divisor
# branches were then checked whole against real costs, same area 81 against 80.5 and across an area
# boundary 73 against 73.5, so the 10 and the 20 are right relative to each other as well.
# The market-access pass caches this edge's figure in cm_ab_rd_ed as it walks, so the read comes
# from there whenever that map is live. Root is the location, expects scope:cm_ab_rd_from and
# scope:cm_ab_rd_to.
cm_ab_rd_unit_distance_value = {
	if = {
		limit = {
			scope:cm_ab_rd_from = {
				has_variable_map = cm_ab_rd_ed
				is_key_in_variable_map = { name = cm_ab_rd_ed target = scope:cm_ab_rd_to }
			}
		}
		value = "scope:cm_ab_rd_from.variable_map(cm_ab_rd_ed|scope:cm_ab_rd_to)"
	}
	else = {
		value = "scope:cm_ab_rd_from.distance_to(scope:cm_ab_rd_to)"
		multiply = {
			value = cm_ab_rd_movement_cost_value
			add = scope:cm_ab_rd_to.cm_ab_rd_movement_cost_value
		}
		if = {
			limit = { area = scope:cm_ab_rd_to.area }
			divide = 10
		}
		else = {
			divide = 20
		}
		multiply = 1.65
	}
	min = 0.01
}

# How much of each construction good one road order demands. Mirrors vanilla
# build_gravel_road_demand: in_game/common/goods_demand/special_construction_demands.txt:28.
# Every road tier demands the same three; railroad adds steel, which is not modelled.
# Read by the cost model below and by the market shortage gate and its commit.
# Taken as a flat per-order draw. Vanilla's own road_types/readme.txt qualifies
# build_time_per_unit_distance and price_per_unit_distance as per unit distance and
# construction_demand as plain "What goods are needed to build this road type", which is the
# evidence for that reading; whether it scales with the span is unverified.
cm_ab_rd_demand_lumber = {
	value = 0.5
}

cm_ab_rd_demand_masonry = {
	value = 1
}

cm_ab_rd_demand_sand = {
	value = 0.25
}

cm_ab_rd_demand_total = {
	value = cm_ab_rd_demand_lumber
	add = cm_ab_rd_demand_masonry
	add = cm_ab_rd_demand_sand
}

# What this market's construction-goods prices do to a road's cost, as a fraction: each good's
# deviation from its default price weighted by how much of it a road demands, summed. Verified
# against the game's own Build Gravel Road price tooltip (2026-07-27, Burgos market: lumber -8.33%,
# masonry -13.79%, sand -4.71%, total -26.83%), which sums the three the same way.
# Root is the location.
cm_ab_rd_goods_adjust_value = {
	value = 0
	add = {
		value = "market.market_price(goods:lumber)"
		divide = "default_price(goods:lumber)"
		subtract = 1
		multiply = cm_ab_rd_demand_lumber
	}
	add = {
		value = "market.market_price(goods:masonry)"
		divide = "default_price(goods:masonry)"
		subtract = 1
		multiply = cm_ab_rd_demand_masonry
	}
	add = {
		value = "market.market_price(goods:sand)"
		divide = "default_price(goods:sand)"
		subtract = 1
		multiply = cm_ab_rd_demand_sand
	}
}

# The same three deviations as a demand-weighted mean rather than a sum, which is the scale the
# min discount settings are stated on: cm_get_standard_construction_cost_modifier_for_building_type
# divides by its own demand total the same way. The sibling above stays a sum because it scales a
# cost, where every weight counts in full. Root is the location.
cm_ab_rd_goods_price_index_value = {
	value = cm_ab_rd_goods_adjust_value
	divide = cm_ab_rd_demand_total
}

# Gold per unit distance for the best road type this country has unlocked, carrying that type's own
# cost modifier and inflation. Root is the country.
cm_ab_rd_tier_cost_value = {
	value = 0
	if = {
		# railroad_advance unlocks railroad: in_game/common/advances/3_road_unlocks.txt:28
		limit = { has_advance = railroad_advance }
		value = 100 # Mirrors vanilla build_railroad price: in_game/common/prices/00_hardcoded.txt:23
		multiply = { value = 1 add = modifier:build_railroad_cost_modifier }
	}
	else_if = {
		# modern_road_advance unlocks modern_road: in_game/common/advances/3_road_unlocks.txt:20
		limit = { has_advance = modern_road_advance }
		value = 50 # Mirrors vanilla build_modern_road price: in_game/common/prices/00_hardcoded.txt:19
		multiply = { value = 1 add = modifier:build_modern_road_cost_modifier }
	}
	else_if = {
		# paved_road_advance unlocks paved_road: in_game/common/advances/3_road_unlocks.txt:12
		limit = { has_advance = paved_road_advance }
		value = 25 # Mirrors vanilla build_paved_road price: in_game/common/prices/00_hardcoded.txt:15
		multiply = { value = 1 add = modifier:build_paved_road_cost_modifier }
	}
	else_if = {
		# road_building unlocks gravel_road: in_game/common/advances/3_road_unlocks.txt:1
		limit = { has_advance = road_building }
		value = 10 # Mirrors vanilla build_gravel_road price: in_game/common/prices/00_hardcoded.txt:11
		multiply = { value = 1 add = modifier:build_gravel_road_cost_modifier }
	}
	multiply = { value = 1 add = inflation }
	min = 0
}

# Estimated gold for one road between this location and scope:cm_ab_rd_to, as distance times the
# tier's gold per unit distance times what the market's construction-goods prices do to it.
# scope:cm_ab_rd_tier is cm_ab_rd_tier_cost_value, saved once a pass because it is the same for
# every corridor. Root is the location, expects scope:cm_ab_rd_from, scope:cm_ab_rd_to and
# scope:cm_ab_rd_tier.
cm_ab_rd_cost_value = {
	value = cm_ab_rd_unit_distance_value
	multiply = scope:cm_ab_rd_tier
	multiply = {
		value = 1
		add = cm_ab_rd_goods_adjust_value
		min = 0
	}
	min = 1
}

# The tax base a road on the edge above this location would unlock, per gold it costs.
# Root is the location, expects scope:cm_ab_rd_from, scope:cm_ab_rd_to and scope:cm_ab_rd_tier.
cm_ab_rd_score_value = {
	value = var:cm_ab_rd_sub
	divide = cm_ab_rd_cost_value
}

# Ordered iterators sort high to low, so negating visits the farthest tiles first, which is every
# child before its parent in a shortest-path tree. Root is the location.
cm_ab_rd_leaf_order = {
	value = 0
	if = {
		limit = { has_variable = cm_pf_pb }
		subtract = var:cm_pf_pb
	}
}

# The tax base this tile alone would gain once the network reaches its predicted proximity.
# Control rises 0.75 per unit of proximity and proximity runs 0-100, hence 0.0075, the same
# conversion cm_gf_taxgain uses. Root is the location.
cm_ab_rd_node_demand = {
	value = var:cm_pf_pb
	subtract = proximity
	min = 0
	multiply = var:cm_pf_w
	multiply = 0.0075
}

# What one unit of distance costs a location in market access once the best unlocked road sits on
# that edge. Mirrors cm_ab_rd_tier_cost_value's ladder branch for branch, since a corridor priced at
# one tier has to be modelled at the same one. Root is the country.
cm_ab_rd_ma_segment_factor = {
	value = 0.005 # MARKET_BASE_DISTANCE_FACTOR: loading_screen/common/defines/00_defines.txt:1782
	if = {
		# railroad_advance unlocks railroad: in_game/common/advances/3_road_unlocks.txt:28
		limit = { has_advance = railroad_advance }
		multiply = 0.6 # railroad market_access -0.4: in_game/common/road_types/00_generic.txt:46
	}
	else_if = {
		# modern_road_advance unlocks modern_road: in_game/common/advances/3_road_unlocks.txt:20
		limit = { has_advance = modern_road_advance }
		multiply = 0.7 # modern_road market_access -0.3: in_game/common/road_types/00_generic.txt:32
	}
	else_if = {
		# paved_road_advance unlocks paved_road: in_game/common/advances/3_road_unlocks.txt:12
		limit = { has_advance = paved_road_advance }
		multiply = 0.8 # paved_road market_access -0.2: in_game/common/road_types/00_generic.txt:18
	}
	else = {
		multiply = 0.9 # gravel_road market_access -0.1: in_game/common/road_types/00_generic.txt:4
	}

# A location's transport loss to its market centre, which is additive along the path. Vanilla renders
# the same decomposition as a 100% base less a transportation distance line, and the local bonus is a
# terminal one at the destination that does not travel, so taking it back out leaves the path alone.
# MARKET_BASE_ACCESS 1: loading_screen/common/defines/00_defines.txt:1780. Where a location already
# sits at full access the cap makes this read high, which only understates what a road would buy.
# Root is the location.
cm_ab_rd_loss_value = {
	value = 1
	add = modifier:local_market_access
	subtract = market_access
	min = 0
}

# Highest predicted loss is farthest from the centre, so ordering by it visits every tile before the

	if = {
		limit = { has_variable = cm_ab_rd_ma_l }
		value = var:cm_ab_rd_ma_l
	}
}
