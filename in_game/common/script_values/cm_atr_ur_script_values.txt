# Auto-Grant Urban Rights, specialization half.

# Root is the country. The fit threshold as a fraction: the specialization_window percent
# setting over 100 (50 -> 0.5).
cm_atr_ur_window_ratio = {
	if = {
		limit = { exists = var:cm_specialization_window }
		value = var:cm_specialization_window
		divide = 100
	}
	else = {
		value = 0.5
	}
}

# The baked best-fit score, safe on locations the map mode never scored.
# Root is the location.
cm_atr_ur_order = {
	value = 0
	if = {
		limit = { exists = var:cm_trmm_best_mil }
		add = var:cm_trmm_best_mil
	}
}
