# Root is the location. Combined laborer+slave pool minus the jobs either pop type can fill.
cm_location_laborers_or_slaves_available = {
  value = num_pop_type:laborers
  add = num_pop_type:slaves
  subtract = cm_location_laborers_needed_for_buildings
  subtract = cm_location_slaves_needed_for_buildings
  subtract = cm_modifier_rgo_level
  subtract = cm_get_rgos_under_construction
}

# Includes potential employment from levels under construction and from buildings that are subsidized but not fully employed
# Expects scope: building
cm_building_employment_required = {
  value = 0
  scope:cm_building = {
    if = { limit = { is_opened = yes }
      if = { limit = { AND = { building_type = { cm_is_building_type_production_building = yes } is_subsidized = yes } }
        add = {
          value = building_level
          multiply = building_type.employment_size
        }
      } else = {
        add = building_employment_size_amount
      }
      add = {
        value = building_levels_under_construction
        multiply = building_type.employment_size
      }
    }
  }
}

# Unemployed pops that can promote into each target type (direct promoters per vanilla). Root is the location.

  value = "unemployed_pops_of_pop_type_in_location(pop_type:peasants)"
}
cm_location_clergy_promotion_source_pool = {
  value = "unemployed_pops_of_pop_type_in_location(pop_type:peasants)"
}
cm_location_burghers_promotion_source_pool = {
  value = "unemployed_pops_of_pop_type_in_location(pop_type:peasants)"
  add = "unemployed_pops_of_pop_type_in_location(pop_type:laborers)"
}

  value = "unemployed_pops_of_pop_type_in_location(pop_type:peasants)"

  value = "unemployed_pops_of_pop_type_in_location(pop_type:peasants)"

cm_location_peasants_promotion_source_pool = {
  value = "unemployed_pops_of_pop_type_in_location(pop_type:tribesmen)"
}

# Projected one-year pop slack for the building pop type: location surplus plus a year of promotion capped
# by the source pool. Negative means the deficit will not fill within a year. Root is the location, expects
# scope:cm_building_type.
cm_location_promotion_buffer_slack = {
  value = 0
  if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:nobles }
    add = cm_location_nobles_available
    add = { value = cm_combined_pop_promotion_speed_for_nobles multiply = 12 max = cm_location_nobles_promotion_source_pool }
  }
  if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:clergy }
    add = cm_location_clergy_available
    add = { value = cm_combined_pop_promotion_speed_for_clergy multiply = 12 max = cm_location_clergy_promotion_source_pool }
  }
  if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:burghers }
    add = cm_location_burghers_available
    add = { value = cm_combined_pop_promotion_speed_for_burghers multiply = 12 max = cm_location_burghers_promotion_source_pool }
  }
  if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:laborers }
    add = cm_location_laborers_available
    add = { value = cm_combined_pop_promotion_speed_for_laborers multiply = 12 max = cm_location_laborers_promotion_source_pool }
  }
  if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:soldiers }
    add = cm_location_soldiers_available
    add = { value = cm_combined_pop_promotion_speed_for_soldiers multiply = 12 max = cm_location_soldiers_promotion_source_pool }
  }
  if = { limit = { "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)" = pop_type:peasants }
    add = cm_location_peasants_available
    add = { value = cm_combined_pop_promotion_speed_for_peasants multiply = 12 max = cm_location_peasants_promotion_source_pool }
  }
}

# Projected one-year laborers/slaves slack for RGO expansion: surplus plus a year of laborers promotion
# capped by the source pool (slaves count toward the surplus but do not promote). Root is the location.
cm_location_rgo_promotion_buffer_slack = {
  value = cm_location_laborers_or_slaves_available
  add = { value = cm_combined_pop_promotion_speed_for_laborers multiply = 12 max = cm_location_laborers_promotion_source_pool }
}
