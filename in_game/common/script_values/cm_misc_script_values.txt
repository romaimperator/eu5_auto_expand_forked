# Root is the country
cm_get_allowed_building_overbuild_levels = {
	if = { limit = { exists = var:cm_allowed_building_overbuild_levels }
		value = var:cm_allowed_building_overbuild_levels
	} else = { value = 0 }
}

# Root is the country

	value = 5
	multiply = {
		value = 1
		add = modifier:grant_town_rights_cost_modifier

	}
}

# Minimum government power needed to grant without dropping below the floor. Root is the country.

	if = { limit = { exists = var:cm_town_rights_min_gov_power }
		value = var:cm_town_rights_min_gov_power
	} else = { value = 0 }
	add = cm_grant_town_rights_cost
}

# Includes potential employment from levels under construction and from buildings that are subsidized but not fully employed
# Expects scope: building
cm_building_employment_required = {
  value = 0
  scope:building = {
    if = { limit = { is_opened = yes }
      if = { limit = { AND = { building_type = { cm_is_building_type_production_building = yes } is_subsidized = yes } }
        add = {
          value = building_level
          multiply = building_type.employment_size
        }
      } else = {
        add = building_employment_size_amount
      }
      add = {
        value = building_levels_under_construction
        multiply = building_type.employment_size
      }
    }
  }
}
