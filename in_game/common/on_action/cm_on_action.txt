cmf_monthly_human_country_pulse = {
	on_actions = {
		cm_unified_auto_expand
		cm_update_auto_build_visibility
		cm_update_auto_town_rights_visibility
		cm_monthly_auto_town_rights
	}
}

# Update auto-build list visibility each month so newly researched buildings appear.
cm_update_auto_build_visibility = {
	effect = {
		cm_update_auto_build_list_visibility = yes
	}
}

# Update auto town rights list visibility each month so newly available rights appear
# (advance researched, estate privilege adopted, tag swap, culture change, etc.).
cm_update_auto_town_rights_visibility = {
	effect = {
		cm_update_auto_town_rights_list_visibility = yes
	}
}

# Standalone monthly hook for auto-grant town rights. Lives outside the priority
# dispatcher because the player toggles it on its own tab, not in the priority list.
# Still respects the master pause toggle.
cm_monthly_auto_town_rights = {
	effect = {
		save_scope_as = country
		if = {
			limit = {
				cm_should_pause_auto_expand = no
				exists = var:cm_auto_grant_town_rights
			}
			cm_run_auto_town_rights = yes
		}
	}
}

# Unified dispatcher: iterates the priority list in order and calls each
# enabled feature's effect. Gated by the master pause toggle.
cm_unified_auto_expand = {
	effect = {
		save_scope_as = country
		# Start a fresh construction queue each cycle, drained once by the queue window.
		every_in_list = {
			variable = cm_q_locations
			save_scope_as = cm_q_loc
			scope:cm_q_loc = { clear_variable_list = cm_q_building_types }
		}
		clear_variable_list = cm_q_locations
		clear_variable_list = cm_q_rgo_locations
		every_in_list = {
			variable = cm_q_ungated_locations
			save_scope_as = cm_q_uloc
			scope:cm_q_uloc = { clear_variable_list = cm_q_ungated_building_types }
		}
		clear_variable_list = cm_q_ungated_locations
		every_in_list = {
			variable = cm_q_urbanize_locations
			save_scope_as = cm_q_urb_loc
			scope:cm_q_urb_loc = { clear_variable_list = cm_q_urbanize_building_types }
		}
		clear_variable_list = cm_q_urbanize_locations
		every_in_list = {
			variable = cm_q_food_locations
			save_scope_as = cm_q_food_loc
			scope:cm_q_food_loc = { clear_variable_list = cm_q_food_building_types }
		}
		clear_variable_list = cm_q_food_locations
		clear_variable_list = cm_q_food_rgo_locations
		every_in_list = {
			variable = cm_gui_upgrade_locations
			save_scope_as = cm_q_up_loc
			scope:cm_q_up_loc = {
				clear_variable_list = cm_gui_upgrade_building_types
				clear_variable_list = cm_q_upgrade_gated_types
			}
		}
		clear_variable_list = cm_gui_upgrade_locations
		if = { limit = { cm_should_pause_auto_expand = no }
			if = { limit = { has_variable_list = cm_priority_features_list }
				every_in_list = {
					variable = cm_priority_features_list
					save_scope_as = feature
					if = { limit = { scope:country = { is_target_in_variable_list = { name = cm_priority_features_enabled target = scope:feature } } }
						switch = {
							trigger = scope:feature
							flag:cm_feature_auto_expand_buildings = { scope:country = { cm_run_auto_expand_buildings = yes } }
							flag:cm_feature_auto_expand_rgos = { scope:country = { cm_run_auto_expand_rgos = yes } }
							flag:cm_feature_upgrade_all_buildings = { scope:country = { cm_run_upgrade_all_buildings = yes } }
							flag:cm_feature_auto_build = { scope:country = { cm_run_auto_build = yes } }
							flag:cm_feature_auto_food = { scope:country = { cm_run_auto_food = yes } }
							flag:cm_feature_auto_urbanize = { scope:country = { cm_run_auto_urbanize = yes } }
							flag:cm_feature_auto_build_capital = { scope:country = { cm_run_auto_build_capital = yes } }
						}
					}
				}
			}
		}
		# Signal the hidden window GUI to process upgrades and do periodic checks
		set_variable = cm_monthly_gui_tick
		# Show the construction-queue drain section; its items construct once via trigger_on_create,
		# then the wait->apply driver clears this flag so the section re-shows (re-fires) next month.
		set_variable = { name = cm_should_construct value = yes }
	}
}
