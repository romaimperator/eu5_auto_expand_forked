cmf_monthly_human_country_pulse = {
	on_actions = {
		# Budget refresh first so this month's funds exist before the dispatcher stages.
		cm_fa_monthly_budget
		cm_unified_auto_expand
		cm_update_all_list_visibility
		cm_monthly_auto_town_rights
	}
}

# Refresh the CMF settings lists each month so newly available items appear.
cm_update_all_list_visibility = {
	effect = {
		cm_update_auto_build_list_visibility = yes
		cm_update_auto_town_rights_list_visibility = yes
		cm_update_auto_food_list_visibility = yes
		cm_update_auto_urbanize_ignore_list_visibility = yes
	}
}

# Standalone monthly hook for auto-grant town rights. Lives outside the priority
# dispatcher because the player toggles it on its own tab, not in the priority list.
# Still respects the master pause toggle.
cm_monthly_auto_town_rights = {
	effect = {
		save_scope_as = cm_country
		if = {
			limit = {
				cm_should_pause_auto_expand = no
				exists = var:cm_auto_grant_town_rights
			}
			cm_run_auto_town_rights = yes
		}
	}
}

# Standalone monthly hook for the full-automation budget and food sampling. Its own
# on_action so a skipped construction cycle never skips them.
cm_fa_monthly_budget = {
	effect = {
		save_scope_as = cm_country
		if = {
			limit = { cm_fa_master_active = yes }
			cm_fa_monthly_budget_refresh = yes
			cm_fa_food_sample_markets = yes
		}
	}
}

# Unified dispatcher: iterates the priority list in order and calls each
# enabled feature's effect. Gated by the master pause toggle.
cm_unified_auto_expand = {
	effect = {
		save_scope_as = cm_country
		if = { limit = { exists = var:cm_should_construct }
			# A cycle can straddle one month tick at high speed; a second straight skip means it is stuck.
			change_variable = { name = cm_q_skipped_pulses add = 1 }
			if = { limit = { var:cm_q_skipped_pulses >= 2 }
				cm_q_reset_cycle_state = yes
				set_variable = { name = cm_q_skipped_pulses value = 0 }
				cmf_log = { action = cm_log_cycle_force_reset }
			} else = {
				cmf_log = { action = cm_log_cycle_skipped }
			}
		} else = {
			set_variable = { name = cm_q_skipped_pulses value = 0 }
			# Start a fresh construction queue each cycle, drained once by the queue window.
			cm_clear_queue_pair = { COUNTRY_LIST = cm_q_locations LOCATION_LIST = cm_q_building_types }
			clear_variable_list = cm_q_rgo_locations
			cm_clear_queue_pair = { COUNTRY_LIST = cm_q_ungated_locations LOCATION_LIST = cm_q_ungated_building_types }
			cm_clear_queue_pair = { COUNTRY_LIST = cm_q_urbanize_locations LOCATION_LIST = cm_q_urbanize_building_types }
			cm_clear_queue_pair = { COUNTRY_LIST = cm_q_food_locations LOCATION_LIST = cm_q_food_building_types }
			clear_variable_list = cm_q_food_rgo_locations
			clear_variable_list = cm_q_rank_locations
			cm_clear_queue_pair = { COUNTRY_LIST = cm_fa_b_locations LOCATION_LIST = cm_fa_b_types }
			cm_clear_queue_pair = { COUNTRY_LIST = cm_fa_u_locations LOCATION_LIST = cm_fa_u_types }
			clear_variable_list = cm_fa_r_locations
			clear_variable_list = cm_fa_food_rgo_locations
			clear_variable_list = cm_fa_collapsar_rgo_locations
			# The upgrade queue carries two extra per-location lists alongside the drained one
			every_in_list = {
				variable = cm_gui_upgrade_locations
				save_scope_as = cm_q_up_loc
				scope:cm_q_up_loc = {
					clear_variable_list = cm_gui_upgrade_building_types
					clear_variable_list = cm_q_upgrade_gated_types
					clear_variable_list = cm_q_upgrade_target_types
				}
			}
			clear_variable_list = cm_gui_upgrade_locations
			cm_clear_queue_pair = { COUNTRY_LIST = cm_q_approved_build_locations LOCATION_LIST = cm_q_approved_build_types }
			cm_clear_queue_pair = { COUNTRY_LIST = cm_q_approved_upgrade_locations LOCATION_LIST = cm_q_approved_upgrade_types }
			clear_variable_list = cm_q_approved_rank_locations
			set_variable = { name = cm_q_planned_spend value = 0 }
			set_variable = { name = cm_q_staged value = 0 }
			set_variable = { name = cm_q_processed value = 0 }
			set_variable = { name = cm_q_progress_last value = 0 }
			set_variable = { name = cm_q_stall_rounds value = 0 }
			cm_reset_market_committed_demand = yes
			if = { limit = { cm_should_pause_auto_expand = no }
				if = { limit = { has_variable_list = cm_priority_features_list }
					every_in_list = {
						variable = cm_priority_features_list
						save_scope_as = cm_feature
						if = { limit = { scope:cm_country = { is_target_in_variable_list = { name = cm_priority_features_enabled target = scope:cm_feature } } }
							switch = {
								trigger = scope:cm_feature
								flag:cm_feature_auto_expand_buildings = { scope:cm_country = { cm_run_auto_expand_buildings = yes } }
								flag:cm_feature_auto_expand_rgos = { scope:cm_country = { cm_run_auto_expand_rgos = yes } }
								flag:cm_feature_upgrade_all_buildings = { scope:cm_country = { cm_run_upgrade_all_buildings = yes } }
								flag:cm_feature_auto_build = { scope:cm_country = { cm_run_auto_build = yes } }
								flag:cm_feature_auto_food = { scope:cm_country = { cm_run_auto_food = yes } }
								flag:cm_feature_auto_urbanize = { scope:cm_country = { cm_run_auto_urbanize = yes } }
								flag:cm_feature_auto_build_capital = { scope:cm_country = { cm_run_auto_build_capital = yes } }
							}
						}
					}
				}
			}
			# Full-automation staging after the priority features, so the player's own
			# features always stage and claim first.
			if = {
				limit = { cm_fa_master_active = yes }
				if = { limit = { exists = var:cm_fa_industry_enabled } cm_fa_run_industry = yes }
				if = { limit = { exists = var:cm_fa_collapsar_enabled } cm_fa_run_collapsar = yes }
				if = { limit = { exists = var:cm_fa_food_enabled } cm_fa_run_food = yes }
				if = { limit = { exists = var:cm_fa_utilities_enabled } cm_fa_run_utilities = yes }
			}
			# Show the construction-queue window to construct this month's staged items.
			# The full-automation RGO lists bypass the staged counter (no GUI scoring), so
			# they arm the window on their own.
			if = {
				limit = {
					OR = {
						var:cm_q_staged >= 1
						AND = {
							has_variable_list = cm_fa_food_rgo_locations
							variable_list_size = { name = cm_fa_food_rgo_locations value >= 1 }
						}
						AND = {
							has_variable_list = cm_fa_collapsar_rgo_locations
							variable_list_size = { name = cm_fa_collapsar_rgo_locations value >= 1 }
						}
					}
				}
				set_variable = { name = cm_should_construct value = yes }
			}
		}
	}
}
