on_location_changed_rank = {
  on_actions = {
    lsq_on_location_changed_rank
  }
}

lsq_on_location_changed_rank = {
  effect = {
    save_temporary_scope_as = location
    # debug_log = "Location changed rank [THIS.GetLocation.GetNameWithNoTooltip]"
    every_in_list = {
      variable = lsq_auto_expand_registered_building_types
      save_temporary_scope_as = building_type
      scope:location = {
        if = { limit = { NOT = { can_build_building = scope:building_type } }
          # debug_log = "Removing building type [THIS.sBuildingType('building_type').GetBuildingType.GetNameWithNoTooltip] from auto expand list"
          remove_list_variable = { name = lsq_auto_expand_registered_building_types target = scope:building_type }
        }
      }
    }
  }
}
