# Yearly hook for the Auto-Build passes that run once a year.
cmf_yearly_human_country_pulse = {
	on_actions = {
		cm_ab_yearly_pulse
	}
}

# Root is the country.
cm_ab_yearly_pulse = {
	effect = {
		cm_ab_food_yearly_evaluation = yes
	}
}

# on_capital_moved's root is the country, per vanilla in_game/common/on_action/_hardcoded.txt:5499.
four_yearly_country_pulse = {
	on_actions = {
		cm_ab_rd_on_plan
	}
}

on_capital_moved = {
	on_actions = {
		cm_ab_rd_on_plan
	}
}

# Root is the country. Both pulses fire for every country in the world. In multiplayer the plan runs
# from Force Re-Plan Roads alone, so a whole-map spread never lands on every machine unasked; a lucky
# nation is off the cadence in both modes and replans from cm_ab_rd_mark_replan instead.
cm_ab_rd_on_plan = {
	trigger = {
		is_ai = no
		is_multiplayer_session = no
	}
	effect = {
		cm_ab_rd_plan = yes
	}
}
