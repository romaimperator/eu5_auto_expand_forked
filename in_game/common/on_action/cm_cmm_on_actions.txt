# Hook this mod into CMF shared registration on_action.
cmf_on_mod_registration = {
	on_actions = {
		cm_on_register_cmf_mod
	}
}

cm_on_register_cmf_mod = {
	effect = {
		cm_register_cmf_mod = yes
		cm_cmm_post_registration = yes
	}
}

# Unified callback hook for setting changes, alert clicks, action bar clicks.
cmf_on_callback = {
	on_actions = {
		cm_on_cmf_callback
	}
}

cm_on_cmf_callback = {
	effect = {
		cm_handle_cmf_callback = yes
	}
}
