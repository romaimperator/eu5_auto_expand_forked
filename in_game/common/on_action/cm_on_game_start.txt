# Populate building-type globals, goods globals, and town-right mapmode scores once per lobby.
on_game_start_after_lobby = {
	on_actions = {
		cm_on_init_lobby_globals
	}
}

on_game_load_after_lobby = {
	on_actions = {
		cm_on_init_lobby_globals
	}
}

cm_on_init_lobby_globals = {
	effect = {
		cm_populate_building_types_to_process = {}
		cm_fb_populate_excluded_types = yes
		cm_pf_populate_river_locations = yes
		cm_initialize_goods_globals = {}
		cm_amenity_populate_grant_maps = yes
		cm_trmm_recompute_all = yes
	}
}

# Seed food data, list visibility, and the first GUI tick at game start so the UI is populated immediately.
on_game_start_after_lobby_human_country = {
	on_actions = {
		cm_on_init_human_country
	}
}

on_game_load_after_lobby_human_country = {
	on_actions = {
		cm_on_init_human_country
	}
}

cm_on_init_human_country = {
	effect = {
		cm_update_food_data_columns = yes
		cm_update_auto_build_list_visibility = yes
		cm_update_auto_town_rights_list_visibility = yes
		cm_update_auto_food_list_visibility = yes
		cm_update_auto_urbanize_ignore_list_visibility = yes
		# A loaded save can have a cycle's flags stuck on, which stops auto-construction.
		cm_q_reset_cycle_state = yes
		# A save taken between the cleanup toggles and their end state would otherwise
		# toggle the same buildings a second time after the load.
		remove_variable = cm_fa_cu_fire
		clear_variable_list = cm_fa_cu_toggle
		set_variable = { name = cm_q_planned_spend value = 0 }
		cm_reset_market_committed_demand = yes
		# Initialize the placement finder scores for the starting country.
		cm_pf_recompute_now = yes
	}
}

# CMF copies CMM setting values and re-runs registration before this fires; only the non-CMM expand-threshold memory is copied here.
cmf_on_country_transfer = {
	on_actions = {
		cm_on_country_transfer
	}
}

cm_on_country_transfer = {
	effect = {
		cm_copy_expand_thresholds = { m = 1 }
		cm_copy_expand_thresholds = { m = 2 }
		cm_copy_expand_thresholds = { m = 3 }
		cm_copy_expand_thresholds = { m = 4 }
		cm_fa_copy_budget_state = yes
		cm_fb_copy_state = yes
		# An interrupted cycle on this country must not resume with stale approvals.
		cm_q_reset_cycle_state = yes
	}
}
