# Yearly hook for the Smart Automations passes that run once a year.
cmf_yearly_human_country_pulse = {
	on_actions = {
		cm_sa_yearly_pulse
	}
}

# Root is the country.
cm_sa_yearly_pulse = {
	effect = {
		cm_sa_food_yearly_evaluation = yes
		cm_sa_cu_yearly_pass = yes
	}
}

# on_capital_moved's root is the country, per vanilla in_game/common/on_action/_hardcoded.txt:5499.
four_yearly_country_pulse = {
	on_actions = {
		cm_sa_rd_on_plan
	}
}

on_capital_moved = {
	on_actions = {
		cm_sa_rd_on_plan
	}
}

# Root is the country. Both pulses fire for every country in the world.
cm_sa_rd_on_plan = {
	trigger = {
		OR = {
			is_ai = no
			cm_ln_is_active = yes
		}
	}
	effect = {
		cm_sa_rd_plan = yes
	}
}
