cm_upgrade_location_rank_on_action = {
	effect = {
		if = {
			limit = {
				has_building = building_type:cm_upgrade_town_to_city_building
			}
			change_building_level_in_location = { building = building_type:cm_upgrade_town_to_city_building value = -1 }
			if = {
				limit = { location_rank = location_rank:town }
				change_location_rank = location_rank:city
			}
		}
		if = {
			limit = {
				has_building = building_type:cm_upgrade_rural_to_town_building
			}
			change_building_level_in_location = { building = building_type:cm_upgrade_rural_to_town_building value = -1 }
			if = {
				limit = { location_rank = location_rank:rural_settlement }
				change_location_rank = location_rank:town
			}
		}
		if = {
			limit = {
				has_building = building_type:cm_upgrade_city_to_megalopolis_building
			}
			change_building_level_in_location = { building = building_type:cm_upgrade_city_to_megalopolis_building value = -1 }
			if = {
				limit = { location_rank = location_rank:city }
				change_location_rank = location_rank:megalopolis
			}
		}
		# A location leaving rural rank leaves the full-automation food system.
		if = {
			limit = {
				exists = var:cm_fa_food_marked
				NOT = { location_rank = location_rank:rural_settlement }
			}
			save_scope_as = cm_fa_left_rural
			owner = { cm_fa_food_drop_pick = { SITE = scope:cm_fa_left_rural } }
		}
	}
}
