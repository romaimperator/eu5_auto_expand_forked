# Keeps a lucky nation's gold out of the court, stability and diplomacy investment sliders.
cm_ln_frugal_government = {
	potential_trigger = {
		cm_ln_is_active = yes
	}

	# Court investment feeds government power (in_game/gui/shared/topbar_tooltips.gui:174 reads
	# court_maintenance into the legitimacy tooltip), so its target is what drives court spending.
	# Base is AI_GOVERNMENT_POWER_LOWER_LIMIT = 50: loading_screen/common/defines/00_defines.txt:1295
	ai_government_power_target_modifier = -50
	# Base is AI_STABILITY_LOWER_LIMIT = -50: loading_screen/common/defines/00_defines.txt:1294
	ai_stability_target_modifier = -50
	# Twenty times the strongest weighting vanilla applies, which spans -0.2 to +0.25
	# (in_game/common/ai_personalities/00_ai_personalities.txt:151, :172).
	stability_importance_modifier = -5
	diplomacy_importance_modifier = -5
	# No court_importance_modifier exists, so gold is the only weight reaching court spending.
	# Vanilla's greediest ruler is 0.2 (in_game/common/traits/00_ruler.txt:773).
	gold_importance_modifier = 5
	# Court and stability cost economical_base * 0.1 / (1 + this), so 99 leaves a hundredth.
	# Exactly -1 is a divide by zero. COURT_SPENDING_FRACTON and STABILIY_EXPENSE_FACTOR (both
	# 0.1, both spelled that way in vanilla): loading_screen/common/defines/00_defines.txt:1862, :1941
	court_spending_efficiency = 99
	# This one is the fraction itself rather than a divisor, so cancelling the country-wide 0.1
	# baseline (in_game/common/auto_modifiers/country.txt:117) leaves nothing.
	diplomatic_spending_cost = -0.1
}

# loading_screen/common/defines/00_defines.txt:1944
