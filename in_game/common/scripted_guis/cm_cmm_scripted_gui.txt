# Auto-Expand (auto_expand) Tab
## Construction Automation (priority_list) Group
cm__priority_list_on_changed = {
	scope = country

	is_shown = {
		always = yes
	}

	effect = {
		cmm_apply_list_change = {
			setting = cm__priority_list
		}
		cm_rebuild_priority_lookups = yes
	}
}
