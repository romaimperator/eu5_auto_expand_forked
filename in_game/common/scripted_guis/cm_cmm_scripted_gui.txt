# Auto Town Rights (auto_town_rights) Tab
## Auto-Grant Town Rights (auto_town_rights) Group
cm__town_rights_free_grants_on_changed = {
	scope = country

	is_shown = {
		cmf_can_use_unrestricted_tools = yes
	}

	# Global rule, so mirror it onto every country's revoke cost.
	effect = {
		if = {
			limit = { cm_town_rights_grants_are_free = yes }
			every_country = {
				add_country_modifier = { modifier = cm_town_rights_free years = 1000 mode = replace }
			}
		}
		else = {
			every_country = {
				remove_country_modifier = cm_town_rights_free
			}
		}
	}
}

cm__town_rights_min_gov_power_on_changed = {
	scope = country

	is_shown = {
		NOT = { cm_town_rights_grants_are_free = yes }
	}

	effect = { }
}
