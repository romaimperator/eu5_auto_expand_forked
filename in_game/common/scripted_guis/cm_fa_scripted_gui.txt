# Full-automation scripted GUIs: the queue scorers the construct queue window invokes per
# staged entry, the Building Cleanup toggle driver, and the vanilla automation card bridge.

# Build scorer for every category's staged entry. Root is the location. Stores the engine
# cost when the engine allows the build; Industry entries additionally score on ROI Tax
# Income, and a blocked Industry candidate still anchors its building type so a worse
# buildable one cannot slip inside the margin.
# scope:cm_cost = GetBuildOrExpandBuildingCost, scope:cm_income = income-to-owner per level,
# scope:cm_profit = profit per level, scope:cm_can_build = CanBuildOrExpandBuilding,
# scope:cm_obsolete_building = the standing obsolete predecessor, unset when none.
# Plus the indirect inputs: cm_indirect_loc, cm_cur_market_access,
# cm_cur_total_income, cm_cur_rgo_profit.
cm_fa_score_build = {
	saved_scopes = { cm_building_type cm_country cm_cost cm_income cm_profit cm_can_build cm_obsolete_building cm_indirect_loc cm_cur_market_access cm_cur_total_income cm_cur_rgo_profit }
	is_valid = {
		always = yes
	}
	effect = {
		cm_fa_score_build_apply = yes
	}
}

# Reserve pull from the Buildings card's gold slider. Polled, so the guards live here: skip
# while a push is pending, while the slider reads the stale zero it can show at card creation,
# when the value already matches, and while the setting sits above the slider's own maximum.
# Writes the CMM value map directly, since the callback would arm a push right back.
# Root is the player. Saved scopes: cm_fa_res (value).
cm_fa_reserve_changed = {
	scope = country
	saved_scopes = { cm_fa_res }
	is_shown = { always = yes }
	effect = {
		set_local_variable = { name = cm_fa_tmp_reserve value = scope:cm_fa_res }
		if = {
			limit = {
				NOT = { has_variable = cm_fa_reserve_from_cmm }
				local_var:cm_fa_tmp_reserve > 0
				NOT = { var:cm_fa_reserve = local_var:cm_fa_tmp_reserve }
				OR = {
					NOT = { has_variable = cm_fa_reserve_slider_max }
					var:cm_fa_reserve <= var:cm_fa_reserve_slider_max
				}
			}
			set_variable = { name = cm_fa_reserve value = scope:cm_fa_res }
			remove_from_variable_map = { name = cmm key = flag:cm__fa_reserve }
			add_to_variable_map = { name = cmm key = flag:cm__fa_reserve value = scope:cm_fa_res }
		}
	}
}

# Records the engine gold slider's scale, which appears in no file and has to be observed.
# Root is the player. Saved scopes: cm_fa_max (value).
cm_fa_reserve_max_changed = {
	scope = country
	saved_scopes = { cm_fa_max }
	is_shown = { always = yes }
	effect = {
		set_variable = { name = cm_fa_reserve_slider_max value = scope:cm_fa_max }
	}
}

# Clears the pending push once the caps have committed the value. Root is the player.
cm_fa_reserve_pushed = {
	scope = country
	is_shown = { always = yes }
	effect = {
		if = {
			limit = { has_variable = cm_fa_reserve_from_cmm }
			remove_variable = cm_fa_reserve_from_cmm
		}
	}
}

# Arms a push on every card creation, so the stored setting wins the open. Root is the player.
cm_fa_reserve_reconcile = {
	scope = country
	is_shown = { always = yes }
	effect = {
		set_variable = cm_fa_reserve_from_cmm
	}
}
