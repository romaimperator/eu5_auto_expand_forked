# Manual-upgrade auto-expand swap. Root is the obsolete building being upgraded;
# scope:cm_new_building_type is its engine replacement tier from the call site, unset when none.
cm_check_for_auto_expand_when_upgrading_building = {
	effect = {
		save_scope_as = cm_the_building
		scope:cm_the_building = { building_type = { save_temporary_scope_as = cm_old_building_type } }
		if = { limit = { exists = scope:cm_new_building_type }
			location = {
				if = { limit = { owner = { is_target_in_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_old_building_type } } }
					# Mass mode: add new type to country mass list, swap exclusion entries
					owner = {
						if = { limit = { NOT = { is_target_in_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_new_building_type } } }
							add_to_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_new_building_type }
						}
					}
					if = { limit = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_old_building_type } }
						remove_list_variable = { name = cm_auto_expand_excluded_building_types target = scope:cm_old_building_type }
						if = { limit = { NOT = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_new_building_type } } }
							add_to_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_new_building_type }
						}
					}
				} else = {
					# Normal mode: swap registration from old to new
					if = { limit = { is_target_in_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_old_building_type } }
						remove_list_variable = { name = cm_auto_expand_registered_building_types target = scope:cm_old_building_type }
						if = { limit = { NOT = { is_target_in_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_new_building_type } } }
							add_to_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_new_building_type }
						}
					}
				}
			}
		}
	}
}
# Called from the construct queue window to approve a building upgrade for the engine-action fire pass.
# Root = location, scope:cm_old_building_type = obsolete type, scope:cm_new_building_type = upgrade target from GetNextReplacementBuilding, scope:cm_country = player
# scope:cm_cost, scope:cm_income, scope:cm_profit = engine values for the replacement type.
# scope:cm_can_build = CanUpgradeToBuilding for the replacement type.
cm_gui_construct_upgrade = {
	is_valid = {
		always = yes
	}
	effect = {
		# The round loop re-triggers every item: run each staged entry once, skip stale widgets,
		# and freeze once the fire pass is armed.
		if = {
			limit = {
				scope:cm_country = { NOT = { exists = var:cm_q_fire_approved } }
				is_target_in_variable_list = { name = cm_gui_upgrade_building_types target = scope:cm_old_building_type }
				NOT = { is_target_in_variable_list = { name = cm_q_done_upgrade_types target = scope:cm_old_building_type } }
			}
			add_to_variable_list = { name = cm_q_done_upgrade_types target = scope:cm_old_building_type }
			scope:cm_country = { change_variable = { name = cm_q_processed add = 1 } }
			if = {
				limit = {
					exists = scope:cm_new_building_type
					scope:cm_can_build = yes
					cm_country_can_afford_staged_cost = yes
					scope:cm_new_building_type = { save_temporary_scope_as = cm_building_type }
					cm_market_has_construction_goods_for_building = yes
					# Auto-expand's obsolete upgrades are gated on the replacement's engine values;
					# upgrade-all entries are not marked gated and skip the metric check.
					OR = {
						NOT = { is_target_in_variable_list = { name = cm_q_upgrade_gated_types target = scope:cm_old_building_type } }
						cm_gui_metric_gate_passes = yes
					}
				}
				if = {
					limit = { NOT = { is_target_in_variable_list = { name = cm_q_approved_upgrade_types target = scope:cm_old_building_type } } }
					add_to_variable_list = { name = cm_q_approved_upgrade_types target = scope:cm_old_building_type }
					save_temporary_scope_as = cm_approved_location
					scope:cm_country = {
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = cm_q_approved_upgrade_locations target = scope:cm_approved_location } } }
							add_to_variable_list = { name = cm_q_approved_upgrade_locations target = scope:cm_approved_location }
							scope:cm_approved_location = { clear_variable_list = cm_q_seen_upgrade_types }
						}
						change_variable = { name = cm_q_planned_spend add = scope:cm_cost }
					}
					scope:cm_new_building_type = { save_temporary_scope_as = cm_building_type }
					cm_commit_market_demand = yes
				}
				# Swap auto-expand tracking from old to new
				if = { limit = { owner = { is_target_in_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_old_building_type } } }
					# Mass mode: add new type to country mass list and swap exclusion entries
					owner = {
						if = { limit = { NOT = { is_target_in_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_new_building_type } } }
							add_to_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_new_building_type }
						}
					}
					if = { limit = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_old_building_type } }
						remove_list_variable = { name = cm_auto_expand_excluded_building_types target = scope:cm_old_building_type }
						if = { limit = { NOT = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_new_building_type } } }
							add_to_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_new_building_type }
						}
					}
				} else = {
					# Normal mode: swap registration entries, only if old type was registered.
					if = { limit = { is_target_in_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_old_building_type } }
						remove_list_variable = { name = cm_auto_expand_registered_building_types target = scope:cm_old_building_type }
						if = { limit = { NOT = { is_target_in_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_new_building_type } } }
							add_to_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_new_building_type }
						}
					}
				}
				# Track that this building type has an upgrade queued so the monthly script won't re-flag it
				if = { limit = { NOT = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:cm_old_building_type } } }
					add_to_variable_list = { name = cm_upgrade_in_progress target = scope:cm_old_building_type }
				}
				# Record the constructed target for the finished-upgrade check
				remove_from_variable_map = { name = cm_upgrade_in_progress_targets key = scope:cm_old_building_type }
				add_to_variable_map = { name = cm_upgrade_in_progress_targets key = scope:cm_old_building_type value = scope:cm_new_building_type }
			}
		}
	}
}
# Construction-queue executor for building kinds. Root is the location.
# Invoked by the construct queue window per queued entry; applies the metric gate
# and the gold-affordability gate using the engine values grabbed in GUI, then approves the build.
# scope:cm_cost = GetBuildOrExpandBuildingCost, scope:cm_income = income-to-owner per level,
# scope:cm_profit = profit per level, scope:cm_can_build = CanBuildOrExpandBuilding.
# scope:cm_obsolete_building = the standing obsolete predecessor from GetObsoleteBuildingInLocation, unset when none.
cm_gui_try_build = {
	saved_scopes = { cm_building_type cm_country cm_cost cm_income cm_profit cm_can_build cm_obsolete_building }
	is_valid = {
		always = yes
	}
	effect = {
		if = {
			limit = {
				scope:cm_country = { NOT = { exists = var:cm_q_fire_approved } }
				is_target_in_variable_list = { name = cm_q_building_types target = scope:cm_building_type }
				NOT = { is_target_in_variable_list = { name = cm_q_done_build_types target = scope:cm_building_type } }
			}
			add_to_variable_list = { name = cm_q_done_build_types target = scope:cm_building_type }
			scope:cm_country = { change_variable = { name = cm_q_processed add = 1 } }
			if = {
				limit = {
					cm_gui_metric_gate_passes = yes
					scope:cm_can_build = yes
					cm_country_can_afford_staged_cost = yes
					cm_market_has_construction_goods_for_building = yes
					# A staged upgrade already constructs this type here
					NOT = { is_target_in_variable_list = { name = cm_q_upgrade_target_types target = scope:cm_building_type } }
					# The engine converts a build into the standing obsolete predecessor's upgrade.
					trigger_if = {
						limit = { exists = scope:cm_obsolete_building }
						scope:cm_obsolete_building = { building_type = { save_temporary_scope_as = cm_build_obsolete_type } }
						NOT = { is_target_in_variable_list = { name = cm_gui_upgrade_building_types target = scope:cm_build_obsolete_type } }
					}
				}
				cm_approve_gui_build = yes
				if = {
					limit = { exists = scope:cm_obsolete_building }
					scope:cm_obsolete_building = { building_type = { save_temporary_scope_as = cm_build_obsolete_type } }
					if = {
						limit = { NOT = { is_target_in_variable_list = { name = cm_upgrade_in_progress target = scope:cm_build_obsolete_type } } }
						add_to_variable_list = { name = cm_upgrade_in_progress target = scope:cm_build_obsolete_type }
					}
					remove_from_variable_map = { name = cm_upgrade_in_progress_targets key = scope:cm_build_obsolete_type }
					add_to_variable_map = { name = cm_upgrade_in_progress_targets key = scope:cm_build_obsolete_type value = scope:cm_building_type }
				}
			}
		}
	}
}
# Construction-queue RGO executor. Root is the location; re-checks RGO eligibility
# including gold, and the metric gate at drain time, then upgrades the RGO.
# Expects scopes: country, location, income, profit. scope:cm_profit = Location.GetRGOProfitPerLevel / Location.GetControl.
# scope:cm_income = scope:cm_profit scaled by the location income-to-wealth ratio.

		always = yes
	}
	effect = {
		if = {
			limit = {
				scope:cm_country = { NOT = { exists = var:cm_q_fire_approved } }
				scope:cm_country = { is_target_in_variable_list = { name = cm_q_rgo_locations target = scope:cm_location } }
				NOT = { has_variable = cm_q_done_rgo }
			}
			set_variable = { name = cm_q_done_rgo value = yes }
			scope:cm_country = { change_variable = { name = cm_q_processed add = 1 } }

					cm_should_rgo_auto_expand = yes
					cm_rgo_metric_gate_passes = yes
					cm_market_has_construction_goods_for_rgo = yes
				}
				construct_rgo_upgrade = { }
			}
		}
	}
}
# Ungated construction-queue executor for auto-build and capital. Root is the location;
# re-checks build-queue slots and affordability against the engine cost, then approves the build.
# Subject locations build with the country as owner and payer. Expects scopes: country, building_type, cost, can_build.
cm_gui_try_ungated = {
	saved_scopes = { cm_country cm_building_type cm_cost cm_can_build }
	is_valid = {
		always = yes
	}
	effect = {
		if = {
			limit = {
				scope:cm_country = { NOT = { exists = var:cm_q_fire_approved } }
				is_target_in_variable_list = { name = cm_q_ungated_building_types target = scope:cm_building_type }
				NOT = { is_target_in_variable_list = { name = cm_q_done_ungated_types target = scope:cm_building_type } }
			}
			add_to_variable_list = { name = cm_q_done_ungated_types target = scope:cm_building_type }
			scope:cm_country = { change_variable = { name = cm_q_processed add = 1 } }
			if = {
				limit = {
					cm_get_open_build_queue_slots > 0
					cm_country_can_afford_staged_cost = yes
					cm_market_has_construction_goods_for_building = yes

						NOT = { owner = scope:cm_country }
						scope:cm_can_build = yes
					}
				}
				if = {
					limit = { owner = scope:cm_country }
					cm_approve_gui_build = yes
				} else = {
					construct_building = { building_type = scope:cm_building_type owner = scope:cm_country payer = scope:cm_country }
					cm_commit_market_demand = yes
				}
			}
		}
	}
}

cm_check_for_auto_expand_when_removing_building = {
	effect = {
		save_scope_as = cm_the_building
		scope:cm_the_building = { building_type = { save_temporary_scope_as = cm_building_type } }
		owner = { save_scope_as = cm_country }
		location = {
			if = { limit = { scope:cm_the_building = { building_level < 2 } }
				scope:cm_building_type = { cm_find_current_building_type = yes }
				# Clean up both lists - only the relevant one will have an entry
				if = { limit = { is_target_in_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_building_type } }
					remove_list_variable = { name = cm_auto_expand_registered_building_types target = scope:cm_building_type }
				}
				if = { limit = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_building_type } }
					remove_list_variable = { name = cm_auto_expand_excluded_building_types target = scope:cm_building_type }
				}
				# Demolishing the last tier of a transferred lineage clears the replacement's entries
				if = {
					limit = {
						scope:cm_country = { building_type_is_obsolete = scope:cm_building_type }
						NOT = { scope:cm_country = { building_type_is_obsolete = scope:cm_current_building_type } }
						NOT = { has_building = scope:cm_current_building_type }
					}
					if = { limit = { is_target_in_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_current_building_type } }
						remove_list_variable = { name = cm_auto_expand_registered_building_types target = scope:cm_current_building_type }
					}
					if = { limit = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_current_building_type } }
						remove_list_variable = { name = cm_auto_expand_excluded_building_types target = scope:cm_current_building_type }
					}
				}
			}
		}
	}
}

	}
}

# Root is the building type. Toggles mass auto-expand for this specific building type.
# When ON: this building type auto-expands at all owned locations, exclusion-based per location.
# When OFF: this building type is removed from mass tracking and location exclusion lists are cleaned.
cm_toggle_mass_auto_expand_for_building_type = {
	saved_scopes = { cm_country }
	effect = {
		save_scope_as = cm_building_type
		if = { limit = { scope:cm_country = { is_target_in_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_building_type } } }
			# Turning OFF - remove from country list and clear exclusion entries for this type
			scope:cm_country = {
				remove_list_variable = { name = cm_mass_auto_expand_building_types target = scope:cm_building_type }
				ordered_owned_location = {
					order_by = max_control
					max = cm_max_guess_at_number_of_locations
					check_range_bounds = no
					limit = { is_target_in_variable_list = { name = cm_auto_expand_excluded_building_types target = scope:cm_building_type } }
					remove_list_variable = { name = cm_auto_expand_excluded_building_types target = scope:cm_building_type }
				}
			}
		} else = {
			# Turning ON - add to country list and clear per-location registrations for this type
			scope:cm_country = {
				add_to_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_building_type }
				ordered_owned_location = {
					order_by = max_control
					max = cm_max_guess_at_number_of_locations
					check_range_bounds = no
					limit = { is_target_in_variable_list = { name = cm_auto_expand_registered_building_types target = scope:cm_building_type } }
					remove_list_variable = { name = cm_auto_expand_registered_building_types target = scope:cm_building_type }
				}
			}
		}
	}
	is_shown = {
		save_temporary_scope_as = cm_building_type
		scope:cm_country = { is_target_in_variable_list = { name = cm_mass_auto_expand_building_types target = scope:cm_building_type } }
	}
}
