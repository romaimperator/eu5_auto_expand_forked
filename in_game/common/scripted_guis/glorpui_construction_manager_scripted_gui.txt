# True when the Construction Manager mod is active.
glorpui_is_cm_active = {
	scope = country

	is_shown = {
		is_target_in_global_variable_list = {
			name = cmf_active_mod_ids
			target = flag:cm
		}
	}
	effect = { }
}

# True when the ROI figures should carry Construction Manager's predicted indirect income.
glorpui_roi_indirect_active = {
	scope = country

	is_shown = {
		is_target_in_global_variable_list = {
			name = cmf_active_mod_ids
			target = flag:cm
		}
		NOT = { has_variable = disableRoiIndirect }
	}
	effect = { }
}
