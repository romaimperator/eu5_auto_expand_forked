# Per-round completion check for the construct window's loop. Root is the player.
# Once every executor has scored (processed >= staged), runs cm_q_approval_pass, then arms
# the fire pass when every approved entry's widget is verified by its probe mark. A cycle
# making no instantiation progress re-arms the scan sweep twice, then fires whatever is
# verified instead of dropping the month.
cm_q_sync_check = {
	is_valid = {
		always = yes
	}
	effect = {
		if = {
			limit = { NOT = { exists = var:cm_q_fire_approved } }
			save_scope_as = cm_q_chk_country
			set_local_variable = { name = cm_q_chk_all_seen value = 1 }
			set_local_variable = { name = cm_q_chk_any value = 0 }
			set_variable = { name = cm_q_seen_count value = 0 }
			if = {
				limit = {
					var:cm_q_processed >= var:cm_q_staged
					NOT = { exists = var:cm_q_pass_done }
				}
				set_variable = { name = cm_q_pass_done value = yes }
				cm_q_approval_pass = yes
			}
			if = {
				limit = { var:cm_q_processed >= var:cm_q_staged }
				every_in_list = {
					variable = cm_q_approved_build_locations
					save_scope_as = cm_q_chk_loc
					scope:cm_q_chk_loc = {
						every_in_list = {
							variable = cm_q_approved_build_types
							save_scope_as = cm_q_chk_type
							scope:cm_q_chk_country = { set_local_variable = { name = cm_q_chk_any value = 1 } }
							if = {
								limit = { scope:cm_q_chk_loc = { is_target_in_variable_list = { name = cm_q_seen_build_types target = scope:cm_q_chk_type } } }
								scope:cm_q_chk_country = { change_variable = { name = cm_q_seen_count add = 1 } }
							} else = {
								scope:cm_q_chk_country = { set_local_variable = { name = cm_q_chk_all_seen value = 0 } }
							}
						}
					}
				}
				every_in_list = {
					variable = cm_q_approved_upgrade_locations
					save_scope_as = cm_q_chk_loc
					scope:cm_q_chk_loc = {
						every_in_list = {
							variable = cm_q_approved_upgrade_types
							save_scope_as = cm_q_chk_type
							scope:cm_q_chk_country = { set_local_variable = { name = cm_q_chk_any value = 1 } }
							if = {
								limit = { scope:cm_q_chk_loc = { is_target_in_variable_list = { name = cm_q_seen_upgrade_types target = scope:cm_q_chk_type } } }
								scope:cm_q_chk_country = { change_variable = { name = cm_q_seen_count add = 1 } }
							} else = {
								scope:cm_q_chk_country = { set_local_variable = { name = cm_q_chk_all_seen value = 0 } }
							}
						}
					}
				}
			}
			every_in_list = {
				variable = cm_q_approved_rank_locations
				save_scope_as = cm_q_chk_loc
				scope:cm_q_chk_country = { set_local_variable = { name = cm_q_chk_any value = 1 } }
				if = {
					limit = { scope:cm_q_chk_loc = { has_variable = cm_q_seen_rank } }
					scope:cm_q_chk_country = { change_variable = { name = cm_q_seen_count add = 1 } }
				} else = {
					scope:cm_q_chk_country = { set_local_variable = { name = cm_q_chk_all_seen value = 0 } }
				}
			}
			if = {
				limit = {
					var:cm_q_processed >= var:cm_q_staged
					local_var:cm_q_chk_all_seen = 1
				}
				if = {
					limit = { local_var:cm_q_chk_any = 1 }
					set_variable = { name = cm_q_fire_approved value = yes }
				} else = {
					cm_q_reset_cycle_state = yes
				}
			} else = {
				# Stall watchdog: progress is executor completions plus verified approved entries.
				set_variable = { name = cm_q_progress_now value = var:cm_q_processed }
				change_variable = { name = cm_q_progress_now add = var:cm_q_seen_count }
				if = {
					limit = { var:cm_q_progress_now > var:cm_q_progress_last }
					set_variable = { name = cm_q_progress_last value = var:cm_q_progress_now }
					set_variable = { name = cm_q_stall_rounds value = 0 }
				} else = {
					change_variable = { name = cm_q_stall_rounds add = 1 }
					if = {
						# Scoring is the only side a re-sweep can help: an item widget that had
						# not instantiated when the one sweep fired has no other way back in,
						# while an unverified approved entry already gets a probe every round.
						limit = {
							var:cm_q_stall_rounds >= 6
							var:cm_q_processed < var:cm_q_staged
							var:cm_q_rescans < 2
							NOT = { exists = var:cm_q_scan_armed }
						}
						change_variable = { name = cm_q_rescans add = 1 }
						set_variable = { name = cm_q_stall_rounds value = 0 }
						set_variable = { name = cm_q_scan_armed value = yes }
					} else_if = {
						limit = { var:cm_q_stall_rounds >= { value = cm_q_give_up_rounds } }
						cmf_log = { action = cm_log_cycle_gave_up }
						# A stalled cycle can give up before reaching the approval pass; approve
						# whatever scored so the month is not dropped outright.
						if = {
							limit = { NOT = { exists = var:cm_q_pass_done } }
							set_variable = { name = cm_q_pass_done value = yes }
							cm_q_approval_pass = yes
						}
						# The fire pass cannot reach an unverified upgrade; unwind its in-progress
						# markers or a fort would stay blocked from re-staging.
						every_in_list = {
							variable = cm_q_approved_upgrade_locations
							save_scope_as = cm_q_chk_loc
							scope:cm_q_chk_loc = {
								every_in_list = {
									variable = cm_q_approved_upgrade_types
									save_scope_as = cm_q_chk_type
									if = {
										limit = { NOT = { scope:cm_q_chk_loc = { is_target_in_variable_list = { name = cm_q_seen_upgrade_types target = scope:cm_q_chk_type } } } }
										scope:cm_q_chk_loc = {
											remove_list_variable = { name = cm_upgrade_in_progress target = scope:cm_q_chk_type }
											remove_from_variable_map = { name = cm_upgrade_in_progress_targets key = scope:cm_q_chk_type }
										}
									}
								}
							}
						}
						set_variable = { name = cm_q_fire_approved value = yes }
					}
				}
			}
		}
	}
}

# Root is the player. Advances the batched river harvest one tick: queues the next batch of source
# locations for the datamodel to instantiate, or, once every location has been queued, stamps the
# harvest complete, tears down the source lists (freeing the per-frame widget cost) and invalidates
# the finder scores computed before the adjacency data existed. The harvest runs once per save, so
# nothing synchronous is chained off it here. 150 sources per tick keeps the record volume under
# the flood threshold. An iterator limit cannot bound the cursor walk: the limit is evaluated
# against local_var:cm_pf_river_i before the body increments it, so every element passes and the
# whole tail of the list is queued in one fire, flooding scripted_gui_command until the game hangs.
cm_pf_river_fill_batch = {
	effect = {
		if = {
			limit = { global_var:cm_pf_river_idx < global_var:cm_pf_river_count }
			clear_global_variable_list = cm_pf_river_batch
			set_local_variable = { name = cm_pf_river_hi value = { value = global_var:cm_pf_river_idx add = 150 } }
			set_local_variable = { name = cm_pf_river_i value = 0 }
			every_in_global_list = {
				variable = cm_pf_river_locations
				if = {
					limit = {
						local_var:cm_pf_river_i >= global_var:cm_pf_river_idx
						local_var:cm_pf_river_i < local_var:cm_pf_river_hi
					}
					add_to_global_variable_list = { name = cm_pf_river_batch target = this }
				}
				change_local_variable = { name = cm_pf_river_i add = 1 }
			}
			change_global_variable = { name = cm_pf_river_idx add = 150 }
		}
		else_if = {
			limit = { has_global_variable = cm_pf_river_process }
			clear_global_variable_list = cm_pf_river_locations
			clear_global_variable_list = cm_pf_river_batch
			set_global_variable = { name = cm_pf_river_idx value = 0 }
			set_global_variable = { name = cm_pf_river_count value = 0 }
			remove_global_variable = cm_pf_river_process
			# Written only here, so a save taken mid-harvest carries no stamp and re-harvests.
			set_global_variable = { name = cm_pf_river_stamp value = cm_pf_river_version_value }
			if = {
				# A mode opened during the harvest primed its edges before the neighbors existed;
				# re-priming lets its own driver rebatch instead of caching that for four years.
				limit = { exists = var:cm_pf_mode_open }
				cm_pf_force_refresh = yes
			}
			else = {
				remove_variable = cm_pf_fresh_g
				remove_variable = cm_pf_fresh_c
			}
			cm_log_setup_complete_if_ready = yes
		}
	}
}

# Root is the country
cm_building_type_section_buildings_init = {
	saved_scopes = { cm_building_type is_producing is_multiple_levels is_fort pop_type upgrade_type }
	effect = {
		# debug_log = "CM: Initializing buildings [TopScope.sBuildingType('building_type').GetNameWithNoTooltip]"

		# Set the which type of building this should be treated as
		if = { limit = { AND = { scope:cm_is_producing = yes scope:cm_is_multiple_levels = yes } }
			add_to_global_variable_map = { name = cm_building_type_to_is_production_building key = scope:cm_building_type value = 1 }
			if = { limit = { scope:cm_is_fort = yes }
				# debug_log = "CM: Building: [TopScope.sBuildingType('building_type').GetNameWithNoTooltip] is_production_building: yes is_fort: yes"
			} else = {
				# debug_log = "CM: Building: [TopScope.sBuildingType('building_type').GetNameWithNoTooltip] is_production_building: yes is_fort: no"
			}
		} else = {
			add_to_global_variable_map = { name = cm_building_type_to_is_production_building key = scope:cm_building_type value = 0 }
		}
		# Set if this building type is a fort for discount purposes
		if = { limit = { scope:cm_is_fort = yes }
			add_to_global_variable_map = { name = cm_building_type_to_is_fort key = scope:cm_building_type value = 1 }
		} else = {
			add_to_global_variable_map = { name = cm_building_type_to_is_fort key = scope:cm_building_type value = 0 }
		}
		# Set the pop type employed by this building type
		if = { limit = { exists = scope:cm_pop_type }
			add_to_global_variable_map = { name = cm_building_type_to_pop_type key = scope:cm_building_type value = scope:cm_pop_type }
		}

		# Set the building type that this building upgrades into if there is one
		if = { limit = { exists = scope:cm_upgrade_type }
			add_to_global_variable_map = { name = cm_building_type_to_upgrade_type key = scope:cm_building_type value = scope:cm_upgrade_type }
	  }

		# Set that this building type has been processed
		add_to_global_variable_map = { name = cm_building_type_processed key = scope:cm_building_type value = 1 }

		# Check if all buildings have been classified
		if = { limit = { AND = {
			has_global_variable = cm_process_buildings
			has_global_variable_map = cm_building_type_processed
			global_variable_map_size = { name = cm_building_type_processed value >= global_var:cm_building_types_to_process_count }
		} }
			remove_global_variable = cm_process_buildings
			# Written only here, so a save taken mid-classification carries no stamp and reclassifies.
			set_global_variable = { name = cm_bt_class_stamp value = cm_bt_class_version_value }
			# The rgob collect data is complete; start the per-province coverage sweep the
			# R.G.O. Bonus filters read. Arm before the readiness check, so a sweep that is
			# about to run still counts as outstanding.
			cm_rgob_arm_sweep = yes
			cm_log_setup_complete_if_ready = yes
		}
	}
}

# R.G.O. Bonus scan wave, fired once per producing production method item.
# Records each building type's best output. A type with no producing method never
# gets a best entry and is skipped by the collect wave.
# cm_bt_output carries the same value past cm_rgob_reset_scan_state, which the collapsar's
# deficit sizing reads to turn a goods shortfall into a count of producer levels.
cm_rgob_scan_pm = {
	saved_scopes = { cm_rgob_bt cm_rgob_out }
	effect = {
		if = {
			limit = { scope:cm_rgob_out > 0 }
			if = {
				limit = { NOT = { is_key_in_global_variable_map = { name = cm_rgob_best target = scope:cm_rgob_bt } } }
				add_to_global_variable_map = { name = cm_rgob_best key = scope:cm_rgob_bt value = scope:cm_rgob_out }
				add_to_global_variable_map = { name = cm_bt_output key = scope:cm_rgob_bt value = scope:cm_rgob_out }
			}
			else_if = {
				limit = { scope:cm_rgob_out > "global_variable_map(cm_rgob_best|scope:cm_rgob_bt)" }
				remove_from_global_variable_map = { name = cm_rgob_best key = scope:cm_rgob_bt }
				add_to_global_variable_map = { name = cm_rgob_best key = scope:cm_rgob_bt value = scope:cm_rgob_out }
				add_to_global_variable_map = { name = cm_bt_output key = scope:cm_rgob_bt value = scope:cm_rgob_out }
			}
		}
	}
}

# Scan wave, fired once per goods demand entry of a producing method. Accumulates the
# method's total input amount. The claim map makes the first building type to fire own
# a shared method's total, so a method referenced by several types is summed exactly
# once (their entry sets are identical).
cm_rgob_scan_entry = {
	saved_scopes = { cm_rgob_bt cm_rgob_pm cm_rgob_amount }
	effect = {
		if = {
			limit = { NOT = { is_key_in_global_variable_map = { name = cm_rgob_total_claim target = scope:cm_rgob_pm } } }
			add_to_global_variable_map = { name = cm_rgob_total_claim key = scope:cm_rgob_pm value = scope:cm_rgob_bt }
		}
		if = {
			limit = { "global_variable_map(cm_rgob_total_claim|scope:cm_rgob_pm)" = scope:cm_rgob_bt }
			if = {
				limit = { is_key_in_global_variable_map = { name = cm_rgob_total target = scope:cm_rgob_pm } }
				set_local_variable = { name = cm_rgob_t value = { value = "global_variable_map(cm_rgob_total|scope:cm_rgob_pm)" add = scope:cm_rgob_amount } }
				remove_from_global_variable_map = { name = cm_rgob_total key = scope:cm_rgob_pm }
				add_to_global_variable_map = { name = cm_rgob_total key = scope:cm_rgob_pm value = local_var:cm_rgob_t }
			}
			else = {
				add_to_global_variable_map = { name = cm_rgob_total key = scope:cm_rgob_pm value = scope:cm_rgob_amount }
			}
		}
	}
}

# Collect wave, fired once per producing production method item. Appends a
# (building type, method) pair for every worth-using method: output at least half
# the type's best output.
cm_rgob_collect_pm = {
	saved_scopes = { cm_rgob_bt cm_rgob_pm cm_rgob_out }
	effect = {
		if = {
			limit = { is_key_in_global_variable_map = { name = cm_rgob_best target = scope:cm_rgob_bt } }
			set_local_variable = { name = cm_rgob_thr value = { value = "global_variable_map(cm_rgob_best|scope:cm_rgob_bt)" multiply = 0.5 } }
			if = {
				limit = { scope:cm_rgob_out >= local_var:cm_rgob_thr }
				change_global_variable = { name = cm_rgob_pair_count add = 1 }
				add_to_global_variable_map = { name = cm_rgob_pair_bt key = global_var:cm_rgob_pair_count value = scope:cm_rgob_bt }
				# Push onto the method's pair chain, so a province walks only the methods it
				# accumulated a sum for instead of all of them. 0 ends a chain, and the counter
				# is pre-incremented, so no real index collides with it.
				set_local_variable = { name = cm_rgob_head value = 0 }
				if = {
					limit = { is_key_in_global_variable_map = { name = cm_rgob_pm_head target = scope:cm_rgob_pm } }
					set_local_variable = { name = cm_rgob_head value = "global_variable_map(cm_rgob_pm_head|scope:cm_rgob_pm)" }
					remove_from_global_variable_map = { name = cm_rgob_pm_head key = scope:cm_rgob_pm }
				}
				add_to_global_variable_map = { name = cm_rgob_pair_next key = global_var:cm_rgob_pair_count value = local_var:cm_rgob_head }
				add_to_global_variable_map = { name = cm_rgob_pm_head key = scope:cm_rgob_pm value = global_var:cm_rgob_pair_count }
			}
		}
	}
}

# Collect wave, fired once per goods demand entry. Appends one row (method, good,
# input share) per raw-material input of a worth-using method. Rows are
# method-intrinsic: the claim map writes a shared method's rows once, and only a
# building type the method is worth-using for can claim, so the rows always exist
# when a pair references them.
cm_rgob_collect_entry = {
	saved_scopes = { cm_rgob_bt cm_rgob_pm cm_rgob_out cm_rgob_good cm_rgob_amount }
	effect = {
		if = {
			limit = { is_key_in_global_variable_map = { name = cm_rgob_best target = scope:cm_rgob_bt } }
			set_local_variable = { name = cm_rgob_thr value = { value = "global_variable_map(cm_rgob_best|scope:cm_rgob_bt)" multiply = 0.5 } }
			if = {
				limit = { scope:cm_rgob_out >= local_var:cm_rgob_thr }
				if = {
					limit = { NOT = { is_key_in_global_variable_map = { name = cm_rgob_row_claim target = scope:cm_rgob_pm } } }
					add_to_global_variable_map = { name = cm_rgob_row_claim key = scope:cm_rgob_pm value = scope:cm_rgob_bt }
				}
				if = {
					limit = {
						"global_variable_map(cm_rgob_row_claim|scope:cm_rgob_pm)" = scope:cm_rgob_bt
						scope:cm_rgob_good = { goods_category = raw_material }
						scope:cm_rgob_amount > 0
						is_key_in_global_variable_map = { name = cm_rgob_total target = scope:cm_rgob_pm }
					}
					set_local_variable = { name = cm_rgob_tot value = "global_variable_map(cm_rgob_total|scope:cm_rgob_pm)" }
					if = {
						limit = { local_var:cm_rgob_tot > 0 }
						set_local_variable = { name = cm_rgob_share value = { value = scope:cm_rgob_amount divide = local_var:cm_rgob_tot } }
						change_global_variable = { name = cm_rgob_row_count add = 1 }
						add_to_global_variable_map = { name = cm_rgob_row_pm key = global_var:cm_rgob_row_count value = scope:cm_rgob_pm }
						add_to_global_variable_map = { name = cm_rgob_row_share key = global_var:cm_rgob_row_count value = local_var:cm_rgob_share }
						# Push onto the good's row chain, so a province walks only the rows for the
						# handful of goods it actually produces instead of all of them.
						set_local_variable = { name = cm_rgob_head value = 0 }
						if = {
							limit = { is_key_in_global_variable_map = { name = cm_rgob_good_head target = scope:cm_rgob_good } }
							set_local_variable = { name = cm_rgob_head value = "global_variable_map(cm_rgob_good_head|scope:cm_rgob_good)" }
							remove_from_global_variable_map = { name = cm_rgob_good_head key = scope:cm_rgob_good }
						}
						add_to_global_variable_map = { name = cm_rgob_row_next key = global_var:cm_rgob_row_count value = local_var:cm_rgob_head }
						add_to_global_variable_map = { name = cm_rgob_good_head key = scope:cm_rgob_good value = global_var:cm_rgob_row_count }
					}
				}
				# Building Cleanup's construction-material flag, riding the same per-entry data.
				# Outside the row claim: rows are method-intrinsic and written once, but every
				# type the method belongs to has to carry the flag. Masonry and glass are
				# produced goods rather than raw materials, so the rows above never see them.
				cm_sa_cu_mark_core_input = yes
			}
		}
	}
}
