# Building Cleanup scripted GUIs: the toggle driver the construct queue window fires from, and
# the vanilla Close and Open Buildings and Destroy Buildings card bridges.

# Gate for the Building Cleanup toggle driver. Root is the player.
cm_cu_fire_gate = {
	is_shown = {
		exists = var:cm_cu_fire
	}
}

# Ends the toggle pass. Root is the player.
cm_cu_end = {
	effect = {
		remove_variable = cm_cu_fire
		clear_variable_list = cm_cu_toggle
	}
}

# Toggles closing and opening from the vanilla Close and Open Buildings card. Root is the player.
cm_cu_toggle_close_open = {
	scope = country
	is_shown = { always = yes }
	effect = {
		cm_cu_toggle_close_open_value = yes
	}
}

# Tick state for the intercepted Close and Open Buildings card. Root is the player.
cm_cu_close_open_is_enabled = {
	scope = country
	is_shown = {
		exists = var:cm_cu_close_open
	}
}

# Toggles destroying from the vanilla Destroy Buildings card. Root is the player.
cm_cu_toggle_destroy = {
	scope = country
	is_shown = { always = yes }
	effect = {
		cm_cu_toggle_destroy_value = yes
	}
}

# Tick state for the intercepted Destroy Buildings card. Root is the player.
cm_cu_destroy_is_enabled = {
	scope = country
	is_shown = {
		exists = var:cm_cu_destroy
	}
}
