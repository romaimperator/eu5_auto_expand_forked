# Root is the location, also expects scope:cm_building_type.
cm_location_has_enough_pops_available = {
	has_global_variable_map = cm_building_type_to_pop_type
	is_key_in_global_variable_map = { name = cm_building_type_to_pop_type target = scope:cm_building_type }
	OR = {
		cm_location_unemployed_for_building_type > 0
		cm_location_promotion_buffer_covers_deficit = yes
	}
}

# Buffer branch of cm_location_has_enough_pops_available (root location, expects scope:cm_building_type):
# the pop deficit fills within six years at the location's own fill rate. The switch confines the
# projection to the six types cm_location_promotion_buffer_slack has a branch for.
cm_location_promotion_buffer_covers_deficit = {
	switch = { trigger = "global_variable_map(cm_building_type_to_pop_type|scope:cm_building_type)"
		pop_type:nobles = { always = yes }
		pop_type:clergy = { always = yes }
		pop_type:burghers = { always = yes }
		pop_type:laborers = { always = yes }
		pop_type:soldiers = { always = yes }
		pop_type:peasants = { always = yes }
		fallback = { always = no }
	}
	cm_location_promotion_buffer_slack >= 0
}

# Root is the location. Some pop type's building jobs, plus the RGO's share, outnumber the pops
# that can fill them. The laborers/slaves pool is read combined so the RGO and the jobs either
# type can take are not counted twice; between them the six values cover all eight pop types.
cm_location_has_worker_deficit = {
	OR = {
		cm_location_nobles_available < 0
		cm_location_clergy_available < 0
		cm_location_burghers_available < 0
		cm_location_soldiers_available < 0
		cm_location_peasants_available < 0
		cm_location_laborers_or_slaves_available < 0
	}
}

# Root is the location. Construction gate for locations auto-food is working: while any pop type
# is short, a new building level would take pops the food buildings and RGO need.
cm_location_auto_food_workers_ok = {
	OR = {
		cm_fa_food_location_effectively_active = no
		cm_location_has_worker_deficit = no
	}
}
