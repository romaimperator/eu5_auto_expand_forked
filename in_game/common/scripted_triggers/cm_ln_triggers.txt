# Lucky Nations triggers.

# Root is the country. is_ai is part of the test because a human who tag-switches onto a
# marked country must take the ordinary player path, not the script-construct one.
cm_ln_is_active = {
	exists = var:cm_ln_active
	is_ai = yes
}

# The engine buildability bools take no country argument and the engine only ever adds an
# explicit country parameter where it supports acting for someone else (Location.CanBuildArmy
# beside Location.CanBuildArmyForCountry), so a lucky nation reads the script mirror instead.
# Root is the location. Expects scope:cm_country, scope:cm_building_type, scope:cm_can_build.
cm_ln_can_place = {
	trigger_if = {
		limit = { scope:cm_country = { cm_ln_is_active = yes } }
		cm_location_can_place_building = { target = scope:cm_building_type }
	}
	trigger_else = {
		scope:cm_can_build = yes
	}
}

# Root is the location. Expects scope:cm_country. The player's own auto-urbanize carries no
# tooltip row for this, so only a lucky nation's is gated.
cm_ln_location_proximity_ok = {
	trigger_if = {
		limit = { scope:cm_country = { cm_ln_is_active = yes } }
		cm_fa_location_has_proximity = yes
	}
}

# Root is the building type. The five main-chain forts, the only ones the fort category builds
# or removes: in_game/common/building_types/forts.txt (stockade :1, castle :49, bastion :97,
# star_fort :145, fortress :190). city_walls and the unique walls are left standing.
cm_ln_ft_is_chain_fort = {
	OR = {
		this = building_type:stockade
		this = building_type:castle
		this = building_type:bastion
		this = building_type:star_fort
		this = building_type:fortress
	}
}

# Root is the location. Expects scope:cm_country. A fort the yearly teardown could remove.
cm_ln_ft_has_chain_fort = {
	any_buildings_in_location = {
		owner = scope:cm_country
		cm_is_not_under_construction = yes
		building_type = { cm_ln_ft_is_chain_fort = yes }
	}
}

# Root is the country. Overshooting the limit applies global_defensive -1.0
# (in_game/common/auto_modifiers/country.txt:1476), so staging stops at the cap.
cm_ln_ft_has_room = {
	used_fort_limit < {
		value = modifier:fort_limit
		subtract = var:cm_ln_ft_claimed
	}
}

# Root is the location. Expects scope:cm_country, scope:cm_building_type.
cm_ln_ft_site_placeable = {
	NOT = { has_variable = cm_ln_ft_zx }
	has_fort = no
	cm_location_allowed_to_build = { target = scope:cm_building_type }
}

# Root is the location. Expects scope:cm_country, scope:cm_building_type.
cm_ln_ft_site_eligible = {
	has_variable = cm_ln_ft_nd
	cm_ln_ft_site_placeable = yes
}

# --- Capital and governor reorganization ---

# Root is the building type.
cm_ln_is_governor_type = {
	OR = {
		this = building_type:local_governor
		this = building_type:naval_governor
	}
}

# Root is the location. A site a land governor could never reach, which is exactly what the
# naval type is for: in_game/common/building_types/capital_buildings.txt:799-803.
cm_ln_site_is_naval = {
	NOT = { is_connected_to_through_realm = owner.capital }
	is_overseas_for_owner = no
	is_port = yes
}

# Root is the location. Mirrors both vanilla governor types minus the road and the city rank,
# which a reorganization builds rather than demands:
# in_game/common/building_types/capital_buildings.txt local_governor :607, naval_governor :760.
# Expects the run's slot counters on the owner.
cm_ln_gov_site_eligible = {
	OR = {
		location_rank = location_rank:city # :620
		location_rank = location_rank:megalopolis # :621
		AND = {
			OR = {
				location_rank = location_rank:rural_settlement
				location_rank = location_rank:town
			}
			# The Found City step, which the whole chain has to reach for a governor to stand here:
			# cm_upgrade_town_to_city_building location_potential population >= 30 and
			# allow cannot_upgrade_location,
			# in_game/common/building_types/cm_location_upgrade_buildings.txt:26 and :23.
			population >= 30
			modifier:cannot_upgrade_location = no
		}
	}
	NOT = { has_variable = cm_ln_plan_ord }
	NOT = { cm_location_deurbanize_would_undo = yes }
	OR = {
		AND = {
			owner = { var:cm_ln_land_left > 0 }
			owner.capital != this # location_potential: :642
			NOT = { has_building = building_type:seljuk_mint } # allow: :634
			is_connected_to = owner.capital
		}
		AND = {
			owner = { var:cm_ln_naval_left > 0 }
			cm_ln_site_is_naval = yes
		}
	}
}

# Root is the location.
cm_ln_target_needs_road = {
	NOT = { has_variable = cm_ln_plan_naval }
	has_road_to_capital = no
}

# Root is the location. Both governor types need city rank, so a planned site below it is
# urbanized first: in_game/common/building_types/capital_buildings.txt:620 and :773.
cm_ln_target_needs_rank = {
	OR = {
		location_rank = location_rank:rural_settlement
		location_rank = location_rank:town
	}
}

# Root is the location. Expects scope:cm_country. A governor ordered here that the engine has not
# finished, which the confirm step waits on rather than ordering a second one.
cm_ln_site_gov_under_construction = {
	any_buildings_in_location = {
		owner = scope:cm_country
		building_type = { cm_ln_is_governor_type = yes }
		building_levels_under_construction > 0
	}
}

# Root is the country. A governor the country has bought and does not have. cm_ln_gov_paid_seats only
# ever rises, so a seated count below it is a seat already paid for however it came to be missing,
# and the next placement is not charged again.
cm_ln_gov_seat_is_prepaid = {
	exists = var:cm_ln_gov_paid_seats
	var:cm_ln_gov_paid_seats > cm_ln_gov_seated_value
}

# Root is the country.
cm_ln_gov_slot_open = {
	OR = {
		cm_fa_gov_has_land_room = yes
		cm_fa_gov_has_naval_room = yes
	}
}

# Root is the country. The harvest is a real-time GUI pass while the opening move's wait counts
# game months, so a fast game can burn the whole wait while it is still running. Its cursor moving
# since the last pulse is what separates that from a harvest that will never finish.
cm_ln_open_river_advancing = {
	has_global_variable = cm_pf_river_process
	OR = {
		NOT = { exists = var:cm_ln_open_idx }
		NOT = { var:cm_ln_open_idx = global_var:cm_pf_river_idx }
	}
}

# Root is the country. cm_pf_n set with cm_pf_all_done absent is a run that was primed and
# never finished; the finder's water state is shared, so a reorganization waits it out.
cm_ln_no_human_search_in_flight = {
	NOT = {
		any_country = {
			is_ai = no
			has_variable = cm_pf_n
			NOT = { has_variable = cm_pf_all_done }
		}
	}
}

# Root is the country. An advance or a rank-up granting a slot reopens the plan without
# waiting out the cadence.
cm_ln_gov_slots_grew = {
	exists = var:cm_ln_slots_seen
	cm_ln_gov_slot_total > var:cm_ln_slots_seen
}
