# Income gate for the GUI construction executor. Expects scopes: country, building_type, income.
# Passes when the building is profit-exempt (non-production, no construction demand, or no
# threshold) or its income-to-owner meets the configured minimum.
cm_gui_income_gate_passes = {
	OR = {
		NOT = { scope:building_type = { cm_is_building_type_production_building = yes } }
		cm_get_construction_demand_total_count = 0
		NOT = { scope:country = { has_variable_map = cm_priority_min_profit } }
		"scope:country.variable_map(cm_priority_min_profit|flag:cm_feature_auto_expand_buildings)" = 0
		scope:income >= "scope:country.variable_map(cm_priority_min_profit|flag:cm_feature_auto_expand_buildings)"
	}
}

# Expects scope:building_type. Checks if the building type produces a good that is also
# one of its construction demand goods (e.g. horse_breeders produces and requires horses).
# Precomputed by the hidden window GUI at game start.
cm_building_type_produces_own_construction_good = {
	AND = {
		has_global_variable_map = cm_building_type_produces_own_construction_good
		is_key_in_global_variable_map = { name = cm_building_type_produces_own_construction_good target = scope:building_type }
		"global_variable_map(cm_building_type_produces_own_construction_good|scope:building_type)" = 1
	}
}

# Checks construction cost adjustment against the min discount threshold
# from the priority list (field slot 2, mapped to cm_priority_min_discount, pre-scaled ×0.01).
# Expects scopes: country, building_type.
# $threshold$ = the full variable_map lookup expression for the discount threshold.
# The 0.005 added to $threshold$ is a 0.5% tolerance so that a setting of -33%
# matches a cost adjustment of -32.5% (avoids float-rounding misses at the boundary).
cm_building_meets_minimum_discount = {
	OR = {
		NOT = { scope:country = { has_variable_map = cm_priority_min_discount } }
		cm_get_construction_demand_total_count = 0
		cm_building_type_produces_own_construction_good = yes
		cm_get_standard_construction_demand_total = 0
		scope:building_type = {
			cm_get_standard_construction_cost_adjustment_for_building_type <= { value = $threshold$ add = 0.005 }
		}
	}
}

# Root is the building. Shared core of the two cm_building_can_auto_expand_* triggers —
# the checks that apply to both normal-path and upgrade-path expansion.
# The upgrade path cannot use can_build_building (fails for obsolete building types);
# the normal path adds it via cm_building_can_auto_expand_other_than_checking_for_upgrades.
cm_building_can_auto_expand_for_upgrade = {
	location = { save_temporary_scope_as = location }
	building_type = { save_temporary_scope_as = building_type }
	owner = { save_temporary_scope_as = country }
	AND = {
		is_opened = yes
		cm_is_building_fully_employed = yes
		is_lacking_goods = no
		cm_building_meets_minimum_discount = { threshold = cm_get_min_discount_for_auto_expand_buildings }
		location = { cm_location_has_enough_pops_available = yes }
		location = { cm_is_location_under_building_limit = yes }
	}
}

# Root is the building. Same checks as cm_building_can_auto_expand_for_upgrade, plus can_build_building.
cm_building_can_auto_expand_other_than_checking_for_upgrades = {
	AND = {
		cm_building_can_auto_expand_for_upgrade = yes
		location = { cm_location_allowed_to_build = { target = scope:building_type } }
	}
}

# Expects scopes: building type, country, and location
cm_location_can_auto_expand_from_zero_levels = {
	scope:location = { cm_location_will_auto_expand_from_zero_levels = yes }
}

# Root is the location, also expects scope:building_type and scope:country.
# Shared core of the two cm_location_will_auto_expand_from_zero_levels_* triggers.
# The upgrade path cannot use can_build_building (fails for obsolete building types);
# the normal path adds it via cm_location_will_auto_expand_from_zero_levels.
cm_location_will_auto_expand_from_zero_levels_for_upgrade = {
	AND = {
		scope:country = { cm_should_pause_auto_expand = no }
		NOT = { integration_level = conquered }
		cm_location_has_enough_pops_available = yes
		cm_is_location_under_building_limit = yes
		save_temporary_scope_as = location
		cm_building_meets_minimum_discount = { threshold = cm_get_min_discount_for_auto_expand_buildings }
	}
}

# Root is the location, also expects scope:building_type and scope:country.
# Same checks as cm_location_will_auto_expand_from_zero_levels_for_upgrade, plus can_build_building.
cm_location_will_auto_expand_from_zero_levels = {
	AND = {
		cm_location_will_auto_expand_from_zero_levels_for_upgrade = yes
		cm_location_allowed_to_build = { target = scope:building_type }
		cm_location_building_type_not_under_construction = yes
	}
}

# Expects scopes: building type, country, and location. Same as cm_location_can_auto_expand_from_zero_levels

# Expects scope:country. Checks that the player has gold above their minimum reserve setting.
# Used for upgrade branches where the exact upgrade cost is resolved by the GUI.
cm_auto_expand_check_minimum_gold_reserve = {
	OR = {
		NOT = { scope:country = { exists = var:cm_minimum_gold } }
		scope:country.gold > scope:country.var:cm_minimum_gold
	}
}
