# Root is the location.
# True when the location's population is at or within 1k of its location cap.
cm_location_pop_at_or_near_cap = {
	location_max_population <= {
		value = population
		add = 1
	}
}

# Root is the location, expects scope:cm_country
cm_location_can_mass_upgrade_to_city = {
	AND = {
		location_rank = location_rank:town
		population >= 30
		NOT = { cm_location_rgo_blocks_urbanize = yes }
		modifier:cannot_upgrade_location = no
		NOT = { has_building = building_type:cm_upgrade_town_to_city_building }
		NOT = { any_buildings_in_location = { building_type = building_type:cm_upgrade_town_to_city_building building_levels_under_construction > 0 } }
		NOT = { integration_level = conquered }
		cm_mass_upgrade_town_to_city_discount_check = yes
		cm_location_allowed_to_build = { target = building_type:cm_upgrade_town_to_city_building }
	}
}

# Root is the location, expects scope:cm_country
cm_location_can_mass_upgrade_to_town = {
	AND = {
		location_rank = location_rank:rural_settlement
		OR = {
			population >= 30
			cm_location_pop_at_or_near_cap = yes
		}
		NOT = { cm_location_rgo_blocks_urbanize = yes }
		# A Smart Automations food pick blocks only the automatic path; manual upgrades stay open.
		NOT = { cm_sa_location_is_system_food_picked = yes }
		modifier:cannot_upgrade_location = no
		NOT = { has_building = building_type:cm_upgrade_rural_to_town_building }
		NOT = { any_buildings_in_location = { building_type = building_type:cm_upgrade_rural_to_town_building building_levels_under_construction > 0 } }
		NOT = { integration_level = conquered }
		cm_mass_upgrade_rural_to_town_discount_check = yes
		cm_location_allowed_to_build = { target = building_type:cm_upgrade_rural_to_town_building }
	}
}

# Root is the location. Reads the owner rather than scope:cm_country because the finder's
# governor filter calls it from a sweep that sets no scope.
# True when the location's RGO good is checked in the owner's do-not-urbanize list.
cm_location_rgo_blocks_urbanize = {
	raw_material = { save_temporary_scope_as = cm_rgo_good }
	owner = { is_target_in_variable_list = { name = cm_auto_urbanize_ignored_rgo_goods target = scope:cm_rgo_good } }
}

# Root is the location.
# True when the location holds a rank the do-not-urbanize list says it should not have.
cm_location_deurbanize_target = {
	NOT = { location_rank = location_rank:rural_settlement }
	NOT = { integration_level = conquered }
	cm_location_rgo_blocks_urbanize = yes
}

# Root is the location.
cm_location_can_deurbanize = {
	cm_location_deurbanize_target = yes
	# Every standing source, the capital's own 100 included:
	# main_menu/common/static_modifiers/location.txt:330.
	NOT = { modifier:local_proximity_source > 0 }
	# The modifier lands only once a source stands, and has_building answers from the moment a
	# level is queued, so a governor merely ordered counts.
	NOT = { cm_location_has_governor = yes }
	# A lucky nation's reorganization urbanizes its own planned governor sites.
	NOT = { has_variable = cm_gov_plan_ord }
}

# Root is the location. Ground the de-urbanizer will take the rank off, asked of a site a capital
# or a governor is about to be placed on. Carries no rank test: a governor site below city rank is
# one the plan urbanizes first, which is exactly the case worth refusing.
cm_location_deurbanize_would_undo = {
	owner = { exists = var:cm_deurbanize }
	cm_location_rgo_blocks_urbanize = yes
}

# Root is the location. A rank the de-urbanizer wants gone that a capital or a governor is sitting
# on, which reopens the reorganization plan ahead of its cadence. cm_gov_deurb_tried is set on every
# match when a run begins, so a seat the plan cannot move reopens it once rather than every month
# for the rest of the game.
cm_location_deurbanize_blocked_by_seat = {
	cm_location_deurbanize_target = yes
	owner = { exists = var:cm_deurbanize }
	NOT = { has_variable = cm_gov_deurb_tried }
	OR = {
		is_capital = yes
		cm_location_has_governor = yes
	}
}

# Root is the location, expects scope:cm_country.
# Checks that the market price impact of city upgrade construction demand is within the player's threshold.
cm_mass_upgrade_town_to_city_discount_check = {
	save_temporary_scope_as = cm_location
	# A temporary cm_building_type clashes with the permanent one the executor provides.
	trigger_if = {
		limit = { NOT = { exists = scope:cm_building_type } }
		building_type:cm_upgrade_town_to_city_building = { save_temporary_scope_as = cm_building_type }
	}
	cm_building_meets_minimum_discount = { threshold = cm_get_min_discount_for_auto_urbanize }
}
