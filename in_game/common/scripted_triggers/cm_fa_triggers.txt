# Affordability gate for full-automation approvals: the category pool covers the cost and
# the treasury stays above the full-automation reserve after this drain's committed spend.
# $POOL$ = the category pool variable name. Expects scope:cm_country, scope:cm_cost.
cm_fa_can_afford = {
	scope:cm_country = {
		var:$POOL$ >= scope:cm_cost
		gold >= { value = scope:cm_cost add = cm_fa_get_reserve add = var:cm_q_planned_spend }
	}
}

# RGO twin of cm_fa_can_afford: the cost is the live glorpui_rgo_expand_cost.
# $POOL$ = the category pool variable name. Root is the location, expects scope:cm_country,
# scope:cm_location.
cm_fa_can_afford_rgo = {
	scope:cm_country.var:$POOL$ >= glorpui_rgo_expand_cost
	cm_check_country_gold_above = { threshold = cm_fa_rgo_gold_threshold }
}

# Industry candidacy: measurable payback inside the max-payback-years cap (0 = no cap).
# The adjusted income adds the amenity gain when the amenity scopes are present, matching
# the metric-3 branch of cm_gui_metric_gate_passes. Root is the item's location.
# Expects scope:cm_country, scope:cm_income; $COST$ = the cost value.
cm_fa_ind_roi_passes = {
	cm_bld_adjusted_income > 0
	OR = {
		NOT = { scope:cm_country = { exists = var:cm_fa_industry_min_roi } }
		scope:cm_country.var:cm_fa_industry_min_roi = 0
		$COST$ <= {
			value = cm_bld_adjusted_income
			multiply = scope:cm_country.var:cm_fa_industry_min_roi
			multiply = 12
		}
	}
}

# The auto_build_list min_rgos registration default (cm_cmm_effects.txt); Utilities runs
# on defaults, never the player's tuned per-building values.
@cm_fa_min_rgos = 5

# The sawmill, windmill, and local_smelters RGO-level requirements at the default
# threshold, mirroring cm_auto_build_check_rgo_level_requirement_met's per-building
# variants. Other types always pass. Root is the location, expects
# scope:cm_auto_build_settings_type as the raw roster key.
cm_fa_utilities_rgo_req_met = {
	switch = {
		trigger = scope:cm_auto_build_settings_type
		building_type:sawmill = {
			OR = {
				AND = {
					raw_material = goods:lumber
					cm_modifier_rgo_level >= @cm_fa_min_rgos
				}
				AND = {
					has_building = building_type:lumber_mill
					"building(building_type:lumber_mill|owner)" = { building_level >= @cm_fa_min_rgos }
				}
			}
		}
		building_type:windmill = {
			AND = {
				OR = {
					raw_material = goods:wheat
					raw_material = goods:millet
					raw_material = goods:rice
				}
				cm_modifier_rgo_level >= @cm_fa_min_rgos
			}
		}
		building_type:local_smelters = {
			AND = {
				OR = {
					raw_material = goods:iron
					raw_material = goods:copper
					raw_material = goods:tin
					raw_material = goods:lead
				}
				cm_modifier_rgo_level >= @cm_fa_min_rgos
			}
		}
		fallback = { always = yes }
	}
}

# A bailiff can go up here right now. The survey and the monthly staging share it so a pick
# is only ever made on the terms it will later be staged on. Root is the location, expects
# scope:cm_country, scope:cm_building_type.
cm_fa_bailiff_site_ok = {
	NOT = { integration_level = conquered }
	NOT = { has_building = scope:cm_building_type }
	cm_location_allowed_to_build = { target = scope:cm_building_type }
	cm_location_has_enough_pops_available = yes
	cm_check_location_meets_min_market_access = yes
	cm_get_open_build_queue_slots > 0
	NOT = { is_target_in_variable_list = { name = cm_q_ungated_building_types target = scope:cm_building_type } }
}

# True when scope:cm_fa_good_t already trades at or below the target price in
# scope:cm_fa_market: market price <= default price * (100 + target markup) / 100,
# the min-discount markup convention (negative = below base, positive = above).
# Expects scope:cm_fa_market, scope:cm_fa_good_t, scope:cm_country.
cm_fa_collapsar_target_met = {
	"scope:cm_fa_market.market_price(scope:cm_fa_good_t)" <= {
		value = "default_price(scope:cm_fa_good_t)"
		multiply = {
			value = 100
			add = scope:cm_country.var:cm_fa_collapsar_target
			divide = 100
		}
	}
}

# The good still trades above its target in scope:cm_fa_market. Root is the country.
# $good$ = the good key.
cm_fa_collapsar_good_wanted = {
	goods:$good$ = { save_temporary_scope_as = cm_fa_good_t }
	NOT = { cm_fa_collapsar_target_met = yes }
}

# Root is the location. Nothing collapsar-relevant is still hiring or building here: the
# RGO is neither hiring nor expanding, and no producer building has levels under
# construction or more than one level's worth of vacancies (the staffing tolerance from
# cm_food_building_type_is_staffed).
cm_fa_collapsar_location_absorbed = {
	NOT = {
		AND = {
			OR = {
				raw_material = goods:lumber
				raw_material = goods:sand
			}
			rgo_workers < rgo_level
		}
	}
	cm_is_rgo_under_construction = no
	NOT = {
		any_buildings_in_location = {
			OR = {
				building_type = building_type:lumber_mill
				building_type = building_type:sand_pit
				building_type = building_type:mason
				building_type = building_type:glass_guild
				building_type = building_type:glass_workshop
				building_type = building_type:glassworks
				building_type = building_type:glass_mill
				building_type = building_type:rural_glassmaker
			}
			OR = {
				building_levels_under_construction > 0
				AND = {
					is_opened = yes
					building_employed_amount <= { value = building_employment_size_amount subtract = building_type.employment_size }
				}
			}
		}
	}
}

# Root is the location. The location can still take another collapsar construction: RGO
# headroom on the collapsar RGO goods, or free building levels.

	}

# Root is the location. The full-automation food system picked this location; membership
# in the owner's list is the authority, so a stale marker after an ownership change is
# inert and never clobbers the new owner's own pick.
cm_fa_location_is_system_food_picked = {
	exists = owner
	exists = var:cm_fa_food_marked
	owner = { is_target_in_variable_list = { name = cm_fa_food_locations target = prev } }
}

# Root is the location. Picked and currently operating.
cm_fa_location_is_system_food_active = {
	cm_fa_location_is_system_food_picked = yes
	location_rank = location_rank:rural_settlement
	owner = {
		exists = var:cm_fa_enabled
		exists = var:cm_fa_food_enabled
	}
}

# Root is the location. Auto-food that actually operates here: a system pick builds under the
# food pool on its own, while a player toggle or mass coverage only counts while the player's
# auto-food feature is enabled - a manual toggle with that feature off builds nothing, so it
# neither counts toward a market's coverage nor blocks the location from being picked.
# Mirrors the gating in cm_location_auto_food_supersedes_rgo. The glow and tooltip keep using
# cm_location_is_auto_food_active, so a manually toggled location still displays as enabled.
cm_fa_food_location_effectively_active = {
	OR = {
		AND = {
			owner = { is_target_in_variable_list = { name = cm_priority_features_enabled target = flag:cm_feature_auto_food } }
			cm_location_is_auto_food_active = yes
		}
		cm_fa_location_is_system_food_active = yes
	}
}

# Root is the location, expects scope:cm_country. The lenient expandability test: a

# Root is the location, expects scope:cm_country. Below cm_fa_ur_granary_open_gold the granary
# right is granted only where the granary stands at its level cap, the right being a percentage
# on local_food_capacity (vanilla granary_town: in_game/common/town_rights/00_traditions.txt:29).
cm_fa_ur_granary_gate_passes = {
	OR = {
		scope:cm_country = { gold >= cm_fa_ur_granary_open_gold }

			has_building = building_type:granary
			"building(building_type:granary|owner)" = { is_at_max_level = yes }
		}
	}
}

# Destroyed on sight, whatever it earns: the goods it draws are worth more elsewhere.

# (in_game/common/building_types/common_buildings.txt:591). Root is the building type.
cm_fa_cu_is_always_delete = {

		this = building_type:fruit_orchard
	}

# Left standing whatever it earns: it is built for what it does to the location, not for the
# goods it happens to make. farming_village grants local_food_capacity 100 and

# Root is the building type.
cm_fa_cu_is_never_removed = {
	OR = {
		this = building_type:farming_village
	}
}

# Root is the building, expects scope:cm_country. Takes pops an auto-food location needs
# without putting food back. building_produced_goods reads the method the building actually
# runs, so a fishing village switched off fishing counts here and a modded food producer does not.
cm_fa_cu_af_is_removable = {
	building_type = { cm_is_building_type_production_building = yes }
	save_temporary_scope_as = cm_fa_cu_af_b
	NOT = {
		scope:cm_country = {
			any_in_list = {
				variable = cm_auto_food_rgo_types_list
				scope:cm_fa_cu_af_b = { building_produced_goods = prev }
			}
		}
	}
}

# This building type's close budget in the market is not yet spent.
# Root is the country, expects scope:cm_building_type.
cm_fa_cu_group_has_room = {
	trigger_if = {
		limit = {
			has_variable_map = cm_fa_cu_acted
			is_key_in_variable_map = { name = cm_fa_cu_acted target = scope:cm_building_type }
		}
		"variable_map(cm_fa_cu_acted|scope:cm_building_type)" < cm_fa_cu_close_cap
	}
	trigger_else = { cm_fa_cu_close_cap >= 1 }
}

# A group reopens only when this pass closed nothing in it and one of its open buildings
# still earns. Root is the country, expects scope:cm_building_type.
cm_fa_cu_group_reopen_room = {

		AND = {
			has_variable_map = cm_fa_cu_acted
			is_key_in_variable_map = { name = cm_fa_cu_acted target = scope:cm_building_type }
		}
	}
	has_variable_map = cm_fa_cu_prof_open
	is_key_in_variable_map = { name = cm_fa_cu_prof_open target = scope:cm_building_type }

		limit = {
			has_variable_map = cm_fa_cu_reop
			is_key_in_variable_map = { name = cm_fa_cu_reop target = scope:cm_building_type }

		"variable_map(cm_fa_cu_reop|scope:cm_building_type)" < cm_fa_cu_reopen_cap
	}
	trigger_else = { cm_fa_cu_reopen_cap >= 1 }

# Governors gates.

# Copied from the vanilla local_governor allow block, where lieutenancies share the pool
# (game in_game/common/building_types/capital_buildings.txt:627-630; lieutenancy repeats it at
# in_game/common/building_types/unique_buildings.txt:1267-1270). Root is the country.
cm_fa_gov_has_land_room = {
	total_building_levels_including_construction:local_governor < {
		value = modifier:num_local_governors
		subtract = total_building_levels_including_construction:lieutenancy
	}
}

# Copied from the vanilla naval_governor allow block
# (game in_game/common/building_types/capital_buildings.txt:792). num_naval_governors has no base
# value, so a country holds none until an advance or a naval reform grants one.
# Root is the country.
cm_fa_gov_has_naval_room = {
	total_building_levels_including_construction:naval_governor < modifier:num_naval_governors
}

# A governor of either kind can go up here. The two are mutually exclusive by location: land
# needs a road to the capital, naval needs the capital to be unreachable by land.
# Root is the location, expects scope:cm_country.
cm_fa_gov_location_placeable = {
	OR = {
		AND = {
			scope:cm_country = { cm_fa_gov_has_land_room = yes }
			cm_location_allowed_to_build = { target = building_type:local_governor }
		}
		AND = {
			scope:cm_country = { cm_fa_gov_has_naval_room = yes }
			cm_location_allowed_to_build = { target = building_type:naval_governor }
		}
	}
}
