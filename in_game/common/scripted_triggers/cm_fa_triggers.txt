# True while the category may buy past its allotment and owe the difference: its ledger is

# Affordability gate for full-automation approvals: anything left in the category pool covers a
# borrowing category's pick, the whole cost otherwise, and the treasury stays above the
# full-automation reserve after this drain's committed spend.
# $POOL$ = the category pool variable name. Expects scope:cm_country, scope:cm_cost.
cm_fa_can_afford = {
	scope:cm_country = {
		trigger_if = {
			limit = { cm_fa_may_borrow = yes }
			var:$POOL$ > 0
		}
		trigger_else = { var:$POOL$ >= scope:cm_cost }
	}
	cm_fa_treasury_covers_cost = yes
}

# The treasury half of cm_fa_can_afford. Expects scope:cm_country, scope:cm_cost.
cm_fa_treasury_covers_cost = {
	scope:cm_country = { gold >= { value = scope:cm_cost add = cm_fa_get_reserve add = var:cm_q_planned_spend } }
}

# Per-level affordability for a full-automation batch: the pool balance the batch has left,
# which cm_fa_fill_build_slots holds in var:cm_fa_copy_pool so one trigger serves every category.
# Expects scope:cm_country, scope:cm_cost.
cm_fa_copy_can_afford = {
	scope:cm_country = {

			var:cm_fa_copy_pool > 0
		}
		trigger_else = { var:cm_fa_copy_pool >= scope:cm_cost }
	}
	cm_fa_treasury_covers_cost = yes
}

# RGO twin of cm_fa_can_afford: the cost is the live glorpui_rgo_expand_cost.
# $POOL$ = the category pool variable name. Root is the location, expects scope:cm_country,
# scope:cm_location.
cm_fa_can_afford_rgo = {
	trigger_if = {
		limit = { scope:cm_country = { cm_fa_may_borrow = yes } }
		scope:cm_country.var:$POOL$ > 0
	}
	trigger_else = { scope:cm_country.var:$POOL$ >= glorpui_rgo_expand_cost }
	cm_check_country_gold_above = { threshold = cm_fa_rgo_gold_threshold }
}

# Industry candidacy: measurable payback inside the max-payback-years cap (0 = no cap).
# The adjusted income adds the amenity gain when the amenity scopes are present, matching
# the metric-3 branch of cm_gui_metric_gate_passes. Root is the item's location.
# Expects scope:cm_country, scope:cm_income; $COST$ = the cost value.
cm_fa_ind_roi_passes = {
	cm_bld_adjusted_income > 0
	OR = {
		NOT = { scope:cm_country = { exists = var:cm_fa_industry_min_roi } }
		scope:cm_country.var:cm_fa_industry_min_roi = 0
		$COST$ <= {
			value = cm_bld_adjusted_income
			multiply = scope:cm_country.var:cm_fa_industry_min_roi
			multiply = 12
		}
	}
}

# The auto_build_list min_rgos registration default (cm_cmm_effects.txt); Utilities runs
# on defaults, never the player's tuned per-building values.
@cm_fa_min_rgos = 5

# The sawmill, windmill, and local_smelters RGO-level requirements at the default
# threshold, mirroring cm_auto_build_check_rgo_level_requirement_met's per-building
# variants. Other types always pass. Root is the location, expects
# scope:cm_auto_build_settings_type as the raw roster key.
cm_fa_utilities_rgo_req_met = {
	switch = {
		trigger = scope:cm_auto_build_settings_type
		building_type:sawmill = {
			OR = {
				AND = {
					raw_material = goods:lumber
					cm_modifier_rgo_level >= @cm_fa_min_rgos
				}
				AND = {
					has_building = building_type:lumber_mill
					"building(building_type:lumber_mill|owner)" = { building_level >= @cm_fa_min_rgos }
				}
			}
		}
		building_type:windmill = {
			AND = {
				OR = {
					raw_material = goods:wheat
					raw_material = goods:millet
					raw_material = goods:rice
				}
				cm_modifier_rgo_level >= @cm_fa_min_rgos
			}
		}
		building_type:local_smelters = {
			AND = {
				OR = {
					raw_material = goods:iron
					raw_material = goods:copper
					raw_material = goods:tin
					raw_material = goods:lead
				}
				cm_modifier_rgo_level >= @cm_fa_min_rgos
			}
		}
		fallback = { always = yes }
	}
}

# True when scope:cm_fa_good_t has reached its goal price in scope:cm_fa_market, either at the
# price it trades at now or at the one it is converging to. Market price closes only 5% of that
# gap a month (MONTHLY_PRICE_CHANGE, game/loading_screen/common/defines/00_defines.txt:1755), so
# without the second branch a market keeps being built into for over a year after its supply is
# already enough. The > 0 guard keeps a target_price of 0 from reading as every goal met.
# Expects scope:cm_fa_market, scope:cm_fa_good_t, scope:cm_country.
cm_fa_collapsar_target_met = {
	OR = {
		"scope:cm_fa_market.market_price(scope:cm_fa_good_t)" <= cm_fa_collapsar_goal_price
		AND = {
			"scope:cm_fa_market.target_price(scope:cm_fa_good_t)" > 0
			"scope:cm_fa_market.target_price(scope:cm_fa_good_t)" <= cm_fa_collapsar_goal_price
		}
	}
}

# Copied from the vanilla local_governor allow block, where lieutenancies share the pool
# (game in_game/common/building_types/capital_buildings.txt:627-630; lieutenancy repeats it at
# in_game/common/building_types/unique_buildings.txt:1267-1270). Root is the country.
cm_fa_gov_has_land_room = {
	total_building_levels_including_construction:local_governor < {
		value = modifier:num_local_governors
		subtract = total_building_levels_including_construction:lieutenancy
	}
}

# Copied from the vanilla naval_governor allow block
# (game in_game/common/building_types/capital_buildings.txt:792). num_naval_governors has no base
# value, so a country holds none until an advance or a naval reform grants one.
# Root is the country.
cm_fa_gov_has_naval_room = {
	total_building_levels_including_construction:naval_governor < modifier:num_naval_governors
}

# A governor of either kind can go up here. The two are mutually exclusive by location: land
# needs a road to the capital, naval needs the capital to be unreachable by land.
# Root is the location, expects scope:cm_country.
cm_fa_gov_location_placeable = {
	OR = {
		AND = {
			scope:cm_country = { cm_fa_gov_has_land_room = yes }
			cm_location_allowed_to_build = { target = building_type:local_governor }
		}
		AND = {
			scope:cm_country = { cm_fa_gov_has_naval_room = yes }
			cm_location_allowed_to_build = { target = building_type:naval_governor }
		}
	}
}
