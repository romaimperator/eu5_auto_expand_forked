# Full-automation gates.

# Root is the country.
cm_fa_master_active = {
	exists = var:cm_fa_enabled
	cm_should_pause_auto_expand = no
}

# Affordability gate for full-automation approvals: the category pool covers the cost and
# the treasury stays above the full-automation reserve after this drain's committed spend.
# $POOL$ = the category pool variable name. Expects scope:cm_country, scope:cm_cost.
cm_fa_can_afford = {
	scope:cm_country = {
		var:$POOL$ >= scope:cm_cost
		gold >= { value = scope:cm_cost add = cm_fa_get_reserve add = var:cm_q_planned_spend }
	}
}

# RGO twin of cm_fa_can_afford: the cost is the live glorpui_rgo_expand_cost.
# $POOL$ = the category pool variable name. Root is the location, expects scope:cm_country,
# scope:cm_location.
cm_fa_can_afford_rgo = {
	scope:cm_country.var:$POOL$ >= glorpui_rgo_expand_cost
	cm_check_country_gold_above = { threshold = cm_fa_rgo_gold_threshold }
}

# Industry candidacy: measurable payback inside the max-payback-years cap (0 = no cap).
# The adjusted income adds the amenity gain when the amenity scopes are present, matching
# the metric-3 branch of cm_gui_metric_gate_passes. Root is the item's location.
# Expects scope:cm_country, scope:cm_income; $COST$ = the cost value.
cm_fa_ind_roi_passes = {
	cm_bld_adjusted_income > 0
	OR = {
		NOT = { scope:cm_country = { exists = var:cm_fa_industry_min_roi } }
		scope:cm_country.var:cm_fa_industry_min_roi = 0
		$COST$ <= {
			value = cm_bld_adjusted_income
			multiply = scope:cm_country.var:cm_fa_industry_min_roi
			multiply = 12
		}
	}
}

# The auto_build_list min_rgos registration default (cm_cmm_effects.txt); Utilities runs
# on defaults, never the player's tuned per-building values.
@cm_fa_min_rgos = 5

# The sawmill, windmill, and local_smelters RGO-level requirements at the default
# threshold, mirroring cm_auto_build_check_rgo_level_requirement_met's per-building
# variants. Other types always pass. Root is the location, expects
# scope:cm_auto_build_settings_type as the raw roster key.
cm_fa_utilities_rgo_req_met = {
	switch = {
		trigger = scope:cm_auto_build_settings_type
		building_type:sawmill = {
			OR = {
				AND = {
					raw_material = goods:lumber
					cm_modifier_rgo_level >= @cm_fa_min_rgos
				}
				AND = {
					has_building = building_type:lumber_mill
					"building(building_type:lumber_mill|owner)" = { building_level >= @cm_fa_min_rgos }
				}
			}
		}
		building_type:windmill = {
			AND = {
				OR = {
					raw_material = goods:wheat
					raw_material = goods:millet
					raw_material = goods:rice
				}
				cm_modifier_rgo_level >= @cm_fa_min_rgos
			}
		}
		building_type:local_smelters = {
			AND = {
				OR = {
					raw_material = goods:iron
					raw_material = goods:copper
					raw_material = goods:tin
					raw_material = goods:lead
				}
				cm_modifier_rgo_level >= @cm_fa_min_rgos
			}
		}
		fallback = { always = yes }
	}
}

# True when scope:cm_fa_good_t already trades at or below the target price in
# scope:cm_fa_market: market price <= default price * (100 + target markup) / 100,
# the min-discount markup convention (negative = below base, positive = above).
# Expects scope:cm_fa_market, scope:cm_fa_good_t, scope:cm_country.
cm_fa_collapsar_target_met = {
	"scope:cm_fa_market.market_price(scope:cm_fa_good_t)" <= {
		value = "default_price(scope:cm_fa_good_t)"
		multiply = {
			value = 100
			add = scope:cm_country.var:cm_fa_collapsar_target
			divide = 100
		}
	}
}

# The good is enabled for the collapsar and still trades above its target in
# scope:cm_fa_market. Root is the country. $good$ = the good key.
cm_fa_collapsar_good_wanted = {
	is_target_in_variable_list = { name = cm_fa_collapsar_goods_enabled target = goods:$good$ }
	goods:$good$ = { save_temporary_scope_as = cm_fa_good_t }
	NOT = { cm_fa_collapsar_target_met = yes }
}

# Root is the location. Nothing collapsar-relevant is still hiring or building here: the
# RGO is neither hiring nor expanding, and no producer building has levels under
# construction or more than one level's worth of vacancies (the staffing tolerance from
# cm_food_building_type_is_staffed).
cm_fa_collapsar_location_absorbed = {
	NOT = {
		AND = {
			OR = {
				raw_material = goods:lumber
				raw_material = goods:sand
			}
			rgo_workers < rgo_level
		}
	}
	cm_is_rgo_under_construction = no
	NOT = {
		any_buildings_in_location = {
			OR = {
				building_type = building_type:lumber_mill
				building_type = building_type:sand_pit
				building_type = building_type:mason
				building_type = building_type:glass_guild
				building_type = building_type:glass_workshop
				building_type = building_type:glassworks
				building_type = building_type:glass_mill
				building_type = building_type:rural_glassmaker
			}
			OR = {
				building_levels_under_construction > 0
				AND = {
					is_opened = yes
					building_employed_amount <= { value = building_employment_size_amount subtract = building_type.employment_size }
				}
			}
		}
	}
}

# Root is the location. The location can still take another collapsar construction: RGO
# headroom on the collapsar RGO goods, or free building levels.
cm_fa_collapsar_location_has_levels = {
	OR = {
		AND = {
			OR = {
				raw_material = goods:lumber
				raw_material = goods:sand
			}
			cm_is_rgo_not_max_level_already = yes
		}
		cm_get_remaining_free_building_levels_allowed >= 1
	}
}

# Root is the location. The full-automation food system picked this location; membership
# in the owner's list is the authority, so a stale marker after an ownership change is
# inert and never clobbers the new owner's own pick.
cm_fa_location_is_system_food_picked = {
	exists = owner
	exists = var:cm_fa_food_marked
	owner = { is_target_in_variable_list = { name = cm_fa_food_locations target = prev } }
}

# Root is the location. Picked and currently operating.
cm_fa_location_is_system_food_active = {
	cm_fa_location_is_system_food_picked = yes
	location_rank = location_rank:rural_settlement
	owner = {
		exists = var:cm_fa_enabled
		exists = var:cm_fa_food_enabled
	}
}

# Root is the location, expects scope:cm_country. The lenient expandability test: a
# location waiting on pop promotion (the *_or_will_be checks inside the buildable chain)
# or on a level under construction still counts as able to expand; only a genuine pop
# shortage or no available levels disqualify.
cm_fa_food_location_counts_as_expandable = {
	OR = {
		cm_food_location_has_buildable_option = yes
		cm_is_rgo_under_construction = yes
		any_buildings_in_location = {
			building_levels_under_construction > 0
			OR = {
				building_type = building_type:farming_village
				building_type = building_type:windmill
				building_type = building_type:irrigation_systems
				building_type = building_type:fishing_village
			}
		}
	}
}

# Root is the location, expects scope:cm_country and scope:cm_food_good. The player's own
# auto-food flow processes this location this pulse: feature enabled plus the eligibility
# OR from cm_food_auto_build_for_rgo_type. Such a location builds through the player's
# queue with player funds.
cm_fa_food_player_flow_covers = {
	scope:cm_country = { is_target_in_variable_list = { name = cm_priority_features_enabled target = flag:cm_feature_auto_food } }
	OR = {
		exists = var:cm_auto_food_location_enabled
		AND = {
			scope:cm_country = { is_target_in_variable_list = { name = cm_auto_food_rgo_types_enabled target = scope:cm_food_good } }
			cm_food_location_meets_river_filter = yes
			NOT = { exists = var:cm_auto_food_location_excluded }
		}
	}
}
