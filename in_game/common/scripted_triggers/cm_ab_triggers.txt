# Root is the location, expects scope:cm_country. The road's own construction goods measured as a

# True while the category may buy past its allotment and owe the difference: its ledger is
# running, which only cm_ab_drain_one starts, and the setting is on. Root is the country.
cm_ab_may_borrow = {
	local_var:cm_ab_ledger = 1
	exists = var:cm_ab_save_unaffordable
}

# Affordability gate for Auto-Build approvals: anything left in the category pool covers a
# borrowing category's pick, the whole cost otherwise, and the treasury stays above the
# full reserve after this drain's committed spend.
# $POOL$ = the category pool variable name. Expects scope:cm_country, scope:cm_cost.
cm_ab_can_afford = {
	scope:cm_country = {
		trigger_if = {
			limit = { cm_ab_may_borrow = yes }
			var:$POOL$ > 0
		}
		trigger_else = { var:$POOL$ >= scope:cm_cost }
	}
	cm_ab_treasury_covers_cost = yes
}

# The treasury half of cm_ab_can_afford. Expects scope:cm_country, scope:cm_cost.
cm_ab_treasury_covers_cost = {
	scope:cm_country = { gold >= { value = scope:cm_cost add = cm_get_full_reserve add = var:cm_q_planned_spend } }
}

# Per-level affordability for an Auto-Build batch: the pool balance the batch has left,
# which cm_ab_fill_build_slots holds in var:cm_ab_copy_pool so one trigger serves every category.
# Expects scope:cm_country, scope:cm_cost.
cm_ab_copy_can_afford = {
	scope:cm_country = {
		trigger_if = {
			limit = { cm_ab_may_borrow = yes }
			var:cm_ab_copy_pool > 0
		}
		trigger_else = { var:cm_ab_copy_pool >= scope:cm_cost }
	}
	cm_ab_treasury_covers_cost = yes
}

# RGO twin of cm_ab_can_afford: the cost is the live glorpui_rgo_expand_cost.
# $POOL$ = the category pool variable name. Root is the location, expects scope:cm_country,
# scope:cm_location.
cm_ab_can_afford_rgo = {
	trigger_if = {
		limit = { scope:cm_country = { cm_ab_may_borrow = yes } }
		scope:cm_country.var:$POOL$ > 0
	}
	trigger_else = { scope:cm_country.var:$POOL$ >= glorpui_rgo_expand_cost }
	cm_check_country_gold_above = { threshold = cm_ab_rgo_gold_threshold }
}

# Production candidacy: measurable payback inside the max-payback-years cap (0 = no cap).
# The adjusted income adds the indirect gain when the indirect scopes are present, matching
# the metric-3 branch of cm_gui_metric_gate_passes. Root is the item's location.
# Expects scope:cm_country, scope:cm_income; $COST$ = the cost value.
cm_ab_prod_roi_passes = {
	cm_bld_adjusted_income > 0
	OR = {
		NOT = { scope:cm_country = { exists = var:cm_ab_production_min_roi } }
		scope:cm_country.var:cm_ab_production_min_roi = 0
		$COST$ <= {
			value = cm_bld_adjusted_income
			multiply = scope:cm_country.var:cm_ab_production_min_roi
			multiply = 12
		}
	}
}

# True when scope:cm_ab_good_t has reached its goal price in scope:cm_ab_market, either at the
# price it trades at now or at the one it is converging to. Market price closes only 5% of that
# gap a month (MONTHLY_PRICE_CHANGE, game/loading_screen/common/defines/00_defines.txt:1755), so
# without the second branch a market keeps being built into for over a year after its supply is
# already enough. The > 0 guard keeps a target_price of 0 from reading as every goal met.
# Expects scope:cm_ab_market, scope:cm_ab_good_t, scope:cm_country.
cm_ab_core_target_met = {
	OR = {
		"scope:cm_ab_market.market_price(scope:cm_ab_good_t)" <= cm_ab_core_goal_price
		AND = {
			"scope:cm_ab_market.target_price(scope:cm_ab_good_t)" > 0
			"scope:cm_ab_market.target_price(scope:cm_ab_good_t)" <= cm_ab_core_goal_price
		}
	}
}

# Root is the location. Auto-food that actually operates here: a system pick builds under the
# food pool on its own, while a player toggle or mass coverage only counts while the player's
# auto-food feature is enabled - a manual toggle with that feature off builds nothing, so it
# neither counts toward a market's coverage nor blocks the location from being picked.
# Mirrors the gating in cm_location_auto_food_supersedes_rgo. The glow and tooltip keep using
# cm_location_is_auto_food_active, so a manually toggled location still displays as enabled.
cm_ab_food_location_effectively_active = {
	OR = {
		AND = {
			owner = { is_target_in_variable_list = { name = cm_priority_features_enabled target = flag:cm_feature_auto_food } }
			cm_location_is_auto_food_active = yes
		}
		cm_ab_location_is_system_food_active = yes
	}
}

# Root is the location, expects scope:cm_country. The lenient expandability test: a

# Governors gates.

# Copied from the vanilla local_governor allow block, where lieutenancies share the pool
# (game in_game/common/building_types/capital_buildings.txt:627-630; lieutenancy repeats it at
# in_game/common/building_types/unique_buildings.txt:1267-1270). Root is the country.
cm_ab_gov_has_land_room = {
	total_building_levels_including_construction:local_governor < {
		value = modifier:num_local_governors
		subtract = total_building_levels_including_construction:lieutenancy
	}
}

# Copied from the vanilla naval_governor allow block
# (game in_game/common/building_types/capital_buildings.txt:792). num_naval_governors has no base
# value, so a country holds none until an advance or a naval reform grants one.
# Root is the country.
cm_ab_gov_has_naval_room = {
	total_building_levels_including_construction:naval_governor < modifier:num_naval_governors
}

# A governor of either kind can go up here. The two are mutually exclusive by location: land
# needs a road to the capital, naval needs the capital to be unreachable by land.
# Root is the location, expects scope:cm_country.
cm_ab_gov_location_placeable = {
	NOT = { cm_location_deurbanize_would_undo = yes }
	OR = {
		AND = {
			scope:cm_country = { cm_ab_gov_has_land_room = yes }
			cm_location_allowed_to_build = { target = building_type:local_governor }
		}
		AND = {
			scope:cm_country = { cm_ab_gov_has_naval_room = yes }
			cm_location_allowed_to_build = { target = building_type:naval_governor }
		}
	}
}
