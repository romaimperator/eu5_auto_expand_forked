# Root is the location. Core RGO expansion prerequisites shared by the GUI-visibility trigger
# (cm_is_able_to_auto_expand_rgo) and the on-action trigger (cm_should_rgo_auto_expand).
cm_rgo_basic_expand_eligibility = {
	AND = {
		cm_is_rgo_advance_researched = yes
		cm_is_rgo_not_max_level_already = yes
		cm_check_location_has_laborers_available_or_will_be = yes
	}
}

# This trigger is used by the GUI to see if the prerequisites for auto expanding RGO are met
cm_is_able_to_auto_expand_rgo = {
	AND = {
		owner = {
			cm_should_pause_auto_expand = no
			is_target_in_variable_list = { name = cm_priority_features_enabled target = flag:cm_feature_auto_expand_rgos }
		}
		cm_rgo_basic_expand_eligibility = yes
		cm_check_country_gold_above = { threshold = cm_expected_gold_available_for_rgo_auto_expand }
		cm_rgo_meets_minimum_profit = yes
		cm_rgo_meets_minimum_discount = { threshold = cm_get_min_discount_for_auto_expand_rgos }
		# cm_is_rgo_under_construction = no # don't check if it's under construction when we're deciding if the button should be green or red
	}
}

# This trigger is used by the on action to determine if the RGO should be auto expanded
# Expects scopes: country, location
cm_should_rgo_auto_expand = {
	scope:location = {
		AND = {
			cm_rgo_basic_expand_eligibility = yes
			cm_is_rgo_under_construction = no
			cm_check_country_gold_above = { threshold = cm_expected_gold_available_for_rgo_auto_expand }
			cm_rgo_meets_minimum_profit = yes
			cm_rgo_meets_minimum_discount = { threshold = cm_get_min_discount_for_auto_expand_rgos }
		}
	}
}

# Root is the location. Expects scopes: country
# Checks estimated RGO profit per level against the min profit threshold
# from the priority list (field slot 2, mapped to cm_priority_min_profit).
cm_rgo_meets_minimum_profit = {
	OR = {
		NOT = { scope:country = { has_variable_map = cm_priority_min_profit } }
		"scope:country.variable_map(cm_priority_min_profit|flag:cm_feature_auto_expand_rgos)" = 0
		cm_rgo_estimated_profit_per_level >= "scope:country.variable_map(cm_priority_min_profit|flag:cm_feature_auto_expand_rgos)"
	}
}

# Root is the location. Expects scopes: country.
# $threshold$ = the full variable_map lookup expression for the discount threshold.
# The 0.005 added to $threshold$ is a 0.5% tolerance so that a setting of -33%
# matches a cost adjustment of -32.5% (avoids float-rounding misses at the boundary).
cm_rgo_meets_minimum_discount = {
	OR = {
		NOT = { scope:country = { has_variable_map = cm_priority_min_discount } }
		cm_get_construction_cost_adjustment_for_rgo <= { value = $threshold$ add = 0.005 }
	}
}

cm_is_rgo_not_max_level_already = {
	cm_modifier_rgo_level < cm_get_max_rgo_workers_floor
}

cm_is_rgo_under_construction = {
	cm_get_rgos_under_construction > 0
}

# Root is location
# Expects scopes: country
cm_is_rgo_advance_researched = {
	trigger_if = { limit = { OR = { raw_material = goods:iron raw_material = goods:coal } }
		scope:country = { has_advance = iron_working }
	} trigger_else_if = { limit = { OR = { raw_material = goods:copper raw_material = goods:tin raw_material = goods:lead } }
		scope:country = { has_advance = bronze_working }
	} trigger_else_if = { limit = { OR = { raw_material = goods:goods_gold raw_material = goods:silver raw_material = goods:mercury } }
		scope:country = { has_advance = fine_metals_mining }
	} trigger_else_if = { limit = { OR = { raw_material = goods:wheat raw_material = goods:maize raw_material = goods:potato raw_material = goods:rice raw_material = goods:millet raw_material = goods:legumes } }
		scope:country = { has_advance = agriculture_advance }
	} trigger_else_if = { limit = { raw_material = goods:horses }
		scope:country = { has_advance = ranching }
	} trigger_else_if = { limit = { OR = { raw_material = goods:stone raw_material = goods:marble } }
		scope:country = { has_advance = mining_advance }
	} trigger_else_if = { limit = { OR = { raw_material = goods:saltpeter raw_material = goods:alum } }
		scope:country = { has_advance = advanced_mining }
	} trigger_else = {
		always = yes
	}
}
