# Root is the location. Core RGO expansion prerequisites shared by the GUI-visibility trigger
# cm_is_able_to_auto_expand_rgo and the on-action trigger cm_should_rgo_auto_expand.
cm_rgo_basic_expand_eligibility = {
	AND = {
		cm_is_rgo_advance_researched = yes
		cm_is_rgo_not_max_level_already = yes
		cm_location_rgo_promotion_buffer_slack >= 0
	}
}

# This trigger is used by the GUI to see if the prerequisites for auto expanding RGO are met
# Expects scopes: country, location, income, profit
cm_is_able_to_auto_expand_rgo = {
	AND = {
		owner = {
			cm_should_pause_auto_expand = no
			is_target_in_variable_list = { name = cm_priority_features_enabled target = flag:cm_feature_auto_expand_rgos }
		}
		cm_rgo_basic_expand_eligibility = yes
		cm_is_rgo_under_construction = no
		cm_check_country_gold_above = { threshold = cm_expected_gold_available_for_rgo_auto_expand }
		cm_rgo_metric_gate_passes = yes
		cm_rgo_meets_minimum_discount = { threshold = cm_get_min_discount_for_auto_expand_rgos }
		cm_market_has_construction_goods_for_rgo = yes
	}
}

# Root is the location.
# The owner feature-enabled check prevents suppressing RGO auto-expand while auto-food is off.
cm_location_auto_food_supersedes_rgo = {
	owner = { is_target_in_variable_list = { name = cm_priority_features_enabled target = flag:cm_feature_auto_food } }
	cm_location_is_auto_food_active = yes
}

# This trigger is used at staging time to determine if the RGO should be queued.
# The metric gate runs at drain time in cm_gui_try_rgo, where the engine profit value exists.
# Expects scopes: country, location
cm_should_rgo_auto_expand = {
	scope:cm_location = {
		AND = {
			NOT = { cm_location_auto_food_supersedes_rgo = yes }
			cm_rgo_basic_expand_eligibility = yes
			cm_is_rgo_under_construction = no
			cm_check_country_gold_above = { threshold = cm_expected_gold_available_for_rgo_auto_expand_plus_planned_spend }
			cm_rgo_meets_minimum_discount = { threshold = cm_get_min_discount_for_auto_expand_rgos }
		}
	}
}

# Root is the location. Expects scopes: country, income, profit.
# Metric gate for RGO expansion. scope:cm_income is the RGO profit scaled by the
# location income-to-wealth ratio. The threshold is monthly gold per level for
# Tax Income and Profit, years to recoup the expansion cost for the ROI metrics.
cm_rgo_metric_gate_passes = {
	OR = {
		NOT = { scope:cm_country = { has_variable_map = cm_priority_min_profit } }
		"scope:cm_country.variable_map(cm_priority_min_profit|flag:cm_feature_auto_expand_rgos)" = 0
		AND = {
			cm_expand_metric_is_tax_income = yes
			scope:cm_income >= "scope:cm_country.variable_map(cm_priority_min_profit|flag:cm_feature_auto_expand_rgos)"
		}
		AND = {
			cm_expand_metric_is = { metric = 2 }
			scope:cm_profit >= "scope:cm_country.variable_map(cm_priority_min_profit|flag:cm_feature_auto_expand_rgos)"
		}
		AND = {
			cm_expand_metric_is = { metric = 3 }
			glorpui_rgo_expand_cost <= {
				value = scope:cm_income
				multiply = "scope:cm_country.variable_map(cm_priority_min_profit|flag:cm_feature_auto_expand_rgos)"
				multiply = 12
			}
		}
		AND = {
			cm_expand_metric_is = { metric = 4 }
			glorpui_rgo_expand_cost <= {
				value = scope:cm_profit
				multiply = "scope:cm_country.variable_map(cm_priority_min_profit|flag:cm_feature_auto_expand_rgos)"
				multiply = 12
			}
		}
	}
}

# Root is the location. Expects scopes: country.
# $threshold$ = the full variable_map lookup expression for the discount threshold.
# The 0.005 added to $threshold$ is a 0.5% tolerance so that a setting of -33%
# matches a cost adjustment of -32.5%, avoiding float-rounding misses at the boundary.
cm_rgo_meets_minimum_discount = {
	OR = {
		NOT = { scope:cm_country = { has_variable_map = cm_priority_min_discount } }
		glorpui_rgo_construction_cost_adjustment <= { value = $threshold$ add = 0.005 }
	}
}

cm_is_rgo_not_max_level_already = {
	cm_modifier_rgo_level < cm_get_max_rgo_workers_floor
}

cm_is_rgo_under_construction = {
	cm_get_rgos_under_construction > 0
}

# Root is location
# Expects scopes: country
cm_is_rgo_advance_researched = {
	trigger_if = { limit = { OR = { raw_material = goods:iron raw_material = goods:coal } }
		scope:cm_country = { has_advance = iron_working }
	} trigger_else_if = { limit = { OR = { raw_material = goods:copper raw_material = goods:tin raw_material = goods:lead } }
		scope:cm_country = { has_advance = bronze_working }
	} trigger_else_if = { limit = { OR = { raw_material = goods:goods_gold raw_material = goods:silver raw_material = goods:mercury } }
		scope:cm_country = { has_advance = fine_metals_mining }
	} trigger_else_if = { limit = { OR = { raw_material = goods:wheat raw_material = goods:maize raw_material = goods:potato raw_material = goods:rice raw_material = goods:millet raw_material = goods:legumes } }
		scope:cm_country = { has_advance = agriculture_advance }
	} trigger_else_if = { limit = { raw_material = goods:horses }
		scope:cm_country = { has_advance = ranching }
	} trigger_else_if = { limit = { OR = { raw_material = goods:stone raw_material = goods:marble } }
		scope:cm_country = { has_advance = mining_advance }
	} trigger_else_if = { limit = { OR = { raw_material = goods:saltpeter raw_material = goods:alum } }
		scope:cm_country = { has_advance = advanced_mining }
	} trigger_else = {
		always = yes
	}
}
