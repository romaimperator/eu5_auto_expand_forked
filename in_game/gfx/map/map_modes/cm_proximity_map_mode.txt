		}
		else_if = {
			limit = { cm_location_proximity_modifier >= -1.0 }

		if = {
			limit = {
				is_land = yes

				limit = { modifier:natural_harbor_suitability >= 0.5 }
				lerp = {
					max_color = define:NMapColors|MAP_COLOR_HIGH
					min_color = define:NMapColors|MAP_COLOR_MID
					factor = {
						value = modifier:natural_harbor_suitability
						subtract = 0.5
						multiply = 2
					}
				}
			}
			else = {
				lerp = {
					max_color = define:NMapColors|MAP_COLOR_MID
					min_color = define:NMapColors|MAP_COLOR_LOW
					factor = {
						value = modifier:natural_harbor_suitability
						multiply = 2
					}
				}
			}
		}
	}
